# ==== Overseas dealings events (India batch 17) ====

# .1: Choose a regional country (top 3 by relations) to sign 10-year foreign investment rights.
nesnd_india_overseas.1 = {
	type = country_event
	event_image = { video = "unspecific_signed_contract" }
	title = nesnd_india_overseas.1.t
	desc = nesnd_india_overseas.1.d
	flavor = nesnd_india_overseas.1.f
	trigger = { exists = scope:india_overseas_dir exists = scope:india_overseas_region }
	immediate = {
		if = { limit = { scope:india_overseas_dir = 1 } nesnd_india_overseas_pick_fir = { REGION = region_himalayas } }
		else_if = { limit = { scope:india_overseas_dir = 2 } nesnd_india_overseas_pick_fir = { REGION = region_indonesia } }
		else_if = { limit = { scope:india_overseas_dir = 3 } nesnd_india_overseas_pick_fir = { REGION = region_east_africa } }
		else_if = { limit = { scope:india_overseas_dir = 4 } nesnd_india_overseas_pick_fir = { REGION = region_arabia } }
		else = { nesnd_india_overseas_pick_fir = { REGION = region_greater_persia } }
	}
	option = {
		name = nesnd_india_overseas.1.a
		trigger = { nesnd_india_overseas_fir_option_ok = { PICK = scope:india_overseas_pick_1 } }
		show_as_unavailable = { exists = scope:india_overseas_pick_1 }
		ai_chance = { base = 1 }
		nesnd_india_overseas_sign_fir = { TARGET = scope:india_overseas_pick_1 }
	}
	option = {
		name = nesnd_india_overseas.1.b
		trigger = { nesnd_india_overseas_fir_option_ok = { PICK = scope:india_overseas_pick_2 } }
		show_as_unavailable = { exists = scope:india_overseas_pick_2 }
		ai_chance = { base = 0 }
		nesnd_india_overseas_sign_fir = { TARGET = scope:india_overseas_pick_2 }
	}
	option = {
		name = nesnd_india_overseas.1.c
		trigger = { nesnd_india_overseas_fir_option_ok = { PICK = scope:india_overseas_pick_3 } }
		show_as_unavailable = { exists = scope:india_overseas_pick_3 }
		ai_chance = { base = 0 }
		nesnd_india_overseas_sign_fir = { TARGET = scope:india_overseas_pick_3 }
	}
	option = {
		name = nesnd_india_overseas.1.d_opt
		default_option = yes
		ai_chance = { base = 1 }
		custom_tooltip = nesnd_india_overseas_not_now_tt
	}
}

# .5: Reaction of a major or great power holding states (directly or via subjects) in the region when
# an Indian empire's dealings first reach 50. Five-year per-power cooldown.
nesnd_india_overseas.5 = {
	type = country_event
	event_image = { video = "unspecific_politicians_arguing" }
	title = nesnd_india_overseas.5.t
	desc = nesnd_india_overseas.5.d
	flavor = nesnd_india_overseas.5.f
	trigger = { exists = scope:india_overseas_actor exists = scope:india_overseas_region }
	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:india_overseas_actor }
			scope:india_overseas_actor = { is_country_alive = no }
		}
	}
	option = {
		name = nesnd_india_overseas.5.a
		ai_chance = {
			base = 40
			modifier = {
				trigger = { gdp > scope:india_overseas_actor.gdp }
				add = -10
			}
		}
		change_relations = { country = scope:india_overseas_actor value = -20 }
		hidden_effect = {
			save_scope_as = india_overseas_reactor
			save_scope_value_as = { name = india_overseas_react_choice value = 1 }
			scope:india_overseas_actor = { trigger_event = { id = nesnd_india_overseas.6 popup = yes } }
		}
	}
	option = {
		name = nesnd_india_overseas.5.b
		ai_chance = {
			base = 10
			modifier = {
				trigger = { gdp > scope:india_overseas_actor.gdp }
				add = 40
			}
		}
		custom_tooltip = {
			text = nesnd_india_overseas.5.b_tt
			set_secret_goal = { country = scope:india_overseas_actor secret_goal = antagonize }
		}
		hidden_effect = {
			save_scope_as = india_overseas_reactor
			save_scope_value_as = { name = india_overseas_react_choice value = 2 }
			scope:india_overseas_actor = { trigger_event = { id = nesnd_india_overseas.6 popup = yes } }
		}
	}
	option = {
		name = nesnd_india_overseas.5.c
		default_option = yes
		ai_chance = {
			base = 50
			modifier = {
				trigger = { gdp > scope:india_overseas_actor.gdp }
				add = -30
			}
		}
		custom_tooltip = nesnd_india_overseas.5.c_tt
		hidden_effect = {
			save_scope_as = india_overseas_reactor
			save_scope_value_as = { name = india_overseas_react_choice value = 3 }
			scope:india_overseas_actor = { trigger_event = { id = nesnd_india_overseas.6 popup = yes } }
		}
	}
}

# .6: Notification to the Indian empire reporting the power's reaction.

	type = country_event
	event_image = { video = "unspecific_politicians_arguing" }
	title = nesnd_india_overseas.6.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { exists = scope:india_overseas_react_choice scope:india_overseas_react_choice = 1 } desc = nesnd_india_overseas.6.d_protest }

	cancellation_trigger = { NOT = { exists = scope:india_overseas_reactor } }

		name = nesnd_india_overseas.6.a

		if = { limit = { exists = scope:india_overseas_react_choice scope:india_overseas_react_choice = 1 } custom_tooltip = nesnd_india_overseas.6.a_protest_tt }
