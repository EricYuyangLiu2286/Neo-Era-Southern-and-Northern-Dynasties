# Fired by the Denglai button nesnd_sha_yan_subsidy when the funds are not traced.
nesnd_cne_yan.6 = {
    type = country_event
    placement = root
    title = nesnd_cne_yan.6.t
    desc = nesnd_cne_yan.6.d
    flavor = nesnd_cne_yan.6.f
    duration = 3
    event_image = { video = "asia_poor_people_moving" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = { c:CNE ?= this any_scope_state = { nesnd_sha_yan_region = yes } }
    immediate = {
        random_scope_state = {
            limit = { nesnd_sha_yan_region = yes }
            save_scope_as = nesnd_cne_yan_unrest_state
        }
    }
    option = {
        name = nesnd_cne_yan.6.a
        default_option = yes
        scope:nesnd_cne_yan_unrest_state ?= {
            add_radicals_in_state = { culture = cu:ce_liaodong value = 0.03 }
            add_radicals_in_state = { culture = cu:manchu value = 0.03 }
        }
    }
}
