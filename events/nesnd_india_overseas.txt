namespace = nesnd_india_overseas

# The Indian state chooses the local government to ask (from a button of the journal nesnd_india_overseas).
# scope:india_overseas_dir is the direction (1-3), scope:india_overseas_kind 10 (trade privilege) or
# 11 (treaty port). Up to four local governments are listed by their relations with us.
nesnd_india_overseas.1 = {
    type = country_event
    event_image = { video = "unspecific_signed_contract" }
    title = nesnd_india_overseas.1.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { scope:india_overseas_kind = 10 scope:india_overseas_dir = 1 } desc = nesnd_india_overseas.1.d_trade_1 }
            triggered_desc = { trigger = { scope:india_overseas_kind = 10 scope:india_overseas_dir = 2 } desc = nesnd_india_overseas.1.d_trade_2 }
            triggered_desc = { trigger = { scope:india_overseas_kind = 10 } desc = nesnd_india_overseas.1.d_trade_3 }
            triggered_desc = { trigger = { scope:india_overseas_dir = 1 } desc = nesnd_india_overseas.1.d_port_1 }
            triggered_desc = { trigger = { scope:india_overseas_dir = 2 } desc = nesnd_india_overseas.1.d_port_2 }
            desc = nesnd_india_overseas.1.d_port_3
        }
    }
    flavor = nesnd_india_overseas.1.f
    trigger = { exists = scope:india_overseas_dir exists = scope:india_overseas_kind }
    immediate = {
        if = { limit = { scope:india_overseas_dir = 1 } nesnd_india_overseas_pick = { DIR = 1 } }
        else_if = { limit = { scope:india_overseas_dir = 2 } nesnd_india_overseas_pick = { DIR = 2 } }
        else = { nesnd_india_overseas_pick = { DIR = 3 } }
    }
    option = {
        name = nesnd_india_overseas.1.a
        trigger = { nesnd_india_overseas_option_ok = { PICK = scope:india_overseas_pick_1 } }
        show_as_unavailable = { exists = scope:india_overseas_pick_1 }
        ai_chance = { base = 1 }
        nesnd_india_overseas_option_tooltips = { PICK = scope:india_overseas_pick_1 PORT = scope:india_overseas_port_1 N = 1 }
        hidden_effect = { nesnd_india_overseas_send = { PICK = scope:india_overseas_pick_1 } }
    }
    option = {
        name = nesnd_india_overseas.1.b
        trigger = { nesnd_india_overseas_option_ok = { PICK = scope:india_overseas_pick_2 } }
        show_as_unavailable = { exists = scope:india_overseas_pick_2 }
        ai_chance = { base = 0 }
        nesnd_india_overseas_option_tooltips = { PICK = scope:india_overseas_pick_2 PORT = scope:india_overseas_port_2 N = 2 }
        hidden_effect = { nesnd_india_overseas_send = { PICK = scope:india_overseas_pick_2 } }
    }
    option = {
        name = nesnd_india_overseas.1.c
        trigger = { nesnd_india_overseas_option_ok = { PICK = scope:india_overseas_pick_3 } }
        show_as_unavailable = { exists = scope:india_overseas_pick_3 }
        ai_chance = { base = 0 }
        nesnd_india_overseas_option_tooltips = { PICK = scope:india_overseas_pick_3 PORT = scope:india_overseas_port_3 N = 3 }
        hidden_effect = { nesnd_india_overseas_send = { PICK = scope:india_overseas_pick_3 } }
    }
    option = {
        name = nesnd_india_overseas.1.d
        trigger = { nesnd_india_overseas_option_ok = { PICK = scope:india_overseas_pick_4 } }
        show_as_unavailable = { exists = scope:india_overseas_pick_4 }
        ai_chance = { base = 0 }
        nesnd_india_overseas_option_tooltips = { PICK = scope:india_overseas_pick_4 PORT = scope:india_overseas_port_4 N = 4 }
        hidden_effect = { nesnd_india_overseas_send = { PICK = scope:india_overseas_pick_4 } }
    }
    option = {
        name = nesnd_india_overseas.1.e
        default_option = yes
        ai_chance = { base = 1 }
        custom_tooltip = nesnd_india_overseas_not_now_tt
    }
}

# The local government agreed: a treaty signed, or a renewal agreed. To the Indian state.
nesnd_india_overseas.2 = {
    type = country_event
    event_image = { video = "unspecific_signed_contract" }
    title = nesnd_india_overseas.2.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { exists = scope:india_matter_kind scope:india_matter_kind = 10 } desc = nesnd_india_overseas.2.d_trade }
            triggered_desc = { trigger = { exists = scope:india_matter_kind scope:india_matter_kind = 11 exists = scope:india_overseas_port_signed } desc = nesnd_india_overseas.2.d_port }
            triggered_desc = { trigger = { exists = scope:india_matter_kind scope:india_matter_kind = 12 } desc = nesnd_india_overseas.2.d_renew_trade }
            desc = nesnd_india_overseas.2.d_renew_port
        }
    }
    flavor = nesnd_india_overseas.2.f
    trigger = { exists = scope:india_matter_to }
    cancellation_trigger = { NOT = { exists = scope:india_matter_to } }
    option = {
        name = nesnd_india_diplomacy_ack
        default_option = yes
        if = { limit = { exists = scope:india_matter_kind scope:india_matter_kind = 10 } custom_tooltip = nesnd_india_overseas_signed_trade_tt }
        else_if = { limit = { exists = scope:india_matter_kind scope:india_matter_kind = 11 } custom_tooltip = nesnd_india_overseas_signed_port_tt }
        else_if = { limit = { exists = scope:india_matter_kind scope:india_matter_kind = 12 } custom_tooltip = nesnd_india_overseas_renewed_trade_tt }
        else = { custom_tooltip = nesnd_india_overseas_renewed_port_tt }
    }
}

# No agreement. To the Indian state.
nesnd_india_overseas.3 = {
    type = country_event
    event_image = { video = "unspecific_signed_contract" }
    title = nesnd_india_overseas.3.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { exists = scope:india_matter_result scope:india_matter_result = 1 exists = scope:india_matter_permit } desc = nesnd_india_overseas.3.d_forbid }
            triggered_desc = { trigger = { exists = scope:india_matter_result scope:india_matter_result = 2 exists = scope:india_matter_kind scope:india_matter_kind >= 12 exists = scope:india_matter_to } desc = nesnd_india_overseas.3.d_renew_decline }
            triggered_desc = { trigger = { exists = scope:india_matter_result scope:india_matter_result = 3 exists = scope:india_matter_kind scope:india_matter_kind >= 12 } desc = nesnd_india_overseas.3.d_renew_silent }
            triggered_desc = { trigger = { exists = scope:india_matter_result scope:india_matter_result = 2 exists = scope:india_matter_to } desc = nesnd_india_overseas.3.d_decline }
            triggered_desc = { trigger = { exists = scope:india_matter_result scope:india_matter_result = 3 } desc = nesnd_india_overseas.3.d_silent }
            desc = nesnd_india_overseas.3.d_void
        }
    }
    flavor = nesnd_india_overseas.3.f
    option = {
        name = nesnd_india_diplomacy_ack
        default_option = yes

        else = { custom_tooltip = nesnd_india_overseas_failed_tt }
    }
}
