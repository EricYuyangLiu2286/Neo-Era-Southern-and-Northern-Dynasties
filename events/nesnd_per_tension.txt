namespace = nesnd_per_tension

# 「波斯之忧」月度风味事件。每个选项留下 36 个月的余波（进度条读取 var:nesnd_per_tension_evN）。

nesnd_per_tension.1 = {
    type = country_event
    title = nesnd_per_tension.1.t
    desc = nesnd_per_tension.1.d
    flavor = nesnd_per_tension.1.f
    event_image = { video = "middleeast_islam" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_protest.dds"
    duration = 3
    cooldown = { days = 3650 }
    trigger = {
        nesnd_per_tension_live = yes
        any_scope_state = { nesnd_per_iranian_state = yes }
    }
    immediate = {
        random_scope_state = {
            limit = { nesnd_per_iranian_state = yes }
            save_scope_as = nesnd_per_tension_state
        }
    }
    option = {
        name = nesnd_per_tension.1.a
        default_option = yes
        custom_tooltip = nesnd_per_tension_ev_calm_tt
        hidden_effect = { set_variable = { name = nesnd_per_tension_ev1 value = -0.04 months = 36 } }
        ig:ig_armed_forces ?= { add_modifier = { name = nesnd_per_ig_resentful months = 24 } }
        ai_chance = { base = 50 }
    }
    option = {
        name = nesnd_per_tension.1.b
        custom_tooltip = nesnd_per_tension_ev_heat_tt
        hidden_effect = { set_variable = { name = nesnd_per_tension_ev1 value = 0.04 months = 36 } }
        ig:ig_devout ?= { add_modifier = { name = nesnd_per_ig_resentful months = 36 } }
            ig:ig_armed_forces ?= { add_modifier = { name = nesnd_per_ig_favored months = 24 } }
        ai_chance = { base = 50 }
    }
}
