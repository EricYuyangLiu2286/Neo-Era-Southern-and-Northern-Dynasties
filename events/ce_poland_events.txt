# Event 2: Diplo play from KRA
ce_poland_events.2 = {
	type = country_event
	placement = root
	title = ce_poland_events.2.t
	desc = ce_poland_events.2.d
	flavor = ce_poland_events.2.f
	duration = 3
	event_image = {
		video = "europenorthamerica_before_the_battle"
	}

	immediate = {
		save_scope_as = ce_claim_state_target
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	option = {
		name = ce_poland_events.2.a
		ai_chance = {
			base = 1
		}
		# Yield
		change_infamy = -5
		
		hidden_effect = {
			if = {
				limit = {
					has_modifier = ce_backoff_on_territorial_claims
				}
				remove_modifier = ce_backoff_on_territorial_claims
			}
		}
		
		add_modifier = {
			name = ce_backoff_on_territorial_claims
			months = 12
		}

		scope:target_diplo_play_state = {
			hidden_effect = {
				if = {
					limit = {
						has_modifier = ce_peaceful_vote_for_polish
					}
					remove_modifier = ce_peaceful_vote_for_polish
				}
			}
			#c:KRA = { set_state_owner = prev }
			set_state_owner = c:KRA
			add_modifier = {
				name = ce_peaceful_vote_for_polish
				months = 12
			}
		}
	}

	option = {
		name = ce_poland_events.2.b
		ai_chance = {
			base = 9
		}
		# Fight!
		default_option = yes
		c:KRA = {
			create_diplomatic_play = {
				#name = ce_poland_diplo_play_name
				escalation = 50
				target_country = root
				type = dp_return_state

				add_war_goal = { holder = root target_country = c:KRA type = annex_country }

				add_war_goal = {
					holder = c:KRA
					type = return_state
					target_state = scope:target_diplo_play_state
					target_country = root
				}
			}
			add_modifier = {
				name = ce_poland_war_prep_diplo_play_buff
				months = 12
			}
		}

		random_diplomatic_play = {
			limit = {
				c:KRA ?= {
					is_diplomatic_play_initiator = yes
				}
				root = {
					is_diplomatic_play_target = yes
				}
			}
			save_scope_as = ce_poland_return_land_dp
		}

		every_country = {
			limit = { has_variable = ce_pro_polish_variable }
			# scope:ce_poland_return_land_dp = {
			# 	add_initiator_backers = { prev }
			# }
			trigger_event = { id = ce_poland_events.7 popup = yes }
		}
	}
}

# Event 10: Victory Complete
ce_poland_events.10 = {
	type = country_event
	placement = root
	title = ce_poland_events.10.t
	desc = ce_poland_events.10.d
	duration = 3
	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	immediate = {
		if = {
			limit = { has_variable = ce_pro_polish_variable }

			every_scope_state = {
				limit = {
					any_scope_pop = { culture = cu:polish }
				}
				

				root = {
add_to_variable_list = {
					name = ce_list_state_effective
					target = prev
				}
}
			}
			
		}
		if = {
			limit = { has_variable = ce_anti_polish_variable }
			hidden_effect = {
				every_scope_state = {
					limit = {
						any_scope_pop = { culture = cu:polish }
					}
					

					root = {
add_to_variable_list = {
						name = ce_list_state_effective
						target = prev
					}
}
				}
			}
			
		}
	}

	option = {
		name = ce_poland_events.10.a
		if = {
			limit = { has_variable = ce_pro_polish_variable }
				
			every_in_list = {
				variable = ce_list_state_effective
				add_loyalists_in_state = {
					value = 0.5
					culture = cu:polish
				}
			}

			clear_variable_list = ce_list_state_effective
		}
		if = {
			limit = { has_variable = ce_anti_polish_variable }

			every_in_list = {
				variable = ce_list_state_effective
				add_radicals_in_state = {
					value = 0.5
					culture = cu:polish
				}
			}

			clear_variable_list = ce_list_state_effective
		}
		if = {
			limit = { exists = c:KRA c:KRA ?= this }
			change_tag = POL

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_WEST_GALICIA } 
					NOT = { has_claim = s:STATE_WEST_GALICIA }
				}
				s:STATE_WEST_GALICIA = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_UPPER_SILESIA } 
					NOT = { has_claim = s:STATE_UPPER_SILESIA }
				}
				s:STATE_UPPER_SILESIA = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_EAST_GALICIA } 
					NOT = { has_claim = s:STATE_EAST_GALICIA }
				}
				s:STATE_EAST_GALICIA = { add_claim = c:POL }
			}
				
			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_POSEN } 
					NOT = { has_claim = s:STATE_POSEN }
				}
				s:STATE_POSEN = { add_claim = c:POL }
			}
			
			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_KAUNAS } 
					NOT = { has_claim = s:STATE_KAUNAS }
				}
				s:STATE_KAUNAS = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_BREST } 
					NOT = { has_claim = s:STATE_BREST }
				}
				s:STATE_BREST = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_VOLHYNIA } 
					NOT = { has_claim = s:STATE_VOLHYNIA }
				}
				s:STATE_VOLHYNIA = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_WEST_PRUSSIA } 
					NOT = { has_claim = s:STATE_WEST_PRUSSIA }
				}
				s:STATE_WEST_PRUSSIA = { add_claim = c:POL }
			}
			
			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_GREATER_POLAND } 
					NOT = { has_claim = s:STATE_GREATER_POLAND }
				}
				s:STATE_GREATER_POLAND = { add_claim = c:POL }
			}

			if = {
				limit = { 
					NOT = { owns_entire_state_region = STATE_LESSER_POLAND } 
					NOT = { has_claim = s:STATE_LESSER_POLAND }
				}
				s:STATE_LESSER_POLAND = { add_claim = c:POL }
			}

			hidden_effect = {
				every_state = {
					limit = {
						culture_percent_state = {
							target = cu:polish
							value >= 0.33
						}	
					}

					root = {
add_to_variable_list = {
						name = ce_list_state_effective
						target = prev
					}
}

				}
			}	

			every_in_list = {
				variable = ce_list_state_effective
				state_region = { add_homeland = cu:polish }
			}
				
			add_modifier = {
				name = ce_poland_victory_buff
				months = 120
			}

			clear_variable_list = ce_list_state_effective
		}
	}
}

# Event 11: Failure
ce_poland_events.11 = {
	type = country_event
	placement = root
	title = ce_poland_events.11.t
	desc = ce_poland_events.11.d
	duration = 3
	event_image = {
		video = "europenorthamerica_before_the_battle"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	immediate = {
		if = {
			limit = { has_variable = ce_pro_polish_variable }
			
			
			every_interest_group = {
				limit = {
					OR = {
						is_interest_group_type = ig_landowners
						is_interest_group_type = ig_intelligentsia
						is_interest_group_type = ig_petty_bourgeoisie
					}
				}
				add_modifier = {
					name = ce_poland_lobby_approval_debuff
					months = 12
				}
			}
		}
		
		
	}

	option = {
		name = ce_poland_events.11.a
		if = {
			limit = { exists = c:KRA c:KRA ?= this }
			add_modifier = {
				name = ce_poland_failure_pro_debuff
				months = 120
			}
			# Independence for Lithuania and Belarus
			if = {
				limit = {
					OR = {
						owns_entire_state_region = STATE_KAUNAS 
						owns_entire_state_region = STATE_BREST
					}
				}
			
				if = {
					limit = { 
						owns_entire_state_region = STATE_KAUNAS 
					}
					hidden_effect = {
						s:STATE_KAUNAS = {
							if = {
								limit = {
									NOT = { is_homeland = cu:lithuanian }
								}
								add_homeland = cu:lithuanian
							}
						}
					}
						
					create_country = {
						tag = LIT
						origin = root
						state = s:STATE_KAUNAS.region_state:KRA
					}

					hidden_effect = {
						c:LIT = { set_capital = STATE_KAUNAS }
						s:STATE_KAUNAS = {
							add_claim = root
						}
					}

					create_diplomatic_play = {
						target_state = c:LIT.capital

						war = yes

						type = dp_annex_war

						add_war_goal = {
							holder = c:KRA
							type = annex_country
							target_country = c:LIT
						}

						add_war_goal = {
							holder = c:LIT
							type = make_protectorate
							target_country = c:KRA
						}
					}
						
				}
				else = {
					every_scope_state = {
						add_radicals_in_state = { value = 0.05 }
					}
				}
				#
				if = {
					limit = { owns_entire_state_region = STATE_BREST }

					hidden_effect = {
						s:STATE_BREST = {
							if = {
								limit = {
									NOT = { is_homeland = cu:byelorussian }
								}
								add_homeland = cu:byelorussian
							}
						}
					}
						
					create_country = {
						tag = BYE
						origin = root
						state = s:STATE_BREST.region_state:KRA
					}
				}
				else = {
					every_scope_state = {
						add_radicals_in_state = { value = 0.05 }
					}
				}
			}
			else = {
				every_scope_state = {
					add_radicals_in_state = { value = 0.1 }
				}
			}
		}

		if = {
			limit = { has_variable = ce_pro_polish_variable }
			add_modifier = {
				name = ce_poland_failure_pro_debuff
				months = 120
			}
			custom_tooltip = {
				text = ce_scope_state_polish_gain_radicals
				every_scope_state = {
					limit = {
						any_scope_pop = { culture = cu:polish }
					}
					add_radicals_in_state = {
						value = 0.10
						culture = cu:polish
					}
				}
			}
			
			custom_tooltip = ce_supporters_of_polish_disappointed
		}

		if = {
			limit = { has_variable = ce_anti_polish_variable }
			add_modifier = {
				name = ce_poland_failure_anti_buff
				months = 120
			}

			s:STATE_WEST_GALICIA = { add_claim = root }
			s:STATE_UPPER_SILESIA = { add_claim = root }
			s:STATE_EAST_GALICIA = { add_claim = root }
			s:STATE_POSEN = { add_claim = root }
			s:STATE_KAUNAS = { add_claim = root }
			s:STATE_BREST = { add_claim = root }
			s:STATE_VOLHYNIA = { add_claim = root }
			s:STATE_WEST_PRUSSIA = { add_claim = root }
			s:STATE_LESSER_POLAND = { add_claim = root }
			
			custom_tooltip = {
				text = ce_polish_lost_homeland_in_our_tag_tt
				every_scope_state = {
					limit = {
						NOT = { state_region = s:STATE_GREATER_POLAND }
						state_region = { is_homeland = cu:polish }
					}
					state_region = { remove_homeland = cu:polish }
				}
			}
				
		}
	}
}

# Event 98: Monthly Pulse
ce_poland_events.98 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = c:KRA
		c:KRA ?= This
	}

	immediate = {
		if = {
			limit = { exists = c:KRA c:KRA ?= this }
			
			# Check war prep progress bar for infamy addition
			if = {
				limit = {
					has_journal_entry = je_ce_polish_war_preparation
					je:je_ce_polish_war_preparation = { "scripted_bar_progress(ce_polish_reconquest_war_prep_bar)" >= 30 }
					infamy < 20
				}
				change_infamy = 0.05
			}
				
			# Calculate owned polish states
			set_variable = {
				name = ce_poland_owned_polish_states
				value = 0
			}
			
			
			if = { limit = { owns_entire_state_region = STATE_WEST_GALICIA } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_UPPER_SILESIA } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_EAST_GALICIA } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_POSEN } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_KAUNAS } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_BREST } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_VOLHYNIA } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_WEST_PRUSSIA } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_GREATER_POLAND } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			if = { limit = { owns_entire_state_region = STATE_LESSER_POLAND } change_variable = { name = ce_poland_owned_polish_states add = 1 } }
			
			set_global_variable = {
				name = ce_poland_owned_polish_states_global
				value = var:ce_poland_owned_polish_states
			}
			
			ce_update_poland_supporter_and_haters_effect = yes

			set_global_variable = {
				name = ce_poland_support_diff
				value = global_var:ce_poland_supporters_count
			}
			change_global_variable = {
				name = ce_poland_support_diff
				subtract = global_var:ce_poland_anti_count
			}

			if = {
				limit = {
					exists = global_var:ce_polish_claim_state_count
					has_journal_entry = je_ce_polish_reconquest
				}
				set_global_variable = {
					name = ce_polish_claim_state_count_inverse
					value = global_var:ce_polish_claim_state_count
				}
				change_global_variable = {
					name = ce_polish_claim_state_count_inverse
					multiply = -1
				}
			}
		}
	}
}
