namespace = nesnd_jp_court
nesnd_jp_court.1 = { type = country_event hidden = yes trigger = { c:JIM ?= this is_country_alive = yes } immediate = { nesnd_jp_refresh_court = yes } }

nesnd_jp_court.2 = {
    type = country_event hidden = yes
    trigger = { nesnd_jp_country = yes }
    immediate = { remove_variable = nesnd_jp_korea_check_pending nesnd_jp_ensure_korea = yes }
}

nesnd_jp_court.3 = {
    type = country_event hidden = yes
    trigger = { nesnd_jp_country = yes }
    immediate = {
        remove_variable = nesnd_jp_statistics_pending
        if = { limit = { total_population > 0 }
            set_variable = nesnd_jp_statistics_ready
            if = { limit = { NOT = { has_variable = nesnd_jp_living_baseline } }
                set_variable = { name = nesnd_jp_sol_previous value = average_sol }
                set_variable = nesnd_jp_living_baseline
            }
            nesnd_jp_bootstrap = yes
        }
    }
}
