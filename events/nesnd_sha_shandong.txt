    }
    option = {
        name = nesnd_sha_qi.2.a

        trigger = {
            NOT = { has_war_with = scope:nesnd_sha_qi_host }
            OR = {

                can_create_treaty = { first_country = c:SHA second_country = scope:nesnd_sha_qi_host is_draft = no articles_to_create = { { article = alliance } } }
            }
        }
        ai_chance = { base = 95 }
        if = {
            limit = { NOT = { has_treaty_alliance_with = { TARGET = scope:nesnd_sha_qi_host } } }
            create_treaty = { first_country = c:SHA second_country = scope:nesnd_sha_qi_host is_draft = no articles_to_create = { { article = alliance } } }
        }
        custom_tooltip = nesnd_sha_qi_joint_tt
        hidden_effect = {
            set_variable = { name = nesnd_sha_shandong_sister value = scope:nesnd_sha_qi_host }
            set_variable = { name = nesnd_sha_sister_pact months = 12 }
        }
    }
    option = {
        name = nesnd_sha_qi.2.b
        ai_chance = { base = 5 }
        if = { limit = { has_modifier = nesnd_sha_sole_republic } remove_modifier = nesnd_sha_sole_republic }
        if = { limit = { has_modifier = nesnd_sha_sister_republic } remove_modifier = nesnd_sha_sister_republic }
        add_radicals = { value = 0.05 }
        nesnd_sha_declare_claim = yes
    }
}

# Sent to the allied republic holding Shandong: Denglai asks for Dengzhou as a treaty port.
nesnd_sha_qi.9 = {
    type = country_event
    placement = root
    title = nesnd_sha_qi.9.t
    desc = nesnd_sha_qi.9.d
    flavor = nesnd_sha_qi.9.f
    duration = 3
    event_image = { video = "unspecific_signed_contract" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = { nesnd_sha_can_answer_dengzhou = yes }
    cancellation_trigger = { nesnd_sha_can_answer_dengzhou = no }
    immediate = { c:SHA = { save_scope_as = nesnd_sha_qi_petitioner } }
    option = {
        name = nesnd_sha_qi.9.a
        trigger = { nesnd_sha_can_answer_dengzhou = yes }
        ai_chance = {
            base = 70
            modifier = { trigger = { "relations(c:SHA)" >= relations_threshold:friendly } add = 20 }
        }
        p:xAC24A6.state = { cede_treaty_port = c:SHA }
        if = {
            limit = { p:xAC24A6.state.owner ?= c:SHA }
            change_relations = { country = c:SHA value = 20 }
            hidden_effect = { c:SHA = { set_variable = nesnd_sha_dengzhou_granted } }
        }
        hidden_effect = { c:SHA = { remove_variable = nesnd_sha_dengzhou_pending } }
    }
    option = {
        name = nesnd_sha_qi.9.b
        trigger = { nesnd_sha_can_answer_dengzhou = yes }
        default_option = yes
        ai_chance = { base = 30 }
        change_relations = { country = c:SHA value = -20 }
        hidden_effect = {
            c:SHA = {
                remove_variable = nesnd_sha_dengzhou_pending
                set_variable = { name = nesnd_sha_dengzhou_request_wait years = 5 }
                trigger_event = { id = nesnd_sha_qi.10 popup = yes }
            }
        }
    }
}
