namespace = nesnd_company

# Charter events for the national companies. Fired from on_action nesnd_co_yearly_charters.
# Option A sets the charter variable that the company type checks. Option B waits ten years.

# Tiangong Kaiwu Hong (Nanjing machine works)
nesnd_company.1 = {
	type = country_event
	placement = root
	title = nesnd_company.1.t
	desc = nesnd_company.1.d
	flavor = nesnd_company.1.f
	duration = 3
	event_image = { video = "unspecific_gears_pistons" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	trigger = {
		NOT = { has_variable = nesnd_co_tiangong_chartered }
		NOT = { has_variable = nesnd_co_tiangong_deferred }
		NOT = { any_country = { has_company = company_type:company_nesnd_tiangong } }
		has_technology_researched = mechanical_tools
		has_building_levels_in_state_region = { region = STATE_NANJING building = building_tooling_workshop levels >= 3 is_incorporated = yes }
	}
	option = {
		name = nesnd_company.1.a
		default_option = yes
		set_variable = nesnd_co_tiangong_chartered
		custom_tooltip = nesnd_co_tiangong_open_tt
		add_modifier = { name = nesnd_co_charter_expenses months = 24 multiplier = money_amount_multiplier_medium }
		ai_chance = { base = 80 }
	}
	option = {
		name = nesnd_company.1.b
		set_variable = { name = nesnd_co_tiangong_deferred years = 10 }
		custom_tooltip = nesnd_co_deferred_tt
		ai_chance = { base = 20 }
	}
}
