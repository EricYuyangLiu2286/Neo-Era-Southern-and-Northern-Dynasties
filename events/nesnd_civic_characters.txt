namespace = nesnd_civic_characters

nesnd_civic_characters.14 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_civic_characters.14.t
    event_image = { video = "unspecific_politicians_arguing" }
    trigger = { nesnd_c2_duan_introduction = yes any_scope_character = { nesnd_c2_duan_politician = yes } }
    immediate = { nesnd_c2_seat_duan_at_academy = yes }
    desc = nesnd_civic_characters.14.d
    flavor = nesnd_civic_characters.14.f
    cancellation_trigger = { NOT = { nesnd_c2_duan_country = yes nesnd_c2_duan_present = yes } }
    option = {
        name = nesnd_civic_characters.14.a
        set_global_variable = nesnd_introduced_usm_duan
        add_treasury = nesnd_civic_tuition_expense
        scope:c2_duan_school = { add_modifier = { name = nesnd_civic_public_tuition months = 24 } }
    }
    option = {
        name = nesnd_civic_characters.14.b
        default_option = yes
        set_global_variable = nesnd_introduced_usm_duan
        scope:c2_duan = { add_modifier = { name = nesnd_duan_lecturer months = 12 } }
    }
    after = { scope:c2_duan ?= { remove_variable = nesnd_story_civic_event } }
}
