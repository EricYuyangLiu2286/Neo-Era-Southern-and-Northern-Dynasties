# Fixed-delay notification is optional to settlement: the global epoch clock is authoritative.
nesnd_hus_registration.10 = {
    type = country_event hidden = yes
    trigger = {
        nesnd_hus_registration_open = yes
        trigger_if = { limit = { has_variable = nesnd_hus_registration_epoch exists = scope:nesnd_hus_delayed_epoch }
            var:nesnd_hus_registration_epoch = scope:nesnd_hus_delayed_epoch
        }
        trigger_else = { always = no }
        nesnd_hus_epoch_matches_record = yes
    }
    immediate = {
        nesnd_hus_registration_settle = yes
        if = { limit = {
            NOT = { has_variable = nesnd_hus_registration_done }
            has_global_variable = nesnd_hus_registration_wait
            NOT = { exists = scope:nesnd_hus_delayed_retry }
        }
            save_scope_value_as = { name = nesnd_hus_delayed_retry value = 1 }
            trigger_event = { id = nesnd_hus_registration.10 days = 1 }
        }
    }
}
