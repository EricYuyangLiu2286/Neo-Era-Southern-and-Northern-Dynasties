# 「伊犁的朝议」的朝争事件。由朝争条满（nesnd_qe1_intrigue_month）或按钮「召开廷议」抽取。
# 选项写入三方份额公式中的一时偏向（nesnd_qe1_sway_*，三年）。君权对应地主与宗教人士，文人对应知识分子与实业家，军队对应军方。

# 1 御前三议：任官之争
nesnd_qe1_intrigue.1 = {
	type = country_event
	placement = root
	title = nesnd_qe1_intrigue.1.t
	desc = nesnd_qe1_intrigue.1.d
	flavor = nesnd_qe1_intrigue.1.f
	duration = 3
	event_image = { video = "votp_monarch_holding_court" }
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_qe1_court_journal }

	option = {
		name = nesnd_qe1_intrigue.1.a
		nesnd_qe1_sway_add = { WHO = crown AMT = 10 }
		nesnd_qe1_sway_sub = { WHO = literati AMT = 10 }
		ai_chance = { base = 20 }
	}
	option = {
		name = nesnd_qe1_intrigue.1.b
		nesnd_qe1_sway_add = { WHO = literati AMT = 10 }
		nesnd_qe1_sway_sub = { WHO = crown AMT = 10 }
		ai_chance = { base = 20 }
	}
	option = {
		name = nesnd_qe1_intrigue.1.c
		default_option = yes
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		nesnd_qe1_intrigue_change = { DIR = add AMT = 30 }
		ai_chance = { base = 10 }
	}
}

# 4 一方独大（某方份额 ≥ 60）
nesnd_qe1_intrigue.4 = {
	type = country_event
	placement = root
	title = nesnd_qe1_intrigue.4.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { has_variable = nesnd_qe1_power_army var:nesnd_qe1_power_army >= 60 } desc = nesnd_qe1_intrigue.4.d_army }
			triggered_desc = { trigger = { has_variable = nesnd_qe1_power_literati var:nesnd_qe1_power_literati >= 60 } desc = nesnd_qe1_intrigue.4.d_literati }
			desc = nesnd_qe1_intrigue.4.d_crown
		}
	}
	flavor = nesnd_qe1_intrigue.4.f
	duration = 3
	event_image = { video = "votp_hostile_court" }
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_qe1_court_journal nesnd_qe1_court_dominant = yes }

	option = {
		name = nesnd_qe1_intrigue.4.a
		default_option = yes
		if = {
			limit = { has_variable = nesnd_qe1_power_army var:nesnd_qe1_power_army >= 60 }
			nesnd_qe1_sway_sub = { WHO = army AMT = 15 }
			if = { limit = { exists = ig:ig_armed_forces } add_radicals = { value = medium_radicals interest_group = ig:ig_armed_forces } }
		}
		else_if = {
			limit = { has_variable = nesnd_qe1_power_literati var:nesnd_qe1_power_literati >= 60 }
			nesnd_qe1_sway_sub = { WHO = literati AMT = 15 }
			if = { limit = { exists = ig:ig_intelligentsia } add_radicals = { value = medium_radicals interest_group = ig:ig_intelligentsia } }
		}
		else = {
			nesnd_qe1_sway_sub = { WHO = crown AMT = 15 }
			if = { limit = { exists = ig:ig_landowners } add_radicals = { value = medium_radicals interest_group = ig:ig_landowners } }
		}
		ai_chance = { base = 20 }
	}
	option = {
		name = nesnd_qe1_intrigue.4.b
		if = {
			limit = { has_variable = nesnd_qe1_power_army var:nesnd_qe1_power_army >= 60 }
			nesnd_qe1_sway_add = { WHO = army AMT = 5 }
		}
		else_if = {
			limit = { has_variable = nesnd_qe1_power_literati var:nesnd_qe1_power_literati >= 60 }
			nesnd_qe1_sway_add = { WHO = literati AMT = 5 }
		}
		else = {
			nesnd_qe1_sway_add = { WHO = crown AMT = 5 }
		}
		add_modifier = { name = nesnd_power_shamed days = short_modifier_time }
		ai_chance = { base = 10 }
	}
}
