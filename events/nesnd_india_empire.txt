nesnd_mug.1 = {
    type = country_event
    event_image = { video = "unspecific_signed_contract" }

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    NOT = { exists = scope:mug_deal_site }
                }
                desc = nesnd_mug.1.d_none
            }
            triggered_desc = {
                trigger = {
                    exists = scope:india_req_kind
                    scope:india_req_kind = 11
                }
                desc = nesnd_mug.1.d_buyback
            }
            desc = nesnd_mug.1.d_sell
        }
    }
    flavor = nesnd_mug.1.f
    trigger = { exists = scope:india_partner }
    cancellation_trigger = {
        NOT = { exists = scope:india_partner }
    }
    option = {
        name = nesnd_india_diplomacy_ack
        default_option = yes
        if = {
            limit = {
                NOT = { exists = scope:mug_deal_site }
            }
            custom_tooltip = nesnd_mug_unchanged_tt
        }
        else_if = {
            limit = {
                exists = scope:india_req_kind
                scope:india_req_kind = 11
            }
            custom_tooltip = nesnd_mug_bought_tt
            if = {
                limit = { exists = scope:mug_deal_price }
                show_as_tooltip = {
                    add_treasury = {
                        value = scope:mug_deal_price
                        multiply = -1
                    }
                }
            }
        }
        else = {
            custom_tooltip = nesnd_mug_sold_tt
            if = {
                limit = { exists = scope:mug_deal_price }
                show_as_tooltip = { add_treasury = scope:mug_deal_price }
            }
        }
    }
}

    }
    option = {
        name = nesnd_mug.12.a
        ai_chance = { base = 0 modifier = { trigger = { has_variable = nesnd_mug_council_ai_yes } add = 100 } }

        else = { custom_tooltip = nesnd_mug_attend_defence_tt }
        hidden_effect = {
            set_variable = { name = nesnd_mug_council_answer value = 1 days = 60 }
            remove_variable = nesnd_mug_council_ai_yes
        }
    }
    option = {
        name = nesnd_mug.12.b
        default_option = yes
        ai_chance = { base = 0 modifier = { trigger = { NOT = { has_variable = nesnd_mug_council_ai_yes } } add = 100 } }

            set_variable = { name = nesnd_mug_council_answer value = 2 days = 60 }
