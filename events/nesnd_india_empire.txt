# Each invited prince attends or declines.
nesnd_mug.12 = {
    type = country_event
    event_image = { video = "asia_politician_parliament_motion" }
    title = nesnd_mug.12.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_mug_council_topic_is = { T = 1 } } desc = nesnd_mug.12.d_rail }
            triggered_desc = { trigger = { nesnd_mug_council_topic_is = { T = 2 } } desc = nesnd_mug.12.d_relief }
            desc = nesnd_mug.12.d_defence
        }
    }
    flavor = nesnd_mug.12.f
    trigger = { exists = var:nesnd_mug_council_invite }
    cancellation_trigger = { NOT = { exists = var:nesnd_mug_council_invite } }
    immediate = {
        remove_variable = nesnd_mug_council_ai_yes
        if = { limit = { is_ai = yes }
            save_temporary_scope_value_as = { name = mug_ai_chance value = nesnd_mug_council_attend_chance }
            random = { chance = scope:mug_ai_chance set_variable = { name = nesnd_mug_council_ai_yes days = 60 } }
        }
    }
    option = {
        name = nesnd_mug.12.a
        ai_chance = { base = 0 modifier = { trigger = { has_variable = nesnd_mug_council_ai_yes } add = 100 } }
        if = { limit = { nesnd_mug_council_topic_is = { T = 1 } } custom_tooltip = nesnd_mug_attend_rail_tt }
        else_if = { limit = { nesnd_mug_council_topic_is = { T = 2 } } custom_tooltip = nesnd_mug_attend_relief_tt }
        else = { custom_tooltip = nesnd_mug_attend_defence_tt }
        hidden_effect = {
            set_variable = { name = nesnd_mug_council_answer value = 1 days = 60 }
            remove_variable = nesnd_mug_council_ai_yes
        }
    }
    option = {
        name = nesnd_mug.12.b
        default_option = yes
        ai_chance = { base = 0 modifier = { trigger = { NOT = { has_variable = nesnd_mug_council_ai_yes } } add = 100 } }
        custom_tooltip = nesnd_mug_decline_council_tt
        hidden_effect = {
            set_variable = { name = nesnd_mug_council_answer value = 2 days = 60 }
            remove_variable = nesnd_mug_council_ai_yes
        }
    }
}
