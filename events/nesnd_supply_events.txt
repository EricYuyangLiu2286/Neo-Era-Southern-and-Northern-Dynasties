namespace = nesnd_supply
nesnd_supply.1 = {
	type = country_event
	placement = root
	duration = 3
	title = nesnd_supply.1.t
	desc = nesnd_supply.1.d
	event_image = {
		video = "gfx/event_pictures/unspecific_politicians_arguing.bk2"
	}
	trigger = {
		nesnd_supply_request_valid = yes
	}
	option = {
		name = nesnd_supply.1.a
		trigger = {
			nesnd_supply_request_valid = yes
			is_at_war = no
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				trigger = {
					trigger_if = {
						limit = {
							exists = scope:nesnd_supply_requester
						}
						"relations(scope:nesnd_supply_requester)" >= relations_threshold:cordial
					}
					trigger_else = {
						always = no
					}
				}
			}
			modifier = {
				add = -50
				trigger = {
					trigger_if = {
						limit = {
							exists = scope:nesnd_supply_requester
						}
						"relations(scope:nesnd_supply_requester)" < relations_threshold:poor
					}
					trigger_else = {
						always = no
					}
				}
			}
		}
		if = {
			limit = {
				nesnd_supply_request_valid = yes
				is_at_war = no
			}
			save_scope_as = nesnd_supply_receiver
			scope:nesnd_supply_requester ?= {
				set_variable = {
					name = nesnd_supply_partner
					value = root
				}
				remove_variable = nesnd_supply_pending
				remove_variable = nesnd_supply_pending_sponsor
				remove_variable = nesnd_supply_self_reliance
				nesnd_supply_save_sponsor = yes
				if = {
					limit = {
						root = {
							is_subject = yes
						}
					}
					set_variable = {
						name = nesnd_supply_partner_sponsor
						value = root.overlord
					}
				}
				else = {
					set_variable = {
						name = nesnd_supply_partner_sponsor
						value = root
					}
				}
				change_relations = {
					country = root
					value = 10
				}
			}
		}
		custom_tooltip = nesnd_supply_accept_tt
	}
	option = {
		name = nesnd_supply.1.b
		default_option = yes
		ai_chance = {
			base = 50
		}
		scope:nesnd_supply_requester ?= {
			if = {
				limit = {
					has_variable = nesnd_supply_pending
					var:nesnd_supply_pending ?= root
				}
				remove_variable = nesnd_supply_pending
				remove_variable = nesnd_supply_pending_sponsor
			}
		}
	}
}
