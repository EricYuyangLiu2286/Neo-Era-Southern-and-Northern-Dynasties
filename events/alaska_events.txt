# An Offer for Alaska
alaska.1 = {
	type = country_event
	placement = ROOT

	title = alaska.1.t
	desc = alaska.1.d
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:alaska_seller = {
						c:RUS ?= this
					}
				}
				desc = alaska.1.f_ru
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = alaska.1.f
			}
		}
	}
	
	cooldown = { days = 180 }

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	trigger = {
	}

	immediate = {
		# Scopes set in alaska.4
	}

	option = {
		name = alaska.1.a
		trigger = { sr:region_pacific_coast = { any_scope_state = { nesnd_is_aixue_state = yes nesnd_aixue_sellers_land = yes } } }
		scope:alaska_buyer = {
			trigger_event = { id = alaska.2 days = 0 popup = yes }
		}
		show_as_tooltip = {
			scope:alaska_buyer = { nesnd_settle_aixue_purchase = yes }
		}
		ai_chance = {
			base = 75
		}
	}
	option = {
		name = alaska.1.b
		default_option = yes
		scope:alaska_buyer = {
			trigger_event = { id = alaska.3 days = 0 popup = yes }
		}
		show_as_tooltip = {
			change_relations = {
				country = scope:alaska_buyer
				value = -20
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					scope:alaska_seller = {
						NOT = {
							has_variable = alaska_sale_cooldown #Unsolicited offer
						}
					}
				}
				add = 25
			}
		}
	}
}

# Making an Offer
alaska.4 = {
	type = country_event
	placement = ROOT

	title = alaska.4.t
	desc = alaska.4.d
	flavor = alaska.4.f

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
	}

	immediate = {
		root = {
			save_scope_as = alaska_buyer
		}
		if = {
		    limit = { exists = scope:alaska_seller }
		    sr:region_pacific_coast = {
		        random_scope_state = {
		            limit = { nesnd_is_aixue_state = yes nesnd_aixue_sellers_land = yes }
		            save_scope_as = alaska_scope
		        }
		    }
		}
		else = {
		    sr:region_pacific_coast = {
		        random_scope_state = {
		            limit = { nesnd_aixue_offer_candidate = yes }
		            save_scope_as = alaska_scope
		            if = {
		                limit = { owner = { is_subject = yes capital = { nesnd_is_aixue_state = yes } } }
		                owner.top_overlord = { save_scope_as = alaska_seller }
		            }
		            else = { owner = { save_scope_as = alaska_seller } }
		        }
		    }
		}
	}
	
	cooldown = { days = 180 }

	option = {
		name = alaska.4.a
		trigger = { exists = scope:alaska_seller exists = scope:alaska_scope }
		ai_chance = {
			base = 1
		}
		custom_tooltip = alaska.4.tt1 # 3 million over a year
		scope:alaska_seller ?= {
			trigger_event = {
				id = alaska.1
				popup = yes
			}
		}
	}
	option = { # perhaps another time
		name = alaska.4.c
		default_option = yes
		ai_chance = {
			base = 0
		}
		scope:alaska_seller ?= {
			post_notification = alaska_negotiations_fail
		}
	}
}
