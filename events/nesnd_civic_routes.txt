    type = country_event
    placement = root
    duration = 3
    title = nesnd_civic_routes.30.t
    event_image = { video = "unspecific_signed_contract" }
    trigger = { nesnd_hus_civic_country = yes has_law_or_variant = law_type:law_slavery_banned }
    immediate = {
        random_scope_state = {
            limit = { state_region = s:STATE_SOUTH_DAYUAN state_population > 0 }
            save_scope_as = c2_dayuan
        }
    }
    desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_c2_dayuan_school_available = yes } desc = nesnd_civic_routes.30.d }
            desc = nesnd_civic_routes.30.national
        }
    }
    flavor = nesnd_civic_routes.30.f
    cancellation_trigger = { NOT = { nesnd_hus_civic_country = yes has_law_or_variant = law_type:law_slavery_banned } }
    option = {
        name = nesnd_civic_routes.30.a
        trigger = { nesnd_c2_dayuan_school_available = yes }
        add_treasury = nesnd_civic_tuition_expense
        scope:c2_dayuan = { add_modifier = { name = nesnd_civic_public_tuition months = 24 } }
    }
    option = {
        name = nesnd_civic_routes.30.b
        trigger = { nesnd_c2_dayuan_school_available = yes }
        default_option = yes
        scope:c2_dayuan = { add_modifier = { name = nesnd_civic_local_tuition months = 12 } }
    }
    option = {
        name = nesnd_civic_routes.30.c
        trigger = { NOT = { nesnd_c2_dayuan_school_available = yes } }
    }
}
