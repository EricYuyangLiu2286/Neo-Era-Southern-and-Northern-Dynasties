# 「统治者与内阁」的事件。政潮条满时由 nesnd_premier_fire_event 抽取（common/scripted_effects/nesnd_premier.txt）。
# 有最大在野集团时，对方是该集团及其领袖；所有集团都在政府中时，对方是其余各派。
# 凡杀死、流放或令人物退休的结果，目标都已通过 nesnd_power_character_removable，且不是统治者。
# 极端结果的概率：首相遇刺身亡 20%、负伤 20%；反对派领袖遇害 20%、被捕流放 15%；调查成功率见 nesnd_premier_arrest_chance。

# 1 内阁改组（首相所在集团离开政府，由政府内最大集团的领袖出任首相）
nesnd_premier.1 = {
	type = country_event
	placement = root
	title = nesnd_premier.1.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { exists = scope:nesnd_premier_old_ig_scope } desc = nesnd_premier.1.d }
			desc = nesnd_premier.1.d_generic
		}
	}
	flavor = nesnd_premier.1.f
	duration = 3
	gui_window = event_window_1char_tabloid
	left_icon = scope:nesnd_premier_pm_scope
	event_image = { video = "unspecific_politicians_arguing" }
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_premier_journal nesnd_premier_has_pm = yes }
	immediate = {
		nesnd_premier_save_scopes = yes
		var:nesnd_premier_old_ig ?= { save_scope_as = nesnd_premier_old_ig_scope }
	}
	option = {
		name = nesnd_premier.1.a
		default_option = yes
		custom_tooltip = nesnd_premier.1.tt
	}
	after = { remove_variable = nesnd_premier_old_ig }
}
