nesnd_sha_republic.7 = {
    type = country_event
    placement = root
    title = nesnd_sha_republic.7.t
    flavor = nesnd_sha_republic.7.f
    desc = nesnd_sha_republic.7.d
    duration = 3
    event_image = { video = "unspecific_steam_ship" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = {
        nesnd_sha_native = yes
        has_journal_entry = nesnd_sha_snow_republic
        exists = c:CNE
        NOT = { has_variable = nesnd_sha_exile_cooldown }
        NOT = { exists = c:NRP }
        OR = {
            has_variable = nesnd_sha_suppression_reported
            any_country = { c:CNE ?= this has_variable = nesnd_cne_republic_settled }
        }
        c:CNE ?= {
            NOT = { has_variable = nesnd_cne_republic_restored }
            any_scope_pop = { has_pop_culture = han total_size >= 2500 }
            any_scope_pop = { has_pop_culture = ce_xibei total_size >= 2500 }
        }
    }
    immediate = {
        set_variable = { name = nesnd_sha_exile_cooldown years = 5 }
        capital = {
            save_scope_as = nesnd_sha_refuge
            set_local_variable = { name = nesnd_sha_before_arrival value = nesnd_sha_refuge_population }
        }
        c:CNE = {
            random_scope_pop = {
                limit = { has_pop_culture = han total_size >= 2500 }
                if = { limit = { total_size <= 5000 } move_pop = scope:nesnd_sha_refuge }
                else = { move_partial_pop = { state = scope:nesnd_sha_refuge population = 5000 } }
            }
            random_scope_pop = {
                limit = { has_pop_culture = ce_xibei total_size >= 2500 }
                if = { limit = { total_size <= 5000 } move_pop = scope:nesnd_sha_refuge }
                else = { move_partial_pop = { state = scope:nesnd_sha_refuge population = 5000 } }
            }
        }
        scope:nesnd_sha_refuge = {
            save_scope_value_as = {
                name = nesnd_sha_exile_count
                value = { value = nesnd_sha_refuge_population subtract = local_var:nesnd_sha_before_arrival }
            }
        }
    }
    option = {
        name = nesnd_sha_republic.7.a
        default_option = yes
        custom_tooltip = nesnd_sha_exiles_arrived_tt
    }
}

nesnd_sha_republic.8 = {
    type = country_event
    placement = root
    title = nesnd_sha_republic.8.t
    flavor = nesnd_sha_republic.8.f
    desc = nesnd_sha_republic.8.d
    duration = 3
    event_image = { video = "unspecific_steam_ship" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = {
        nesnd_sha_native = yes
        exists = s:STATE_OUTER_MANCHURIA.region_state:SHA
        NOT = { has_variable = nesnd_sha_qingdao_capital }
    }
    cancellation_trigger = {
        OR = {
            nesnd_sha_native = no
            NOT = { exists = s:STATE_OUTER_MANCHURIA.region_state:SHA }
            has_variable = nesnd_sha_qingdao_capital
        }
    }
    option = {
        name = nesnd_sha_republic.8.a
        trigger = {
            nesnd_sha_native = yes
            exists = s:STATE_OUTER_MANCHURIA.region_state:SHA
            NOR = { has_variable = nesnd_sha_qingdao_capital has_modifier = declared_bankruptcy }
        }
        ai_chance = { base = 100 }
        add_treasury = nesnd_sha_capital_cost
        set_capital = STATE_OUTER_MANCHURIA
        s:STATE_OUTER_MANCHURIA.region_state:SHA ?= {
            save_scope_as = nesnd_sha_new_capital
            add_modifier = { name = nesnd_sha_new_port years = 10 }
        }
        custom_tooltip = nesnd_sha_transfer_north_tt
        hidden_effect = {
            if = {
                limit = { exists = scope:nesnd_sha_new_capital }
                s:STATE_KAMCHATKA.region_state:SHA ?= {
                    if = {
                        limit = { has_building = building_government_administration }
                        set_local_variable = {
                            name = nesnd_sha_admin_levels
                            value = {
                                value = nesnd_small_administration
                                scope:nesnd_sha_new_capital = { add = nesnd_small_administration }
                            }
                        }
                        if = {
                            limit = { has_local_variable = nesnd_sha_admin_levels }
                            scope:nesnd_sha_new_capital = {
                                create_building = { building = building_government_administration level = local_var:nesnd_sha_admin_levels }
                            }
                            remove_building = building_government_administration
                        }
                    }
                }
                every_scope_state = {
                    limit = { OR = { state_region = s:STATE_KAMCHATKA state_region = s:STATE_OKHOTSK } }
                    set_state_type = unincorporated
                    every_scope_pop = {
                        limit = { has_pop_culture = han }
                        move_pop = scope:nesnd_sha_new_capital
                    }
                }
            }
        }
    }
    option = {
        name = nesnd_sha_republic.8.b
        default_option = yes
        ai_chance = { base = 0 }
    }
}
