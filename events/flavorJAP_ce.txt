flavorJAP_ce.1 = {
    type = country_event
    placement = root
    title = flavorJAP_ce.1.t
    # 明祚难延
    desc = flavorJAP_ce.1.d
    #

    flavor = flavorJAP_ce.1.f
    #
    duration = 3

    event_image = {
        texture = "gfx/event_pictures/event_ce_mordern_chinese_6.dds"
    }


    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    #icon = "gfx/interface/icons/event_icons/event_skull.dds"

    trigger = {
        exists = c:JAP
        c:JAP ?= This
        OR = {
            has_law_or_variant = law_type:law_isolationism
            has_law = law_type:law_sakoku
        }
        NOT = { has_variable = flavorJAP_ce_1_var }
        is_subject = no
    }

    immediate = {
        hidden_effect = {
            set_variable = flavorJAP_ce_1_var
            ruler = {
                save_scope_as = japanese_ruler
            }
        }
    }

    option = { #
        name = flavorJAP_ce.1.1
        default_option = yes

        custom_tooltip = {
            text = end_sakoku_button_now_available
            set_variable = je_sakoku_maybe_we_should_open_up
            nesnd_jp_ensure_nation = yes
        }

        every_interest_group = {
            limit = {
                law_stance = {
                    law = law_type:law_isolationism
                    value > neutral
                }
            }
            add_modifier = {
                name = ce_caved_to_OPEN_pressure_opposition
                days = normal_modifier_time
            }
        }

        every_interest_group = {
            limit = {
                is_in_government = yes
                # whether the IG supported an end to sakoku/isolation or not, they were part of the government that signed off on this
            }
            add_modifier = {
                name = ce_caved_to_OPEN_pressure_government
                days = normal_modifier_time
            }
        }

        ai_chance = {
            base = 1
        }
    }
    after = {
        hidden_effect = {

        }
    }
}

        }
        else_if = {
            limit = {
                any_scope_character = { is_character_alive = yes is_in_exile_pool = no is_ruler_of_own_country = no NOT = { has_template = JAP_toyotomi_heikatsu_alt }
                    NOT = { has_template = JAP_toyotomi_hideharu }
                    NOT = { has_template = JAP_toyotomi_shimei }

                }
            }
            random_scope_character = {
                limit = { is_character_alive = yes is_in_exile_pool = no is_ruler_of_own_country = no NOT = { has_template = JAP_toyotomi_heikatsu_alt }
                    NOT = { has_template = JAP_toyotomi_hideharu }
                    NOT = { has_template = JAP_toyotomi_shimei }
                    has_variable = house_toyotomi
                }
                save_scope_as = japanese_heir
            }
        }
        else = {
            set_variable = heir_shimei_spawned
            if = { limit = { NOT = { any_character = { has_template = JAP_toyotomi_shimei } } } create_character = {
                    template = JAP_toyotomi_shimei
                    on_created = {
                        if = {
                            limit = {
                                NOT = { has_variable = house_toyotomi }
                            }
                            set_variable = house_toyotomi
                        }
                        save_scope_as = japanese_heir
                    }
                }
            }
        }
    }
    cancellation_trigger = { NOT = { nesnd_jp_story_succession_valid = yes } }

    option = {
        name = flavorJAP_ce.2.1
        default_option = yes
        ai_chance = {
            base = 1
        }
        if = { limit = { nesnd_jp_story_succession_valid = yes }
            scope:japanese_heir = {
                set_character_as_ruler = yes
            }
            scope:japanese_shogun = {
                kill_character = yes
            }
            ig:ig_landowners = {
                if = {
                    limit = {
                        has_ideology = ideology:ideology_hierarchic
                    }
                    remove_ideology = ideology_hierarchic
                }
                else_if = {
                    limit = {
                        has_ideology = ideology:ideology_japan_hierarchic
                    }
                    remove_ideology = ideology_japan_hierarchic
                }
                else_if = {
                    limit = {
                        has_ideology = ideology:ideology_japan_hierarchic_2
                    }
                    remove_ideology = ideology_japan_hierarchic_2
                }

                if = {
                    limit = {
                        has_ideology = ideology:ideology_paternalistic
                    }
                    remove_ideology = ideology_paternalistic
                }
                else_if = {
                    limit = {
                        has_ideology = ideology:ideology_japan_paternalistic
                    }
                    remove_ideology = ideology_japan_paternalistic
                }
                #
                add_ideology = ideology_confucian
                add_ideology = ideology_colonialist

                add_modifier = {
                    name = ce_land_owner_change_sides
                    years = 5
                    is_decaying = yes
                }
            }
            japan_generate_scripted_heirs = yes
            nesnd_jp_bootstrap = yes
        }
    }

    option = {
