# 「宫廷力量」的宫廷事件。由 nesnd_power_month 在宫廷事件条满时抽取（common/scripted_effects/nesnd_power.txt）。
# 情形 A：宠臣不是最大利益集团的领袖（宫廷派系相争）。情形 B：宠臣就是最大集团的领袖（大权臣）。
# 凡杀死、流放、撤去人物的结果，目标都已通过 nesnd_power_character_removable。
# 极端结果的概率（作者规定）：遇害 20%，被逐或被捕流放 15%，脱身 65%；按钮清除成功 45%。

# 11 御前争执
nesnd_palace.11 = {
	type = country_event
	placement = root
	title = nesnd_palace.11.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { nesnd_power_case_b = yes } desc = nesnd_palace.11.d_b }
			desc = nesnd_palace.11.d_a
		}
	}
	flavor = nesnd_palace.11.f
	duration = 3
	gui_window = event_window_1char_tabloid
	left_icon = scope:nesnd_power_holder_scope
	event_image = { video = "votp_monarch_holding_court" }
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_palace_journal nesnd_palace_ev_quarrel_ok = yes }
	immediate = { nesnd_power_save_scopes = yes }

	option = {
		name = nesnd_palace.11.a
		nesnd_power_add = { BAR = power AMT = 5 }
		nesnd_power_add = { BAR = balance AMT = 10 }
		if = {
			limit = { nesnd_power_case_a = yes }
			scope:nesnd_power_top_ig_scope ?= { add_modifier = { name = nesnd_power_humbled days = short_modifier_time } }
		}
		else = {
			every_interest_group = {
				limit = { is_in_government = yes NOT = { scope:nesnd_power_holder_ig_scope ?= this } }
				add_modifier = { name = nesnd_power_slighted days = short_modifier_time }
			}
		}
		ai_chance = { base = 15 modifier = { trigger = { nesnd_power_b <= 40 } add = 20 } }
	}
	option = {
		name = nesnd_palace.11.b
		default_option = yes
		nesnd_power_sub = { BAR = power AMT = 5 }
		nesnd_power_center = { AMT = 10 }
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		ai_chance = { base = 30 modifier = { trigger = { nesnd_power_p >= 60 } add = 40 } }
	}
	option = {
		name = nesnd_palace.11.c
		nesnd_power_sub = { BAR = power AMT = 10 }
		nesnd_power_sub = { BAR = balance AMT = 10 }
		if = {
			limit = { nesnd_power_case_a = yes }
			scope:nesnd_power_top_ig_scope ?= { add_modifier = { name = nesnd_power_gratified days = short_modifier_time } }
		}
		else = {
			every_interest_group = {
				limit = { is_in_government = yes NOT = { scope:nesnd_power_holder_ig_scope ?= this } }
				add_modifier = { name = nesnd_power_gratified days = short_modifier_time }
			}
		}
		ai_chance = { base = 10 modifier = { trigger = { nesnd_power_b >= 70 } add = 30 } }
	}
}

# 21 宠臣遭难（对方压倒宠臣一方时）
nesnd_palace.21 = {
	type = country_event
	placement = root
	title = nesnd_palace.21.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { var:nesnd_power_ev_outcome ?= 1 var:nesnd_power_ev_case ?= 2 } desc = nesnd_palace.21.d_b_dead }
			triggered_desc = { trigger = { var:nesnd_power_ev_outcome ?= 1 } desc = nesnd_palace.21.d_a_dead }
			triggered_desc = { trigger = { var:nesnd_power_ev_outcome ?= 2 var:nesnd_power_ev_case ?= 2 } desc = nesnd_palace.21.d_b_ousted }
			triggered_desc = { trigger = { var:nesnd_power_ev_outcome ?= 2 } desc = nesnd_palace.21.d_a_ousted }
			triggered_desc = { trigger = { var:nesnd_power_ev_case ?= 2 } desc = nesnd_palace.21.d_b_escaped }
			desc = nesnd_palace.21.d_a_escaped
		}
	}
	flavor = nesnd_palace.21.f
	duration = 3
	gui_window = event_window_1char_tabloid
	left_icon = scope:nesnd_power_holder_scope
	event_image = { video = "southamerica_public_figure_assassination" }
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_palace_journal nesnd_palace_ev_strike_ok = yes }
	immediate = {
		nesnd_power_save_scopes = yes
		set_variable = { name = nesnd_power_ev_case value = var:nesnd_power_case }
		random_list = {
			20 = { set_variable = { name = nesnd_power_ev_outcome value = 1 } }
			15 = { set_variable = { name = nesnd_power_ev_outcome value = 2 } }
			65 = { set_variable = { name = nesnd_power_ev_outcome value = 3 } }
		}
		if = {
			limit = { var:nesnd_power_ev_outcome = 1 }
			scope:nesnd_power_holder_scope = { kill_character = yes }
			nesnd_power_after_removal = { REGIME = palace }
		}
		else_if = {
			limit = { var:nesnd_power_ev_outcome = 2 }
			scope:nesnd_power_holder_scope = { exile_character = yes }
			nesnd_power_after_removal = { REGIME = palace }
		}
	}
	option = {
		name = nesnd_palace.21.a
		nesnd_power_add = { BAR = balance AMT = 20 }
		if = {
			limit = { var:nesnd_power_ev_case ?= 1 exists = scope:nesnd_power_top_ig_scope }
			add_radicals = { value = medium_radicals interest_group = scope:nesnd_power_top_ig_scope }
		}
		else = {
			add_radicals = { value = small_radicals strata = upper }
		}
		ai_chance = { base = 15 }
	}
	option = {
		name = nesnd_palace.21.b
		default_option = yes
		nesnd_power_sub = { BAR = power AMT = 10 }
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		ai_chance = { base = 25 }
	}
	after = { remove_variable = nesnd_power_ev_outcome remove_variable = nesnd_power_ev_case }
}
