# 「波斯之忧」≥ 60：伊朗诸州的矿工罢工。余波记在「波斯之忧」的 nesnd_per_tension_ev7 一行。
nesnd_per_mines.1 = {
    type = country_event
    title = nesnd_per_mines.1.t
    desc = nesnd_per_mines.1.d
    flavor = nesnd_per_mines.1.f
    event_image = { video = "middleeast_engineer_blueprint" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_protest.dds"
    duration = 3
    trigger = {
        nesnd_per_mines_unrest = yes
        NOT = { has_variable = nesnd_per_tension_ev7 }
        any_scope_state = { nesnd_per_iranian_state = yes any_scope_building = { is_building_group = bg_mining } }
    }
    immediate = { nesnd_per_mines_pick_mine_state = yes }
    option = {
        name = nesnd_per_mines.1.a
        default_option = yes
        scope:nesnd_per_mines_state ?= { add_radicals_in_state = { value = 0.1 } }
        custom_tooltip = nesnd_per_mines.1.a.tt
        hidden_effect = { set_variable = { name = nesnd_per_tension_ev7 value = 0.04 months = 36 } }
        ai_chance = { base = 50 }
    }
    option = {
        name = nesnd_per_mines.1.b
        add_treasury = { value = gdp multiply = -0.01 }
        custom_tooltip = nesnd_per_mines.1.b.tt
        hidden_effect = { set_variable = { name = nesnd_per_tension_ev7 value = -0.04 months = 36 } }
        ai_chance = { base = 50 }
    }
}
