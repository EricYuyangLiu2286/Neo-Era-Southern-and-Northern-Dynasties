namespace = nesnd_cne_funding
nesnd_cne_funding.1 = {
	type = country_event
	placement = scope:nesnd_cne_funding_state
	title = nesnd_cne_funding.1.t
	desc = nesnd_cne_funding.1.d
	duration = 1
	event_image = { video = "gfx/event_pictures/unspecific_politicians_arguing.bk2" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { nesnd_cne_funding_target_valid = yes }
	cancellation_trigger = { NOT = { nesnd_cne_funding_target_valid = yes } }
	option = {
		name = nesnd_cne_funding.accept
		ai_chance = { base = 100 }
		trigger = { nesnd_cne_funding_target_valid = yes nesnd_reform_can_invest = yes }
		if = { limit = { nesnd_cne_funding_target_valid = yes nesnd_reform_can_invest = yes }
			add_treasury = nesnd_reform_investment_debit
			set_variable = { name = nesnd_reform_investment_cooldown months = 12 }
			scope:nesnd_cne_funding_state = { add_modifier = { name = nesnd_reform_public_workshops months = 12 } }
			remove_variable = nesnd_cne_funding_pending
		}
	}
	option = { name = nesnd_cne_funding.cancel default_option = yes ai_chance = { base = 0 } remove_variable = nesnd_cne_funding_pending }
}
