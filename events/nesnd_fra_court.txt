namespace = nesnd_fra_court

# 维持拿破仑王朝期间的宫廷事件，挂在 je_ce_unstable_imperium 的月度随机事件中，调整王朝稳定度。
# 稳定度提示沿用原有键 ce_french_stability_increase_*_tt / ce_french_stability_decrease_*_tt。

# 1 皇帝的近臣
nesnd_fra_court.1 = {
	type = country_event
	placement = root
	title = nesnd_fra_court.1.t
	desc = nesnd_fra_court.1.d
	flavor = nesnd_fra_court.1.f
	duration = 3
	event_image = { video = "votp_monarch_holding_court" }
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	cooldown = { days = short_modifier_time }
	cancellation_trigger = { NOT = { has_journal_entry = je_ce_unstable_imperium } }
	trigger = { has_journal_entry = je_ce_unstable_imperium exists = ruler }
	immediate = { ruler = { save_scope_as = nesnd_fra_court_ruler } }

	option = {
		name = nesnd_fra_court.1.a
		custom_tooltip = {
			text = ce_french_stability_increase_50_tt
			je:je_ce_unstable_imperium = { add_progress = { value = 50 name = ce_monarchists_bonapartist_stability_progress_bar } }
		}
		add_modifier = { name = nesnd_power_patronage_posts days = short_modifier_time }
		ai_chance = { base = 15 }
	}
	option = {
		name = nesnd_fra_court.1.b
		default_option = yes
		custom_tooltip = {
			text = ce_french_stability_decrease_25_tt
			je:je_ce_unstable_imperium = { add_progress = { value = -25 name = ce_monarchists_bonapartist_stability_progress_bar } }
		}
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		ai_chance = { base = 10 }
	}
}
