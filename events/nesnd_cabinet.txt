# 「执政者与派系」的政潮事件。由 nesnd_power_month 在政潮条满时抽取（common/scripted_effects/nesnd_power.txt）。
# 情形 A：执政者不是最大利益集团的领袖（执政者与反对党）。情形 B：执政者兼领最大集团（执政者与各派）。
# 凡杀死、流放人物的结果，目标都已通过 nesnd_power_character_removable。
# 极端结果的概率：执政者遇刺身亡 15%、负伤 20%；反对党领袖遇害 20%、被捕流放 15%；调查成功 45%。

# 11 政纲之争
nesnd_cabinet.11 = {
	type = country_event
	placement = root
	title = nesnd_cabinet.11.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { nesnd_power_case_b = yes } desc = nesnd_cabinet.11.d_b }
			desc = nesnd_cabinet.11.d_a
		}
	}
	flavor = nesnd_cabinet.11.f
	duration = 3
	gui_window = event_window_1char_tabloid
	left_icon = scope:nesnd_power_holder_scope
	event_image = { video = "unspecific_politicians_arguing" }
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_cabinet_journal nesnd_cabinet_ev_debate_ok = yes }
	immediate = { nesnd_power_save_scopes = yes }

	option = {
		name = nesnd_cabinet.11.a
		nesnd_power_add = { BAR = power AMT = 5 }
		nesnd_power_add = { BAR = balance AMT = 10 }
		if = {
			limit = { nesnd_power_case_a = yes }
			scope:nesnd_power_top_ig_scope ?= { add_modifier = { name = nesnd_power_humbled days = short_modifier_time } }
		}
		else = {
			every_interest_group = {
				limit = { ig_clout >= 0.05 NOT = { scope:nesnd_power_holder_ig_scope ?= this } }
				add_modifier = { name = nesnd_power_slighted days = short_modifier_time }
			}
		}
		ai_chance = { base = 15 modifier = { trigger = { nesnd_power_b <= 40 } add = 20 } }
	}
	option = {
		name = nesnd_cabinet.11.b
		default_option = yes
		nesnd_power_sub = { BAR = power AMT = 5 }
		nesnd_power_center = { AMT = 10 }
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		ai_chance = { base = 30 modifier = { trigger = { nesnd_power_p >= 60 } add = 40 } }
	}
	option = {
		name = nesnd_cabinet.11.c
		nesnd_power_sub = { BAR = power AMT = 10 }
		nesnd_power_sub = { BAR = balance AMT = 10 }
		if = {
			limit = { nesnd_power_case_a = yes }
			scope:nesnd_power_top_ig_scope ?= { add_modifier = { name = nesnd_power_gratified days = short_modifier_time } }
		}
		else = {
			every_interest_group = {
				limit = { ig_clout >= 0.05 NOT = { scope:nesnd_power_holder_ig_scope ?= this } }
				add_modifier = { name = nesnd_power_gratified days = short_modifier_time }
			}
		}
		ai_chance = { base = 10 modifier = { trigger = { nesnd_power_b >= 70 } add = 30 } }
	}
}

	}
	flavor = nesnd_cabinet.21.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = { has_journal_entry = nesnd_cabinet_journal nesnd_cabinet_ev_strike_ok = yes }
	immediate = {
		nesnd_power_save_scopes = yes
		random_list = {
			15 = { set_variable = { name = nesnd_power_ev_outcome value = 1 } }
			20 = { set_variable = { name = nesnd_power_ev_outcome value = 2 } }

			limit = { var:nesnd_power_ev_outcome = 1 }

		}
		else_if = {
			limit = { var:nesnd_power_ev_outcome = 2 }
			scope:nesnd_power_holder_scope = { add_trait = wounded }
		}
	}
	option = {
		name = nesnd_cabinet.21.a
		nesnd_power_add = { BAR = power AMT = 10 }
		nesnd_power_add = { BAR = balance AMT = 15 }
		add_modifier = { name = nesnd_power_shamed days = short_modifier_time }
		ai_chance = { base = 15 }
	}
	option = {
		name = nesnd_cabinet.21.b
		default_option = yes
		nesnd_power_sub = { BAR = power AMT = 10 }
		add_modifier = { name = nesnd_power_mediation days = short_modifier_time }
		ai_chance = { base = 25 }
	}
	after = {
		remove_variable = nesnd_power_ev_outcome
		nesnd_power_refresh = { REGIME = cabinet }
	}
}
