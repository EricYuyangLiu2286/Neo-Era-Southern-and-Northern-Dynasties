# 「波斯之忧」满 100：伊朗人起义。迎立沙阿（广义伊朗人过半时）、平叛，或（仅玩家）平叛后改投起义方。
nesnd_per_rising.1 = {
    type = country_event
    title = nesnd_per_rising.1.t
    desc = nesnd_per_rising.1.d
    flavor = nesnd_per_rising.1.f
    event_image = { video = "middleeast_courtroom_upheaval" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_fire.dds"
    duration = 3
    trigger = { nesnd_per_tang_regime = yes }

    option = {
        name = nesnd_per_rising.1.a
        trigger = { nesnd_per_rising_shah_possible = yes }
        custom_tooltip = nesnd_per_rising.1.a.tt
        hidden_effect = { nesnd_per_shah_takeover = yes }
        ai_chance = {
            base = 0
            modifier = {
                trigger = {
                    has_variable = nesnd_per_iran_share_var
                    var:nesnd_per_iran_share_var >= 0.55
                    government_legitimacy < 40
                }
                add = 100
            }
        }
    }
    option = {
        name = nesnd_per_rising.1.b
        default_option = yes
        if = {
            limit = { nesnd_per_rising_civil_war = yes }
            custom_tooltip = nesnd_per_rising.1.b.civil_tt
        }
        else = { custom_tooltip = nesnd_per_rising.1.b.independence_tt }
        hidden_effect = { nesnd_per_split = yes }
        ai_chance = { base = 10 }
    }
    option = {
        name = nesnd_per_rising.1.c
        trigger = { is_player = yes }
        custom_tooltip = nesnd_per_rising.1.c.tt
        hidden_effect = {
            nesnd_per_split = yes
            if = { limit = { exists = c:IRN } play_as = c:IRN }
        }
        ai_chance = { base = 0 }
    }
}
