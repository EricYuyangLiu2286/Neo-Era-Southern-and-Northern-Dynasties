namespace = nesnd_native_khanate
nesnd_native_khanate.1 = {
 type = country_event hidden = yes
 trigger = { c:NKH ?= this has_variable = nesnd_native_khanate }
 immediate = {
  if = { limit = { any_scope_state = { count = 4 state_population > 0 } }
   nesnd_khanate_prepare = yes
   c:USA ?= { nesnd_khanate_start_war = yes }
   if = { limit = { exists = var:nesnd_khanate_original_hus } var:nesnd_khanate_original_hus = { nesnd_khanate_start_war = yes } }
  }
  else = { set_variable = nesnd_khanate_legacy_invalid }
 }
}
