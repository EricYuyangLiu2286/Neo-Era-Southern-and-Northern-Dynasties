# 唐人团结度过低：一州的部分西域汉羌转入广义伊朗文化。
nesnd_per_unity.1 = {
    type = country_event
    title = nesnd_per_unity.1.t
    desc = nesnd_per_unity.1.d
    flavor = nesnd_per_unity.1.f
    event_image = { video = "asia_poor_people_moving" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_map.dds"
    duration = 3
    trigger = { exists = scope:nesnd_per_unity_state }
    cancellation_trigger = { NOT = { exists = scope:nesnd_per_unity_state } }
    option = {
        name = nesnd_per_unity.1.a
        default_option = yes
        custom_tooltip = nesnd_per_unity.1.a.tt
        hidden_effect = {
            scope:nesnd_per_unity_state ?= {
                if = { limit = { owner = root nesnd_per_unity_decay_state = yes }
                    # 用上月抽定、日志已显示的文化。旧档的预告没有这一项时当场抽。
                    if = { limit = { exists = scope:nesnd_per_unity_planned_culture }
                        scope:nesnd_per_unity_planned_culture = { save_temporary_scope_as = nesnd_per_unity_target_culture }
                    }
                    else = {
                        random_scope_pop = {
                            limit = { nesnd_per_unity_iranian_target_pop = yes }
                            culture = { save_temporary_scope_as = nesnd_per_unity_target_culture }
                        }
                    }
                    every_scope_pop = {
                        limit = { has_pop_culture = ce_xiyu }
                        change_pop_culture = { target = scope:nesnd_per_unity_target_culture value = 0.25 }
                    }
                }
            }
        }
        ai_chance = { base = 50 }
    }
    option = {
        name = nesnd_per_unity.1.b
        custom_tooltip = nesnd_per_unity.1.b.tt
        add_treasury = { value = nesnd_per_shrine_cost multiply = -1 }
        ai_chance = { base = 50 }
    }
}
