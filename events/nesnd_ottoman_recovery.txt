nesnd_ottoman_recovery.2 = {
    type = country_event
    placement = root
    title = nesnd_ottoman_recovery.2.t
    desc = {
        first_valid = {
            triggered_desc = {
                desc = nesnd_ottoman_recovery.2.d
                trigger = { exists = scope:nesnd_qin_country }
            }
            triggered_desc = {
                desc = nesnd_ottoman_recovery.2.d_fallen
                trigger = { always = yes }
            }
        }
    }
    flavor = nesnd_ottoman_recovery.2.f
    event_image = { video = "asia_politician_parliament_motion" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    duration = 3
    trigger = {
        nesnd_ottoman_reform_active = yes
        any_interest_group = { nesnd_ottoman_qinist_candidate = yes }
    }
    immediate = {
        c:QE1 ?= { save_scope_as = nesnd_qin_country }
        random_interest_group = {
            limit = { nesnd_ottoman_qinist_candidate = yes }
            save_scope_as = nesnd_qinist_ig
            leader = { save_scope_as = nesnd_qinist_leader }
        }
    }
    option = {
        name = nesnd_ottoman_recovery.2.a
        default_option = yes
        # The variable goes on the group too: interest_group_leader_trigger may be evaluated with either root.
        scope:nesnd_qinist_ig ?= { set_variable = nesnd_qinist_allowed }
        scope:nesnd_qinist_leader ?= {
            set_variable = nesnd_qinist_allowed
            set_ideology = ideology:ideology_ce_qin_royalist_leader
        }
    }
}

        name = nesnd_ottoman_recovery.4.a

        set_capital = STATE_ANKARA
        ai_chance = { base = 50 }
    }
    option = {
        name = nesnd_ottoman_recovery.4.b
        default_option = yes
        if = {
            limit = {
                exists = scope:nesnd_thrace_taker
                any_country = { NOT = { this = root } has_diplomatic_pact = { who = scope:nesnd_thrace_taker type = rivalry } }
            }
            every_country = {
                limit = { NOT = { this = root } has_diplomatic_pact = { who = scope:nesnd_thrace_taker type = rivalry } }
                change_relations = { country = root value = 30 }
            }
        }
        else = { custom_tooltip = nesnd_ottoman_recovery.4.no_rival_tt }

    }
}

# Journal outcome "Peoples United" (nesnd_ottoman_unity).
nesnd_ottoman_recovery.7 = {
    type = country_event
    placement = root
    title = nesnd_ottoman_recovery.7.t
    desc = nesnd_ottoman_recovery.7.d
    flavor = nesnd_ottoman_recovery.7.f
    event_image = { video = "unspecific_ruler_speaking_to_people" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    duration = 3
    immediate = {
        ruler ?= { save_scope_as = nesnd_ottoman_ruler }
        # Largest culture of each incorporated state (GAME common/scripted_effects/05_russian_civil_war_scripted_effects.txt:2259-2264),
        # kept once each in a variable list (GAME events/iberia_events/vernacular_events.txt:282, :293-297).
        clear_variable_list = nesnd_concord_cultures
        every_scope_state = {
            limit = { is_incorporated = yes }
            save_temporary_scope_as = nesnd_concord_state
            ordered_scope_culture = {
                order_by = "scope:nesnd_concord_state.culture_percent_state(this)"
                position = 0
                check_range_bounds = no
                if = {
                    limit = {
                        NOT = { is_primary_culture_of = root }
                        root = { NOT = { is_target_in_variable_list = { name = nesnd_concord_cultures target = prev } } }
                    }
                    root = { add_to_variable_list = { name = nesnd_concord_cultures target = prev } }
                }
            }
        }
    }
    option = {
        name = nesnd_ottoman_recovery.7.a
        default_option = yes
        if = { limit = { any_in_list = { variable = nesnd_concord_cultures } }
            every_in_list = {
                variable = nesnd_concord_cultures
                save_temporary_scope_as = nesnd_concord_culture
                root = { add_culture_acceptance_modifier = { culture = scope:nesnd_concord_culture multiplier = 20 } }
            }
        }
        else = { custom_tooltip = nesnd_ottoman_recovery.7.no_state_tt }
        hidden_effect = { clear_variable_list = nesnd_concord_cultures }
    }
}

            text = nesnd_ottoman_charter_bonus_qin_gain_tt
            set_variable = { name = nesnd_ottoman_qin_bonus months = 60 }

        ai_chance = {
            base = 10
            modifier = { trigger = { scope:nesnd_route_army_ig ?= { is_in_government = yes } } add = 10 }
        }
    }
    option = {
        name = nesnd_ottoman_recovery.10.b
        scope:nesnd_barracks_leader ?= { add_modifier = { name = nesnd_ottoman_qinist_rebuffed months = 24 } }
        scope:nesnd_route_army_ig ?= { add_modifier = { name = nesnd_ottoman_army_vexed months = 24 } }

                scope:nesnd_route_army_ig ?= { add_modifier = { name = nesnd_ottoman_barracks_grievance } }
            }
        }
        else = {
            custom_tooltip = nesnd_ottoman_recovery.10.count_tt

    }
