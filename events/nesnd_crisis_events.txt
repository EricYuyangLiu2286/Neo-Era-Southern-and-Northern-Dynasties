    } option = { name = nesnd_crisis.20.c

        set_variable = nesnd_north_intervention add_modifier = { name = nesnd_foreign_commitment months = 60 } if = { limit = { has_variable = flavorCNE_land_reform_value NOT = { has_variable = ce_CNE_land_reform_done } } custom_tooltip = nesnd_cne_feedback_disorder_up_15

    flavor = nesnd_crisis.20.f
