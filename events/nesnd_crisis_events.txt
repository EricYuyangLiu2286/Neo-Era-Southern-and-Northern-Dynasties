namespace = nesnd_crisis

# ==================================================================
# 靖难：外国对南朝变局的反应
# ==================================================================

# ------------------------------------------------------------------
# .20 伪朝禅代（中华帝国的对策）
# 触发：nesnd_notify_foreign_countries（nesnd_crisis_effects.txt），南方开战时发给 CNE，一局一次。
# ------------------------------------------------------------------
nesnd_crisis.20 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_crisis.20.t
    desc = nesnd_crisis.20.d
    flavor = nesnd_crisis.20.f
    event_image = { video = "gfx/event_pictures/unspecific_politicians_arguing.bk2" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    trigger = {
        c:CNE ?= this
        exists = c:USM
        NOT = { has_variable = nesnd_foreign_policy_seen }
    }
    immediate = { set_variable = nesnd_foreign_policy_seen }

    # a. 先整顿国内：只记下立场
    option = {
        name = nesnd_crisis.20.a
        default_option = yes
        ai_chance = { base = 65 }
        set_variable = nesnd_foreign_observer
    }
    # b. 接济湖广军需：南方事务支出 36 个月，湖广获境外物资援助 36 个月；
    #    地权革命进行中时地方失序 +8
    option = {
        name = nesnd_crisis.20.b
        ai_chance = { base = 25 }
        trigger = {
            exists = c:HNA
            NOT = { has_modifier = declared_bankruptcy }
        }
        set_variable = nesnd_foreign_aid
        add_modifier = { name = nesnd_foreign_commitment months = 36 }
        c:HNA ?= { add_modifier = { name = nesnd_foreign_supplies months = 36 } }
        if = {
            limit = {
                has_variable = flavorCNE_land_reform_value
                NOT = { has_variable = ce_CNE_land_reform_done }
            }
            custom_tooltip = nesnd_cne_feedback_disorder_up_8
            change_variable = { name = cne_landreform_failure_tracker add = 8 }
        }
    }
    # c. 准备南下：南方事务支出 60 个月，加入反吴一方的博弈；
    #    地权革命进行中时地方失序 +15
    option = {
        name = nesnd_crisis.20.c
        ai_chance = { base = 10 }
        trigger = {
            bureaucracy > 0
            is_at_war = no
            NOT = { has_modifier = declared_bankruptcy }
        }
        set_variable = nesnd_north_intervention
        add_modifier = { name = nesnd_foreign_commitment months = 60 }
        if = {
            limit = {
                has_variable = flavorCNE_land_reform_value
                NOT = { has_variable = ce_CNE_land_reform_done }
            }
            custom_tooltip = nesnd_cne_feedback_disorder_up_15
            change_variable = { name = cne_landreform_failure_tracker add = 15 }
        }
        nesnd_join_opposition_play = yes
    }
}

# ------------------------------------------------------------------
# .21 转发入口（隐藏）
# 其他国家的反应已移到 nesnd_external.1；这里只负责转发，
# 因为 nesnd_notify_foreign_countries 仍按此 ID 发送。
# ------------------------------------------------------------------
nesnd_crisis.21 = {
    type = country_event
    hidden = yes
    trigger = { NOT = { has_variable = nesnd_foreign_policy_seen } }
    immediate = {
        set_variable = nesnd_foreign_policy_seen
        trigger_event = { id = nesnd_external.1 popup = yes }
    }
}

# ==================================================================
# 吴国新朝的善后
# ==================================================================

# ------------------------------------------------------------------
# .40 经始新朝
# 触发：吴国在南方内战中全胜，由 usm_civilwar.8 的 after 发出。
# 两个选项都给新朝财赋整理 120 个月；另选招抚或清洗。
# ------------------------------------------------------------------
nesnd_crisis.40 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_crisis.40.t
    desc = nesnd_crisis.40.d
    flavor = nesnd_crisis.40.f
    event_image = { video = "gfx/event_pictures/unspecific_politicians_arguing.bk2" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    trigger = { NOT = { has_variable = nesnd_wu_reconstruction_seen } }
    immediate = { set_variable = nesnd_wu_reconstruction_seen }

    # a. 分案审理：分案审理与招抚 120 个月
    option = {
        name = nesnd_crisis.40.a
        default_option = yes
        ai_chance = { base = 70 }
        add_modifier = { name = nesnd_wu_administration months = 120 }
        add_modifier = { name = nesnd_wu_clemency months = 120 }
    }
    # b. 清查旧党：军政清洗 120 个月，激进 +8%
    option = {
        name = nesnd_crisis.40.b
        ai_chance = { base = 30 }
        add_modifier = { name = nesnd_wu_administration months = 120 }
        add_modifier = { name = nesnd_wu_repression months = 120 }
        add_radicals = { value = 0.08 }
    }
}
