        ai_chance = { base = 10 }
    }
    option = {
        name = nesnd_sha_overseas.1.b
        trigger = { nesnd_sha_landing_available = yes nesnd_sha_can_claim_madagascar = yes }

        ai_chance = { base = 10 }
    }
    option = {
        name = nesnd_sha_overseas.1.c
        trigger = { nesnd_sha_landing_available = yes nesnd_sha_can_claim_kenya = yes }
        nesnd_sha_file_landing = { REGION = s:STATE_KENYA CANDIDATE = nesnd_sha_can_claim_kenya }
        ai_chance = { base = 20 }
    }
    option = {
        name = nesnd_sha_overseas.1.wait
        default_option = yes
        ai_chance = { base = 1 }
    }

nesnd_sha_overseas.3 = {
    type = country_event
    placement = root
    title = nesnd_sha_overseas.3.t
    desc = nesnd_sha_overseas.3.d
    flavor = nesnd_sha_overseas.3.f
    duration = 3
    event_image = { video = "unspecific_steam_ship" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    option = {
        name = nesnd_sha_overseas.3.a
        default_option = yes
        trigger = { scope:nesnd_sha_harbor ?= { owner ?= root } }
        add_treasury = { value = gdp multiply = 0.01 min = 1000 multiply = -1 }
        scope:nesnd_sha_harbor = { add_loyalists_in_state = { value = 0.02 } }
    }
    option = {
        name = nesnd_sha_overseas.3.b
        if = {
            limit = { scope:nesnd_sha_harbor ?= { owner ?= root } }
            scope:nesnd_sha_harbor = { add_radicals_in_state = { value = 0.02 } }
        }
    }
}
