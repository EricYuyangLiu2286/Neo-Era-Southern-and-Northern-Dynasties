namespace = nesnd_sih_jingnan

# Shu (SIH) in the Second Jingnan War, and the "Unify the South" journal that follows any southern settlement.
# .1 is fired by the Shu army journal's weekly pulse (nesnd_war_collapse.txt), which also sets nesnd_sih_regime_pending.

# Who governs Shu now that the war turns against it.
nesnd_sih_jingnan.1 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_sih_jingnan.1.t
    desc = nesnd_sih_jingnan.1.d
    event_image = { video = "gfx/event_pictures/unspecific_politicians_arguing.bk2" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = {
        c:SIH ?= this
        is_country_alive = yes
        has_journal_entry = nesnd_shu_army_journal
        NOT = { has_variable = nesnd_sih_regime }
    }
    # Liu Zhongming and Xiao Huli stay in charge. The court must then decide whether to submit.
    option = {
        name = nesnd_sih_jingnan.1.a
        default_option = yes
        ai_chance = { base = 60 }
        custom_tooltip = nesnd_sih_jingnan.1.a_tt
        hidden_effect = { trigger_event = { id = nesnd_sih_jingnan.2 popup = yes } }
    }
    # Wei Jian removes the party of surrender and takes the throne.
    option = {
        name = nesnd_sih_jingnan.1.b
        trigger = { nesnd_sih_wei_available = yes }
        ai_chance = { base = 40 }
        custom_tooltip = nesnd_sih_jingnan.wei_seizes_tt
        add_modifier = { name = nesnd_sih_wei_rules }
        custom_tooltip = nesnd_sih_jingnan.rally_tt
        hidden_effect = { nesnd_sih_wei_seizes = yes }
    }
}

# Liu Zhongming's court: submit to Wu, or let Xiao Huli fight on.
nesnd_sih_jingnan.2 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_sih_jingnan.2.t
    desc = nesnd_sih_jingnan.2.d
    event_image = { texture = "gfx/event_pictures/event_ce_mordern_chinese_7.dds" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = {
        c:SIH ?= this
        is_country_alive = yes
        has_journal_entry = nesnd_shu_army_journal
        NOT = { has_variable = nesnd_sih_regime }
    }
    immediate = { c:USM ?= { save_scope_as = nesnd_sih_wu } }
    # Liu Zhongming submits.
    option = {
        name = nesnd_sih_jingnan.2.a
        trigger = { any_scope_state = { state_region = s:STATE_SICHUAN } exists = c:USM }
        ai_chance = { base = 25 }
        custom_tooltip = nesnd_sih_jingnan.surrender_tt
        hidden_effect = {
            nesnd_sih_surrender = yes
            trigger_event = { id = nesnd_sih_jingnan.3 popup = yes }
        }
    }
    # Xiao Huli fights on.
    option = {
        name = nesnd_sih_jingnan.2.b
        default_option = yes
        ai_chance = { base = 75 }
        add_modifier = { name = nesnd_sih_xiao_holds }
        custom_tooltip = nesnd_sih_jingnan.rally_tt
        hidden_effect = { nesnd_sih_xiao_stands = yes }
    }
}

# Where the Shu officers went after the surrender.
nesnd_sih_jingnan.3 = {
    type = country_event
    placement = root
    duration = 3
    title = nesnd_sih_jingnan.3.t
    desc = nesnd_sih_jingnan.3.d
    event_image = { texture = "gfx/event_pictures/event_ce_mordern_chinese_1.dds" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    trigger = { is_country_alive = yes has_variable = nesnd_sih_surrendered }
    immediate = { c:USM ?= { save_scope_as = nesnd_sih_wu } }
    option = {
        name = nesnd_sih_jingnan.3.a
        default_option = yes
        if = { limit = { has_variable = nesnd_sih_fate_wei_jian var:nesnd_sih_fate_wei_jian = 1 } custom_tooltip = nesnd_sih_jingnan.3.wei_jian_fled }
        else_if = { limit = { has_variable = nesnd_sih_fate_wei_jian var:nesnd_sih_fate_wei_jian = 2 } custom_tooltip = nesnd_sih_jingnan.3.wei_jian_killed }
        else_if = { limit = { has_variable = nesnd_sih_fate_wei_jian var:nesnd_sih_fate_wei_jian = 3 } custom_tooltip = nesnd_sih_jingnan.3.wei_jian_stayed }
        if = { limit = { has_variable = nesnd_sih_fate_xiao_huli var:nesnd_sih_fate_xiao_huli = 1 } custom_tooltip = nesnd_sih_jingnan.3.xiao_huli_fled }
        else_if = { limit = { has_variable = nesnd_sih_fate_xiao_huli var:nesnd_sih_fate_xiao_huli = 2 } custom_tooltip = nesnd_sih_jingnan.3.xiao_huli_killed }
        else_if = { limit = { has_variable = nesnd_sih_fate_xiao_huli var:nesnd_sih_fate_xiao_huli = 3 } custom_tooltip = nesnd_sih_jingnan.3.xiao_huli_stayed }
        if = { limit = { has_variable = nesnd_sih_fate_xiao_xinghua var:nesnd_sih_fate_xiao_xinghua = 1 } custom_tooltip = nesnd_sih_jingnan.3.xiao_xinghua_fled }
        else_if = { limit = { has_variable = nesnd_sih_fate_xiao_xinghua var:nesnd_sih_fate_xiao_xinghua = 2 } custom_tooltip = nesnd_sih_jingnan.3.xiao_xinghua_killed }
        else_if = { limit = { has_variable = nesnd_sih_fate_xiao_xinghua var:nesnd_sih_fate_xiao_xinghua = 3 } custom_tooltip = nesnd_sih_jingnan.3.xiao_xinghua_stayed }
    }
    after = {
        remove_variable = nesnd_sih_fate_wei_jian
        remove_variable = nesnd_sih_fate_xiao_huli
        remove_variable = nesnd_sih_fate_xiao_xinghua
    }
}
