namespace = nesnd_crimea

# Yearly player check on Russia (10b): choose whether to launch a pre-emptive Crimean War play against TUR.
nesnd_crimea.1 = {
    type = country_event
    placement = root
    title = nesnd_crimea.1.t
    desc = nesnd_crimea.1.d
    flavor = nesnd_crimea.1.f
    event_image = { video = "unspecific_military_parade" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    duration = 3
    trigger = { nesnd_crimea_rus_yearly_eligible = yes }
    immediate = {
        c:TUR ?= { save_scope_as = nesnd_tur_country }
    }
    option = {
        name = nesnd_crimea.1.a
        nesnd_crimea_rus_launch_play = yes
        ai_chance = {
            base = 5
            modifier = { trigger = { "relations(c:TUR)" <= relations_threshold:poor } add = 8 }
            modifier = { trigger = { "relations(c:TUR)" <= relations_threshold:cold } add = 5 }
            modifier = { trigger = { "relations(c:TUR)" <= relations_threshold:hostile } add = 7 }
            modifier = { trigger = { game_date < 1851.1.1 } add = -20 }
        }
    }
    option = {
        name = nesnd_crimea.1.b
        default_option = yes
        custom_tooltip = nesnd_crimea.1.b_tt
        hidden_effect = { set_variable = { name = nesnd_crimea_rus_defer months = 36 } }
        ai_chance = {
            base = 100
            modifier = { trigger = { game_date < 1851.1.1 } add = 400 }
        }
    }
}

# France receives the Ottoman call to arms in the Crimean War (11).
nesnd_crimea.2 = {
    type = country_event
    placement = root
    title = nesnd_crimea.2.t
    desc = nesnd_crimea.2.d
    flavor = nesnd_crimea.2.f
    event_image = { video = "unspecific_signed_contract" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    duration = 3
    trigger = { nesnd_crimea_play_before_war = yes }
    immediate = {
        c:TUR ?= { save_scope_as = nesnd_tur_country }
        c:RUS ?= { save_scope_as = nesnd_rus_country }
    }
    option = {
        name = nesnd_crimea.2.a
        default_option = yes
        nesnd_crimea_join_tur_play = yes
        c:TUR ?= { change_relations = { country = root value = 25 } }
        ai_chance = {
            base = 20
            modifier = { trigger = { exists = c:RUS "relations(c:RUS)" <= relations_threshold:poor } add = 15 }
            modifier = { trigger = { exists = c:RUS "relations(c:RUS)" <= relations_threshold:cold } add = 25 }
            modifier = { trigger = { exists = c:RUS "relations(c:RUS)" <= relations_threshold:hostile } add = 35 }
            modifier = { trigger = { exists = c:RUS has_diplomatic_pact = { who = c:RUS type = rivalry } } add = 20 }
            modifier = { trigger = { exists = c:TUR "relations(c:TUR)" >= relations_threshold:cordial } add = 15 }
            modifier = { trigger = { exists = c:TUR "relations(c:TUR)" >= relations_threshold:amicable } add = 25 }
            modifier = {
                trigger = {
                    exists = c:TUR
                    OR = {
                        has_treaty_defensive_pact_with = { TARGET = c:TUR }
                        has_treaty_alliance_with = { TARGET = c:TUR }
                    }
                }
                add = 30
            }
        }
    }
    option = {
        name = nesnd_crimea.2.b
        c:TUR ?= { change_relations = { country = root value = -20 } }
        ai_chance = {
            base = 30
            modifier = { trigger = { exists = c:RUS "relations(c:RUS)" >= relations_threshold:cordial } add = 30 }
        }
    }
}
