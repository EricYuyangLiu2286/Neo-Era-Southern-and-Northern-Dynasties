namespace = nesnd_uni

# A claim on a region that is still not held fades away.
nesnd_uni.1 = {
	type = country_event
	placement = ROOT

	title = nesnd_uni.1.t
	desc = nesnd_uni.1.d
	flavor = nesnd_uni.1.f

	duration = 3

	event_image = {
		video = "asia_union_leader"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		exists = scope:nesnd_uni_lost_region
	}

	cancellation_trigger = {
		is_at_war = yes
	}

	option = {
		name = nesnd_uni.1.a
		default_option = yes
		if = {
			limit = {
				is_at_war = no
			}
			scope:nesnd_uni_lost_region = {
				remove_claim = root
			}
			custom_tooltip = {
				text = nesnd_uni_morale_reset_tt
				je:nesnd_uni_country ?= {
					set_bar_progress = {
						name = nesnd_uni_morale_bar
						value = 100
					}
				}
			}
		}
		else = {
			custom_tooltip = nesnd_uni_claim_kept_tt
		}
	}
}
