namespace = nesnd_phi_finance

nesnd_phi_finance.1 = {
	type = country_event
	placement = root
	duration = 2
	title = nesnd_phi_finance.1.t
	desc = nesnd_phi_finance.1.d
	flavor = nesnd_phi_finance.1.f
	event_image = {
		video = "ep2_active_shipyard"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = {
		c:USM ?= this
		c:PHI ?= {
			nesnd_phi_pending_valid = yes
			var:nesnd_phi_stage = 1
			var:nesnd_phi_requested_route = 1
		}
		c:PHI = {
			nesnd_phi_request_event_current = yes
			NOT = {
				has_variable = nesnd_phi_requested_cash
			}
		}
	}
	option = {
		name = nesnd_phi_finance.1.a
		trigger = {
			c:PHI = {
				nesnd_phi_pending_valid = yes
				var:nesnd_phi_stage = 1
				var:nesnd_phi_requested_route = 1
			}
			c:PHI = {
				nesnd_phi_request_event_current = yes
				NOT = {
					has_variable = nesnd_phi_requested_cash
				}
			}
		}
		ai_chance = {
			base = 70
			modifier = {
				trigger = {
					net_fixed_income < 20000
				}
				factor = 0.25
			}
		}
		if = {
			limit = {
				c:PHI = {
					nesnd_phi_request_event_current = yes
					NOT = {
						has_variable = nesnd_phi_requested_cash
					}
				}
			}
			hidden_effect = {
				c:PHI = {
					if = {
						limit = {
							nesnd_phi_pending_valid = yes
							var:nesnd_phi_stage = 1
						}
						set_variable = {
							name = nesnd_phi_stage
							value = 2
						}
						trigger_event = {
							id = nesnd_phi_finance.3
							popup = yes
						}
					}
				}
			}
			custom_tooltip = nesnd_phi_usm_accept_tt
		}
	}
	option = {
		name = nesnd_phi_finance.reject
		ai_chance = {
			base = 15
		}
		if = {
			limit = {
				c:PHI = {
					nesnd_phi_request_event_current = yes
					NOT = {
						has_variable = nesnd_phi_requested_cash
					}
				}
			}
			hidden_effect = {
				c:PHI ?= {
					if = {
						limit = {
							has_variable = nesnd_phi_pending
							var:nesnd_phi_stage = 1
						}
						nesnd_phi_clear_pending = yes
						set_variable = {
							name = nesnd_phi_cooldown
							days = 90
						}
					}
				}
			}
			custom_tooltip = nesnd_phi_reject_tt
		}
	}
	option = {
		name = nesnd_phi_finance.defer
		default_option = yes
		ai_chance = {
			base = 15
		}
		if = {
			limit = {
				c:PHI = {
					nesnd_phi_request_event_current = yes
					NOT = {
						has_variable = nesnd_phi_requested_cash
					}
				}
			}
			hidden_effect = {
				c:PHI ?= {
					if = {
						limit = {
							has_variable = nesnd_phi_pending
							var:nesnd_phi_stage = 1
						}
						nesnd_phi_clear_pending = yes
						set_variable = {
							name = nesnd_phi_cooldown
							days = 30
						}
					}
				}
			}
			custom_tooltip = nesnd_phi_defer_tt
		}
	}
}

    type = country_event hidden = yes
    trigger = {
        c:PHI ?= this
        has_variable = nesnd_phi_active
        has_variable = nesnd_phi_contract_serial
        exists = scope:nesnd_phi_binding_recheck_serial
        var:nesnd_phi_contract_serial = scope:nesnd_phi_binding_recheck_serial
        exists = scope:nesnd_phi_binding_recheck_treaty
        exists = var:nesnd_phi_treaty
        var:nesnd_phi_treaty = scope:nesnd_phi_binding_recheck_treaty
    }
    immediate = {
        remove_variable = nesnd_phi_binding_recheck_pending
        if = { limit = { nesnd_phi_owned_treaty_unamended = yes }
            nesnd_phi_expire_contract = yes
        }
        else = { nesnd_phi_month = yes }
    }
}
