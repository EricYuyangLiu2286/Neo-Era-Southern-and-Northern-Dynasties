nesnd_polish_windows.3 = {
	type = country_event
	placement = root
	title = nesnd_polish_windows.3.t
	desc = nesnd_polish_windows.3.d
	duration = 14
	event_image = {
		video = "unspecific_signed_contract"
	}
	trigger = {
		nesnd_poland_active = yes
		exists = c:FRA
		exists = global_var:nesnd_poland_window_enemy
		global_var:nesnd_poland_window_play ?= {
			is_war = yes
			nesnd_poland_external_play = yes
		}
	}
	cancellation_trigger = {
		NOT = {
			nesnd_poland_active = yes
			exists = c:FRA
			exists = global_var:nesnd_poland_window_enemy
			global_var:nesnd_poland_window_play ?= {
				is_war = yes
				nesnd_poland_external_play = yes
			}
		}
	}
	immediate = {
		global_var:nesnd_poland_window_enemy = {
			save_scope_as = window_enemy
		}
		c:FRA = {
			save_scope_as = window_france
		}
	}
	option = {
		name = nesnd_polish_windows.3.a
		default_option = yes
		ai_chance = {
			base = 6
		}
		save_scope_as = window_neutral
		custom_tooltip = nesnd_polish_windows.3.a_tt
		c:FRA = {
			trigger_event = {
				id = nesnd_polish_windows.10
				popup = yes
			}
		}
	}
	option = {
		name = nesnd_polish_windows.3.b
		ai_chance = {
			base = 2
		}
		trigger = {
			nesnd_poland_window_joinable = yes
		}
		save_scope_as = window_intervener
		global_var:nesnd_poland_window_play = {
			war = {
				join_war = {
					target = scope:window_intervener
					side = global_var:nesnd_poland_window_enemy
				}
				nesnd_poland_snapshot_war_sides = yes
			}
		}
		change_relations = {
			country = c:FRA
			value = -20
		}
	}
	option = {
		name = nesnd_polish_windows.3.c
		ai_chance = {
			base = 2
		}
		trigger = {
			hidden_trigger = { c:RUS ?= this global_var:nesnd_poland_window ?= 3 }
			nesnd_poland_window_joinable = yes
			# Claude audit 2026-11-17: named condition instead of raw global-variable rows.
			custom_tooltip = {
				text = nesnd_poland_krakow_in_war_tt
				global_var:nesnd_poland_window_play = {
					war = {
						is_war_participant = c:KRA
						nesnd_poland_krakow_joined_france = yes
					}
				}
			}
			c:KRA = {
				any_scope_state = {
					OR = {
						state_region = s:STATE_MAZOVIA
						state_region = s:STATE_GREATER_POLAND
					}
				}
			}
		}
		save_scope_as = window_intervener
		global_var:nesnd_poland_window_play = {
			war = {
				join_war = {
					target = scope:window_intervener
					side = global_var:nesnd_poland_window_enemy
				}
				nesnd_poland_snapshot_war_sides = yes
			}
		}
		if = {
			limit = {
				global_var:nesnd_poland_window_play ?= {
					war = {
						nesnd_poland_krakow_joined_france = yes
						nesnd_poland_opposing_war_sides = {
							FIRST = c:RUS
							SECOND = c:KRA
						}
						nesnd_poland_opposing_war_sides = {
							FIRST = c:RUS
							SECOND = c:FRA
						}
					}
				}
			}
			c:KRA = {
				random_scope_state = {
					limit = {
						OR = {
							state_region = s:STATE_MAZOVIA
							state_region = s:STATE_GREATER_POLAND
						}
					}
					save_scope_as = window_russian_demand
					if = {
						limit = {
							has_claim_by = c:RUS
						}
						global_var:nesnd_poland_window_play = {
							add_war_goal = {
								holder = c:RUS
								type = return_state
								target_state = scope:window_russian_demand
								target_country = c:KRA
							}
						}
					}
					else = {
						global_var:nesnd_poland_window_play = {
							add_war_goal = {
								holder = c:RUS
								type = conquer_state
								target_state = scope:window_russian_demand
								target_country = c:KRA
							}
						}
					}
				}
			}
		}
	}
}

# Called only by exact-play on_war_end. Never infer victory merely from peace.
nesnd_polish_windows.90 = {
	type = country_event
	hidden = yes
	trigger = {
		nesnd_poland_active = yes
		OR = {
			global_var:nesnd_poland_window ?= 2
			global_var:nesnd_poland_window ?= 3
		}
	}
	immediate = {
		if = {
			limit = {
				NOT = { has_global_variable = nesnd_poland_war_unclassified_goal }
				exists = global_var:nesnd_poland_window_enemy
				exists = c:FRA
				NOT = {
					has_global_variable = nesnd_poland_war_france_lost
				}
				OR = {
					has_global_variable = nesnd_poland_war_france_won
					AND = {
						has_global_variable = nesnd_poland_war_from_start
						has_global_variable = nesnd_poland_war_signed
						NOR = {
							has_global_variable = nesnd_poland_war_any_goal
							has_global_variable = nesnd_poland_war_capitulation
						}
					}
				}
			}
			set_global_variable = {
				name = nesnd_poland_window
				value = 4
			}
			set_global_variable = {
				name = nesnd_poland_peace_target
				value = global_var:nesnd_poland_window_enemy
			}
			if = {
				limit = {
					has_global_variable = nesnd_poland_war_france_won
				}
				set_global_variable = {
					name = nesnd_poland_peace_defeated
					value = yes
				}
			}
			else = {
				set_global_variable = {
					name = nesnd_poland_peace_defeated
					value = no
				}
			}
			trigger_event = {
				id = nesnd_polish_windows.7
				popup = yes
			}
