amendment_nesnd_cne_estate_ownership = {
    parent = law_nesnd_cne_estate_rights
    allowed_laws = { law_tenant_farmers law_serfdom law_peasant_proprietorship law_collectivized_agriculture }
    modifier = {
        interest_group_ig_landowners_pol_str_mult = 0.10
        interest_group_ig_rural_folk_pol_str_mult = 0.10
        state_literacy_growth_add = -0.0002
        state_radicals_and_loyalists_from_sol_change_mult = 0.05
        state_bureaucracy_population_base_cost_factor_mult = -0.05
    }
    possible = {
        trigger_if = {
            limit = { c:CNE ?= this }
            custom_tooltip = { text = nesnd_cne_estate_settled_tt has_variable = nesnd_cne_estate_settled }
        }
        OR = {
            ig:ig_landowners ?= { OR = { is_in_government = yes is_powerful = yes } }
            any_primary_culture = { has_discrimination_trait = tradition_sinosphere }
        }
    }
    would_sponsor = { nesnd_cne_estate_backer = yes }
    can_repeal = {
        nesnd_cne_estate_laws_unlocked = yes
        legitimacy >= legitimacy_to_repeal_amendment
    }
    ai_will_revoke = { always = yes }
    on_activate = { nesnd_cne_back_estate_rights = yes }
    on_deactivate = {
        nesnd_cne_end_estate_backing = {
            OTHER_GROUP = lawgroup_army_model
            OTHER_AMENDMENT = amendment_nesnd_cne_chartered_districts
        }
    }
}
