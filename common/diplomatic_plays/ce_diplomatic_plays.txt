dp_usm_civil_war = {
	war_goal = wg_nesnd_usm_annex
	mirror_war_goal = yes
	add_infamy_for_starting_initiator_wargoals = no
	enable_switch_sides = no
	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	is_epic = yes
	selectable_in_lens = {
		always = no
	}
	possible = {
		scope:target_country = {
			has_variable = usm_opposition_country
		}
	}
	additional_involvement_trigger = {
		NOT = {
			is_country_type = decentralized
		}
		OR = {
			has_variable = usm_opposition_country
			has_variable = nesnd_north_intervention
			has_variable = ce_usm_cw_ally_with_mng
			AND = {
				NOT = {
					c:CNE ?= this
				}
				aggressive_diplomatic_plays_permitted = yes
			}
		}
	}
	on_weekly_pulse = {

	}
	on_war_begins = {
        every_country = { limit = { nesnd_south_mainland_holdout = yes has_war_with = c:USM }
            set_variable = { name = nesnd_holdout_enrolled_round value = nesnd_south_round_value }
        }
    }
	on_war_end = {

	}
}
