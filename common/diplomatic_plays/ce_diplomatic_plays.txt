dp_usm_civil_war = {
	war_goal = wg_nesnd_usm_annex
	mirror_war_goal = yes
	allow_negotiated_peace = no
	blocked_by_diplomatic_status = no
	requires_interest_marker = no
	add_infamy_for_starting_initiator_wargoals = no
	enable_switch_sides = no
	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	is_epic = yes
	selectable_in_lens = {
		always = no
	}
	possible = {
        trigger_if = { limit = { exists = scope:initiator exists = scope:target_country }
            scope:initiator = { c:USM ?= this }
            scope:target_country = { OR = { custom_tooltip = { text = nesnd_ui_usm_opposition_country_tt has_variable = usm_opposition_country } nesnd_south_campaign_holdout = yes } NOT = { c:USM ?= this } }
        }
        trigger_else = { always = no }
    }
	additional_involvement_trigger = {
        NOT = { c:JAP ?= this custom_tooltip = { text = nesnd_ui_japan_neutral_tt has_variable = nesnd_japan_neutral } }
        # Japan against Wu fights in the Eastern Intervention (nesnd_open_japan_jingnan_war), never in this play (2026-11-17).
        NOT = { c:JAP ?= this custom_tooltip = { text = nesnd_ui_usm_opposition_country_tt has_variable = usm_opposition_country } }
        # Respect AI HNA's independent answer to the opening summons.
        NOT = {
            AND = {
                c:HNA ?= this
                is_ai = yes
                custom_tooltip = { text = nesnd_ui_south_answered_tt has_variable = nesnd_south_answered }
                custom_tooltip = { text = nesnd_ui_not_usm_opposition_country_tt NOT = { has_variable = usm_opposition_country } }
            }
        }

		NOT = {
			is_country_type = decentralized
		}
		OR = {
			custom_tooltip = { text = nesnd_ui_usm_opposition_country_tt has_variable = usm_opposition_country }
			custom_tooltip = { text = nesnd_ui_north_intervention_tt has_variable = nesnd_north_intervention }
			custom_tooltip = { text = nesnd_ui_ce_usm_cw_ally_with_mng_tt has_variable = ce_usm_cw_ally_with_mng }
			AND = {
				NOT = {
					c:CNE ?= this
				}
				aggressive_diplomatic_plays_permitted = yes
			}
		}
	}
	on_weekly_pulse = {

	}
	on_war_begins = {
        nesnd_muster_jingnan_war = yes
    }
	on_war_end = {

	}
}
