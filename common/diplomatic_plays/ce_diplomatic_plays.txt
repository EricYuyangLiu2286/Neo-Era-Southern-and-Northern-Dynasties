dp_usm_civil_war = {
	war_goal = annex_country
	mirror_war_goal = yes
	add_infamy_for_starting_initiator_wargoals = no
	enable_switch_sides = no
	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	
	is_epic = yes
	
	selectable_in_lens = {
		always = no
	}
	
	possible = {
		scope:target_country = {
			has_variable = usm_opposition_country
		}
	}
	
	additional_involvement_trigger = {
		NOT = { is_country_type = decentralized }
		OR = {
			custom_tooltip = {
				text = is_opposition_usm.tt
				has_variable = usm_opposition_country
			}
			NOT = { has_journal_entry = ce_cne_land_reform }
		}
		aggressive_diplomatic_plays_permitted = yes
	
		scope:target_country = {
			capital = {
				region = {
					save_temporary_scope_as = target_capital_region
				}
			}
		}
		
		OR = {
			is_home_country_for = scope:target_country
			is_subject_of = c:USM
			any_scope_ally = {
				this = c:USM
				has_interest_marker_in_region = scope:target_capital_region
			}
			any_scope_ally = {
				this = scope:target_country
				has_interest_marker_in_region = scope:target_capital_region
			}
			custom_tooltip = {
				text = is_opposition_usm.tt
				has_variable = usm_opposition_country
			}
		}
	}
	
	on_weekly_pulse = {}
	
	on_war_begins = {}

	on_war_end = {}
}
