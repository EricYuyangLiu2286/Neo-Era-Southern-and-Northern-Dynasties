    selectable = { nesnd_sha_gold_open = yes }

    possible = { nesnd_sha_shares_available = yes }

        evaluation_chance = { value = 0.25 }
        propose_score = { value = 20 }
        will_propose = { nesnd_sha_shares_available = yes }
        accept_score = {
            value = 0
            add = { desc = DIPLOMATIC_ACCEPTANCE_BASE value = -10 }
            if = { limit = { "relations(c:SHA)" >= relations_threshold:amicable } add = { desc = nesnd_sha_score_warm value = 20 } }
            if = { limit = { c:SHA = { nesnd_sha_gold_levels >= 4 } } add = { desc = nesnd_sha_score_four_mines value = 15 } }
