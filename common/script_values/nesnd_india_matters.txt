# Indian affairs: one government asks, another answers. Variables on the asking country:
#   nesnd_india_matter_kind     1 local project, 2 trade privilege, 3 military access,
#                               4 renewal of a trade privilege treaty, 5 renewal of a military access treaty,
#                               6 a subject leaves its overlord to join Mughal (consent only; nesnd_india_union.txt),
#                               7 Mughal asks its overlord for a loan, 8 Mughal asks to redeem its subject status,
#                               9 a prince asks Mughal or its overlord for famine aid (nesnd_india_mughal.txt),
#                               10 an Indian state asks an overseas local government for a trade privilege,
#                               11 for a treaty port, 12 and 13 their renewals (nesnd_india_overseas.txt)
#   nesnd_india_matter_stage    0 being chosen (projects), 2 waiting for an overlord's consent,
#                               3 waiting for the other government's answer, 4 under construction (projects)
#   nesnd_india_matter_to       the other government
#   nesnd_india_matter_await    the government asked now: an overlord at stage 2, the other government at stage 3
#   nesnd_india_matter_side     at stage 2: 1 our own overlord is asked, 2 the other government's overlord
#   nesnd_india_matter_ok_own, nesnd_india_matter_ok_other   the overlord that consented for each side
#   nesnd_india_matter_amount   the sum named in the matter
#   nesnd_india_matter_clock    the matter lapses unanswered when this runs out
# On the answering government: nesnd_india_matter_from (whose matter), nesnd_india_matter_answered.
nesnd_india_matter_kind_value = { value = 0 if = { limit = { has_variable = nesnd_india_matter_kind } value = var:nesnd_india_matter_kind } }
nesnd_india_matter_stage_value = { value = 0 if = { limit = { has_variable = nesnd_india_matter_stage } value = var:nesnd_india_matter_stage } }
nesnd_india_matter_side_value = { value = 0 if = { limit = { has_variable = nesnd_india_matter_side } value = var:nesnd_india_matter_side } }
nesnd_india_matter_amount_value = { value = 0 if = { limit = { has_variable = nesnd_india_matter_amount } value = var:nesnd_india_matter_amount } }

# Answering government scope. The sum named in the matter put to us.
nesnd_india_matter_quote = {
    value = 0
    if = { limit = { exists = var:nesnd_india_matter_from } value = var:nesnd_india_matter_from.nesnd_india_matter_amount_value }
}

# Answering government scope, per cent. Each kind of matter keeps its own weights.
nesnd_india_matter_consent_chance = {
    value = 0
    if = { limit = { nesnd_india_matter_is = { KIND = 1 } } value = nesnd_india_project_consent_chance }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 6 } } value = nesnd_india_union_consent_chance }
    else_if = { limit = { OR = { nesnd_india_matter_is = { KIND = 10 } nesnd_india_matter_is = { KIND = 11 } nesnd_india_matter_is = { KIND = 12 } nesnd_india_matter_is = { KIND = 13 } } } value = nesnd_india_overseas_consent_chance }
    else = { value = nesnd_india_frontier_consent_chance }
}
nesnd_india_matter_answer_chance = {
    value = 0
    if = { limit = { nesnd_india_matter_is = { KIND = 1 } } value = nesnd_india_project_fund_chance }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 3 } } value = nesnd_india_frontier_access_chance }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 7 } } value = nesnd_mug_loan_answer_chance }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 8 } } value = nesnd_mug_redeem_answer_chance }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 9 } } value = nesnd_mug_aid_answer_chance }
    else_if = { limit = { OR = { nesnd_india_matter_is = { KIND = 10 } nesnd_india_matter_is = { KIND = 12 } } } value = nesnd_india_overseas_trade_answer_chance }
    else_if = { limit = { OR = { nesnd_india_matter_is = { KIND = 11 } nesnd_india_matter_is = { KIND = 13 } } } value = nesnd_india_overseas_port_answer_chance }
    else = { value = nesnd_india_frontier_trade_chance }
}
