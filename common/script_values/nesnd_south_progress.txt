# Country scope. Tooltip evaluation must not depend on effects being executed.
nesnd_south_pressure_value = {
    if = { limit = { has_variable = usm_conflicts_within_tracker } value = var:usm_conflicts_within_tracker }
    else = { value = 0 }
}
nesnd_south_compact_value = {
    if = { limit = { has_variable = nesnd_compact_progress } value = var:nesnd_compact_progress }
    else = { value = 0 }
}

# Read-only display of the existing JE monthly effect; no second increment.
nesnd_south_monthly_pressure_value = {
 value = 0
 if = { limit = { has_journal_entry = ce_usm_conflicts_within }
  if = { limit = { has_modifier = nesnd_federal_compact } add = 0.4 }
  else = { add = 1 }
 }
 if = { limit = { has_variable = nesnd_south_reconciliation_active NOT = { has_variable = nesnd_south_reconciliation_first_month } nesnd_south_reconciliation_basis = yes } add = -3 }
}
