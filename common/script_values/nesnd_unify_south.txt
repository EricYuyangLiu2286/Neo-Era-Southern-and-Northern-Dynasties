# Country scope. Key points of nesnd_unify_south_state held entirely by the country or its subjects.
nesnd_unify_south_held = {
    value = 0
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_GUANGDONG } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_SICHUAN } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_CHONGQING } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_HUNAN } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_EASTERN_HUBEI } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_NANJING } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_FUJIAN } add = 1 }
    if = { limit = { country_or_subject_owns_entire_state_region = STATE_JIANGXI } add = 1 }
}

# Country scope. Key points where the country itself owns at least one state.
nesnd_unify_south_owned = {
    value = 0
    every_scope_state = { limit = { nesnd_unify_south_state = yes } add = 1 }
}

# Country scope. The bar is the only writer: each week it adds the gap between the held count and its current value.
nesnd_unify_south_bar_delta = {
    value = 0
    if = {
        limit = { has_journal_entry = nesnd_unify_south }
        value = nesnd_unify_south_held
        subtract = "je:nesnd_unify_south.scripted_bar_progress(nesnd_unify_south_bar)"
    }
}
