nesnd_congress_votes = {
	value = 1
	if = {
		limit = { country_rank >= rank_value:great_power }
		add = 1
	}
	if = {
		limit = { global_var:nesnd_congress_chair ?= this }
		if = {
			limit = {
				has_global_variable = nesnd_poland_france_will
				global_var:nesnd_poland_france_will >= 60
			}
			add = 2
		}
		else = { add = 1 }
	}
}

nesnd_congress_order_value = {
	value = 70
	if = {
		limit = {
			c:FRA ?= {
				has_variable = nesnd_congress_guarantee_pledge
				nesnd_poland_guarantees_krakow = yes
			}
		}
		add = 5
	}
	if = {
		limit = {
			c:FRA ?= { country_rank >= rank_value:great_power }
		}
		add = 10
	}
	else = { subtract = 20 }
	if = {
		limit = {
			c:FRA ?= { is_power_bloc_leader = yes }
		}
		add = 10
	}
	every_country = {
		limit = { nesnd_congress_member = yes has_variable = nesnd_congress_grievances }
		subtract = { value = var:nesnd_congress_grievances multiply = 2 }
	}
	if = {
		limit = { has_global_variable = nesnd_congress_last_rate }
		add = { value = global_var:nesnd_congress_last_rate subtract = 50 divide = 5 }
	}
	if = {
		limit = { has_global_variable = peoples_springtime_happening }
		subtract = 20
	}
	c:FRA ?= {
		je:je_forest_of_bayonets ?= {
			subtract = {
				value = "scripted_bar_progress(forest_of_bayonets_progress_bar)"
				divide = 5
			}
		}
	}
	min = 0
	max = 100
}
