# Local projects of the Indian princes. A project asked of another government is a matter of kind 1;
# its stage, the government asked, the sum and the deadline are the matter's (nesnd_india_matters.txt).
# Project variables on the prince:
#   nesnd_india_project_type    1 university, 2 tooling workshop, 3 railway, 4 survey and training
#   nesnd_india_project_site    the chosen state
#   nesnd_india_project_base    levels standing at the site when the work was ordered
#   nesnd_india_project_choice  the thirty days in which the prince chooses
#   nesnd_india_project_shown   the last project chosen, kept after it ends so that reports can name it
nesnd_india_project_type_value = { value = 0 if = { limit = { has_variable = nesnd_india_project_type } value = var:nesnd_india_project_type } }
nesnd_india_project_base_value = { value = 0 if = { limit = { has_variable = nesnd_india_project_base } value = var:nesnd_india_project_base } }
nesnd_india_project_shown_value = { value = 0 if = { limit = { has_variable = nesnd_india_project_shown } value = var:nesnd_india_project_shown } }

nesnd_india_survey_fee = 5000

# Prince scope. The sum named when another government is asked: the survey fee the prince pays,
# or the grant the other government pays.
nesnd_india_project_ask_amount = {
    value = nesnd_india_grant
    if = { limit = { nesnd_india_project_type_value = 4 } value = nesnd_india_survey_fee }
}

# State scope. Levels of the building standing there now.
nesnd_india_project_level_1 = { value = 0 if = { limit = { has_building = building_university } add = b:building_university.level } }
nesnd_india_project_level_2 = { value = 0 if = { limit = { has_building = building_tooling_workshop } add = b:building_tooling_workshop.level } }
nesnd_india_project_level_3 = { value = 0 if = { limit = { has_building = building_railway } add = b:building_railway.level } }

# Answering government scope, per cent. An overlord asked to let another government take a stake.
nesnd_india_project_consent_chance = {
    value = 60
    if = { limit = { nesnd_india_project_ai_funder_ours = yes } add = 20 }
    if = { limit = { nesnd_india_project_ai_funder_friendly = yes } add = 15 }
    if = { limit = { nesnd_india_project_ai_funder_hostile = yes } add = -40 }
    if = { limit = { nesnd_india_project_ai_owner_restless = yes } add = -20 }
    min = 5
    max = 95
}
# Answering government scope, per cent. A government asked for money, or for its engineers.
nesnd_india_project_fund_chance = {
    value = 30
    if = { limit = { nesnd_india_project_ai_owner_friendly = yes } add = 25 }
    if = { limit = { nesnd_india_project_ai_has_interest = yes } add = 20 }
    if = { limit = { nesnd_india_empire = yes } add = 20 }
    if = { limit = { nesnd_india_project_ai_owner_ours = yes } add = 15 }
    if = { limit = { nesnd_india_project_offer_survey = yes } add = 30 }
    min = 5
    max = 95
}
