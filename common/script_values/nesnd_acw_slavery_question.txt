# Slavery question helpers. The crisis bar's sources are inline in
# scripted_progress_bars/nesnd_acw_slavery_question.txt so the journal panel lists them.

# COUNTRY scope (USA). Movement radicalism, read by the secession crisis.
nesnd_acw_rad_anti = {
    value = 0
    every_political_movement = {
        limit = { is_political_movement_type = movement_anti_slavery }
        add = this.political_movement_radicalism
    }
}
nesnd_acw_rad_pro = {
    value = 0
    every_political_movement = {
        limit = { is_political_movement_type = movement_pro_slavery }
        add = this.political_movement_radicalism
    }
}

# One year of income; also the AI's reserve test (1.5 years).
nesnd_acw_buyout_cost = {
    value = income
    multiply = 52
    min = 0
}
nesnd_acw_buyout_ai_reserve = {
    value = nesnd_acw_buyout_cost
    multiply = 1.5
}

# COUNTRY scope (HUS). A quarter of income for the settler fund.
nesnd_hus_west_funding_cost = {
    value = income
    multiply = 13
    min = 0
}

# STATE scope. Escapees in a state, and a batch size: 0.5% of border escapees, 5,000–20,000.
nesnd_hus_west_state_escapees = {
    value = 0
    every_scope_pop = { limit = { nesnd_hus_west_escapee = yes } add = total_size }
}
# COUNTRY scope (USA).
nesnd_hus_west_batch_size = {
    value = 0
    every_scope_state = {
        limit = { nesnd_hus_west_departure = yes }
        add = nesnd_hus_west_state_escapees
    }
    multiply = 0.005
    min = 5000
    max = 20000
}
