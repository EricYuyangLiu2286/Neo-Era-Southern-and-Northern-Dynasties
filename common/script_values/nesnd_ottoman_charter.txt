# Country scope. Scores of the two agendas of nesnd_ottoman_charter, recomputed from laws whenever read (no stored copy;
# same approach as vanilla tenpo_total_agenda_goals_completed, GAME common/script_values/ep2_japan_values.txt:76).
nesnd_ottoman_charter_score = {
    value = 0
    if = { limit = { nesnd_ottoman_charter_c1 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c2 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c3 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c4 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c5 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c6 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c7 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_c8 = yes } add = 1 }
}
nesnd_ottoman_qin_score = {
    value = 0
    if = { limit = { nesnd_ottoman_charter_q1 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q2 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q3 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q4 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q5 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q6 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q7 = yes } add = 1 }
    if = { limit = { nesnd_ottoman_charter_q8 = yes } add = 1 }
}
nesnd_ottoman_charter_lead = { value = nesnd_ottoman_charter_score subtract = nesnd_ottoman_qin_score }
nesnd_ottoman_qinist_leader_count = {
    value = 0
    # Sum inside an iterator as in nesnd_ottoman_enthusiasm_qinist (GAME common/script_values/ip4_je_values.txt:122-135).
    every_interest_group = { limit = { leader ?= { has_ideology = ideology:ideology_ce_qin_royalist_leader } } add = 1 }
}
nesnd_ottoman_barracks_grant = { value = gdp multiply = 0.01 round = yes min = 1000 }
