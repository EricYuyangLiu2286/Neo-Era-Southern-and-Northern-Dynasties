# CNE country scope; the monthly bar only uses this when the four required state pieces exist.
nesnd_sha_yan_mean_turmoil = {
    value = 0
    every_scope_state = { limit = { nesnd_sha_yan_region = yes } add = turmoil }
    divide = nesnd_sha_yan_region_states
}
nesnd_sha_yan_region_states = {
    value = 0
    every_scope_state = { limit = { nesnd_sha_yan_region = yes } add = 1 }
    min = 1
}

nesnd_sha_recruitment_cost = {
    value = nesnd_sha_medium_cost
    if = {
        limit = {
            c:CNE ?= {
                OR = {
                    is_at_war = yes
                    trigger_if = { limit = { has_variable = cne_landreform_failure_tracker } var:cne_landreform_failure_tracker >= 30 }
                    trigger_else = { always = no }
                }
            }
        }
        multiply = 0.5
    }
}

# State-region scope, all owners. Lower-strata size and size-weighted average standard of living
# (GAME common/script_values/ip5_je_values.txt:515 counts the lower strata the same way).
nesnd_sha_yan_lower_pop = {
    value = 0
    every_scope_state = { every_scope_pop = { limit = { strata = lower } add = total_size } }
}
nesnd_sha_yan_lower_sol = {
    value = 0
    every_scope_state = { every_scope_pop = { limit = { strata = lower } add = { value = standard_of_living multiply = total_size } } }
    divide = { value = nesnd_sha_yan_lower_pop min = 1 }
}
# Any scope. Number of the seven Yan state regions (nesnd_sha_yan_region) in lower-strata hardship.
nesnd_sha_yan_hardship_states = {
    value = 0
    if = { limit = { s:STATE_SHENGJING = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_SOUTHERN_MANCHURIA = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_SONGJIANG = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_NORTHERN_MANCHURIA = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_AMUR = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_OUTER_MANCHURIA = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
    if = { limit = { s:STATE_BOLI = { nesnd_sha_yan_region_hardship = yes } } add = 1 }
}
