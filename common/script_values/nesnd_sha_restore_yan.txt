nesnd_sha_yan_mean_turmoil = {
    value = 0
    every_scope_state = { limit = { nesnd_sha_yan_region = yes } add = turmoil }
    divide = nesnd_sha_yan_region_states
}
nesnd_sha_yan_region_states = {
    value = 0
    every_scope_state = { limit = { nesnd_sha_yan_region = yes } add = 1 }
    min = 1
}

nesnd_sha_recruitment_cost = {
    value = nesnd_sha_medium_cost
    if = {
        limit = {
            c:CNE ?= {
                OR = {
                    is_at_war = yes
                    trigger_if = { limit = { has_variable = cne_landreform_failure_tracker } var:cne_landreform_failure_tracker >= 30 }
                    trigger_else = { always = no }
                }
            }
        }
        multiply = 0.5
    }
}
