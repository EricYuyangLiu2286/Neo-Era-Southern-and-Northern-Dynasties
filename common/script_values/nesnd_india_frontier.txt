# Bhutan and Nepal with Mughal. Records on Bhutan or Nepal, one set per treaty (T is trade or access):
#   nesnd_india_frontier_T_with     the Mughal party of the treaty
#   nesnd_india_frontier_T_term     runs out on the day the treaty expires (five years after signing)
#   nesnd_india_frontier_T_notice   runs out 180 days before that day
#   nesnd_india_frontier_T_asked    the renewal question has been put for this term
#   nesnd_india_frontier_T_renewed  Mughal agreed to renew: at expiry the term starts again for five years
#   nesnd_india_frontier_empire     the Mughal government Bhutan or Nepal deals with (the richest), set monthly
# Default sums and chances; to be tuned after in-game tests.

# Paid once by Mughal to Bhutan or Nepal when either treaty is signed.
nesnd_india_frontier_payment = 10000

# Days of a record T (nesnd_india_frontier_record): the term runs to expiry, the notice to 180 days
# before it. The overseas journal's records (overseas_trade, overseas_port) are in
# common/script_values/nesnd_india_overseas.txt.
nesnd_india_frontier_trade_days = 1825
nesnd_india_frontier_trade_notice_days = 1645
nesnd_india_frontier_access_days = 1825
nesnd_india_frontier_access_notice_days = 1645

# Bhutan or Nepal scope, per cent a month: an AI asks Mughal for the trade privilege.
nesnd_india_frontier_trade_ask_chance = {
    value = 3
    if = { limit = { exists = var:nesnd_india_frontier_empire is_adjacent_to_country = var:nesnd_india_frontier_empire } value = 6 }
}
# Mughal scope, per cent a month for each country: an AI asks for military access.
nesnd_india_frontier_access_ask_chance = 1

# Answering government scope, per cent. Mughal asked for the trade privilege, or to renew a treaty.
nesnd_india_frontier_trade_chance = {
    value = 60
    if = { limit = { nesnd_india_frontier_ai_asker_relations = { VALUE = 20 } } add = 20 }
    min = 5
    max = 95
}
# Answering government scope, per cent. An overlord asked to consent.
nesnd_india_frontier_consent_chance = {
    value = 40
    if = { limit = { nesnd_india_frontier_ai_empire_friendly = yes } add = 20 }
    if = { limit = { nesnd_india_frontier_ai_empire_hostile = yes } add = -30 }
    if = { limit = { nesnd_india_frontier_ai_subject_restless = yes } add = -10 }
    min = 5
    max = 95
}
# Answering government scope, per cent. Bhutan or Nepal asked for military access.
nesnd_india_frontier_access_chance = {
    value = 25
    if = { limit = { nesnd_india_frontier_ai_asker_relations = { VALUE = 50 } } add = 25 }
    if = { limit = { nesnd_india_frontier_ai_has_trade = yes } add = 15 }
    min = 5
    max = 95
}
