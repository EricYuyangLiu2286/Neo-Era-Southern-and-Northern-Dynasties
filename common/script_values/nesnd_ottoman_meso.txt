# Script values for Ottoman B9 "Mesopotamia" journal (nesnd_ottoman_meso).
# Country scope (evaluated country).

# Mesopotamia foe selection score (Rule 3): +20 per state in target regions held directly or via subjects; +10 historical weight for c:PER / c:IRN (R11 class a).
nesnd_ottoman_meso_foe_score = {
    value = 0
    every_scope_state = {
        limit = {
            state_region = {
                nesnd_ottoman_meso_is_target_region = yes
            }
        }
        add = 20
    }
    every_subject_or_below = {
        every_scope_state = {
            limit = {
                state_region = {
                    nesnd_ottoman_meso_is_target_region = yes
                }
            }
            add = 20
        }
    }
    if = {
        limit = {
            OR = {
                c:PER ?= this
                c:IRN ?= this
            }
        }
        add = 10
    }
}

# Direct target-region state count for selecting the alarm event (.23) recipient (Rule 11).
nesnd_ottoman_meso_held_states_count = {
    value = 0
    every_scope_state = {
        limit = {
            state_region = {
                nesnd_ottoman_meso_is_target_region = yes
            }
        }
        add = 1
    }
}
