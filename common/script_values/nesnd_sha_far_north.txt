# State scope: the furthest secession among the non-Russian peoples living in this state.

    value = 0
    ordered_scope_culture = {
        limit = { NOT = { this = cu:russian } }
        order_by = "culture_secession_progress(c:RUS)"
        add = "culture_secession_progress(c:RUS)"
    }

# The furthest secession among the peoples of the Russian borderland, in percent.
nesnd_sha_borderland_progress = {
    value = 0
    if = {
        limit = { exists = c:RUS }
        c:RUS = {
            ordered_scope_state = {
                limit = { nesnd_sha_russian_borderland = yes }
                order_by = nesnd_sha_state_secession_progress
                add = nesnd_sha_state_secession_progress
            }
        }
        multiply = 100
    }
}
