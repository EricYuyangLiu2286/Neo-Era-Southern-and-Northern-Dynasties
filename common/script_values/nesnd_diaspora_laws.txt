# Both rates are evaluated on the country from which this particular party is leaving.
nesnd_diaspora_citizenship_rate = {
    value = 0
    if = { limit = { has_law = law_type:law_ethnostate } value = 4 }
    else_if = { limit = { has_law = law_type:law_national_supremacy } value = 3 }
    else_if = { limit = { has_law = law_type:law_racial_segregation } value = 2 }
    else_if = {
        limit = {
            OR = {
                has_law = law_type:law_cultural_exclusion
                has_law = law_type:law_nesnd_hus_hierarchy
                has_law = law_type:law_nesnd_hus_registry_manchu
                has_law = law_type:law_nesnd_hus_registry_native
            }
        }
        value = 1
    }
}

nesnd_diaspora_distribution_rate = {
    value = 0
    if = {
        limit = {
            OR = {
                has_law = law_type:law_autocracy
                has_law = law_type:law_neo_absolutism
                has_law = law_type:law_bakufu
                has_law = law_type:law_single_party_state
            }
        }
        value = 3
    }
    else_if = {
        limit = {
            OR = {
                has_law = law_type:law_oligarchy
                has_law = law_type:law_technocracy
                has_law = law_type:law_organic_regulation
            }
        }
        value = 2
    }
    else_if = {
        limit = {
            OR = {
                has_law = law_type:law_landed_voting
                has_law = law_type:law_wealth_voting
                has_law = law_type:law_nesnd_keju_voting
            }
        }
        value = 1
    }
}
