# Shared script values of the Ottoman mid/late-game line (F3).
# Country scope (evaluated country).

# Intervener selection score for Rumelia and Eastern Provinces (F3 Rule 3).
nesnd_ottoman_intervener_score = {
    value = 0
    if = {
        limit = { c:RUS ?= this }
        add = 40
    }
    if = {
        limit = { c:AUS ?= this }
        add = 30
    }
    if = {
        limit = {
            exists = c:TUR
            "relations(c:TUR)" <= relations_threshold:poor
        }
        add = 10
    }
    if = {
        limit = {
            exists = c:TUR
            "relations(c:TUR)" <= relations_threshold:cold
        }
        add = 20
    }
    if = {
        limit = {
            exists = c:TUR
            has_diplomatic_pact = { who = c:TUR type = rivalry }
        }
        add = 20
    }
    if = {
        limit = { country_rank >= rank_value:great_power }
        add = 10
    }
}

# European Concert partner selection score for event nesnd_ottoman_recovery.12 option b (F3 Rule 11).
nesnd_ottoman_concert_score = {
    value = 0
    if = {
        limit = { exists = c:TUR }
        add = "relations(c:TUR)"
    }
    if = {
        limit = {
            OR = {
                c:GBR ?= this
                c:FRA ?= this
            }
        }
        add = 15
    }
}
