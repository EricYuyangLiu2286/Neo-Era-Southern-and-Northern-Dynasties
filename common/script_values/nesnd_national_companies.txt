# Building levels the scoped country holds in one area, for first-come companies.
# Evaluated in country scope; company possible blocks compare these.

nesnd_co_coffee_south_india = {
	value = 0
	every_scope_state = {
		limit = { region = sr:region_south_india }
		every_scope_building = {
			limit = { is_building_type = building_coffee_plantation }
			add = this.level
		}
	}
}
