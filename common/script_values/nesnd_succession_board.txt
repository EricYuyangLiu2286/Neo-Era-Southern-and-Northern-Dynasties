# Contextless values, not country ROOT. Political shares do not estimate military strength.
nesnd_board_ming_share = {
	value = 0
	if = {
		limit = {
			has_global_variable = nesnd_board_members
			nesnd_preview_g_board_members_value > 0
		}
		add = nesnd_preview_g_board_ming_members_value
		multiply = 100
		divide = nesnd_preview_g_board_members_divisor
	}
}
