# ==== The request channel: unified event-based requests across Indian states and powers ====
# Variables on the asking country:
#   nesnd_india_request_kind    1 teachers, 2 engineers, 3 officers, 4 autonomy, 5 defensive pact,
#                               6 empire invites prince, 8 prince asks empire, 9 local project,
#                               10 Mughal sells property, 11 Mughal buys back property, 12 famine aid.
#   nesnd_india_request_stage   1 awaiting overlord consent, 2 awaiting partner reply
nesnd_india_request_serial = { value = 0 if = { limit = { has_variable = nesnd_india_request_serial } value = var:nesnd_india_request_serial } }
nesnd_india_request_kind = { value = 0 if = { limit = { has_variable = nesnd_india_request_kind } value = var:nesnd_india_request_kind } }
nesnd_india_request_quote = { value = 0 if = { limit = { has_variable = nesnd_india_request_quote } value = var:nesnd_india_request_quote } }
nesnd_india_request_stage_value = {
    value = 0
    if = {
        limit = { has_variable = nesnd_india_request_stage }
        value = var:nesnd_india_request_stage
    }
}
nesnd_india_request_consent_side_value = {
    value = 0
    if = {
        limit = { has_variable = nesnd_india_request_consent_side }
        value = var:nesnd_india_request_consent_side
    }
}
nesnd_india_incoming_kind = { value = 0 if = { limit = { exists = scope:india_sender } value = scope:india_sender.nesnd_india_request_kind } }
nesnd_india_incoming_quote = {
    value = 0
    if = {
        limit = { exists = scope:india_sender }
        value = scope:india_sender.nesnd_india_request_quote
    }
}
nesnd_india_req_consent_chance = {
    value = 40
    if = {
        limit = { nesnd_india_incoming_kind = 9 }
        value = nesnd_india_project_consent_chance
    }
    else_if = {
        limit = {
            OR = {
                nesnd_india_incoming_kind = 6
                nesnd_india_incoming_kind = 8
            }
        }
        value = nesnd_india_union_consent_chance
    }
    else_if = {
        limit = { nesnd_india_incoming_kind = 12 }
        value = nesnd_mug_aid_consent_chance
    }
}
nesnd_india_req_consent_forbid_chance = {
    value = 100
    subtract = nesnd_india_req_consent_chance
    min = 0
}
nesnd_india_req_answer_chance = {
    value = 0
    if = {
        limit = { nesnd_india_incoming_kind = 9 }
        value = nesnd_india_project_fund_chance
    }
    else_if = {
        limit = { nesnd_india_incoming_kind = 10 }
        value = nesnd_mug_sell_answer_chance
    }
    else_if = {
        limit = { nesnd_india_incoming_kind = 11 }
        value = nesnd_mug_buyback_answer_chance
    }
    else_if = {
        limit = { nesnd_india_incoming_kind = 12 }
        value = nesnd_mug_aid_answer_chance
    }
}
nesnd_india_req_answer_decline_chance = {
    value = 100
    subtract = nesnd_india_req_answer_chance
    min = 0
}
nesnd_india_grant = { value = net_fixed_income multiply = 4 min = 25000 max = 125000 }
nesnd_india_regime = {
    value = 0
    if = { limit = { has_law = law_type:law_council_republic } value = 6 }
    else_if = { limit = { has_law = law_type:law_theocracy } value = 7 }
    else_if = { limit = { OR = { has_law = law_type:law_presidential_republic has_law = law_type:law_parliamentary_republic } } value = 4 }
    else_if = { limit = { country_has_monarchy_law = yes }
        value = 5
        if = { limit = { exists = ruler }
            if = { limit = { ruler = { nesnd_character_of_dynasty = { HOUSE = house_bonaparte } } } value = 1 }
            else_if = { limit = { ruler = { nesnd_character_of_dynasty = { HOUSE = house_orleans } } } value = 2 }
            else_if = { limit = { ruler = { OR = { nesnd_character_of_dynasty = { HOUSE = house_bourbon } has_variable = house_bourbon_leader has_variable = legitimist_leader } } } value = 3 }
        }
    }
}

# Answering government scope, per cent. A government asked for money, or for its engineers.

    value = 30

        limit = {
            nesnd_india_req_ai_asker_relations = { VALUE = 50 }
        }
        add = 25
    }
    if = {
        limit = { nesnd_india_project_ai_has_interest = yes }
        add = 20
    }
    if = {
        limit = { nesnd_india_empire = yes }
        add = 20
    }
    if = {
        limit = { nesnd_india_project_ai_owner_ours = yes }
        add = 15
    }
    if = {
        limit = {
            nesnd_india_project_offer_is = { TYPE = 4 }
        }
        add = 30
    }
    min = 5
    max = 95
}

nesnd_india_union_mode_dominion = 2

# Country scope. Populated Indian states held by this country and its subjects, and in all the world.
# On both sides of the share a holder counts only if its capital lies in an Indian state and it is not
# the subject of an overlord whose capital lies outside India (nesnd_mug_foreign_overlord): a foreign
# holder (FRA, POR, GBR and the like) and a foreign holder's subjects are left out. Our own states always
# count, even while we are ourselves the subject of a foreign overlord.
nesnd_india_held_states = {
    value = 0
    if = { limit = { capital ?= { nesnd_india_state = yes } }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
    every_subject_or_below = {
        limit = { capital ?= { nesnd_india_state = yes } NOT = { nesnd_mug_foreign_overlord = yes } }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
}
nesnd_india_all_states = {
    value = 0
    every_country = {
        limit = {
            is_country_alive = yes
            capital ?= { nesnd_india_state = yes }
            OR = { this = root NOT = { nesnd_mug_foreign_overlord = yes } }
        }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
}
# The share the journal nesnd_india_unified asks for.
nesnd_india_unified_share_needed = 75
# Per cent of the populated Indian states held by this country and its subjects.
nesnd_india_held_share = {
    value = 0
    if = { limit = { nesnd_india_all_states > 0 }
        value = nesnd_india_held_states
        multiply = 100
        divide = nesnd_india_all_states
    }
}

# Monthly AI roll after the ordinary independence-play conditions pass.
nesnd_india_indep_ai_play_chance = {
    value = 2
    if = {
        limit = {
            OR = {
                # Each stockpiled batch can equip two battalions.
                army_size >= {
                    value = overlord.army_size
                    multiply = 0.5
                    subtract = { value = nesnd_india_indep_stores multiply = 2 }
                }
                overlord = {
                    OR = {
                        has_revolution = yes
                        any_enemy_in_war = { NOT = { this = root } }
                    }
                }
            }
        }
        value = 30
    }
}
