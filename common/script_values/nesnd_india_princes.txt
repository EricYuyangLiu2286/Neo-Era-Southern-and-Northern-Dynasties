# ==== Princes join Mughal (formerly nesnd_india_union.txt) ====
# Princes joining Mughal (see common/scripted_triggers/nesnd_india_princes.txt).
# Variables:
#   nesnd_india_union_host   on a prince: the Mughal government it would ask to join (the richest), set monthly
#   nesnd_india_union_mode   on a prince that asked to join: 1 merged into Mughal, 2 Mughal's dominion
# Default sums and chances; to be tuned after in-game tests.

# The two ways of joining. The settlement takes MODE = merge or dominion and reads these.
nesnd_india_union_mode_merge = 1
nesnd_india_union_mode_dominion = 2
nesnd_india_union_mode_value = { value = 0 if = { limit = { has_variable = nesnd_india_union_mode } value = var:nesnd_india_union_mode } }

# Any scope with scope:india_sender and scope:india_partner (request kinds 6 and 8). What Mughal pays
# in all: the quote, and the same again to the prince's overlord when the prince leaves it.
nesnd_india_union_due = {
    value = 0
    if = { limit = { exists = scope:india_sender }
        value = scope:india_sender.nesnd_india_request_quote
        if = { limit = { nesnd_india_union_compensation_due = yes } multiply = 2 }
    }
}

# Country scope. Populated Indian states held by this country and its subjects, and in all the world.
# On both sides of the share a holder counts only if its capital lies in an Indian state and it is not
# the subject of an overlord whose capital lies outside India (nesnd_mug_foreign_overlord): a foreign
# holder (FRA, POR, GBR and the like) and a foreign holder's subjects are left out. Our own states always
# count, even while we are ourselves the subject of a foreign overlord.
nesnd_india_held_states = {
    value = 0
    if = { limit = { capital ?= { nesnd_india_state = yes } }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
    every_subject_or_below = {
        limit = { capital ?= { nesnd_india_state = yes } NOT = { nesnd_mug_foreign_overlord = yes } }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
}
nesnd_india_all_states = {
    value = 0
    every_country = {
        limit = {
            is_country_alive = yes
            capital ?= { nesnd_india_state = yes }
            OR = { this = root NOT = { nesnd_mug_foreign_overlord = yes } }
        }
        every_scope_state = { limit = { nesnd_india_state = yes state_population > 0 } add = 1 }
    }
}
# The share the journal nesnd_india_unified asks for.
nesnd_india_unified_share_needed = 75
# Per cent of the populated Indian states held by this country and its subjects.
nesnd_india_held_share = {
    value = 0
    if = { limit = { nesnd_india_all_states > 0 }
        value = nesnd_india_held_states
        multiply = 100
        divide = nesnd_india_all_states
    }
}

# Monthly AI roll after the ordinary independence-play conditions pass.
nesnd_india_indep_ai_play_chance = {
    value = 2
    if = {
        limit = {
            OR = {
                # Each stockpiled batch can equip two battalions.
                army_size >= {
                    value = overlord.army_size
                    multiply = 0.5
                    subtract = { value = nesnd_india_indep_stores multiply = 2 }
                }
                overlord = {
                    OR = {
                        has_revolution = yes
                        any_enemy_in_war = { NOT = { this = root } }
                    }
                }
            }
        }
        value = 30
    }
}
