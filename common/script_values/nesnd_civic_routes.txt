nesnd_usm_mandate_institution_points = 10
nesnd_usm_mandate_policy_points = 20
nesnd_usm_mandate_quarterly_points = 2

nesnd_usm_mandate_settled_value = {
    value = 50
    if = { limit = { has_law_or_variant = law_type:law_nesnd_keju_voting } add = 5 }
    if = { limit = { has_law_or_variant = law_type:law_appointed_bureaucrats } add = 10 }
    if = {
        limit = { ig:ig_rural_folk = { is_in_government = yes is_powerful = yes } }
        subtract = 10
    }
    if = {
        limit = { nesnd_usm_civic_country = yes }
        if = {
            limit = { has_variable = nesnd_c2_usm_constitution_done has_variable = nesnd_c2_usm_constitution_choice }
            if = { limit = { OR = { var:nesnd_c2_usm_constitution_choice = 1 var:nesnd_c2_usm_constitution_choice = 2 } } add = nesnd_usm_mandate_institution_points }
            else_if = { limit = { var:nesnd_c2_usm_constitution_choice = 3 } subtract = nesnd_usm_mandate_institution_points }
        }
        if = {
            limit = { has_variable = nesnd_c2_usm_distribution_done has_variable = nesnd_c2_usm_distribution_choice }
            if = { limit = { var:nesnd_c2_usm_distribution_choice = 1 } add = nesnd_usm_mandate_institution_points }
            else_if = { limit = { OR = { var:nesnd_c2_usm_distribution_choice = 2 var:nesnd_c2_usm_distribution_choice = 3 } } subtract = nesnd_usm_mandate_institution_points }
        }
    }
    min = 0
    max = 100
}

nesnd_usm_mandate_target = {
    value = nesnd_usm_mandate_settled_value
    if = { limit = { has_variable = nesnd_usm_mandate_central } add = nesnd_usm_mandate_policy_points }
    else_if = { limit = { has_variable = nesnd_usm_mandate_provincial } subtract = nesnd_usm_mandate_policy_points }
    min = 0
    max = 100
    # AI cabinets pursue moderate quarterly targets. Completed projects still transfer actual authority.
    if = { limit = { is_ai = yes } min = 30 max = 70 }
}
