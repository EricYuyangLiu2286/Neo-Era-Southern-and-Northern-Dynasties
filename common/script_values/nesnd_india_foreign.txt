# ==== Overseas dealings (India batch 17) ====
# Five land strategic regions:
#   1 himalayas   -> sr:region_himalayas
#   2 indonesia   -> sr:region_indonesia
#   3 east_africa -> sr:region_east_africa
#   4 arabia      -> sr:region_arabia
#   5 persia      -> sr:region_greater_persia
# Progress is kept solely on the five scripted progress bars (common/scripted_progress_bars/nesnd_india_foreign.txt);
# no country variable mirrors them.
#
# Variables on the Indian country:
#   nesnd_india_ocean_trade              one-time milestone "Indian Ocean Trade" has fired
#   nesnd_india_overseas_above50_<dir>   tracks whether the direction's bar is currently at or above 50
#   nesnd_india_overseas_warned_<dir>    5-year timed variable list of powers already notified for <dir>

# ---------------------------------------------------------------------------------------------
# Signing price: a quarter of weekly income (13 weeks), floor £10,000.
# ---------------------------------------------------------------------------------------------
nesnd_india_overseas_fir_cost = {
	value = income
	multiply = 13
	min = 10000
}
nesnd_india_overseas_ai_reserve = {
	value = nesnd_india_overseas_fir_cost
	multiply = 2
}
