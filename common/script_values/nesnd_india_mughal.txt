# Mughal's own affairs: the old agreement with its overlord, the council of princes, and famine.
# "Mughal" in these files is only the country holding nesnd_india_mughal_court (set on c:MUG by
# nesnd_india_initialize; it stays through BHT or DRA). See common/scripted_effects/nesnd_india_mughal.txt.
#
# Variables on Mughal:
#   nesnd_mug_loan_from        the government that lent the money (the overlord at the time)
#   nesnd_mug_loan_term        runs out on the day the last instalment falls due (five years)
#   nesnd_mug_loan_by_script   the repayment treaty could not be created; we pay each month by script
#   nesnd_mug_council_wait     ten years before the next council
#   nesnd_mug_council_topic    while a council sits: 1 railways, 2 famine relief, 3 common defence
#   nesnd_mug_council_yes, _no, _unsigned   the count of the last council, read by its report
# On a prince:
#   nesnd_mug_council_invite   the Mughal government that invited it to the sitting council
#   nesnd_mug_council_answer   1 attends, 2 declines
#   nesnd_mug_council_attended ten years after attending (or after Mughal's famine aid)
# On any Indian government that received a famine event: nesnd_mug_famine_wait (24 months).
# On a state struck by a harvest condition that cuts food: nesnd_mug_famine_state (120 days).
#
# Matter kinds of Indian affairs (common/script_values/nesnd_india_matters.txt): 7 Mughal asks its
# overlord for a loan, 8 Mughal asks to redeem its subject status, 9 a prince asks for famine aid.
# Default sums and chances; to be tuned after in-game tests.

nesnd_mug_loan_sum = 100000
# Paid back over five years by a money transfer treaty, each week.
nesnd_mug_loan_weekly = 460
# The treaty inputs in nesnd_mug_loan_possible and nesnd_mug_lend write 460 as a number; change all three together.
nesnd_mug_loan_total = 120000
# Paid back by script each month when the treaty could not be created (60 months).
nesnd_mug_loan_monthly = 2000
nesnd_mug_redeem_price = 250000
nesnd_mug_relief_per_state = 10000

# Mughal scope. Building levels in our states held by our overlord (its government or its
# investors). levels_owned_by_country only compares, so each building is counted level by level up
# to ten. The overlord is read from each building (state.owner is the country holding it), so the
# value means the same whichever country evaluates it.
nesnd_mug_overlord_levels = {
    value = 0
    if = { limit = { is_subject = yes }
        every_scope_state = {
            every_scope_building = {
                limit = { level > 0 exists = state.owner.overlord }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 1 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 2 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 3 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 4 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 5 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 6 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 7 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 8 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 9 } } add = 1 }
                if = { limit = { levels_owned_by_country = { target = state.owner.overlord value >= 10 } } add = 1 }
            }
        }
    }
}

# Country scope. States marked by the latest famine check, and the relief they cost.
nesnd_mug_famine_states = {
    value = 0
    every_scope_state = { limit = { has_variable = nesnd_mug_famine_state } add = 1 }
}
nesnd_mug_famine_cost = {
    value = nesnd_mug_famine_states
    multiply = nesnd_mug_relief_per_state
}

# Council counts, for the report event.
nesnd_mug_council_yes_value = { value = 0 if = { limit = { has_variable = nesnd_mug_council_yes } value = var:nesnd_mug_council_yes } }
nesnd_mug_council_no_value = { value = 0 if = { limit = { has_variable = nesnd_mug_council_no } value = var:nesnd_mug_council_no } }
nesnd_mug_council_unsigned_value = { value = 0 if = { limit = { has_variable = nesnd_mug_council_unsigned } value = var:nesnd_mug_council_unsigned } }

# ---------------------------------------------------------------------------------------------
# AI. Per cent. The asking chances are monthly; the answering chances are held between 5 and 95
# like those of common/script_values/nesnd_india_frontier.txt.
# ---------------------------------------------------------------------------------------------
# Mughal scope, each month.
nesnd_mug_loan_ask_chance = {
    value = 0.5
    if = { limit = { net_fixed_income < 0 } value = 5 }
}
nesnd_mug_redeem_ask_chance = {
    value = 0
    if = { limit = { gold_reserves >= { value = nesnd_mug_redeem_price multiply = 2 } liberty_desire >= 50 } value = 2 }
}
nesnd_mug_council_ask_chance = 2

# Answering government scope (nesnd_india_matter_from is the asking country).
nesnd_mug_loan_answer_chance = {
    value = 60
    if = { limit = { OR = { in_default = yes net_fixed_income < 0 } } add = -40 }
    if = { limit = { nesnd_india_frontier_ai_asker_relations = { VALUE = 20 } } add = 20 }
    min = 5
    max = 95
}
nesnd_mug_redeem_answer_chance = {
    value = 15
    if = { limit = { is_at_war = yes } add = 30 }
    if = { limit = { net_fixed_income < 0 } add = 20 }
    if = { limit = { nesnd_mug_ai_many_levels = yes } add = -20 }
    min = 5
    max = 95
}
nesnd_mug_aid_answer_chance = {
    value = 40
    if = { limit = { has_variable = nesnd_india_mughal_court } value = 50 }
    if = { limit = { net_fixed_income < 0 } add = -30 }
    min = 5
    max = 95
}
# Prince scope, invited to a council (event nesnd_mug.12).
nesnd_mug_council_attend_chance = {
    value = 50
    if = { limit = { nesnd_mug_council_host_relations = { VALUE = 25 } } add = 20 }
    if = { limit = { nesnd_mug_council_host_hostile = yes } add = -40 }
    if = { limit = { nesnd_mug_council_other_subject = yes } add = -20 }
    min = 5
    max = 95
}
