nesnd_ottoman_enthusiasm_value = {
    value = 0
    if = { limit = { has_variable = nesnd_ottoman_enthusiasm } value = var:nesnd_ottoman_enthusiasm }
    min = 0 max = 100
}
nesnd_ottoman_claim_cost = 25
nesnd_ottoman_enthusiasm_base = {
    value = 1
    if = { limit = { nesnd_ottoman_tanzimat_success = yes } value = 2 }
    else_if = { limit = { nesnd_ottoman_tanzimat_failed = yes } value = 0.75 }
}
nesnd_ottoman_enthusiasm_army = {
    value = 0
    if = { limit = { army_size >= 100 } add = 0.5 }
}
nesnd_ottoman_enthusiasm_government = {
    value = 0
    if = { limit = { ig:ig_armed_forces = { is_in_government = yes } government_legitimacy >= 50 } add = 0.5 }
}
nesnd_ottoman_enthusiasm_reform = {
    value = 0
    if = { limit = { nesnd_ottoman_reform_active = yes } add = -0.5 }
}
nesnd_ottoman_enthusiasm_monthly = {
    value = nesnd_ottoman_enthusiasm_base
    add = nesnd_ottoman_enthusiasm_army
    add = nesnd_ottoman_enthusiasm_government
    add = nesnd_ottoman_enthusiasm_reform
}
