# Script values for Ottoman B11 "The Holy Cities" journal (nesnd_ottoman_holy_cities).
# Country scope (evaluated country).

# Holy Cities foe selection score (Rule 5): +20 per state in STATE_HEDJAZ held directly or via subjects; +30 if rival caliph.
nesnd_ottoman_holy_foe_score = {
    value = 0
    every_scope_state = {
        limit = {
            state_region = s:STATE_HEDJAZ
        }
        add = 20
    }
    every_subject_or_below = {
        every_scope_state = {
            limit = {
                state_region = s:STATE_HEDJAZ
            }
            add = 20
        }
    }
    if = {
        limit = {
            nesnd_rival_caliph = yes
        }
        add = 30
    }
}

# Direct state count in STATE_HEDJAZ for selecting the protectorate offer (.32) recipient (Rule 12).
nesnd_ottoman_holy_held_states_count = {
    value = 0
    every_scope_state = {
        limit = {
            state_region = s:STATE_HEDJAZ
        }
        add = 1
    }
}
