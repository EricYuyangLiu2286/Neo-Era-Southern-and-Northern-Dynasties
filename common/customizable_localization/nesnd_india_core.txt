# Names of the journals nesnd_india_industry and nesnd_india_unified, read on the journal's country.
nesnd_india_industry_name = {
    type = country
    text = {
        trigger = { nesnd_india_prince = yes }
        localization_key = nesnd_india_industry_name_prince
    }
    text = { localization_key = nesnd_india_industry_name_empire }
}
nesnd_india_unified_name = {
    type = country
    text = {
        trigger = {
            OR = {
                country_has_state_religion = rel:hindu
                country_has_state_religion = rel:mahayana
                country_has_state_religion = rel:theravada
                country_has_state_religion = rel:gelugpa
            }
        }
        localization_key = nesnd_india_unified_name_dharmic
    }
    text = {
        trigger = {
            religion = { has_discrimination_trait = heritage_islamic }
        }
        localization_key = nesnd_india_unified_name_islamic
    }
    text = { localization_key = nesnd_india_unified_name_other }
}
