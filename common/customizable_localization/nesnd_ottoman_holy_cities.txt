# Dynamic opponent name for the Ottoman "The Holy Cities" journal (Rule 5).
# Resolves to the current global_var:nesnd_holy_foe country name when alive, or a generic fallback key.

nesnd_ottoman_holy_foe_name = {
    type = country
    text = {
        trigger = {
            has_global_variable = nesnd_holy_foe
            global_var:nesnd_holy_foe ?= {
                is_country_alive = yes
            }
        }
        localization_key = nesnd_ottoman_holy_foe_name_active
    }
    text = {
        localization_key = nesnd_ottoman_holy_foe_name_fallback
    }
}
