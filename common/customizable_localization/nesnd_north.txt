# Dynamic titles for the northern war journal entries (Rule 24, R11 class c).
# Uses historical names when Russia is the selected northern foe, and generalized titles otherwise.

nesnd_crimea_title = {
    type = country
    text = {
        trigger = {
            has_global_variable = nesnd_north_foe
            global_var:nesnd_north_foe ?= { c:RUS ?= this }
        }
        localization_key = nesnd_crimea_title_rus
    }
    text = {
        localization_key = nesnd_crimea_title_north
    }
}

nesnd_rtw_title = {
    type = country
    text = {
        trigger = {
            has_global_variable = nesnd_north_foe
            global_var:nesnd_north_foe ?= { c:RUS ?= this }
        }
        localization_key = nesnd_rtw_title_rus
    }
    text = {
        localization_key = nesnd_rtw_title_north
    }
}
