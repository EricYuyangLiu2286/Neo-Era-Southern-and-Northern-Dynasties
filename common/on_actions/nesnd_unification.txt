# Monthly global cleanup: countries that kept unification modifiers after leaving the trend or becoming subjects.
on_monthly_pulse = {
	on_actions = { nesnd_uni_global_cleanup }
}

nesnd_uni_global_cleanup = {
	effect = {
		every_country = {
			limit = {
				OR = {
					has_modifier = nesnd_uni_order
					has_modifier = nesnd_uni_tripod
					has_modifier = nesnd_uni_standoff_power
				}
				OR = {
					NOT = { is_involved_in_journal_entry = nesnd_uni_trend }
					is_subject = yes
				}
			}
			nesnd_uni_clear_country = yes
		}
	}
}
