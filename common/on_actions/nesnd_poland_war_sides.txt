# Root = diplomatic play. These callbacks refresh allies joining after the initial snapshot.
on_diplo_play_join_side = {
	on_actions = {
		nesnd_poland_resample_war_sides
	}
}

on_diplo_play_switch_sides = {
	on_actions = {
		nesnd_poland_resample_war_sides
	}
}

nesnd_poland_resample_war_sides = {
	trigger = {
		trigger_if = {
			limit = {
				has_global_variable = nesnd_poland_window_play
				exists = global_var:nesnd_poland_window_play
			}
			this = global_var:nesnd_poland_window_play
		}
		trigger_else = {
			always = no
		}
	}
	effect = {
		nesnd_poland_snapshot_war_sides = yes
	}
}

# Some forced joins need not produce a diplomatic-play invitation callback.
on_monthly_pulse = {
	on_actions = {
		nesnd_poland_monthly_war_sides
	}
}

nesnd_poland_monthly_war_sides = {
	trigger = {
		exists = global_var:nesnd_poland_window_play
	}
	effect = {
		nesnd_poland_snapshot_war_sides = yes
	}
}
