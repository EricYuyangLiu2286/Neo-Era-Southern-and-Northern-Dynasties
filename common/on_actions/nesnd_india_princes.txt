# ==== The Indian uprising (formerly nesnd_india_uprising.txt) ====

on_monthly_pulse = {
    on_actions = { nesnd_india_global_cases }
}

nesnd_india_global_cases = {
    effect = {
        s:STATE_CIRCARS = {
            nesnd_india_case_maintain = { R = circars STATE = STATE_CIRCARS TAG = IR1 }
        }
        s:STATE_KURNOOL = {
            nesnd_india_case_maintain = { R = kurnool STATE = STATE_KURNOOL TAG = IR2 }
        }
        s:STATE_MADRAS = {
            nesnd_india_case_maintain = { R = madras STATE = STATE_MADRAS TAG = IR3 }
        }
        s:STATE_MYSORE = {
            nesnd_india_case_maintain = { R = mysore STATE = STATE_MYSORE TAG = IR4 }
        }
        s:STATE_TRAVANCORE = {
            nesnd_india_case_maintain = { R = travancore STATE = STATE_TRAVANCORE TAG = IR5 }
        }
        s:STATE_GUJARAT = {
            nesnd_india_case_maintain = { R = gujarat STATE = STATE_GUJARAT TAG = IR6 }
        }
    }
}

on_state_owner_change = {
    on_actions = { nesnd_india_revolt_territory_changed }
}

on_state_created = {
    on_actions = { nesnd_india_revolt_territory_changed }
}

nesnd_india_revolt_territory_changed = {
    effect = {
        owner ?= {
            if = {
                limit = { has_variable = nesnd_india_revolt_actor }
                set_variable = nesnd_india_revolt_territory_changed
            }
        }
    }
}
