on_weekly_pulse_country = { on_actions = { nesnd_usm_western_weekly } }
nesnd_usm_western_weekly = { effect = { nesnd_usm_western_maintenance = yes } }
# Native victory callbacks: root is the victorious country. They also cover engine-enforced annexation.
on_wargoal_enforced = { on_actions = { nesnd_usm_western_after_victory } }
on_enemy_capitulated_notification = { on_actions = { nesnd_usm_western_after_victory } }
nesnd_usm_western_after_victory = {
    effect = {
        if = { limit = { nesnd_usm_western_country = yes NOT = { has_global_variable = nesnd_south_settled }
                trigger_if = { limit = { exists = scope:target } scope:target = { has_variable = usm_opposition_country has_variable = nesnd_holdout_enrolled_round is_country_alive = no } }
                trigger_else = { always = no }
            }
            nesnd_usm_integrate_recovered_mainland = yes
        }
    }
}
