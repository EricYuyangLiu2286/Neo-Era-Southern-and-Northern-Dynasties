# On-actions for the Crimean / Northern War yearly foe check and the shared northern war outcome signals (Rule 18).
on_yearly_pulse_country = { on_actions = { nesnd_crimea_rus_yearly } }
on_wargoal_enforced = { on_actions = { nesnd_north_on_wargoal } }

nesnd_crimea_rus_yearly = {
    trigger = { nesnd_crimea_rus_yearly_eligible = yes }
    effect = { nesnd_crimea_rus_yearly_check = yes }
}

# Shared northern war outcome hook (covering both Crimean War and Russo-Turkish War via nesnd_north_handle_wargoal_enforced).
nesnd_north_on_wargoal = {
    trigger = {
        OR = {
            has_global_variable = nesnd_crimea_active
            has_global_variable = nesnd_rtw_active
        }
    }
    effect = { nesnd_north_handle_wargoal_enforced = yes }
}
