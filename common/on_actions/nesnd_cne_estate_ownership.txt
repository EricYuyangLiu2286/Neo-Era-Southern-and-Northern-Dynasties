# GAME 00_code_on_actions: ROOT is a state for both callbacks.
on_state_owner_change = { on_actions = { nesnd_cne_estate_owner_changed } }
on_state_created = { on_actions = { nesnd_cne_estate_owner_changed } }
nesnd_cne_estate_owner_changed = {
 effect = {
  if = { limit = { exists = c:CNE }
   if = { limit = { c:CNE = { nesnd_cne_programs_live = yes } nesnd_cne_initial_reform_state = yes }
    if = { limit = { NOT = { owner = c:CNE } has_variable = nesnd_cne_camp_state_active } nesnd_cne_remove_state_estate_benefits = yes }
    c:CNE = { nesnd_cne_queue_estate_ownership_check = yes }
   }
  }
 }
}
