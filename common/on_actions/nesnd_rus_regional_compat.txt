# Only a case that actually ran and retained the original Kenesary can resume.
on_yearly_pulse_country = { on_actions = { nesnd_rus_review_steppe } }
nesnd_rus_review_steppe = {
    trigger = { c:RUS ?= this has_variable = nesnd_rus_steppe_deferred }
    effect = {
        if = { limit = { NOT = { any_character_in_void = { is_character_alive = yes has_variable = kenesary_variable } } }
            remove_variable = nesnd_rus_steppe_deferred
        }
        else_if = { limit = { nesnd_rus_steppe_participants = yes nesnd_rus_kazakh_seed = yes }
            remove_variable = nesnd_rus_steppe_deferred
            trigger_event = { id = kazakhstan_events.1 }
            random_character_in_void = { limit = { has_variable = kenesary_variable } kill_character = { hidden = yes } }
        }
    }
}
