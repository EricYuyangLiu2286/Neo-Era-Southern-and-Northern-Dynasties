on_monthly_pulse_country = { on_actions = { nesnd_crisis_monthly } }
nesnd_crisis_monthly = {
    effect = {
        if = { limit = { c:USM ?= this has_variable = nesnd_chen_protected NOT = { has_variable = nesnd_assassination_resolved }
            var:usm_conflicts_within_tracker > 24 any_scope_character = { has_template = USM_chen_dongyang is_character_alive = yes }
        }
            set_variable = nesnd_assassination_resolved
            trigger_event = { id = nesnd_crisis.11 popup = yes }
        }
        if = { limit = { has_variable = nesnd_north_intervention NOT = { has_global_variable = nesnd_south_settled } } nesnd_join_opposition_play = yes }
        if = { limit = { has_variable = nesnd_north_intervention has_global_variable = nesnd_south_settled } remove_variable = nesnd_north_intervention }
    }
}
