on_secession_start = { on_actions = { nesnd_qe1_fr_secession_start } }

	trigger = {
		c:QE1 ?= this
		has_journal_entry = nesnd_qe1_frontier_journal
		OR = { c:AFG ?= scope:target c:UZB ?= scope:target }
	}
	effect = { nesnd_qe1_fr_claim_risings = yes }
}
