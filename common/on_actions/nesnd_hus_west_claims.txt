# The native owner-change hook has the state as its root. Only visit provinces of a registered claim.
on_state_owner_change = { on_actions = { nesnd_hus_west_claim_changed_hands } }
nesnd_hus_west_claim_changed_hands = {
    trigger = { state_region = { has_variable_list = nesnd_hus_west_land } }
    effect = { state_region = { every_in_list = { variable = nesnd_hus_west_land nesnd_hus_west_observe_owner = yes } } }
}

on_war_end = { on_actions = { nesnd_hus_west_close_awards } }
nesnd_hus_west_close_awards = {
    trigger = { has_variable = nesnd_hus_west_has_award }
    effect = { set_variable = nesnd_hus_west_peace_signed }
}
