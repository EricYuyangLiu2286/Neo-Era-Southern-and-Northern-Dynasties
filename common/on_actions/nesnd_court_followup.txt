on_monthly_pulse_country = {
	on_actions = { nesnd_court_followup_monthly }
}
nesnd_court_followup_monthly = {
	effect = {
		if = {
			limit = {
				c:USM ?= this
			}
			nesnd_ming_mark_retained_royals = yes
			nesnd_ming_notice_retained_royals = yes
			nesnd_south_deadlock_month = yes
			if = {
				limit = {
					has_variable = nesnd_ming_royals_retained
					NOT = {
						any_scope_character = {
							has_role = character_role_nesnd_ming_refuge
							is_character_alive = yes
						}
					}
				}
				remove_variable = nesnd_ming_royals_retained
				if = {
					limit = {
						has_modifier = nesnd_ming_royal_allowance
					}
					remove_modifier = nesnd_ming_royal_allowance
				}
			}
		}
	}
}



# The throne's successor state and a court in exile both carry the Ming name after the war: checked monthly, decided once.
on_monthly_pulse_country = {
	on_actions = { nesnd_ming_court_reunion_check }
}
nesnd_ming_court_reunion_check = {
	trigger = { c:USM ?= this }
	effect = {
		if = {
			limit = { nesnd_ming_court_reunion_due = yes }
			trigger_event = { id = nesnd_court_followup.20 popup = yes }
		}
	}
}
