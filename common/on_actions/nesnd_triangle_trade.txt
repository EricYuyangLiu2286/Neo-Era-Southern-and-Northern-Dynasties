# 法律生效时立即撤销两个航线上的买卖身份；其余参与资格由各日志月结刷新。
on_law_activated = { on_actions = { nesnd_triangle_abolition } }

nesnd_triangle_abolition = {
	trigger = {
		owner ?= {
			has_law_or_variant = law_type:law_slavery_banned
			OR = {
				has_variable = nesnd_north_trade_buying
				has_variable = nesnd_north_trade_selling
				has_variable = nesnd_west_trade_buying
				has_variable = nesnd_west_trade_selling
			}
		}
	}
	effect = {
		owner = {
			nesnd_triangle_clear_mode = { KEY = north }
			nesnd_triangle_clear_mode = { KEY = west }
		}
	}
}
