on_monthly_pulse_country = {
    on_actions = { nesnd_small_country_month }
}
nesnd_small_country_month = {
    trigger = {
        OR = {
            nesnd_small_state_country = yes
            has_variable = nesnd_small_lineage
        }
    }
    effect = {
        nesnd_small_market_verify = yes
        nesnd_small_monthly = yes
    }
}
on_revolution_start = {
    on_actions = { nesnd_small_revolution_start }
}
# The revolutionary side inherits only the identity of the country, never a completed or rewarded record.
nesnd_small_revolution_start = {
    trigger = {
        has_variable = nesnd_small_lineage
        exists = scope:target
    }
    effect = {
        if = {
            limit = { exists = scope:target }
            save_scope_as = small_original_country
            nesnd_small_close_episode = yes
            if = {
                limit = {
                    OR = {
                        nesnd_canon_request_kind = 5
                        nesnd_canon_request_kind = 6
                        nesnd_canon_request_kind = 8
                        nesnd_canon_request_kind = 9
                    }
                }
                nesnd_canon_clear_request = yes
            }
            scope:target = {
                set_variable = nesnd_small_lineage
                set_variable = nesnd_small_provisional
                if = {
                    limit = {
                        scope:small_original_country = { has_variable = nesnd_small_indian_line }
                    }
                    set_variable = nesnd_small_indian_line
                }
                if = {
                    limit = {
                        scope:small_original_country = { has_variable = nesnd_small_southern_line }
                    }
                    set_variable = nesnd_small_southern_line
                }
                if = {
                    limit = {
                        scope:small_original_country = { has_variable = nesnd_small_far_south_line }
                    }
                    set_variable = nesnd_small_far_south_line
                }
                if = {
                    limit = {
                        scope:small_original_country = { has_variable = nesnd_small_burma_line }
                    }
                    set_variable = nesnd_small_burma_line
                }
                if = {
                    limit = {
                        scope:small_original_country = { has_variable = nesnd_small_siam_line }
                    }
                    set_variable = nesnd_small_siam_line
                }
            }
            clear_saved_scope = small_original_country
        }
    }
}
on_civil_war_won = {
    on_actions = { nesnd_small_civil_war_won }
}
nesnd_small_civil_war_won = {
    trigger = { has_variable = nesnd_small_lineage }
    effect = {
        nesnd_small_close_episode = yes
        if = {
            limit = { has_variable = nesnd_small_provisional }
            remove_variable = nesnd_small_provisional
        }
    }
}
