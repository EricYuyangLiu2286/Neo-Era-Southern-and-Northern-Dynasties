nesnd_small_country_month = { trigger = { OR = { nesnd_small_state_country = yes has_variable = nesnd_small_lineage } } effect = { nesnd_small_market_verify = yes nesnd_small_monthly = yes nesnd_small_sync_bars = yes } }
on_revolution_start = { on_actions = { nesnd_small_revolution_start } }
nesnd_small_revolution_start = {
    trigger = { has_variable = nesnd_small_lineage exists = scope:target }
    effect = {
        if = { limit = { exists = scope:target }
        save_scope_as = small_original_country
        nesnd_small_close_episode = yes
        if = { limit = { OR = { nesnd_canon_request_kind = 5 nesnd_canon_request_kind = 6 nesnd_canon_request_kind = 7 nesnd_canon_request_kind = 8 nesnd_canon_request_kind = 9 } } nesnd_canon_clear_request = yes }
        scope:target = {
            set_variable = nesnd_small_lineage set_variable = nesnd_small_provisional
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_indian_line } } set_variable = nesnd_small_indian_line }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_southern_line } } set_variable = nesnd_small_southern_line }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_far_south_line } } set_variable = nesnd_small_far_south_line }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_burma_line } } set_variable = nesnd_small_burma_line }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_siam_line } } set_variable = nesnd_small_siam_line }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_economy_done } } set_variable = nesnd_small_economy_done }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_economy_rewarded } } set_variable = nesnd_small_economy_rewarded }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_governance_done } } set_variable = nesnd_small_governance_done }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_governance_rewarded } } set_variable = nesnd_small_governance_rewarded }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_order_done } } set_variable = nesnd_small_order_done }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_order_rewarded } } set_variable = nesnd_small_order_rewarded }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_capital_awarded } } set_variable = nesnd_small_capital_awarded }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_reform_law_1 } } set_variable = nesnd_small_reform_law_1 }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_reform_law_2 } } set_variable = nesnd_small_reform_law_2 }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_reform_law_3 } } set_variable = nesnd_small_reform_law_3 }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_reform_law_4 } } set_variable = nesnd_small_reform_law_4 }
            if = { limit = { scope:small_original_country = { has_variable = nesnd_small_reform_law_5 } } set_variable = nesnd_small_reform_law_5 }
            set_variable = { name = nesnd_small_economy_months value = scope:small_original_country.nesnd_small_economy_months }
            set_variable = { name = nesnd_small_governance_months value = scope:small_original_country.nesnd_small_governance_months }
        }
        clear_saved_scope = small_original_country
        }
    }
}
on_civil_war_won = { on_actions = { nesnd_small_civil_war_won } }
nesnd_small_civil_war_won = {
    trigger = { has_variable = nesnd_small_lineage }
    effect = {
        set_variable = nesnd_small_domestic_settlement
        nesnd_small_close_episode = yes
        nesnd_small_refresh_candidates = yes
        if = { limit = { has_variable = nesnd_small_provisional }
            remove_variable = nesnd_small_provisional
            if = { limit = { nesnd_small_candidate_valid = yes } set_variable = nesnd_small_regional nesnd_small_register_targets = yes }
            else = { remove_variable = nesnd_small_regional }
        }
    }
}
