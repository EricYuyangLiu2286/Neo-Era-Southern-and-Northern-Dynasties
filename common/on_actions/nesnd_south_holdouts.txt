# on_weekly_pulse_country is not fired by the engine (docs/on_actions.log: "From Code: No"), so this never ran
# until 2026-11-17. Monthly is the shortest country pulse the engine fires.
on_monthly_pulse_country = { on_actions = { nesnd_south_holdout_weekly } }
nesnd_south_holdout_weekly = { effect = {
    if = { limit = { exists = var:nesnd_jingnan_call } nesnd_fulfil_jingnan_call = yes }
    nesnd_south_holdout_country_tick = yes
    nesnd_south_holdout_wu_tick = yes
    }
}
