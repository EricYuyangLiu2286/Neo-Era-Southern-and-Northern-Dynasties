# ==== Identity, monthly refresh and the reform journals (formerly nesnd_india_core.txt) ====
on_monthly_pulse_country = { on_actions = { nesnd_india_maintenance } }
on_country_formed = { on_actions = { nesnd_india_maintenance } }
on_game_started_after_lobby = { on_actions = { nesnd_india_startup } }
nesnd_india_maintenance = { trigger = { OR = { has_variable = nesnd_india_unified_done nesnd_small_state_country = yes nesnd_india_country = yes nesnd_india_himalaya_country = yes has_variable = nesnd_india_order_lord has_variable = nesnd_mug_loan_from c:USM ?= this c:YU1 ?= this c:TZ1 ?= this has_variable = nesnd_india_request_partner has_variable = nesnd_india_adviser_partner has_modifier = nesnd_india_french_advisers } } effect = { nesnd_india_refresh = yes } }
nesnd_india_startup = { effect = { every_country = { limit = { OR = { nesnd_small_state_country = yes nesnd_india_country = yes nesnd_india_himalaya_country = yes has_variable = nesnd_india_order_lord has_variable = nesnd_mug_loan_from c:USM ?= this c:YU1 ?= this c:TZ1 ?= this has_variable = nesnd_india_request_partner has_variable = nesnd_india_adviser_partner has_modifier = nesnd_india_french_advisers } } nesnd_india_refresh = yes } } }

# ==== Independence and the direct overlord ====
on_monthly_pulse_country = { on_actions = { nesnd_india_indep_admission } }
on_game_started_after_lobby = { on_actions = { nesnd_india_indep_start } }
nesnd_india_indep_start = {
    effect = { every_country = { limit = { nesnd_india_indep_holder = yes } nesnd_india_indep_admit = yes } }
}
nesnd_india_indep_admission = {
    trigger = { nesnd_india_indep_holder = yes }
    effect = { nesnd_india_indep_admit = yes }
}
on_diplo_play_war_start = { on_actions = { nesnd_india_indep_open_armouries } }
nesnd_india_indep_open_armouries = {
    effect = {
        every_country = {
            limit = { has_variable = nesnd_india_indep_stock is_subject = yes exists = overlord has_war_with = overlord }
            nesnd_india_indep_muster = yes
        }
    }
}
