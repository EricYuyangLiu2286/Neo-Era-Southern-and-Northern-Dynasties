on_monthly_pulse_country = { on_actions = { nesnd_prussian_monthly } }
nesnd_prussian_monthly = { trigger = { OR = { nesnd_prussian_identity = yes has_modifier = nesnd_prussian_joint_staff } } effect = { nesnd_prussia_maintenance = yes } }
on_weekly_pulse_country = { on_actions = { nesnd_europe_america_weekly } }
nesnd_europe_america_weekly = {
    trigger = { OR = { c:AUS ?= this c:USA ?= this } }
    effect = {
        if = { limit = { c:AUS ?= this } nesnd_austria_maintenance = yes }
        if = { limit = { c:USA ?= this } nesnd_usa_maintenance = yes }
    }
}

nesnd_europe_new_play = { effect = { nesnd_europe_play_started = yes } }
on_wargoal_enforced = { on_actions = { nesnd_europe_goal_result } }
