# Scripted activations can bypass the law group's enactment check.
on_law_activated = { on_actions = { nesnd_cne_check_estate_law } }

nesnd_cne_check_estate_law = {
    trigger = {
        exists = owner
        owner = { c:CNE ?= this has_modifier = nesnd_cne_estate_settlement }
        NOR = { type = law_type:law_tenant_farmers type = law_type:law_professional_army }
        type = {
            OR = {
                is_same_law_group_as = law_type:law_tenant_farmers
                is_same_law_group_as = law_type:law_professional_army
            }
        }
    }
    effect = { owner = { nesnd_cne_enforce_estate_laws = yes } }
}
