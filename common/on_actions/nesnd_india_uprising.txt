on_monthly_pulse = { on_actions = { nesnd_india_global_cases } }
nesnd_india_global_cases = { effect = {
        s:STATE_CIRCARS = { nesnd_india_circars_maintain = yes }
        s:STATE_KURNOOL = { nesnd_india_kurnool_maintain = yes }
        s:STATE_MADRAS = { nesnd_india_madras_maintain = yes }
        s:STATE_MYSORE = { nesnd_india_mysore_maintain = yes }
        s:STATE_TRAVANCORE = { nesnd_india_travancore_maintain = yes }
        s:STATE_GUJARAT = { nesnd_india_gujarat_maintain = yes }
    }
}

on_state_owner_change = { on_actions = { nesnd_india_revolt_territory_changed } }
on_state_created = { on_actions = { nesnd_india_revolt_territory_changed } }
nesnd_india_revolt_territory_changed = { effect = {
        owner ?= { if = { limit = { has_variable = nesnd_india_revolt_actor } set_variable = nesnd_india_revolt_territory_changed } }
    }
}
