on_monthly_pulse_state = { on_actions = { nesnd_ottoman_buildings_monthly } }
on_state_owner_change = { on_actions = { nesnd_ottoman_buildings_changed } }
nesnd_ottoman_buildings_monthly = {
    trigger = { any_scope_building = { OR = { has_modifier = nesnd_ottoman_qin_supply has_modifier = nesnd_ottoman_qin_supply_success has_modifier = nesnd_ottoman_qin_supply_failed } } }
    effect = { nesnd_ottoman_refresh_barracks = yes }
}
nesnd_ottoman_buildings_changed = { effect = { nesnd_ottoman_refresh_barracks = yes } }
# Was on_weekly_pulse_country, which the engine never fires (2026-11-17).
# nesnd_ottoman_route_fallback: the milestone normally fires from the journal's monthly pulse; this covers a journal that concluded first.
# nesnd_ottoman_charter_fallback: opens "Charter and Barracks" for a save that passed event nesnd_ottoman_recovery.5 before B3.
# nesnd_ottoman_islam_leader_fallback: opens "Leader of the Islamic World" for a save that passed event nesnd_ottoman_recovery.5 before B4.
# nesnd_ottoman_academy_fallback: opens "Academies and the Press" once the constitutional route is set (events .5 adds it; this is the safety net).
# nesnd_ottoman_rumelia_fallback: opens "Rumelia" on the sultanic failure route once the vanilla Great Eastern Crisis has started, running or over.
# nesnd_ottoman_gec_monthly: records the Balkan holdings at the start of that crisis and how it ended.
# nesnd_ottoman_eastern_provinces_fallback: opens "Eastern Provinces" on the sultanic failure route when eastern minority movements arise.
# nesnd_north_foe_refresh: selects or refreshes global_var:nesnd_north_foe when no northern war is active (F2 Rule 4).
# nesnd_ottoman_straits_check: fires "The Price of the Straits" (.20) if TUR is allied with its would-be northern foe (F2 Rule 6).
# nesnd_north_war_roster_monthly: maintains wartime belligerent side rosters during an active northern war (F2 Rule 17).
# nesnd_rtw_fallback: opens the generalized RTW / Northern Threat journals on the sultanic failure route once Rumelia closes or eastern provinces are lost.
# nesnd_ottoman_revenge_fallback: fires "Where the Sword Turns" (.19) if Rumelia was lost and RTW ended in held/standoff.
# nesnd_ottoman_autocracy_fallback: opens "Mastery of the Autocracy" on the sultanic failure route (event .6 adds it; this covers older saves), and clears any lingering nesnd_ottoman_capital_fallen modifier if the journal is no longer active.
on_monthly_pulse_country = { on_actions = { nesnd_ottoman_reform_weekly nesnd_ottoman_route_fallback nesnd_ottoman_charter_fallback nesnd_ottoman_islam_leader_fallback nesnd_ottoman_academy_fallback nesnd_ottoman_rumelia_fallback nesnd_ottoman_eastern_provinces_fallback nesnd_north_foe_refresh nesnd_ottoman_straits_check nesnd_north_war_roster_monthly nesnd_rtw_fallback nesnd_ottoman_revenge_fallback nesnd_ottoman_gec_monthly nesnd_ottoman_autocracy_fallback } }
nesnd_ottoman_reform_weekly = {
    trigger = { OR = { has_variable = nesnd_ottoman_reform_pending nesnd_ottoman_reform_active = yes } }
    effect = { nesnd_ottoman_reform_maintenance = yes }
}
nesnd_ottoman_route_fallback = {
    trigger = { nesnd_ottoman_recovery_open = yes NOT = { has_journal_entry = nesnd_ottoman_recovery } NOT = { has_variable = nesnd_ottoman_route } }
    effect = { nesnd_ottoman_route_check = yes }
}
nesnd_ottoman_charter_fallback = {
    trigger = { nesnd_ottoman_charter_fallback_due = yes }
    effect = { add_journal_entry = { type = nesnd_ottoman_charter } }
}
nesnd_ottoman_islam_leader_fallback = {
    trigger = { nesnd_ottoman_islam_leader_fallback_due = yes }
    effect = { add_journal_entry = { type = nesnd_ottoman_islam_leader } }
}
nesnd_ottoman_academy_fallback = {
    trigger = { nesnd_ottoman_academy_fallback_due = yes }
    effect = { add_journal_entry = { type = nesnd_ottoman_academy_1 } }
}
nesnd_ottoman_rumelia_fallback = {
    trigger = { nesnd_ottoman_rumelia_fallback_due = yes }
    effect = { add_journal_entry = { type = nesnd_ottoman_rumelia } }
}
nesnd_ottoman_eastern_provinces_fallback = {
    trigger = { nesnd_ottoman_eastern_provinces_fallback_due = yes }
    effect = { add_journal_entry = { type = nesnd_ottoman_eastern_provinces } }
}
nesnd_north_foe_refresh = {
    trigger = { nesnd_north_foe_refresh_due = yes }
    effect = { nesnd_north_refresh_foe = yes }
}
nesnd_ottoman_straits_check = {
    trigger = { nesnd_ottoman_straits_due = yes }
    effect = { trigger_event = { id = nesnd_ottoman_recovery.20 popup = yes } }
}
nesnd_north_war_roster_monthly = {
    trigger = { nesnd_north_war_roster_due = yes }
    effect = { nesnd_north_war_roster_update = yes }
}
nesnd_rtw_fallback = {
    trigger = { nesnd_rtw_start_due = yes }
    effect = { nesnd_rtw_start = yes }
}
nesnd_ottoman_revenge_fallback = {
    trigger = { nesnd_ottoman_revenge_fallback_due = yes }
    effect = {
        set_variable = nesnd_ottoman_revenge_asked
        trigger_event = { id = nesnd_ottoman_recovery.19 popup = yes }
    }
}
nesnd_ottoman_gec_monthly = {
    trigger = { nesnd_ottoman_gec_record_due = yes }
    effect = { nesnd_ottoman_gec_record = yes }
}
nesnd_ottoman_autocracy_fallback = {
    trigger = {
        OR = {
            nesnd_ottoman_autocracy_fallback_due = yes
            AND = { has_modifier = nesnd_ottoman_capital_fallen NOT = { has_journal_entry = nesnd_ottoman_autocracy } }
        }
    }
    effect = {
        if = { limit = { nesnd_ottoman_autocracy_fallback_due = yes } add_journal_entry = { type = nesnd_ottoman_autocracy } }
        else = { nesnd_ottoman_autocracy_clear_capital_fallen = yes }
    }
}
