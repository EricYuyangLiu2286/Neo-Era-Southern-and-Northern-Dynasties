on_monthly_pulse_state = { on_actions = { nesnd_ottoman_buildings_monthly } }
on_state_owner_change = { on_actions = { nesnd_ottoman_buildings_changed } }
nesnd_ottoman_buildings_monthly = {
    trigger = { any_scope_building = { OR = { has_modifier = nesnd_ottoman_qin_supply has_modifier = nesnd_ottoman_qin_supply_success has_modifier = nesnd_ottoman_qin_supply_failed } } }
    effect = { nesnd_ottoman_refresh_barracks = yes }
}
nesnd_ottoman_buildings_changed = { effect = { nesnd_ottoman_refresh_barracks = yes } }
on_weekly_pulse_country = { on_actions = { nesnd_ottoman_reform_weekly } }
nesnd_ottoman_reform_weekly = {
    trigger = { OR = { has_variable = nesnd_ottoman_reform_pending nesnd_ottoman_reform_active = yes } }
    effect = { nesnd_ottoman_reform_maintenance = yes }
}
