on_monthly_pulse_country = { on_actions = { nesnd_c2_monthly_country nesnd_usm_mandate_lost_eligibility } }
nesnd_usm_mandate_lost_eligibility = {
    trigger = {
        has_variable = nesnd_usm_mandate_victory
        OR = { has_modifier = nesnd_usm_mandate_central_2 has_modifier = nesnd_usm_mandate_provincial_2 }
        NOT = { nesnd_usm_civic_country = yes }
    }
    effect = { nesnd_usm_mandate_clear_modifiers = yes }
}
nesnd_c2_monthly_country = { trigger = { OR = { nesnd_c2_country_live = yes has_variable = nesnd_c2_route_seen } } effect = { if = { limit = { nesnd_c2_country_live = yes } set_variable = nesnd_c2_route_seen nesnd_c2_monthly = yes } else = { nesnd_c2_cleanup_country = yes } } }
on_game_started_after_lobby = { on_actions = { nesnd_c2_startup } }
nesnd_c2_startup = { effect = { every_country = { limit = { nesnd_c2_country_live = yes } set_variable = nesnd_c2_route_seen nesnd_c2_refresh_usm_outcome = yes nesnd_c2_prepare_targets = yes } } }
on_country_formed = { on_actions = { nesnd_c2_formed } }
nesnd_c2_formed = { trigger = { nesnd_c2_country_live = yes } effect = { nesnd_c2_refresh_usm_outcome = yes nesnd_c2_prepare_targets = yes } }
