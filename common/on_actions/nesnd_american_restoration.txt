on_character_death = { on_actions = { nesnd_a2_character_death } }
nesnd_a2_character_death = {
 trigger = { has_variable = nesnd_a2_case_character }
 effect = {
  if = { limit = { has_variable = nesnd_a2_itzcoatl } set_global_variable = nesnd_a2_itzcoatl_dead }
  if = { limit = { has_variable = nesnd_a2_amaru } set_global_variable = nesnd_a2_amaru_dead }
  if = { limit = { has_variable = nesnd_a2_rumi } set_global_variable = nesnd_a2_rumi_dead }
  if = { limit = { has_variable = nesnd_a2_tatanka } set_global_variable = nesnd_a2_tatanka_dead }
 }
}

 }
