on_yearly_pulse_country = {
	on_actions = {
		ce_yearly_pulse_country
	}
}

ce_yearly_pulse_country = {
	effect = {
		if = {
			limit = {
				NOT = { 
					has_journal_entry = je_ce_polish_reconquest 
				}
				NOT = {
					any_country = {
						has_journal_entry = je_ce_polish_reconquest 
					}
				}
				has_global_variable = ce_polish_claim_state_count
			}
			remove_global_variable = ce_polish_claim_state_count
		}
	}
}

on_new_ruler = {
	on_actions = {
		ce_new_ruler
	}
}
ce_new_ruler = {
	effect = {
		if = {
			limit = {
				is_ruler_of_own_country = yes
				exists = c:CNE
				exists = owner
				owner ?= {
					c:CNE ?= This
					has_law_or_variant = law_type:law_monarchy
					has_variable = the_dynasty_of_mu_variable
					
				}
				NOT = { has_global_variable = ce_cne_events_12_happened_flag }
			}
			if = {
				limit = {
					has_template = CNE_mu_chimin
				}
				owner = {
					trigger_event = { id = flavorCNE.12 popup = yes }
				}
			}
			else = {
                owner = { trigger_event = { id = flavorCNE.13 popup = yes } }
            }
		}
	}

}


on_diplomatic_play_started = {
	on_actions = {
		on_ce_poland_war_change_side
	}
}

on_ce_poland_war_change_side = {
	effect = {
		if = {
			limit = { 
				exists = c:KRA
				scope:initiator = { c:KRA ?= This }
				scope:target = {
					has_variable = ce_pro_polish_variable
				}
			}
			scope:target = {
				remove_variable = ce_pro_polish_variable
				set_variable = ce_anti_polish_variable
				ce_update_poland_supporter_and_haters_effect = yes
				
# Journal entry activates from its possible trigger.

			}
		}

		if = {
			limit = { 
				exists = c:KRA
				scope:target = { c:KRA ?= This }
				scope:initiator = {
					has_variable = ce_pro_polish_variable
				}
			}
			scope:initiator = {
				remove_variable = ce_pro_polish_variable
				set_variable = ce_anti_polish_variable
				ce_update_poland_supporter_and_haters_effect = yes
				
# Journal entry activates from its possible trigger.

			}
		}
	}
}

	}

			name = ryukyu_rivalry_training_incident
			years = 5
		}
		random_list = {
			1 = {
				set_global_variable = {
					name = ryukyu_rivalry_fire
				}
			}
			1 = {
				set_global_variable = {
					name = ryukyu_rivalry_accident
				}
			}
			1 = {
				set_global_variable = {
					name = ryukyu_rivalry_brawl
				}
			}
		}
		hidden_effect = { c:JAP ?= { trigger_event = { id = ryukyu_rivalry.6 days = 1 popup = yes } } } # Consider notification
		hidden_effect = { c:CHI ?= { trigger_event = { id = ryukyu_rivalry.6 days = 1 popup = yes } } } # Consider notification
