on_character_death = { on_actions = { nesnd_xq_on_death } }

# 根作用域：去世的人物。向每个玩家国家发「追认圣徒」，日志在下次更新时隐藏失效。
nesnd_xq_on_death = {
	trigger = {
		has_template = nesnd_xiuquan_template
		NOT = { has_global_variable = nesnd_xq_dead }
	}
	effect = { nesnd_xq_commemorate = yes }
}

# 只处理旅途所在州的易主。若吞并已将人物交给新东道国，只登记新所属国。
on_state_owner_change = { on_actions = { nesnd_xq_host_changed } }
nesnd_xq_host_changed = {
	trigger = {
		exists = global_var:nesnd_xq_state
		this = global_var:nesnd_xq_state
		exists = owner
		exists = global_var:nesnd_xq_host_country
		NOT = { owner = global_var:nesnd_xq_host_country }
		exists = global_var:nesnd_xq_char
		global_var:nesnd_xq_char = {
			nesnd_xq_active_missionary = yes
			OR = {
				owner = global_var:nesnd_xq_host_country
				owner = global_var:nesnd_xq_state.owner
			}
		}
	}
	effect = { owner = { nesnd_xq_enter_host = yes } }
}
