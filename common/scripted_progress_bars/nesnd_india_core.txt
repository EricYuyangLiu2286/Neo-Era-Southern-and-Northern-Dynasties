# ==== Old treaty dispute (India batch 15) ====
# Shared 0-100 negotiation bar (starts at 50) of the global journal nesnd_india_treaty_dispute, which Mughal
# and its overlord (France) read together. The monthly progress comes from the bilateral relations of the
# two countries kept in the global variables nesnd_india_dispute_g_subject and nesnd_india_dispute_g_lord,
# and the buttons of both sides shift the same bar.
nesnd_india_treaty_dispute_bar = {
    name = nesnd_india_treaty_dispute_bar
    desc = nesnd_india_treaty_dispute_bar_renew_desc
    second_desc = nesnd_india_treaty_dispute_bar_void_desc
    progress_texture = "gfx/interface/progressbar/progressbar_green.dds"
    background_texture = "gfx/interface/progressbar/progressbar_bad.dds"
    glow_texture = "gfx/interface/progressbar/progressbar_green_glow.dds"
    glow_line_texture = "gfx/interface/progressbar/progressbar_green_glow_line.dds"
    marker = yes
    start_value = 50
    min_value = 0
    max_value = 100
    monthly_progress = {
        add = {
            desc = nesnd_india_dispute_rel_good_prog
            value = {
                value = 0
                if = {
                    limit = {
                        global_var:nesnd_india_dispute_g_subject ?= { nesnd_india_dispute_relations_high = yes }
                    }
                    value = 0.5
                }
            }
        }
        add = {
            desc = nesnd_india_dispute_rel_bad_prog
            value = {
                value = 0
                if = {
                    limit = {
                        global_var:nesnd_india_dispute_g_subject ?= { nesnd_india_dispute_relations_low = yes }
                    }
                    value = -0.5
                }
            }
        }
    }
}

# ==== Frontier defence against Persia's eastern claims (India batch 16) ====
# Single-direction bar 0-100, starting at 0. Passively +0.5/month while at peace with Persia and
# holding every tracked claimed region (200 months to fill passively; mainly driven by buttons).
nesnd_india_north_bar = {
    name = nesnd_india_north_bar
    desc = nesnd_india_north_bar_readout
    default_green = yes
    start_value = 0
    min_value = 0
    max_value = 100
    monthly_progress = {
        add = {
            desc = nesnd_india_north_bar_monthly_desc
            value = {
                value = 0
                if = {
                    limit = {
                        owner = {
                            exists = c:PER
                            c:PER = { is_country_alive = yes }
                            NOT = { has_war_with = c:PER }
                            nesnd_india_north_all_regions_held = yes
                        }
                    }
                    value = 0.5
                }
            }
        }
    }
}
