# 成交进度：每月由下列来源累加（唯一写入者），满 100 时日志周脉冲结算并清零。
# 来源数值由日志月脉冲 nesnd_triangle_mark 统计后存入全局变量。
nesnd_north_trade_bar = {
	name = nesnd_north_trade_bar
	desc = nesnd_north_trade_bar_desc
	default_bad = yes
	is_inverted = yes
	min_value = 0
	max_value = 100
	start_value = 0

	monthly_progress = {
		if = {
			limit = {
				NOT = { has_global_variable = nesnd_north_trade_ended }
				has_global_variable = nesnd_north_trade_rate_law
				has_global_variable = nesnd_north_trade_rate_navy
			}
			add = { desc = nesnd_triangle_trade_base_row value = 1 }
			add = { desc = nesnd_triangle_trade_law_row value = global_var:nesnd_north_trade_rate_law }
			add = { desc = nesnd_triangle_trade_navy_row value = global_var:nesnd_north_trade_rate_navy }
		}
	}
}
