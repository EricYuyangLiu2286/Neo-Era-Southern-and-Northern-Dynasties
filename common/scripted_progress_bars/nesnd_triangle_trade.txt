# 每单一条成交进度条。只在该槽位有订单时累加，满 100 时日志周脉冲结算这一单并清零。
# 速度 = 运输效率（排单时随机）+ 卖方奴隶制法律 + 买方奴隶制法律。
nesnd_north_trade_bar_1 = {
	name = nesnd_triangle_trade_bar_1
	desc = nesnd_triangle_trade_bar_1_desc
	default_bad = yes
	is_inverted = yes
	min_value = 0
	max_value = 100
	start_value = 0

	monthly_progress = {
		if = {
			limit = {
				NOT = { has_global_variable = nesnd_north_trade_ended }
				has_global_variable = nesnd_north_trade_amount_1
				has_global_variable = nesnd_north_trade_speed_1
				has_global_variable = nesnd_north_trade_seller_speed_1
				has_global_variable = nesnd_north_trade_buyer_speed_1
			}
			add = { desc = nesnd_triangle_trade_speed_row value = global_var:nesnd_north_trade_speed_1 }
			add = { desc = nesnd_triangle_trade_seller_law_row value = global_var:nesnd_north_trade_seller_speed_1 }
			add = { desc = nesnd_triangle_trade_buyer_law_row value = global_var:nesnd_north_trade_buyer_speed_1 }
		}
	}
}
