# ==== Overseas dealings progress bars (India batch 17) ====
# Five bars, one per land strategic region (0-100, start 0, default_green).
# No country variable mirrors these bars; script values read them directly.

nesnd_india_overseas_himalayas_bar = {
	name = nesnd_india_overseas_himalayas_bar
	desc = nesnd_india_overseas_himalayas_bar_readout
	default_green = yes
	start_value = 0
	min_value = 0
	max_value = 100
	monthly_progress = {
		progress_condition = {
			limit = { has_strategic_region_interest_tier = { strategic_region = sr:region_himalayas value >= 2 } }
			add = { desc = nesnd_india_overseas_prog_interest_2 value = 1 }
		}
		progress_condition = {
			limit = { has_strategic_region_interest_tier = { strategic_region = sr:region_himalayas value >= 3 } }
			add = { desc = nesnd_india_overseas_prog_interest_3 value = 1 }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_treaty_count_himalayas >= 1 }
			add = { desc = nesnd_india_overseas_prog_treaties value = nesnd_india_overseas_treaty_prog_himalayas }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_cordial_count_himalayas >= 1 }
			add = { desc = nesnd_india_overseas_prog_relations value = nesnd_india_overseas_cordial_prog_himalayas }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_trade_sum_himalayas >= 1000 }
			add = { desc = nesnd_india_overseas_prog_trade_1000 value = 0.5 }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_trade_sum_himalayas >= 5000 }
			add = { desc = nesnd_india_overseas_prog_trade_5000 value = 0.5 }
		}
		progress_condition = {
			limit = { has_variable = nesnd_india_unified_done }
			add = { desc = nesnd_india_overseas_prog_unified value = 0.5 }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_rival_pressure = { REGION = region_himalayas } }
			add = { desc = nesnd_india_overseas_prog_competition value = -0.5 }
		}
		progress_condition = {
			limit = { nesnd_india_overseas_at_war_in_region = { REGION = region_himalayas } }
			add = { desc = nesnd_india_overseas_prog_war value = -2 }
		}
	}
}
