nesnd_sha_yan_banner = {
    name = nesnd_sha_yan_banner
    desc = nesnd_sha_yan_banner_desc
    default = yes
    min_value = 0
    max_value = 100
    start_value = 0
    monthly_progress = {
        if = {
            limit = { exists = c:CNE c:SHA ?= { nesnd_sha_yan_preparing = yes } }
            c:CNE = {
                if = { limit = { is_at_war = yes } add = { desc = nesnd_sha_yan_at_war value = 0.5 } }
                if = {
                    limit = { has_variable = cne_landreform_failure_tracker }
                    if = { limit = { var:cne_landreform_failure_tracker >= 60 } add = { desc = nesnd_sha_yan_disorder_high value = 1 } }
                    else_if = { limit = { var:cne_landreform_failure_tracker >= 30 } add = { desc = nesnd_sha_yan_disorder_low value = 0.5 } }
                }
                if = {
                    limit = {
                        nesnd_sha_yan_four_states = yes
                        nesnd_sha_yan_mean_turmoil >= 0.25
                    }
                    add = { desc = nesnd_sha_yan_turmoil_row value = 0.5 }
                }
            }
            c:SHA = {
                if = { limit = { has_variable = nesnd_sha_korean_compact exists = c:KOR } add = { desc = nesnd_sha_yan_compact_row value = 0.25 } }
                if = { limit = { has_modifier = nesnd_sha_yan_funding } add = { desc = nesnd_sha_yan_funding value = 1 } }
            }
        }
    }
}
