# USM. Counts months while some Wu opponent is not at war with Wu; ce_usm_suppress_rebellion resets it
# when none is left and opens the war (nesnd_continue_war_open) when it is full.
nesnd_continue_war_bar = {
    name = nesnd_continue_war_bar_name
    desc = nesnd_continue_war_bar_desc
    min_value = 0 max_value = 5 start_value = 0
    default = yes
    monthly_progress = {
        progress_condition = {
            limit = { any_country = { nesnd_continue_war_candidate = yes } }
            add = { desc = nesnd_continue_war_bar_month value = 1 }
        }
    }
}
