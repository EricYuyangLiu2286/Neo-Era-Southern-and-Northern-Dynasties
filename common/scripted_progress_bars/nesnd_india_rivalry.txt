# ==== Franco-Ming Rivalry in India (India batch 15) ====
# Tug-of-war bar (-100 to 100, start 0). Positive favours France; negative favours United Federation of Ming.
# desc is drawn on the left over the filled part (vanilla gui/journal_entry.gui, je_ryukyu_rivalry_score_bar),
# so it names the power that wins at the high end.
nesnd_india_rivalry_bar = {
    name = nesnd_india_rivalry_bar
    desc = nesnd_india_rivalry_bar_fra_desc
    second_desc = nesnd_india_rivalry_bar_usm_desc
    progress_texture = "gfx/interface/progressbar/progressbar_default.dds"
    background_texture = "gfx/interface/progressbar/progressbar_gold.dds"
    glow_texture = "gfx/interface/progressbar/progressbar_default_glow.dds"
    glow_line_texture = "gfx/interface/progressbar/progressbar_default_glow_line.dds"
    marker = yes
    start_value = 0
    min_value = -100
    max_value = 100
    monthly_progress = {
        add = {
            desc = nesnd_india_rivalry_prog_fra_subjects
            value = {
                value = 0
                if = {
                    limit = { nesnd_india_rivalry_fra_subjects > nesnd_india_rivalry_usm_subjects }
                    value = 0.5
                }
            }
        }
        add = {
            desc = nesnd_india_rivalry_prog_usm_subjects
            value = {
                value = 0
                if = {
                    limit = { nesnd_india_rivalry_usm_subjects > nesnd_india_rivalry_fra_subjects }
                    value = -0.5
                }
            }
        }
        add = {
            desc = nesnd_india_rivalry_prog_fra_states
            value = {
                value = 0
                if = {
                    limit = { nesnd_india_rivalry_fra_states > nesnd_india_rivalry_usm_states }
                    value = 0.5
                }
            }
        }
        add = {
            desc = nesnd_india_rivalry_prog_usm_states
            value = {
                value = 0
                if = {
                    limit = { nesnd_india_rivalry_usm_states > nesnd_india_rivalry_fra_states }
                    value = -0.5
                }
            }
        }
    }
}
