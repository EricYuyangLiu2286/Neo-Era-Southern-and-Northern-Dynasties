nesnd_sha_sailings = {
    name = nesnd_sha_sailings
    desc = nesnd_sha_sailings_desc
    default = yes
    min_value = 0
    max_value = 100
    start_value = 0
    weekly_progress = {
        if = {
            # The counters exist only while the settlement journal is running.
            limit = { c:SHA ?= { has_variable = nesnd_sha_misses has_variable = nesnd_sha_sailing_level var:nesnd_sha_misses < 3 } }
            add = { desc = nesnd_sha_sailings_base value = 2 }
            if = {
                limit = { c:SHA = { var:nesnd_sha_sailing_level > 0 } }
                add = { desc = nesnd_sha_sailings_extra value = c:SHA.var:nesnd_sha_sailing_level }
            }
        }
    }
}
