ce_war_to_opium_progress_bar = {
	name = "ce_war_to_opium_progress_bar"
	desc = "ce_war_to_opium_progress_bar_desc"

	default_bad = yes
	start_value = 50
	min_value = 0
	max_value = 100

	monthly_progress = {
		add = {
			if = {
				limit = {
					any_market = {
						owner = root
						mg:opium ?= { market_goods_sell_orders >= 400 }
					}
				}
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_sell_orders_high"
				}
			}
			else_if = {
				limit = {
					any_market = {
						owner = root
						mg:opium ?= { market_goods_sell_orders < 400 }
					}
				}
				add = {
					value = -1
					desc = "ce_war_to_opium_pb_sell_orders_low"
				}
			}
			if = {
				limit = { is_banning_goods = g:opium }
				add = {
					value = -1
					desc = "ce_war_to_opium_pb_banning"
				}
			}
			if = {
				limit = { average_sol_for_primary_cultures < 8 }
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_low_sol"
				}
			}
			if = {
				limit = { has_modifier = declared_bankruptcy }
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_bankrupt"
				}
			}
			if = {
				limit = { is_in_civil_war = yes }
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_civil_war"
				}
			}
			if = {
				limit = { country_turmoil > 0.5 }
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_turmoil"
				}
			}
			if = {
				limit = { country_rank >= rank_value:major_power }
				add = {
					value = -1
					desc = "ce_war_to_opium_pb_major_power"
				}
			}
			if = {
				limit = { has_law_or_variant = law_type:law_council_republic }
				add = {
					value = -1
					desc = "ce_war_to_opium_pb_council"
				}
			}
			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_laissez_faire
						has_law_or_variant = law_type:law_cooperative_ownership
						has_law_or_variant = law_type:law_extraction_economy
					}
				}
				add = {
					value = 1
					desc = "ce_war_to_opium_pb_bad_laws"
				}
			}
			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_command_economy
						has_law_or_variant = law_type:law_traditionalism
						has_law_or_variant = law_type:law_interventionism
					}
				}
				add = {
					value = -1
					desc = "ce_war_to_opium_pb_good_laws"
				}
			}
			if = {
				limit = { has_variable = AI_default_opium_je_success }
				add = {
					value = -3
					desc = "ce_war_to_opium_pb_ai"
				}
			}
		}
	}
}
