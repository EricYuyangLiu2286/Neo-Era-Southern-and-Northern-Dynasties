nesnd_jp_korea_bar = {
    name = nesnd_jp_korea_bar_name
    desc = nesnd_jp_korea_bar_desc
    default = yes
    start_value = 0
    min_value = 0
    max_value = 100
    monthly_progress = {
        owner = {
            add = {
                desc = nesnd_jp_korea_bar_grievance
                value = -3
            }
            # 科举，每行 +1
            if = {
                limit = {
                    nesnd_jp_korea_policy_value = 1
                }
                progress_condition = {
                    limit = {
                        bureaucracy >= 0
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_exam_admin
                        value = 1
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            nesnd_jp_korea_southern_state = yes
                            controller = owner
                            count >= 2
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_exam_control
                        value = 1
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            nesnd_jp_korea_southern_state = yes
                            turmoil < 0.25
                            count >= 2
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_exam_order
                        value = 1
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            nesnd_jp_korea_southern_state = yes
                            any_scope_building = {
                                is_building_type = building_government_administration
                                level >= 2
                                occupancy >= 0.5
                            }
                            count >= 2
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_exam_offices
                        value = 1
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            nesnd_jp_korea_southern_state = yes
                            any_scope_building = {
                                is_building_type = building_university
                                level >= 1
                            }
                            count >= 2
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_exam_university
                        value = 1
                    }
                }
            }
            # 关白，每行 +2；动乱 -10
            if = {
                limit = {
                    nesnd_jp_korea_policy_value = 2
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            state_region = s:STATE_BUSAN
                            has_decree = decree_promote_social_mobility
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_kampaku_busan_mobility
                        value = 2
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            state_region = s:STATE_BUSAN
                            has_decree = decree_greener_grass_campaign
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_kampaku_busan_grass
                        value = 2
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            state_region = s:STATE_YANGHO
                            has_decree = decree_promote_social_mobility
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_kampaku_yangho_mobility
                        value = 2
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            state_region = s:STATE_YANGHO
                            has_decree = decree_greener_grass_campaign
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_kampaku_yangho_grass
                        value = 2
                    }
                }
                progress_condition = {
                    limit = {
                        any_scope_state = {
                            nesnd_jp_korea_southern_state = yes
                            turmoil >= 0.25
                        }
                    }
                    add = {
                        desc = nesnd_jp_korea_bar_kampaku_turmoil
                        value = -10
                    }
                }
            }
            # 朝鲜州过多，两条路线和未选路线都起作用
            progress_condition = {
                limit = {
                    any_scope_state = {
                        state_region = {
                            is_homeland = cu:korean
                        }
                        count >= 3
                    }
                }
                add = {
                    desc = nesnd_jp_korea_bar_states_3
                    value = -0.5
                }
            }
            progress_condition = {
                limit = {
                    any_scope_state = {
                        state_region = {
                            is_homeland = cu:korean
                        }
                        count >= 4
                    }
                }
                add = {
                    desc = nesnd_jp_korea_bar_states_4
                    value = -0.5
                }
            }
            progress_condition = {
                limit = {
                    any_scope_state = {
                        state_region = {
                            is_homeland = cu:korean
                        }
                        count >= 5
                    }
                }
                add = {
                    desc = nesnd_jp_korea_bar_states_5
                    value = -0.5
                }
            }
        }
    }
}
