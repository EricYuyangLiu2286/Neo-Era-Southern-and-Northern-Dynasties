ce_french_empire_republican_movement_progress_bar = {
	name = "ce_french_empire_republican_movement_progress_bar_name"

	start_value = 25
	min_value = 0
	max_value = 100
	
	default_bad = yes
	
	monthly_progress = {
		add = {
			desc = "ce_incor_state_avg_sol"
			if = {
				limit = { 
					exists = var:ce_french_empire_sol_count
					has_variable = ce_french_empire_sol_count 
				}
				add = {
					value = var:ce_french_empire_sol_count
					multiply = -1
					divide = 3
				}
			}
		}
		add = {
			desc = "ce_french_revolution_base"
			value = 1
		}
		add = {
			if = {
				limit = {
					year >= 1840
					year <= 1850
				}
				add = {
					desc = "ce_french_year_of_crisis"
					value = 2
				}
					
			}
		}
		add = {
			if = {
				limit = {
					OR = {
						is_involved_in_journal_entry = je_springtime_of_the_peoples
						has_journal_entry = je_springtime_of_the_peoples
					}
				}
				
				add = {
					desc = "je_springtime_of_the_peoples"
					value = 2
				}
					
			}
		}
		add = {
			if = {
				limit = {
					any_political_movement = {
						OR = {
							movement_is_liberal = yes
							movement_is_socialist = yes
						}
						political_movement_support > 0.15
					}
				}
				
				add = {
					desc = "ce_republican_movement_support"
					value = 2
				}
			}
		}
		add = {
			if = {
				limit = {
					AND = {
						ig:ig_intelligentsia = { ig_clout >= 0.15 }
						ig:ig_trade_unions = { ig_clout >= 0.15 }
					}
				}
				
				add = {
					desc = "ce_french_ig_support_liberal"
					value = 2
				}
			}
			else_if = {
				limit = {
					OR = {
						ig:ig_intelligentsia = { ig_clout >= 0.15 }
						ig:ig_trade_unions = { ig_clout >= 0.15 }
					}
				}
				
				add = {
					desc = "ce_french_ig_support_liberal"
					value = 1
				}
			}
		}
		add = {
			if = {
				limit = {
					any_political_movement = {
						has_modifier = ce_french_republican_helped
					}
				}
				add = {
					desc = "ce_republican_helped"
					value = 2
				}
			}
		}
		add = {
			if = {
				limit = {
					has_global_variable = ce_orleanist_come_back
				}
				add = {
					desc = "Louis-Philippe"
					value = 1
				}
			}
		}
	}
}
