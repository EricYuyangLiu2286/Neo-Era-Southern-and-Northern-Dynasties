# USM, on ce_usm_suppress_rebellion. Wu morale in the Second Jingnan War, 0-100, starting at 30.
# Moves only while Wu is in a Jingnan war; the journal's monthly pulse turns it into a scaled modifier.
nesnd_wu_resolve_bar = {
    name = nesnd_wu_resolve_bar_name
    desc = nesnd_wu_resolve_bar_desc
    min_value = 0 max_value = 100 start_value = 30
    default_green = yes
    monthly_progress = {
        progress_condition = {
            limit = {
                nesnd_wu_jingnan_at_war = yes
                has_variable = nesnd_wu_resolve_months
                var:nesnd_wu_resolve_months >= 24
            }
            subtract = { desc = nesnd_wu_resolve_drag_long value = 1.5 }
            else = {
                limit = { nesnd_wu_jingnan_at_war = yes }
                subtract = { desc = nesnd_wu_resolve_drag value = 1 }
            }
        }
        progress_condition = {
            limit = {
                nesnd_wu_jingnan_at_war = yes
                nesnd_wu_resolve_foreign_value > 0
            }
            add = { desc = nesnd_wu_resolve_foreign value = nesnd_wu_resolve_foreign_value }
        }
        progress_condition = {
            limit = {
                nesnd_wu_jingnan_at_war = yes
                nesnd_wu_resolve_fresh_value > 0
            }
            add = { desc = nesnd_wu_resolve_fresh value = nesnd_wu_resolve_fresh_value }
        }
    }
}
