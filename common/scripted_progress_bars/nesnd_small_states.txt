# Counts the months that pass "no default, market access, profitable production". A month that does not qualify adds nothing and takes nothing away.
nesnd_small_economy_bar = {
    name = nesnd_small_observation_months
    desc = nesnd_small_economy_bar_readout
    default_green = yes
    min_value = 0
    max_value = 12
    start_value = 0
    monthly_progress = {
        add = {
            desc = nesnd_small_qualified_month
            value = {
                value = 0
                if = {
                    limit = {
                        owner = { nesnd_small_economy_qualified = yes }
                    }
                    value = 1
                }
            }
        }
    }
}
