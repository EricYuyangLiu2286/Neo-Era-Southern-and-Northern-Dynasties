ce_polish_reconquest_energy_bar = {
	name = "ce_polish_reconquest_energy_bar_name"
	desc = "ce_polish_reconquest_energy_bar_desc"
	
	start_value = 0
	min_value = 0
	max_value = 100
	
	default_green = yes
	
	monthly_progress = {
		add = {
			desc = "ce_polish_war_prep_allies"
			if = {
				limit = { has_global_variable = ce_poland_support_diff }
				add = {
					value = global_var:ce_poland_support_diff
					max = 3
					min = -3
				}
			}
		}
		add = {
			if = {
				limit = {
					infamy > 10
				}
				add = {
					value = -1
					desc = "ce_polish_energy_infamy"
				}
			}
		}
		add = {
			if = {
				limit = {
					has_modifier = ce_pro_poland_spend_buff
				}
				add = {
					value = 2
					desc = "ce_pro_poland_spend_buff"
				}
			}
		}
		add = {
			if = {
				limit = {
					radical_fraction = {
						value >= 0.2
					}
				}
				add = {
					value = -1
					desc = "ce_polish_energy_radicals_1"
				}
			}
		}
		add = {
			if = {
				limit = {
					radical_fraction = {
						value >= 0.4
					}
				}
				add = {
					value = -1
					desc = "ce_polish_energy_radicals_2"
				}
			}
		}
		add = {
			if = {
				limit = {
					radical_fraction = {
						value >= 0.8
					}
				}
				add = {
					value = -1
					desc = "ce_polish_energy_radicals_3"
				}
			}
		}
		add = {
			if = {
				limit = {
					gdp_ownership_ratio = {
						target = c:KRA
						value < 0.5
					}
				}
				add = {
					value = -2
					desc = "ce_polish_energy_gdp"
				}
			}
		}
	}
}
