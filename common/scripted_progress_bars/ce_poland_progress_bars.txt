# Root journal je_ce_the_polish_problem (contextless). Strength of the 1829 settlement, 0-100.
# Only writer: this monthly_progress. Inputs are global variables written by nesnd_poland_root_month.
# Phase thresholds (60 / 30) are read in nesnd_poland_root_month.
nesnd_poland_consensus_bar = {
	name = "nesnd_poland_consensus_bar_name"
	desc = "nesnd_poland_consensus_bar_desc"

	start_value = 70
	min_value = 0
	max_value = 100

	default_green = yes

	monthly_progress = {
		add = {
			desc = nesnd_poland_order_from_congress
			if = { limit = { has_global_variable = nesnd_congress_order }
				add = { value = global_var:nesnd_congress_order subtract = 75 divide = 100 }
			}
		}
		add = { desc = nesnd_poland_rw_pressure
			if = { limit = { c:KRA ?= { je:je_ce_polish_reconquest ?= { "scripted_bar_progress(nesnd_poland_red_white_bar)" > 60 } } } add = -1 }
		}
		add = {
			desc = "nesnd_poland_consensus_drift"
			value = -2
		}
		add = {
			desc = "nesnd_poland_consensus_guarantees"
			if = {
				limit = { has_global_variable = nesnd_poland_guarantees }
				add = {
					value = global_var:nesnd_poland_guarantees
					multiply = 0.5
				}
			}
		}
		add = {
			desc = "nesnd_poland_consensus_bloc"
			if = {
				limit = {
					exists = c:KRA
					exists = c:FRA
					c:KRA = { is_in_same_power_bloc = c:FRA }
				}
				add = 0.5
			}
		}
		add = {
			desc = "nesnd_poland_consensus_paris_strong"
			if = {
				limit = {
					has_global_variable = nesnd_poland_france_will
					global_var:nesnd_poland_france_will >= 75
				}
				add = 0.5
			}
		}
		add = {
			desc = "nesnd_poland_consensus_paris_weak"
			if = {
				limit = {
					has_global_variable = nesnd_poland_france_will
					global_var:nesnd_poland_france_will < 40
				}
				add = -1
			}
		}
		add = {
			desc = "nesnd_poland_consensus_france_rank"
			if = {
				limit = {
					exists = c:FRA
					c:FRA = { country_rank < rank_value:great_power }
				}
				add = -2
			}
		}
		add = {
			desc = "nesnd_poland_consensus_warsaw"
			if = {
				limit = {
					exists = c:KRA
					c:KRA = { capital ?= { state_region = s:STATE_GREATER_POLAND } }
				}
				add = -1
			}
		}
		add = {
			desc = "nesnd_poland_consensus_war"
			if = {
				limit = {
					exists = c:FRA
					c:FRA = {
						OR = {
							trigger_if = { limit = { exists = c:RUS } has_war_with = c:RUS } trigger_else = { always = no }
							trigger_if = { limit = { exists = c:PRU } has_war_with = c:PRU } trigger_else = { always = no }
							trigger_if = { limit = { exists = c:AUS } has_war_with = c:AUS } trigger_else = { always = no }
						}
					}
				}
				add = -2
			}
		}
		add = {
			desc = "nesnd_poland_consensus_backers"
			if = {
				limit = {
					has_global_variable = ce_poland_support_diff
					global_var:ce_poland_support_diff > 0
				}
				add = 0.5
			}
			else_if = {
				limit = {
					has_global_variable = ce_poland_support_diff
					global_var:ce_poland_support_diff < 0
				}
				add = -0.5
			}
		}
	}
}
