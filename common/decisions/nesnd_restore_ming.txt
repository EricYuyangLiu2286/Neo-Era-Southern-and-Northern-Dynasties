# Rebuild the Ming. Any Ming loyalist that holds the south's core after the Jingnan War has ended in a stalemate,
# including one that has annexed Shu or Great Han. It reuses the Ming Federation founding of nesnd_celestial_collapse.3.
nesnd_restore_ming = {
    is_shown = {
        nesnd_ming_federation_loyalist = yes
        OR = { has_global_variable = nesnd_south_settled has_global_variable = usm_civilwar_status_quo }
        NOT = { has_variable = nesnd_ming_federation_founded }
    }
    possible = {
        custom_tooltip = { text = nesnd_restore_ming_no_court_tt NOT = { any_country = { has_variable = nesnd_ming_court } } }
        custom_tooltip = { text = nesnd_restore_ming_no_wu_tt NOT = { c:USM ?= { is_country_alive = yes has_variable = tag_WUU_var } } }
        custom_tooltip = { text = nesnd_restore_ming_stalemate_tt nesnd_unify_south_stalemate = yes }
        custom_tooltip = { text = nesnd_restore_ming_core_tt OR = { owns_entire_state_region = STATE_NANJING nesnd_unify_south_owned >= 8 } }
        is_subject = no
        is_at_war = no
    }
    when_taken = {
        custom_tooltip = nesnd_restore_ming_tt
        hidden_effect = {
            if = {
                limit = { nesnd_restore_ming_possible = yes }
                set_variable = nesnd_ming_court
                set_variable = nesnd_restoration_side
                set_variable = nesnd_south_unifier
                set_global_variable = { name = nesnd_ming_federation_host value = this }
                trigger_event = { id = nesnd_celestial_collapse.3 popup = yes }
            }
        }
    }
    ai_chance = { value = 100 }
}
