# Before a venue exists, members can still request it from the Decisions tab.
nesnd_request_first_congress = {
	is_shown = {
		nesnd_poland_active = yes
		nesnd_congress_member = yes
		NOT = { has_global_variable = nesnd_congress_started }
	}
	possible = {
		custom_tooltip = { text = nesnd_congress_no_request_tt NOT = { has_global_variable = nesnd_congress_requested } }
		custom_tooltip = { text = nesnd_congress_request_ready_tt NOT = { has_variable = nesnd_congress_request_cooldown } }
	}
	when_taken = { nesnd_congress_request_sitting = yes }
	# AI openings follow diplomatic events; a routine roll must not open the venue in 1836.
	ai_chance = { value = 0 }
}
