nesnd_usm_recognise_japanese = {
    is_shown = { c:USM ?= this NOT = { country_has_primary_culture = cu:japanese } }
    possible = {
        custom_tooltip = {
            text = nesnd_usm_native_insurrection_tt
            any_diplomatic_play = {
                is_war = yes
                OR = {
                    AND = { target = c:USM initiator ?= { nesnd_usm_native_challenger = yes } }
                    AND = { initiator = c:USM target ?= { nesnd_usm_native_challenger = yes } }
                }
                OR = {
                    is_diplomatic_play_type = dp_secession
                    AND = {
                        is_diplomatic_play_type = dp_revolution
                        OR = {
                            initiator ?= { nesnd_usm_republican_or_socialist = yes }
                            target ?= { nesnd_usm_republican_or_socialist = yes }
                        }
                    }
                }
            }
        }
        custom_tooltip = {
            text = nesnd_usm_japanese_home_state_tt
            any_scope_state = {
                is_incorporated = yes
                controller = c:USM
                is_in_geographic_region = geographic_region_japan
                NOR = { state_region = s:STATE_SAKHALIN state_region = s:STATE_RYUKYU_ISLANDS }
            }
        }
        custom_tooltip = {
            text = nesnd_usm_no_japanese_secession_tt
            cu:japanese = { culture_secession_progress = { target = c:USM value = 0 } }
            NOT = { any_political_movement = { exists = culture culture ?= cu:japanese is_secessionist = yes } }
        }
    }
    when_taken = { add_primary_culture = cu:japanese }
    ai_chance = { value = 100 }
}
