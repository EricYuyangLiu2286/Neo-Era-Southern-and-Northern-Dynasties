nesnd_sha_vladivostok = {
    is_shown = {
        nesnd_sha_native = yes
        exists = s:STATE_OUTER_MANCHURIA.region_state:SHA
        NOR = {
            has_variable = nesnd_sha_vladivostok_considered
            has_variable = nesnd_sha_qingdao_capital
        }
    }
    possible = { NOT = { has_modifier = declared_bankruptcy } }
    when_taken = {
        hidden_effect = { set_variable = nesnd_sha_vladivostok_considered }
        trigger_event = { id = nesnd_sha_republic.8 popup = yes }
    }
    ai_chance = { value = 100 }
}

nesnd_sha_qingdao = {
    is_shown = {
        nesnd_sha_native = yes
        exists = s:STATE_SHANDONG.region_state:SHA
        NOT = { has_variable = nesnd_sha_qingdao_capital }
    }
    possible = { NOT = { has_modifier = declared_bankruptcy } }
    when_taken = {
        hidden_effect = { set_variable = nesnd_sha_qingdao_capital }
        add_treasury = nesnd_sha_capital_cost
        set_capital = STATE_SHANDONG
        add_modifier = { name = nesnd_sha_homecoming years = 20 }
        add_loyalists = { value = 0.10 }
    }
    ai_chance = { value = 100 }
}
