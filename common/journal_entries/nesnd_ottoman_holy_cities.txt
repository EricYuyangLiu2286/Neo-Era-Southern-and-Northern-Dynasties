# Ottoman B11 foreign-affairs journal: "The Holy Cities" (both success and sultanic failure routes).
# Added only by nesnd_ottoman_holy_start (event .12 option a and monthly fallback nesnd_ottoman_holy_fallback),
# so it defines neither possible nor is_shown_when_inactive.
# Non-event outcomes omit event_outcome_desc so effects display natively.

nesnd_ottoman_holy_cities = {
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    group = je_group_foreign_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes

    scripted_button = nesnd_ottoman_holy_plan
    scripted_button = nesnd_ottoman_holy_offer

    immediate = {
        set_variable = {
            name = nesnd_ottoman_holy_deadline
            months = 240
        }
    }

    on_monthly_pulse = {
        effect = {
            nesnd_ottoman_holy_update_dependent_months = yes
        }
    }

    # 1. Hidden invalidation (tag changed away from TUR).
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_ottoman_holy_invalid_conclusion
        trigger = {
            NOT = { c:TUR ?= this }
        }
        effect = {
            if = {
                limit = {
                    has_variable = nesnd_ottoman_caliph_claimed
                }
                remove_variable = nesnd_ottoman_caliph_claimed
            }
            set_variable = nesnd_ottoman_holy_closed
            nesnd_ottoman_holy_cleanup = yes
        }
    }

    # 2. Lost autonomy (R13: unable to act abroad for 12 consecutive months; may reopen after 24-month cooldown).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_holy_dependent_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_holy_dependent_tt
                has_variable = nesnd_ottoman_holy_dependent_months
                var:nesnd_ottoman_holy_dependent_months >= 12
            }
        }
        effect = {
            if = {
                limit = {
                    NOT = { has_modifier = nesnd_ottoman_lost_autonomy }
                }
                add_modifier = {
                    name = nesnd_ottoman_lost_autonomy
                }
            }
            custom_tooltip = nesnd_ottoman_holy_reopen_cd_tt
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_holy_outcome
                    value = flag:dependent
                }
                set_variable = {
                    name = nesnd_ottoman_holy_reopen_cd
                    months = 24
                }
                nesnd_ottoman_holy_cleanup = yes
            }
        }
    }

    # 3. Holy Cities secured (R10: all states in STATE_HEDJAZ held by TUR or its subjects).
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_ottoman_holy_won_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_holy_won_tt
                nesnd_ottoman_holds_holy_cities = yes
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_caliphate
            }
            if = {
                limit = {
                    has_journal_entry = nesnd_ottoman_recovery
                }
                je:nesnd_ottoman_recovery ?= {
                    add_progress = {
                        name = nesnd_ottoman_enthusiasm_bar
                        value = 20
                    }
                }
            }
            hidden_effect = {
                if = {
                    limit = {
                        has_variable = nesnd_ottoman_caliph_claimed
                    }
                    remove_variable = nesnd_ottoman_caliph_claimed
                }
                set_variable = {
                    name = nesnd_ottoman_holy_outcome
                    value = flag:won
                }
                set_variable = nesnd_ottoman_holy_closed
                nesnd_ottoman_holy_cleanup = yes
            }
        }
    }

    # 4. Situation changed (R12: foe no longer valid and not currently at war with it; may reopen after 24-month cooldown).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_holy_changed_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_holy_changed_tt
                nesnd_ottoman_holy_situation_changed = yes
            }
        }
        effect = {
            custom_tooltip = nesnd_ottoman_holy_claims_kept_tt
            custom_tooltip = nesnd_ottoman_holy_reopen_cd_tt
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_holy_outcome
                    value = flag:changed
                }
                set_variable = {
                    name = nesnd_ottoman_holy_reopen_cd
                    months = 24
                }
                nesnd_ottoman_holy_cleanup = yes
            }
        }
    }

    # 5. Holy Cities remain under a rival caliph on deadline (240-month deadline elapsed while not in a play or war with the foe).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_holy_rival_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_holy_rival_tt
                NOT = { has_variable = nesnd_ottoman_holy_deadline }
                nesnd_ottoman_holy_play_in_progress = no
                nesnd_ottoman_holy_at_war_with_foe = no
                nesnd_rival_caliph_holds_holy_cities = yes
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_holy_lost_mod
            }
            ig:ig_devout ?= {
                add_modifier = {
                    name = nesnd_ottoman_holy_devout_anger
                    months = 120
                }
            }
            hidden_effect = {
                if = {
                    limit = {
                        has_variable = nesnd_ottoman_caliph_claimed
                    }
                    remove_variable = nesnd_ottoman_caliph_claimed
                }
                set_variable = {
                    name = nesnd_ottoman_holy_outcome
                    value = flag:rival
                }
                set_variable = nesnd_ottoman_holy_closed
                nesnd_ottoman_holy_cleanup = yes
            }
        }
    }

    # 6. Holy Cities unrecovered (240-month deadline elapsed while not in a play or war with the foe, and not wholly held by a rival caliph).
    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_holy_shelved_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_holy_timeout_tt
                NOT = { has_variable = nesnd_ottoman_holy_deadline }
                nesnd_ottoman_holy_play_in_progress = no
                nesnd_ottoman_holy_at_war_with_foe = no
                nesnd_rival_caliph_holds_holy_cities = no
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_holy_shelved_mod
            }
            hidden_effect = {
                if = {
                    limit = {
                        has_variable = nesnd_ottoman_caliph_claimed
                    }
                    remove_variable = nesnd_ottoman_caliph_claimed
                }
                set_variable = {
                    name = nesnd_ottoman_holy_outcome
                    value = flag:shelved
                }
                set_variable = nesnd_ottoman_holy_closed
                nesnd_ottoman_holy_cleanup = yes
            }
        }
    }

    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    nesnd_ottoman_holy_at_war_with_foe = yes
                }
                desc = nesnd_ottoman_holy_status_war
            }
            triggered_desc = {
                trigger = {
                    nesnd_ottoman_holy_play_in_progress = yes
                }
                desc = nesnd_ottoman_holy_status_play
            }
            triggered_desc = {
                trigger = {
                    any_country = {
                        nesnd_rival_caliph = yes
                    }
                }
                desc = nesnd_ottoman_holy_status_rival_caliph
            }
            desc = nesnd_ottoman_holy_status_standoff
        }
    }
}
