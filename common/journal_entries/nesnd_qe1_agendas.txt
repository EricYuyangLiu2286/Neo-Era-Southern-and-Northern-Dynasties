	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 1000

	scripted_progress_bar = nesnd_qe1_balance_bar
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		    custom_tooltip = { text = nesnd_qe1_condition_states_4 num_states >= 4 }
		    OR = {
		        nesnd_qe1_economic_victory = yes
		        nesnd_qe1_military_victory = yes
		    }
		}
		effect = { custom_tooltip = nesnd_sweep_qin_rewards hidden_effect = { nesnd_qe1_settle_agenda = yes } }
	}
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		    custom_tooltip = { text = nesnd_qe1_condition_states_4 num_states >= 4 }
		    hidden_trigger = { has_variable = nesnd_qe1_agenda_started }
		    custom_tooltip = { text = nesnd_qe1_condition_agenda_deadline NOT = { has_variable = nesnd_qe1_agenda_clock } }
		    NOR = { nesnd_qe1_economic_victory = yes nesnd_qe1_military_victory = yes }
		}
		effect = { nesnd_qe1_agenda_timeout = yes }
	}
	timeout = 3650
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = { nesnd_qe1_agenda_timeout = yes }
	}
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { NOT = { nesnd_qe1_agenda_eligible = yes } }
		effect = { nesnd_qe1_close_agenda = yes }
	}
	on_monthly_pulse = { effect = { nesnd_qe1_agenda_monthly = yes } }
	on_weekly_pulse = { effect = { nesnd_qe1_watch_endpoints = yes nesnd_qe1_check_civil_conflict = yes nesnd_qe1_ensure_agenda_children = yes } }
	status_desc = { first_valid = {
			triggered_desc = { trigger = { var:nesnd_qe1_balance < -10 } desc = nesnd_qe1_agenda_status_econ }
			triggered_desc = { trigger = { var:nesnd_qe1_balance > 10 } desc = nesnd_qe1_agenda_status_army }
			desc = nesnd_qe1_agenda_status_neutral
		} }

}
