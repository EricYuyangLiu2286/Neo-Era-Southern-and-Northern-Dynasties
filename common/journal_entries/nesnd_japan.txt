nesnd_wu_eastern_question = {
    icon = "gfx/interface/icons/politics_icons/revolution.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    possible = {
        c:USM ?= this has_variable = tag_WUU_var
        has_global_variable = nesnd_south_settled
        trigger_if = { limit = { exists = c:JAP exists = c:USM } c:JAP = { is_country_alive = yes } }
        trigger_else = { always = no }
        NOT = { has_variable = nesnd_japan_dossier_entered }
        NOT = { has_global_variable = nesnd_japan_dossier_closed }
    }
    is_shown_when_inactive = { c:USM ?= this has_variable = tag_WUU_var }
    immediate = { set_variable = nesnd_japan_dossier_entered nesnd_japan_consider_terms_once = yes set_variable = nesnd_eastern_absolute_v2  scope:journal_entry ?= { set_bar_progress = { name = nesnd_eastern_readiness_bar value = owner.nesnd_japan_preparation_score } } }
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = nesnd_japan_war_started } desc = nesnd_wu_eastern_question_status_war }
            triggered_desc = { trigger = { has_global_variable = nesnd_japan_offer_sent } desc = nesnd_wu_eastern_question_status_offer }
            desc = nesnd_wu_eastern_question_status
        }
    }
    scripted_progress_bar = nesnd_eastern_readiness_bar
    scripted_button = nesnd_japan_send_terms
    scripted_button = nesnd_japan_accept_status_quo
    outcome = {
    	header = je_standard_completion_header
    	effect_header = je_standard_on_completion_header
    	color = "gold_positive"
    	conclusion_desc = je_standard_completion_conclusion_desc
    	trigger = { custom_tooltip = { text = nesnd_eastern_question_resolved_tt has_global_variable = nesnd_japan_dossier_closed } }
    	effect = { }
    }
    outcome = {
    	flags = { hidden invalid_outcome }
    	conclusion_desc = je_standard_failure_conclusion_desc
    	trigger = { OR = { NOT = { has_global_variable = nesnd_south_settled } NOT = { has_variable = tag_WUU_var } NOT = { has_variable = nesnd_eastern_absolute_v2 } } }
    	effect = { if = { limit = { NOT = { has_variable = tag_WUU_var } } set_global_variable = nesnd_japan_dossier_closed } else = { remove_variable = nesnd_japan_dossier_entered } }
    }
    on_monthly_pulse = { effect = {
            nesnd_japan_consider_terms_once = yes
            nesnd_japan_observe_other_house = yes
            if = { limit = {
                    NOT = { has_global_variable = nesnd_japan_dossier_closed }
                    OR = {
                        NOT = { c:JAP ?= { is_country_alive = yes } }
                        AND = {
                            has_global_variable = nesnd_japan_war_started
                            NOT = { has_variable = nesnd_japan_war_grace }
                            NOT = { has_war_with = c:JAP }
                            NOT = { any_diplomatic_play = { has_variable = nesnd_eastern_intervention_play } }
                        }
                    }
                }
                set_global_variable = nesnd_japan_dossier_closed
                trigger_event = { id = nesnd_japan.4 popup = yes }
            }
        }
    }
    weight = 900
    should_be_pinned_by_default_uninvolved_or_context = yes
}
