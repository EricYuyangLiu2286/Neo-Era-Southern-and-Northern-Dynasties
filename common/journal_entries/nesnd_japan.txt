# Wu holds part of the home islands while Japan survives. Six months of peace open a war of annexation.
nesnd_wu_eastern_expedition = {
    icon = "gfx/interface/icons/event_icons/nesnd_usm_war_junk.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    possible = { nesnd_eastern_expedition_ready = yes }
    is_shown_when_inactive = { nesnd_eastern_expedition_context = yes }
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_eastern_expedition_war_running = yes } desc = nesnd_wu_eastern_expedition_status_war }
            triggered_desc = { trigger = { nesnd_eastern_expedition_ready = yes } desc = nesnd_wu_eastern_expedition_status_count }
            desc = nesnd_wu_eastern_expedition_status_wait
        }
    }
    scripted_progress_bar = nesnd_eastern_expedition_bar
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_eastern_expedition_done_conclusion
        trigger = { custom_tooltip = {
            text = nesnd_eastern_expedition_done_tt
            NOT = { c:JAP ?= { is_country_alive = yes } }
        } }
        effect = { nesnd_eastern_expedition_concluded = yes }
        event_outcome_desc = { desc = nesnd_eastern_expedition_done_effect_tt }
    }
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_eastern_expedition_invalid_conclusion
        trigger = {
            c:JAP ?= { is_country_alive = yes }
            NOT = { nesnd_eastern_expedition_context = yes }
        }
    }
    # The bar has already counted this month. A full bar waits for peace, then opens the war and starts again.
    on_monthly_pulse = { effect = {
        if = { limit = {
                scope:journal_entry ?= { "scripted_bar_progress(nesnd_eastern_expedition_bar)" >= 6 }
                nesnd_eastern_expedition_ready = yes
            }
            nesnd_eastern_expedition_open = yes
            scope:journal_entry ?= { set_bar_progress = { name = nesnd_eastern_expedition_bar value = 0 } }
        }
    } }
    weight = 890
    should_be_pinned_by_default_uninvolved_or_context = yes
}
