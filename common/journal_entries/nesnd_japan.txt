nesnd_wu_eastern_question = {
    icon = "gfx/interface/icons/politics_icons/revolution.dds"
    group = je_group_historical_content
    can_revolution_inherit = no
    possible = {
        c:USM ?= this has_variable = tag_WUU_var
        has_global_variable = nesnd_south_settled
        c:JAP ?= { is_country_alive = yes }
        NOT = { has_variable = nesnd_japan_dossier_entered }
        NOT = { has_global_variable = nesnd_japan_dossier_closed }
    }
    is_shown_when_inactive = { c:USM ?= this has_variable = tag_WUU_var }
    immediate = { set_variable = nesnd_japan_dossier_entered set_variable = nesnd_eastern_absolute_v2  scope:journal_entry ?= { set_bar_progress = { name = nesnd_eastern_readiness_bar value = owner.nesnd_japan_preparation_score } } }
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = nesnd_japan_war_started } desc = nesnd_wu_eastern_question_status_war }
            triggered_desc = { trigger = { has_global_variable = nesnd_japan_offer_sent } desc = nesnd_wu_eastern_question_status_offer }
            desc = nesnd_wu_eastern_question_status
        }
    }
    scripted_progress_bar = nesnd_eastern_readiness_bar
    scripted_button = nesnd_japan_send_terms
    scripted_button = nesnd_japan_accept_status_quo
    complete = { custom_tooltip = { text = nesnd_eastern_question_resolved_tt has_global_variable = nesnd_japan_dossier_closed } }
    on_complete = { }
    invalid = { OR = { NOT = { has_global_variable = nesnd_south_settled } NOT = { has_variable = tag_WUU_var } NOT = { has_variable = nesnd_eastern_absolute_v2 } } }
    on_invalid = { if = { limit = { NOT = { has_variable = tag_WUU_var } } set_global_variable = nesnd_japan_dossier_closed } else = { remove_variable = nesnd_japan_dossier_entered } }
    on_monthly_pulse = { effect = {
            if = { limit = {
                    NOT = { has_global_variable = nesnd_japan_dossier_closed }
                    OR = {
                        NOT = { c:JAP ?= { is_country_alive = yes } }
                        AND = {
                            has_global_variable = nesnd_japan_war_started
                            NOT = { has_variable = nesnd_japan_war_grace }
                            NOT = { has_war_with = c:JAP }
                            NOT = { any_diplomatic_play = { has_variable = nesnd_eastern_intervention_play } }
                        }
                    }
                }
                set_global_variable = nesnd_japan_dossier_closed
                trigger_event = { id = nesnd_japan.4 popup = yes }
            }
        }
    }
    weight = 900
    should_be_pinned_by_default_uninvolved_or_context = yes
}
