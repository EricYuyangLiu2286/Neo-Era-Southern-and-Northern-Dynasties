# Given to a government that must answer an Indian prince: the overlord whose consent is needed,
# or the government asked for money. Added only by nesnd_india_matter_receive; it has neither
# possible nor is_shown_when_inactive, so no instance exists until then.
# The matter itself lives on the prince (nesnd_india_project_*); this government only holds
# nesnd_india_matter_from. Thirty days without an answer count as no answer.
nesnd_india_foreign_affairs = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_theodolite.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    timeout = 30
    weight = 500
    should_be_pinned_by_default_uninvolved_or_context = yes

    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_india_matter_consent_role = yes nesnd_india_matter_is_survey = yes } desc = nesnd_india_matter_status_consent_survey }
            triggered_desc = { trigger = { nesnd_india_matter_consent_role = yes nesnd_india_matter_is_grant = yes } desc = nesnd_india_matter_status_consent_grant }
            triggered_desc = { trigger = { nesnd_india_matter_consent_role = yes } desc = nesnd_india_matter_status_consent_stake }
            triggered_desc = { trigger = { nesnd_india_matter_fund_role = yes nesnd_india_matter_is_survey = yes } desc = nesnd_india_matter_status_fund_survey }
            triggered_desc = { trigger = { nesnd_india_matter_fund_role = yes nesnd_india_matter_is_grant = yes } desc = nesnd_india_matter_status_fund_grant }
            triggered_desc = { trigger = { nesnd_india_matter_fund_role = yes } desc = nesnd_india_matter_status_fund_stake }
            desc = nesnd_india_matter_status_done
        }
    }

    scripted_button = nesnd_india_matter_permit_button
    scripted_button = nesnd_india_matter_forbid_button
    scripted_button = nesnd_india_matter_fund_button
    scripted_button = nesnd_india_matter_decline_button

    outcome = {
        header = nesnd_india_matter_answered_header
        effect_header = nesnd_india_matter_answered_effect_header
        color = "gold_positive"
        conclusion_desc = nesnd_india_matter_answered_conclusion
        trigger = { custom_tooltip = { text = nesnd_india_matter_answered_tt has_variable = nesnd_india_matter_answered } }
        effect = {
            custom_tooltip = nesnd_india_matter_closed_tt
            hidden_effect = { remove_variable = nesnd_india_matter_answered remove_variable = nesnd_india_matter_from }
        }
    }
    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        flags = { timeout }
        conclusion_desc = nesnd_india_matter_timeout_conclusion
        effect = {
            custom_tooltip = nesnd_india_matter_timeout_tt
            hidden_effect = { nesnd_india_matter_lapse = yes }
        }
    }
    # The prince withdrew, or the matter no longer stands.
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_india_matter_void_conclusion
        trigger = { NOT = { has_variable = nesnd_india_matter_answered } NOT = { nesnd_india_matter_live = yes } }
        effect = { remove_variable = nesnd_india_matter_from }
    }

    on_weekly_pulse = {
        effect = {
            if = {
                limit = { is_ai = yes NOT = { has_variable = nesnd_india_matter_answered } nesnd_india_matter_live = yes }
                nesnd_india_matter_ai_answer = yes
            }
        }
    }
}
