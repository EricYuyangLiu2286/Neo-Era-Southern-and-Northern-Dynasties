# Ottoman success line, block 3: "Academies and the Press" (four-ring sequential milestone chain, vanilla 00_liberalism.txt shape).
# Ring 1 is added by event nesnd_ottoman_recovery.5 (on_action nesnd_ottoman_academy_fallback as safety net); rings 2-4 are added in sequence by the previous ring's completion.
# No possible or is_shown_when_inactive on any ring. All four rings share a single 300-month window (nesnd_ottoman_academy_window).

# Ring 1: The University
nesnd_ottoman_academy_1 = {
    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
    group = je_group_internal_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes

    immediate = {
        set_variable = { name = nesnd_ottoman_academy_window months = 300 }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_ottoman_academy_1_win_conclusion
        trigger = {
            OR = {
                # The vanilla education goal of the Tanzimat already did this work (variable set by je_sick_man_education,
                # GAME common/journal_entries/00_sick_man.txt:355, never cleared).
                custom_tooltip = { text = nesnd_ottoman_academy_1_schools_done_tt has_variable = completed_sick_man_education_je }
                AND = {
                    any_scope_building = {
                        is_building_type = building_university
                        level >= 10
                    }
                    literacy_rate >= 0.4
                }
            }
        }
        effect = {
            add_modifier = { name = nesnd_ottoman_academy_ring1 }
            add_modifier = { name = nesnd_ottoman_academy_ulema_backlash months = 60 }
            add_journal_entry = { type = nesnd_ottoman_academy_2 }
            if = {
                limit = {
                    year >= 1860
                    year <= 1888
                    NOT = { has_variable = nesnd_ottoman_namik_kemal_fired }
                }
                trigger_event = { id = nesnd_ottoman_recovery.16 }
            }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = {
                    trigger = {
                        year >= 1860
                        year <= 1888
                        NOT = { has_variable = nesnd_ottoman_namik_kemal_fired }
                    }
                    desc = nesnd_ottoman_academy_1_win_event_kemal_tt
                }
                triggered_desc = { desc = nesnd_ottoman_academy_1_win_event_tt }
            }
        }
    }

    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        color = "blue_negative"
        flags = { timeout }
        conclusion_desc = nesnd_ottoman_academy_ulema_conclusion
        effect = {
            nesnd_ottoman_academy_ulema_win = yes
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_ottoman_academy_ulema_event_tt }
            }
        }
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_ottoman_academy_invalid_conclusion
        trigger = { OR = { NOT = { c:TUR ?= this } NOT = { nesnd_ottoman_tanzimat_success = yes } } }
        effect = { if = { limit = { has_variable = nesnd_ottoman_academy_window } remove_variable = nesnd_ottoman_academy_window } set_variable = nesnd_ottoman_academy_closed }
    }

    timeout = 9125

    status_desc = {
        first_valid = {
            desc = nesnd_ottoman_academy_1_status
        }
    }
}

# Ring 4: The Arts Academy (final ring)
nesnd_ottoman_academy_4 = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_ottoman_academy_4_win_conclusion
        trigger = {
            any_scope_building = {
                is_building_type = building_art_academy
                level >= 1
            }
        }
        effect = {
            add_modifier = { name = nesnd_ottoman_academy_ring4 }
            add_modifier = { name = nesnd_ottoman_academy_ulema_backlash months = 60 }
            add_modifier = { name = nesnd_ottoman_enlightenment }
            ig:ig_intelligentsia ?= {
                if = {
                    limit = { has_ideology = ideology:ideology_liberal }
                    remove_ideology = ideology_liberal
                    add_ideology = ideology_liberal_modern
                }
                else_if = {
                    limit = { NOT = { has_ideology = ideology:ideology_egalitarian } }
                    add_ideology = ideology_egalitarian
                }
            }
            if = {
                limit = { has_journal_entry = nesnd_ottoman_charter }
                custom_tooltip = {
                    text = nesnd_ottoman_charter_bonus_charter_gain_tt
                    set_variable = { name = nesnd_ottoman_charter_bonus months = 60 }
                }
            }
            set_variable = { name = nesnd_ottoman_academy_outcome value = flag:complete }
            set_variable = nesnd_ottoman_academy_closed
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_ottoman_academy_4_win_event_tt }

        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_academy_ulema_conclusion

            custom_tooltip = {
                text = nesnd_ottoman_academy_timeout_tt
                NOT = { has_variable = nesnd_ottoman_academy_window }

        }
        effect = {
            nesnd_ottoman_academy_ulema_win = yes
        }
        event_outcome_desc = {
            first_valid = {
