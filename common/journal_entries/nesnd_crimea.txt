# Crimean War global crisis journals (pattern #27): four country journals (TUR, RUS, FRA, QE1) added only by nesnd_crimea_start.

nesnd_crimea_tur = {
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    group = je_group_foreign_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes
    scripted_button = nesnd_crimea_invite_fra
    scripted_button = nesnd_crimea_invite_qe1

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_crimea_tur_win_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_crimea_tur_win_tt
                nesnd_crimea_tur_won = yes
            }
        }
        effect = {
            set_global_variable = { name = nesnd_crimea_result value = flag:ottoman }
            if = { limit = { has_global_variable = nesnd_crimea_active } remove_global_variable = nesnd_crimea_active }
            add_modifier = { name = nesnd_crimea_victory }
            nesnd_crimea_claim_remaining_targets = yes
            trigger_event = { id = nesnd_ottoman_recovery.14 }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_crimea_tur_win_event_tt }
            }
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_crimea_tur_loss_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_crimea_rus_win_tt
                nesnd_crimea_rus_won = yes
            }
        }
        effect = {
            set_global_variable = { name = nesnd_crimea_result value = flag:russia }
            if = { limit = { has_global_variable = nesnd_crimea_active } remove_global_variable = nesnd_crimea_active }
            add_modifier = { name = nesnd_crimea_defeat }
            if = {
                limit = { has_journal_entry = nesnd_ottoman_charter }
                custom_tooltip = {
                    text = nesnd_ottoman_charter_bonus_charter_gain_tt
                    set_variable = { name = nesnd_ottoman_charter_bonus months = 60 }
                }
            }
            trigger_event = { id = nesnd_ottoman_recovery.15 }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_crimea_tur_loss_event_tt }
            }
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_crimea_inconclusive_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_crimea_ended_inconclusive_tt
                exists = c:RUS
                nesnd_crimea_play_in_progress = no
                NOT = { has_war_with = c:RUS }
                nesnd_crimea_tur_won = no
                nesnd_crimea_rus_won = no
            }
        }
        effect = {
            set_global_variable = { name = nesnd_crimea_result value = flag:none }
            if = { limit = { has_global_variable = nesnd_crimea_active } remove_global_variable = nesnd_crimea_active }
            add_modifier = { name = nesnd_crimea_stalemate }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_crimea_stalemate_event_tt }
            }
        }
    }

    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        color = "blue_negative"
        flags = { timeout }
        conclusion_desc = nesnd_crimea_stalemate_conclusion
        effect = {
            set_global_variable = { name = nesnd_crimea_result value = flag:none }
            if = { limit = { has_global_variable = nesnd_crimea_active } remove_global_variable = nesnd_crimea_active }
            add_modifier = { name = nesnd_crimea_stalemate }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_crimea_stalemate_event_tt }
            }
        }
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_crimea_invalid_conclusion
        trigger = { NOT = { c:TUR ?= this } }
        effect = {
            if = { limit = { has_global_variable = nesnd_crimea_active } remove_global_variable = nesnd_crimea_active }
            if = { limit = { NOT = { has_global_variable = nesnd_crimea_result } } set_global_variable = { name = nesnd_crimea_result value = flag:none } }
        }
    }

    timeout = 3650

    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { exists = c:RUS has_war_with = c:RUS } desc = nesnd_crimea_status_war }
            triggered_desc = { trigger = { exists = c:RUS is_diplomatic_play_enemy_of = c:RUS } desc = nesnd_crimea_status_play }
            desc = nesnd_crimea_status_lull
        }
    }
}

    icon = "gfx/interface/icons/event_icons/event_military.dds"
    group = je_group_foreign_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes

    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = nesnd_crimea_joined_tur }
                    exists = c:TUR
                    exists = c:RUS
                    OR = {
                        is_in_war_together = c:TUR
                        is_diplomatic_play_ally_of = c:TUR
                    }
                    OR = {
                        has_war_with = c:RUS
                        is_diplomatic_play_enemy_of = c:RUS

            if = {
                limit = { has_variable = nesnd_crimea_joined_tur }
                add_modifier = { name = nesnd_crimea_ally_victory }
                c:TUR ?= { change_relations = { country = root value = 25 } }
