    immediate = { nesnd_ten_initialize = yes }

    scripted_button = nesnd_ten_abolish_button

    scripted_progress_bar = nesnd_ten_continuity_bar

    } }

    on_invalid = { nesnd_ten_cleanup = yes }
    weight = 150 should_be_pinned_by_default_uninvolved_or_context = yes
