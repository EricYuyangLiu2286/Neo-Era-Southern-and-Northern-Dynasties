nesnd_tianzhu_governance = { icon = "gfx/interface/icons/event_icons/event_newspaper.dds" group = je_group_historical_content can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_tianzhu_governance_live = yes } possible = { nesnd_tianzhu_governance_live = yes }
    complete = { hidden_trigger = { nesnd_tianzhu_governance_live = yes }
        institution_investment_level = { institution = institution_schools value >= 1 } literacy_rate >= 0.2
        country_has_building_type_levels = { target = bt:building_port value >= 3 }
        country_has_building_type_levels = { target = bt:building_government_administration value >= 5 }
        bureaucracy >= 0
    }
    on_complete = { if = { limit = { nesnd_tianzhu_governance_live = yes } set_variable = nesnd_commission_governance_done add_modifier = { name = nesnd_commission_governance_reward months = 120 } } }
    invalid = { NOT = { nesnd_tianzhu_governance_live = yes } }
    weight = 650
    scripted_button = nesnd_india_circars_suppress
    scripted_button = nesnd_india_kurnool_suppress
    scripted_button = nesnd_india_madras_suppress
    scripted_button = nesnd_india_mysore_suppress
