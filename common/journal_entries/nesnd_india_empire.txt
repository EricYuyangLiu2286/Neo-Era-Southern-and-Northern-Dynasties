# ==== The centre and the states (formerly nesnd_india_order.txt) ====
# Mughal ("Mughal" here is any country for which nesnd_india_empire holds) and its Indian subjects: a
# central or a local arrangement in tax, in the army and foreign affairs, and in schools and offices.
# Added and taken back each month by nesnd_india_order_upkeep from nesnd_india_refresh; it has neither
# possible nor is_shown_when_inactive, so no instance exists until then. Its text is the single key
# nesnd_india_order_reason; the arrangement now in force is shown by the three pairs of buttons and
# their modifiers.
nesnd_india_order = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_weights.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 750
    should_be_pinned_by_default_uninvolved_or_context = yes
    scripted_button = nesnd_india_order_tax_central_button
    scripted_button = nesnd_india_order_tax_local_button
    scripted_button = nesnd_india_order_army_central_button
    scripted_button = nesnd_india_order_army_local_button
    scripted_button = nesnd_india_order_school_central_button
    scripted_button = nesnd_india_order_school_local_button
    # Ten years with all three matters central. The arrangement stays as it is and the journal does not
    # come back (nesnd_india_order_settled). The event nesnd_india_core.4 asks, member by member, to annex it or keep it.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_india_order_central_conclusion
        trigger = {
            hidden_trigger = { nesnd_india_order_live = yes }
            custom_tooltip = {
                text = nesnd_india_order_central_ten_tt
                has_variable = nesnd_india_order_central_since
                NOT = { has_variable = nesnd_india_order_clock_wait }
            }
        }
        effect = {
            set_variable = {
                name = nesnd_india_order_settled
                value = 1
            }
            remove_variable = nesnd_india_order_central_since
            add_modifier = { name = nesnd_india_order_central_settled }
            save_scope_as = india_order_lord
            every_subject_or_below = {
                limit = {
                    nesnd_india_order_member = { LORD = scope:india_order_lord }
                }
                root = {
                    add_to_variable_list = {
                        name = nesnd_india_order_annex_queue
                        target = prev
                    }
                }
            }
            trigger_event = {
                id = nesnd_india_core.4
                popup = yes
            }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_india_core.4_outcome_tt }
            }
        }
    }
    # Ten years with all three matters local.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_india_order_federal_conclusion
        trigger = {
            hidden_trigger = { nesnd_india_order_live = yes }
            custom_tooltip = {
                text = nesnd_india_order_local_ten_tt
                has_variable = nesnd_india_order_local_since
                NOT = { has_variable = nesnd_india_order_clock_wait }
            }
        }
        effect = {
            set_variable = {
                name = nesnd_india_order_settled
                value = 2
            }
            remove_variable = nesnd_india_order_local_since
            add_modifier = { name = nesnd_india_order_federal_settled }
            save_scope_as = india_lobby_lord
            every_subject_or_below = {
                limit = {
                    nesnd_india_order_member = { LORD = scope:india_lobby_lord }
                }
                change_relations = {
                    country = scope:india_lobby_lord
                    value = 30
                }
                custom_tooltip = {
                    text = nesnd_india_back_empire_lobby_tt
                    nesnd_india_back_empire_lobby = yes
                }
            }
        }
    }
    # Mughal no longer holds it (no Indian subject left, or no longer Mughal): everything is cleared.
    # The journal comes back by nesnd_india_order_upkeep once Mughal holds it again.
    outcome = {
        flags = {
            hidden
            invalid_outcome
        }
        conclusion_desc = nesnd_india_order_void_conclusion
        trigger = {
            NOT = { nesnd_india_order_live = yes }
        }
        effect = { nesnd_india_order_end = yes }
    }
    on_monthly_pulse = {
        effect = {
            nesnd_india_order_ai = yes
            nesnd_india_order_clock = yes
        }
    }
}

# ==== Mughal's old agreement, the council of princes, famine relief (formerly nesnd_india_mughal.txt) ====
# Mughal's own journals (common/scripted_effects/nesnd_india_empire.txt). Only the country holding
# nesnd_india_mughal_court gets them. Both are added each month by nesnd_mug_upkeep from
# nesnd_india_refresh; they have neither possible nor is_shown_when_inactive, so no instance exists
# until then.

# The old agreement with the overlord: selling one building level to it, or buying one back.
nesnd_mug_france = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_slate.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 760
    should_be_pinned_by_default_uninvolved_or_context = yes

    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    nesnd_india_request_kind = 10
                    has_variable = nesnd_india_request_clock
                }
                desc = nesnd_mug_france_status_sell_asked
            }
            triggered_desc = {
                trigger = {
                    nesnd_india_request_kind = 11
                    has_variable = nesnd_india_request_clock
                }
                desc = nesnd_mug_france_status_buyback_asked
            }
        }
    }

    scripted_button = nesnd_mug_sell_button
    scripted_button = nesnd_mug_buyback_button
    # No longer a subject, or the overlord's capital is now in an Indian state (nesnd_mug_foreign_overlord).
    # The old overlord is the one remembered by nesnd_mug_upkeep.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_mug_france_ended_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_mug_france_closure_tt
                NOT = { nesnd_mug_france_live = yes }
            }
        }
        effect = {
            add_modifier = { name = nesnd_mug_old_agreement_ended }
            if = {
                limit = {
                    exists = var:nesnd_mug_france_lord
                    var:nesnd_mug_france_lord = { is_country_alive = yes }
                }
                var:nesnd_mug_france_lord = { save_scope_as = mug_old_lord }
                change_relations = {
                    country = scope:mug_old_lord
                    value = -30
                }
            }
            else = { custom_tooltip = nesnd_journal_effect_revealed_tt }
            remove_variable = nesnd_mug_france_lord
        }
    }

    on_monthly_pulse = {
        effect = { nesnd_mug_france_ai = yes }
    }
}

# The council of princes, after the reform: railways, famine relief or common defence. The button
# nesnd_india_request_6 invites a prince to join the empire from here. The journal ends when India has no
# independent prince left, and none that belongs to another country.
nesnd_mug_council = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_chhatri.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 755
    should_be_pinned_by_default_uninvolved_or_context = yes
    reason = {
        first_valid = {
            triggered_desc = {
                trigger = { has_variable = nesnd_mug_council_topic }
                desc = nesnd_mug_council_reason_sitting
            }
            triggered_desc = {
                trigger = { has_variable = nesnd_mug_council_wait }
                desc = nesnd_mug_council_reason_rest
            }
            desc = nesnd_mug_council_reason_ready
        }
    }
    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    nesnd_india_request_kind = 6
                    nesnd_india_request_stage_value = 1
                    exists = var:nesnd_india_request_await
                    exists = var:nesnd_india_request_partner
                }
                desc = nesnd_india_union_status_invite_consent
            }
            triggered_desc = {
                trigger = {
                    nesnd_india_request_kind = 6
                    nesnd_india_request_stage_value = 2
                    exists = var:nesnd_india_request_partner
                }
                desc = nesnd_india_union_status_invite_sent
            }
        }
    }
    scripted_button = nesnd_mug_council_button
    scripted_button = nesnd_india_request_6
    # No independent prince is left in India, and none belongs to another country.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_mug_council_complete_conclusion
        trigger = {
            hidden_trigger = { nesnd_mug_council_live = yes }
            custom_tooltip = {
                text = nesnd_mug_council_no_free_prince_tt
                NOT = {
                    any_country = {
                        nesnd_india_prince = yes
                        NOT = { is_subject_of = root }
                    }
                }
            }
        }
        effect = {
            set_variable = nesnd_mug_council_done
            add_modifier = { name = nesnd_mug_council_established }
        }
    }
    # No longer Mughal past the reform.
    outcome = {
        flags = {
            hidden
            invalid_outcome
        }
        conclusion_desc = nesnd_mug_council_void_conclusion
        trigger = {
            NOT = { nesnd_mug_council_live = yes }
        }
    }
    on_monthly_pulse = {
        effect = { nesnd_mug_council_ai = yes }
    }
}

# ==== Old treaty dispute (India batch 15, one global journal in the optimisation batch 2) ====
# One global journal that Mughal and its overlord read together. The event nesnd_mug.30 adds it with
# add_contextless_journal_entry once France questions the old treaty; the two countries are kept in the global
# variables nesnd_india_dispute_g_subject and nesnd_india_dispute_g_lord, and in the country variables
# nesnd_india_dispute_overlord (on Mughal) and nesnd_india_dispute_subject (on France). Mughal sees the
# buttons petition and withhold, the overlord the buttons reaffirm and concede. It has neither possible nor
# is_shown_when_inactive, so it can reopen twenty years after a renewal.
nesnd_india_treaty_dispute = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_slate.dds"
    group = je_group_global_international_situations
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 770
    should_be_pinned_by_default_involved = yes
    should_be_pinned_by_default_uninvolved_or_context = no
    is_shown_when_inactive = { always = no }
    should_be_involved = {
        OR = {
            has_variable = nesnd_india_dispute_overlord
            has_variable = nesnd_india_dispute_subject
        }
    }
    should_show_when_not_involved = { always = no }
    reason = {
        first_valid = {
            triggered_desc = {
                trigger = { has_variable = nesnd_india_dispute_overlord }
                desc = nesnd_india_treaty_dispute_reason
            }
            desc = nesnd_india_treaty_dispute_overlord_reason
        }
    }
    scripted_progress_bar = nesnd_india_treaty_dispute_bar
    scripted_button = nesnd_india_dispute_petition_button
    scripted_button = nesnd_india_dispute_withhold_button
    scripted_button = nesnd_india_dispute_reaffirm_button
    scripted_button = nesnd_india_dispute_concede_button
    # Bar reaches 100: treaty renewed.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_india_dispute_renew_conclusion
        trigger = {
            hidden_trigger = { nesnd_india_dispute_active = yes }
            custom_tooltip = {
                text = nesnd_india_dispute_renew_trigger_tt
                scope:journal_entry = { "scripted_bar_progress(nesnd_india_treaty_dispute_bar)" >= 100 }
            }
        }
        effect = {
            global_var:nesnd_india_dispute_g_subject ?= { save_scope_as = india_dispute_subject }
            global_var:nesnd_india_dispute_g_lord ?= { save_scope_as = india_dispute_lord }
            # Mughal: the renewal, once for both countries' relations.
            scope:india_dispute_subject ?= {
                add_modifier = { name = nesnd_india_treaty_renewed }
                change_relations = {
                    country = scope:india_dispute_lord
                    value = 30
                }
                custom_tooltip = nesnd_india_dispute_renew_wait_tt
            }
            # The overlord: influence, and involvement in an Indian strategic region where it holds no state (at start, region_south_india).
            scope:india_dispute_lord ?= {
                add_modifier = { name = nesnd_india_treaty_renewed_overlord }
                if = {
                    limit = {
                        OR = {
                            NOT = {
                                any_scope_state = { region = sr:region_south_india }
                            }
                            NOT = {
                                any_scope_state = { region = sr:region_north_india }
                            }
                        }
                    }
                    custom_tooltip = {
                        text = nesnd_india_dispute_renew_involvement_tt
                        if = {
                            limit = {
                                NOT = {
                                    any_scope_state = { region = sr:region_south_india }
                                }
                            }
                            add_involvement = {
                                strategic_region = sr:region_south_india
                                value = 1000
                            }
                        }
                        if = {
                            limit = {
                                NOT = {
                                    any_scope_state = { region = sr:region_north_india }
                                }
                            }
                            add_involvement = {
                                strategic_region = sr:region_north_india
                                value = 1000
                            }
                        }
                    }
                }
                else = { custom_tooltip = nesnd_journal_effect_revealed_tt }
            }
            scope:india_dispute_subject ?= {
                clear_saved_scope = mug_dispute_partner
                scope:india_dispute_lord ?= { save_scope_as = mug_dispute_partner }
                trigger_event = {
                    id = nesnd_mug.31
                    popup = yes
                }
            }
            scope:india_dispute_lord ?= {
                clear_saved_scope = mug_dispute_partner
                scope:india_dispute_subject ?= { save_scope_as = mug_dispute_partner }
                trigger_event = {
                    id = nesnd_mug.31
                    popup = yes
                }
            }
            hidden_effect = {
                set_global_variable = {
                    name = nesnd_india_treaty_dispute_wait
                    days = 7300
                }
                set_global_variable = nesnd_india_treaty_dispute_resolved
                scope:india_dispute_subject ?= { nesnd_india_dispute_cleanup_subject = yes }
                scope:india_dispute_lord ?= { nesnd_india_dispute_cleanup_overlord = yes }
                remove_global_variable = nesnd_india_dispute_g_subject
                remove_global_variable = nesnd_india_dispute_g_lord
            }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_mug.31_outcome_tt }
            }
        }
    }
    # Bar reaches 0: treaty voided.
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_india_dispute_void_conclusion
        trigger = {
            hidden_trigger = { nesnd_india_dispute_active = yes }
            custom_tooltip = {
                text = nesnd_india_dispute_void_trigger_tt
                scope:journal_entry = { "scripted_bar_progress(nesnd_india_treaty_dispute_bar)" <= 0 }
            }
        }
        effect = {
            global_var:nesnd_india_dispute_g_subject ?= { save_scope_as = india_dispute_subject }
            global_var:nesnd_india_dispute_g_lord ?= { save_scope_as = india_dispute_lord }
            # Mughal: the repudiation, and the relations of both countries once.
            scope:india_dispute_subject ?= {
                add_modifier = {
                    name = nesnd_india_treaty_void
                    days = long_modifier_time
                }
                change_relations = {
                    country = scope:india_dispute_lord
                    value = -30
                }
                custom_tooltip = {
                    text = nesnd_india_dispute_void_fervor_tt
                    save_scope_as = mug_dispute_fervor_country
                    every_primary_culture = {
                        scope:mug_dispute_fervor_country = {
                            add_fervor_target_modifier = {
                                culture = prev
                                days = 3650
                                multiplier = 10
                            }
                        }
                    }
                }
                if = {
                    limit = { has_journal_entry = nesnd_india_indep }
                    custom_tooltip = {
                        text = nesnd_india_dispute_void_indep_tt
                        je:nesnd_india_indep ?= {
                            add_progress = {
                                name = nesnd_india_indep_economy_bar
                                value = 10
                            }
                            add_progress = {
                                name = nesnd_india_indep_military_bar
                                value = 10
                            }
                        }
                    }
                }
                else = { custom_tooltip = nesnd_journal_effect_revealed_tt }
            }
            # The overlord: a claim on Mughal's capital state, a negative catalyst, and the AI's secret goal.
            scope:india_dispute_lord ?= {
                custom_tooltip = {
                    text = nesnd_india_dispute_void_claim_tt
                    scope:india_dispute_subject ?= {
                        capital ?= {
                            state_region = {
                                if = {
                                    limit = {
                                        NOT = { has_claim_by = scope:india_dispute_lord }
                                    }
                                    add_claim = scope:india_dispute_lord
                                }
                            }
                        }
                    }
                }
                create_diplomatic_catalyst = {
                    type = catalyst_event_negative
                    target = scope:india_dispute_subject
                }
                hidden_effect = {
                    if = {
                        limit = { is_ai = yes }
                        # While Mughal is still our subject, use dominate: vanilla sets it on an overlord's own subject
                        # (austria_federalism.txt:2483) and weights direct subjects for it (00_default_strategy.txt:9120-9129).
                        if = {
                            limit = {
                                scope:india_dispute_subject = { is_subject_of = scope:india_dispute_lord }
                            }
                            set_secret_goal = {
                                country = scope:india_dispute_subject
                                secret_goal = dominate
                            }
                        }
                        else = {
                            set_secret_goal = {
                                country = scope:india_dispute_subject
                                secret_goal = conquer
                            }
                        }
                    }
                }
            }
            scope:india_dispute_subject ?= {
                clear_saved_scope = mug_dispute_partner
                scope:india_dispute_lord ?= { save_scope_as = mug_dispute_partner }
                trigger_event = {
                    id = nesnd_mug.32
                    popup = yes
                }
            }
            scope:india_dispute_lord ?= {
                clear_saved_scope = mug_dispute_partner
                scope:india_dispute_subject ?= { save_scope_as = mug_dispute_partner }
                trigger_event = {
                    id = nesnd_mug.32
                    popup = yes
                }
            }
            hidden_effect = {
                set_global_variable = {
                    name = nesnd_india_treaty_dispute_wait
                    days = 3650
                }
                set_global_variable = nesnd_india_treaty_dispute_resolved
                scope:india_dispute_subject ?= { nesnd_india_dispute_cleanup_subject = yes }
                scope:india_dispute_lord ?= { nesnd_india_dispute_cleanup_overlord = yes }
                remove_global_variable = nesnd_india_dispute_g_subject
                remove_global_variable = nesnd_india_dispute_g_lord
            }
        }
        event_outcome_desc = {
            first_valid = {
                triggered_desc = { desc = nesnd_mug.32_outcome_tt }
            }
        }
    }
    # Mughal is no longer a subject of France during the dispute, or one of them is gone: silent invalidation;
    # nesnd_mug_france settles on its own.
    outcome = {
        flags = {
            hidden
            invalid_outcome
        }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            NOT = { nesnd_india_dispute_active = yes }
        }
        effect = {
            global_var:nesnd_india_dispute_g_subject ?= { nesnd_india_dispute_cleanup_subject = yes }
            global_var:nesnd_india_dispute_g_lord ?= { nesnd_india_dispute_cleanup_overlord = yes }
            remove_global_variable = nesnd_india_dispute_g_subject
            remove_global_variable = nesnd_india_dispute_g_lord
        }
    }
    on_monthly_pulse = {
        effect = {
            global_var:nesnd_india_dispute_g_subject ?= { nesnd_india_dispute_subject_ai = yes }
            global_var:nesnd_india_dispute_g_lord ?= { nesnd_india_dispute_overlord_ai = yes }
        }
    }
}
