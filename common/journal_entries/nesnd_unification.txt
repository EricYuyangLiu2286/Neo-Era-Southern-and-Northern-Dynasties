# Global journal entry: the trend towards unifying the 98 regions. Every qualified country is involved.
nesnd_uni_trend = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_global_international_situations

	scripted_progress_bar = nesnd_uni_lead_bar
	scripted_button = nesnd_uni_join_button
	scripted_button = nesnd_uni_declare_button

	is_shown_when_inactive = {
		always = yes
	}

	possible = {
		NOT = { has_global_variable = nesnd_uni_declared }
		any_country = {
			nesnd_uni_qualified = yes
		}
	}

	can_deactivate = yes

	should_be_involved = {
		nesnd_uni_qualified = yes
	}

	should_show_when_not_involved = {
		OR = {
			nesnd_uni_capital_in_scope = yes
			any_scope_state = {
				state_region = {
					nesnd_uni_scope_region = yes
				}
			}
		}
	}

	should_be_pinned_by_default_involved = yes

	immediate = {
		nesnd_uni_refresh = yes
	}

	on_monthly_pulse = {
		effect = {
			nesnd_uni_refresh = yes
		}
	}

	on_no_longer_involved = {
		nesnd_uni_clear_country = yes
	}

	reason = nesnd_uni_trend_reason

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = nesnd_uni_trend_status_none
				trigger = {
					NOT = { has_global_variable = nesnd_uni_leader }
				}
			}
			triggered_desc = {
				desc = nesnd_uni_trend_status_order
				trigger = {
					has_global_variable = nesnd_uni_many
				}
			}
			desc = nesnd_uni_trend_status_tripod
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = nesnd_uni_trend_done
		trigger = {
			custom_tooltip = {
				text = nesnd_uni_trend_done_tt
				has_global_variable = nesnd_uni_declared
			}
		}
		effect = {
			global_var:nesnd_uni_declared ?= {
				add_modifier = {
					name = nesnd_uni_celestial
					months = 240
				}
			}
			custom_tooltip = nesnd_uni_trend_done_others_tt
		}
		effect_all_involved = {
			nesnd_uni_clear_country = yes
			if = {
				limit = {
					has_variable = nesnd_uni_joined
				}
				remove_variable = nesnd_uni_joined
			}
			if = {
				limit = {
					is_subject = no
					NOT = { global_var:nesnd_uni_declared ?= this }
				}
				add_modifier = {
					name = nesnd_uni_lost
					months = 120
				}
			}
		}
	}
}
