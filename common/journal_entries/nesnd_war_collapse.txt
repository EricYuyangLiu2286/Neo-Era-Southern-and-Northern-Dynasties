nesnd_japan_war_collapse = {
    icon = "gfx/interface/icons/politics_icons/revolution.dds"
    group = je_group_global_international_situations
    activation_description = { always = yes }
    is_shown_when_inactive = {
        NOT = { has_global_variable = nesnd_japan_collapse_journal_started }
        NOT = { has_global_variable = nesnd_japan_partitioned }
        c:USM ?= { has_variable = tag_WUU_var }
    }
    possible = {
        NOT = { has_global_variable = nesnd_japan_collapse_journal_started }
        NOT = { has_global_variable = nesnd_japan_partitioned }
        nesnd_japan_at_war_with_wu = yes
    }
    should_be_involved = { nesnd_japan_collapse_viewer = yes }
    should_show_when_not_involved = { nesnd_japan_collapse_viewer = yes }
    immediate = { set_global_variable = nesnd_japan_collapse_journal_started }
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = nesnd_japan_collapse_committed } desc = nesnd_japan_war_collapse_fallen }
            desc = nesnd_japan_war_collapse_status
        }
    }
    on_weekly_pulse = { effect = { nesnd_japan_collapse_refresh = yes c:USM ?= { nesnd_check_war_collapses = yes } } }
    outcome = {
    	header = je_standard_completion_header
    	effect_header = je_standard_on_completion_header
    	color = "gold_positive"
    	conclusion_desc = je_standard_completion_conclusion_desc
    	trigger = { has_global_variable = nesnd_japan_partitioned }
    	effect = { }
    }
    weight = 1200
    should_be_pinned_by_default_involved = yes
    should_be_pinned_by_default_uninvolved_or_context = yes
}
