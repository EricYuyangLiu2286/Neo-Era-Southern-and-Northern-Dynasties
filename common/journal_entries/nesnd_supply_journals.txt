nesnd_supply_preparations = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	can_revolution_inherit = no
	is_shown_when_inactive = {
		nesnd_supply_country = yes
		NOT = {
			c:HUS ?= this
		}
		NOT = {
			has_variable = nesnd_supply_done
		}
		NOT = {
			has_variable = nesnd_supply_dismissed
		}
	}
	possible = {
		is_country_alive = yes
	}
	immediate = {
		nesnd_supply_initialize = yes
	}
	scripted_progress_bar = nesnd_supply_readiness_bar
	scripted_progress_bar = nesnd_supply_continuity_bar
	scripted_button = nesnd_supply_invite_hna
	scripted_button = nesnd_supply_invite_sih
	scripted_button = nesnd_supply_invite_gng
	scripted_button = nesnd_supply_invite_yn1
	scripted_button = nesnd_supply_invite_cne
	scripted_button = nesnd_supply_invite_fra
	scripted_button = nesnd_supply_self
	scripted_button = nesnd_supply_end_contact
	scripted_button = nesnd_supply_trade
	scripted_button = nesnd_supply_finish
	scripted_button = nesnd_supply_finish_shared
	scripted_button = nesnd_supply_finish_autonomous
	scripted_button = nesnd_supply_dismiss
	on_weekly_pulse = {
		effect = {
			nesnd_supply_week = yes
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					is_at_war = yes
				}
				desc = nesnd_supply_status_war
			}
			triggered_desc = {
				trigger = {
					has_modifier = declared_bankruptcy
				}
				desc = nesnd_supply_status_bankruptcy
			}
			triggered_desc = {
				trigger = {
					NOT = {
						nesnd_supply_inputs_ready = yes
					}
				}
				desc = nesnd_supply_status_inputs
			}
			triggered_desc = {
				trigger = {
					c:HNA ?= this
					NOT = {
						nesnd_supply_jobs_ready = yes
					}
				}
				desc = nesnd_supply_status_jobs_hna
			}
			triggered_desc = {
				trigger = {
					c:SIH ?= this
					NOT = {
						nesnd_supply_jobs_ready = yes
					}
				}
				desc = nesnd_supply_status_jobs_sih
			}
			triggered_desc = {
				trigger = {
					c:GNG ?= this
					NOT = {
						nesnd_supply_jobs_ready = yes
					}
				}
				desc = nesnd_supply_status_jobs_gng
			}
			triggered_desc = {
				trigger = {
					c:HUS ?= this
					NOT = {
						nesnd_supply_jobs_ready = yes
					}
				}
				desc = nesnd_supply_status_jobs_hus
			}
			triggered_desc = {
				trigger = {
					NOT = {
						nesnd_supply_jobs_ready = yes
					}
				}
				desc = nesnd_supply_status_jobs
			}
			triggered_desc = {
				trigger = {
					NOT = {
						nesnd_supply_livelihood_ready = yes
					}
				}
				desc = nesnd_supply_status_livelihood
			}
			triggered_desc = {
				trigger = {
					NOT = {
						nesnd_supply_diplomacy_ready = yes
					}
				}
				desc = nesnd_supply_status_diplomacy
			}
			desc = nesnd_supply_status_stable
		}
	}
	complete = {
		has_variable = nesnd_supply_done
	}
	on_complete = {
		trigger_event = {
			id = nesnd_supply.3
			popup = yes
		}
		nesnd_supply_cleanup = yes
	}
	invalid = {
		OR = {
			has_variable = nesnd_supply_dismissed
			NOT = {
				nesnd_supply_country = yes
			}
		}
	}
	on_invalid = {
		nesnd_supply_cleanup = yes
	}
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
