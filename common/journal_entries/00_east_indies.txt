je_reform_indies = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		AND = {
			exists = c:DEI
			c:DEI = THIS
		}
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_free_trade
			has_law_or_variant = law_type:law_mercantilism
			NOT = { has_law_or_variant = law_type:law_land_based_taxation }
			NOT = { has_law_or_variant = law_type:law_serfdom }
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			OR = {
				has_law_or_variant = law_type:law_free_trade
				has_law_or_variant = law_type:law_mercantilism
			}
			NOT = { has_law_or_variant = law_type:law_land_based_taxation }
			NOT = { has_law_or_variant = law_type:law_serfdom }
			has_law_or_variant = law_type:law_slavery_banned
		}
		effect = {
			trigger_event = { id = east_indies.2 }
			show_as_tooltip = {
				add_journal_entry = {
					type = je_ethical_policy
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = bic_collapse_var
		}
		effect = {
			je_reform_indies_cleanup = yes
		}
	}

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.5
	}

	progressbar = yes

	weight = 100
}
