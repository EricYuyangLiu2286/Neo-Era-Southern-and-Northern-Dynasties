nesnd_rg_pru = {
    icon = "gfx/interface/icons/event_icons/event_industry.dds" group = je_group_historical_content can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_rg_pru_entry = yes } possible = { nesnd_rg_pru_entry = yes }
    immediate = { set_variable = nesnd_rg_prussian_origin }
    scripted_progress_bar = nesnd_rg_work
    complete = {
        hidden_trigger = { nesnd_rg_open = yes }
        nesnd_rg_pru_ready = yes
        custom_tooltip = { text = nesnd_rg_twelve_months scope:journal_entry = { "scripted_bar_progress(nesnd_rg_work)" >= 12 } }
    }
    on_complete = { nesnd_rg_finish = { EVENT = nesnd_regional.1 } }
    fail = { nesnd_rg_collapsed = yes } on_fail = { nesnd_rg_end_without_reward = yes }
    invalid = { OR = { NOT = { nesnd_rg_pru_identity = yes } has_variable = nesnd_rg_closed } }
    on_invalid = { nesnd_rg_archive = { IDENTITY = nesnd_rg_pru_identity } }
    timeout = 5475 on_timeout = { nesnd_rg_end_without_reward = yes }
    should_be_pinned_by_default_uninvolved_or_context = yes weight = 600
}
