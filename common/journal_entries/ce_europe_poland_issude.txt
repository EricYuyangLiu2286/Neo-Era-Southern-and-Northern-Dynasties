je_ce_the_polish_problem = {
	icon = "gfx/interface/icons/event_icons/nesnd_poland_settlement.dds"
	group = je_ce_poland_unification
	activation_description = { always = yes }

	is_shown_when_inactive = {
		any_country = { has_variable = ce_participated_in_polish_problem }
		NOT = {
			has_global_variable = ce_poland_problem_end
		}
		exists = c:KRA
	}

	# Claude audit 2026-11-17: guard hidden; the end flag is internal.
	possible = {
		hidden_trigger = { NOT = { has_global_variable = ce_poland_problem_end } }
		exists = c:KRA
	}

	is_shown_in_lobby = {
		custom_tooltip = { text = nesnd_ui_ce_participated_in_polish_problem_tt has_variable = ce_participated_in_polish_problem }
		custom_tooltip = { text = nesnd_ui_not_ce_poland_problem_end_tt NOT = { has_global_variable = ce_poland_problem_end } }
	}

	should_be_involved = {
		OR = {
			has_variable = ce_participated_in_polish_problem
			AND = { exists = c:KRA c:KRA ?= This }
		}
		NOT = { has_global_variable = ce_poland_problem_end }
	}

	should_show_when_not_involved = {
		always = no
	}

	# Phase is written monthly by nesnd_poland_root_month from the consensus bar (60 / 30).
	status_desc = {
		first_valid = {
			triggered_desc = { trigger = { global_var:nesnd_poland_window ?= 1 } desc = nesnd_poland_window_status_1 }
			triggered_desc = { trigger = { global_var:nesnd_poland_window ?= 2 } desc = nesnd_poland_window_status_2 }
			triggered_desc = { trigger = { global_var:nesnd_poland_window ?= 3 } desc = nesnd_poland_window_status_3 }
			triggered_desc = { trigger = { global_var:nesnd_poland_window ?= 4 } desc = nesnd_poland_window_status_4 }
			triggered_desc = { trigger = { global_var:nesnd_poland_window ?= 5 } desc = nesnd_poland_window_status_5 }
			triggered_desc = {
				desc = nesnd_poland_status_decision
				trigger = { has_global_variable = nesnd_poland_phase global_var:nesnd_poland_phase >= 3 }
			}
			triggered_desc = {
				desc = nesnd_poland_status_shaken
				trigger = { has_global_variable = nesnd_poland_phase global_var:nesnd_poland_phase = 2 }
			}
			triggered_desc = {
				desc = nesnd_poland_status_consensus
				trigger = { always = yes }
			}
		}
	}

	scripted_progress_bar = nesnd_poland_consensus_bar
	scripted_progress_bar = nesnd_poland_france_will_bar

	# 1836-01-01 + 8766 days = 1860-01-01 (author 2026-11-17: the question ends in 1860).
	timeout = 8766

	immediate = {
		# Triggered when it starts
		if = {
			limit = {
				c:KRA = { NOT = { has_variable = ce_poland_owned_polish_states } }
			}
			c:KRA = { set_variable = { name = ce_poland_owned_polish_states value = 0 } }
		}
		if = {
			limit = {
				NOT = { has_global_variable = ce_poland_supporters_count }
			}
			set_global_variable = { name = ce_poland_supporters_count value = 0 }
		}
		if = {
			limit = {
				NOT = { has_global_variable = ce_poland_allied_supporters_count }
			}
			set_global_variable = { name = ce_poland_allied_supporters_count value = 0 }
		}
		if = {
			limit = {
				NOT = { has_global_variable = ce_poland_anti_count }
			}
			set_global_variable = { name = ce_poland_anti_count value = 0 }
		}
		nesnd_poland_root_month = { ADVANCE_WINDOWS = no }
		hidden_effect = {
			c:KRA = { trigger_event = { id = ce_poland_events.98 } }
		}
	}

	immediate_all_involved = { nesnd_poland_enrol_country = yes }

	on_monthly_pulse = {
		effect = {
			nesnd_poland_refresh_participants = yes
			nesnd_poland_root_month = { ADVANCE_WINDOWS = yes }
			nesnd_poland_civil_month = yes
			nesnd_polish_exile_month = yes
			c:KRA ?= { trigger_event = { id = ce_poland_events.98 } }
		}
	}

	on_yearly_pulse = { effect = { nesnd_poland_conspiracy_year = yes } }

	# Map mode. Root scope = state.
	map_color = {
		if = {
			limit = {
				nesnd_poland_restoration_state = yes
				owner ?= { OR = { c:KRA ?= this c:POL ?= this } }
			}
			value = "nesnd_poland_held"
		}
		else_if = {
			limit = { nesnd_poland_restoration_state = yes }
			value = "nesnd_poland_missing"
		}
		else_if = {
			limit = { nesnd_poland_outer_state = yes }
			value = "nesnd_poland_outer"
		}
	}

	map_color_second = {
		if = { limit = { nesnd_poland_window_map_state = yes } value = "nesnd_poland_claim" }
	}

	map_tooltip = {
		if = { limit = { nesnd_poland_window_map_state = yes } custom_tooltip_no_bullet = nesnd_poland_window_map_tt }
		if = {
			limit = {
				nesnd_poland_restoration_state = yes
				owner ?= { OR = { c:KRA ?= this c:POL ?= this } }
			}
			custom_tooltip_no_bullet = nesnd_poland_map_held_tt
		}
		else_if = {
			limit = { nesnd_poland_restoration_state = yes }
			custom_tooltip_no_bullet = nesnd_poland_map_missing_tt
		}
		else_if = {
			limit = { nesnd_poland_outer_state = yes }
			custom_tooltip_no_bullet = nesnd_poland_map_outer_tt
		}
		if = {
			limit = {
				OR = { nesnd_poland_restoration_state = yes nesnd_poland_outer_state = yes }
				exists = c:KRA
				has_claim_by = c:KRA
				NOT = { owner ?= c:KRA }
			}
			custom_tooltip_no_bullet = nesnd_poland_map_claimed_tt
		}
		if = {
			limit = {
				OR = { nesnd_poland_restoration_state = yes nesnd_poland_outer_state = yes }
				owner ?= { custom_tooltip = { text = nesnd_ui_ce_pro_polish_variable_tt has_variable = ce_pro_polish_variable } }
			}
			custom_tooltip_no_bullet = nesnd_poland_map_owner_pro_tt
		}
		else_if = {
			limit = {
				OR = { nesnd_poland_restoration_state = yes nesnd_poland_outer_state = yes }
				owner ?= { custom_tooltip = { text = nesnd_ui_ce_anti_polish_variable_tt has_variable = ce_anti_polish_variable } }
			}
			custom_tooltip_no_bullet = nesnd_poland_map_owner_anti_tt
		}
	}

	# Restoration: Krakow declared victory (button needs the five states and Warsaw as capital).
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		# Claude audit 2026-11-17: the panel showed the three partition tests (negated) and a raw
		# "country has variable" row in this list. Partition precedence and the POL check are hidden
		# guards; the player sees the five states, Warsaw and the declaration as named lines.
		trigger = {
			hidden_trigger = {
				exists = c:KRA
				NOT = { exists = c:POL }
				nesnd_poland_partitioned = no
			}
			c:KRA ?= {
				nesnd_poland_restoration_met = yes
				custom_tooltip = { text = nesnd_poland_victory_declared_tt has_variable = ce_polish_victory_declared }
			}
		}
		effect = {
			nesnd_poland_set_result = { RESULT = restored }
		}
		effect_all_involved = {
			nesnd_poland_snapshot_stance = yes
			trigger_event = { id = ce_poland_events.10 popup = yes }
			if = {
				limit = { custom_tooltip = { text = nesnd_ui_ce_participated_in_polish_problem_tt has_variable = ce_participated_in_polish_problem } }
				remove_variable = ce_participated_in_polish_problem
			}
		}
	}

	# Partition: Krakow gone, a subject of Russia, Prussia or Austria, or left without its core.
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			nesnd_poland_partitioned = yes
		}
		effect = {
			nesnd_poland_set_result = { RESULT = partition }
		}
		effect_all_involved = {
			nesnd_poland_snapshot_stance = yes
			trigger_event = { id = ce_poland_events.11 popup = yes }
			if = {
				limit = { custom_tooltip = { text = nesnd_ui_ce_participated_in_polish_problem_tt has_variable = ce_participated_in_polish_problem } }
				remove_variable = ce_participated_in_polish_problem
			}
		}
	}

	# 1860: the settlement outlived the question.
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		# The deadline may fall on the same tick as annexation or an already
		# declared victory. Resolve once, before dispatch can change the KRA tag.
		effect = {
			if = { limit = { nesnd_poland_partitioned = yes }
				nesnd_poland_set_result = { RESULT = partition }
			}
			else_if = { limit = { nesnd_poland_declared_restoration = yes }
				nesnd_poland_set_result = { RESULT = restored }
			}
			else = { nesnd_poland_set_result = { RESULT = consensus } }
			scope:journal_entry = {
				every_scope_je_involved = {
					nesnd_poland_snapshot_stance = yes
					if = { limit = { nesnd_poland_result_is = { RESULT = partition } }
						trigger_event = { id = ce_poland_events.11 popup = yes }
					}
					else_if = { limit = { nesnd_poland_result_is = { RESULT = restored } }
						trigger_event = { id = ce_poland_events.10 popup = yes }
					}
					else = { trigger_event = { id = ce_poland_events.16 popup = yes } }
					if = { limit = { custom_tooltip = { text = nesnd_ui_ce_participated_in_polish_problem_tt has_variable = ce_participated_in_polish_problem } }
						remove_variable = ce_participated_in_polish_problem
					}
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			nesnd_poland_partitioned = no
			nesnd_poland_declared_restoration = no
			OR = {

				custom_tooltip = {
					text = ce_je_poland_1_invalid_tt_0
					has_global_variable = ce_poland_problem_end
				}
				nesnd_poland_existing_plc = yes
				custom_tooltip = {
					text = ce_je_poland_1_invalid_tt_2
					NOT = {
						any_country = {
							exists = c:KRA
							NOT = { c:KRA ?= This }
							OR = {
								has_variable = ce_participated_in_polish_problem
								any_state = {
									OR = {
										state_region = s:STATE_WEST_GALICIA
										state_region = s:STATE_UPPER_SILESIA
										state_region = s:STATE_EAST_GALICIA
										state_region = s:STATE_POSEN
										state_region = s:STATE_KAUNAS
										state_region = s:STATE_BREST
										state_region = s:STATE_VOLHYNIA
										state_region = s:STATE_WEST_PRUSSIA
										state_region = s:STATE_GREATER_POLAND
										state_region = s:STATE_LESSER_POLAND
									}
									owner = prev
								}
							}
						}
					}

				}
			}
		}
		effect = { nesnd_poland_windows_cleanup = yes nesnd_poland_civil_cleanup = yes set_global_variable = ce_poland_problem_end }
		effect_all_involved = {
			nesnd_poland_snapshot_stance = yes

			trigger_event = { id = ce_poland_events.12 popup = yes }
			if = {
				limit = { custom_tooltip = { text = nesnd_ui_ce_participated_in_polish_problem_tt has_variable = ce_participated_in_polish_problem } }
				remove_variable = ce_participated_in_polish_problem
			}
		}
	}

	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	on_become_involved_after_activation = { nesnd_poland_enrol_country = yes }
}
