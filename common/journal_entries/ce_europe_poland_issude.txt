			# Below 30 threshold - remove all
			else = {
				if = {
					limit = { scope:journal_entry = { has_modifier = ce_anti_poland_block_30 } }
					scope:journal_entry = { remove_modifier = ce_anti_poland_block_30 }
				}
				if = {
					limit = { scope:journal_entry = { has_modifier = ce_anti_poland_block_50 } }
					scope:journal_entry = { remove_modifier = ce_anti_poland_block_50 }
				}
				if = {
					limit = { scope:journal_entry = { has_modifier = ce_anti_poland_block_90 } }
					scope:journal_entry = { remove_modifier = ce_anti_poland_block_90 }
				}
			}
		}
