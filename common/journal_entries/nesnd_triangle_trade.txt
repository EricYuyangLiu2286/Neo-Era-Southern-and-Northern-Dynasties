nesnd_north_triangle_trade = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	scripted_button = nesnd_north_trade_export_button
	scripted_button = nesnd_north_trade_refuse_button

		effect = {
			scope:journal_entry = {
				if = {
					limit = {
						NOT = { has_global_variable = nesnd_north_trade_ended }
						OR = {
							"scripted_bar_progress(nesnd_north_trade_bar_1)" >= 100
							"scripted_bar_progress(nesnd_north_trade_bar_2)" >= 100
							"scripted_bar_progress(nesnd_north_trade_bar_3)" >= 100
						}

					if = {
						limit = { "scripted_bar_progress(nesnd_north_trade_bar_1)" >= 100 }

					}
					if = {
						limit = { "scripted_bar_progress(nesnd_north_trade_bar_2)" >= 100 }
						nesnd_triangle_settle = { KEY = north SLOT = 2 }
					}
					if = {
						limit = { "scripted_bar_progress(nesnd_north_trade_bar_3)" >= 100 }
						nesnd_triangle_settle = { KEY = north SLOT = 3 }

					nesnd_triangle_fill_slots = { KEY = north }

	on_monthly_pulse = {
		effect = { scope:journal_entry = { nesnd_triangle_month = { KEY = north } } }

	on_yearly_pulse = {
		effect = { nesnd_triangle_ai = { KEY = north } }

		}
	}

			triggered_desc = {
