# 北洋 / 西洋三角贸易。两个日志共用 nesnd_triangle_* 的 trigger 与 effect，开局由 history/global 挂出。

nesnd_north_triangle_trade = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_global_international_situations
	activation_description = { always = yes }
	weight = 1400
	can_revolution_inherit = no
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
	is_shown_when_inactive = { always = no }
	possible = { always = no }
	should_be_involved = {
		has_variable = nesnd_north_trade_participant
	}
	should_show_when_not_involved = {
		nesnd_triangle_region_country = { KEY = north }
	}

	scripted_progress_bar = nesnd_north_trade_bar

	scripted_button = nesnd_north_trade_buy_button
	scripted_button = nesnd_north_trade_sell_button
	scripted_button = nesnd_north_trade_none_button
	scripted_button = nesnd_north_trade_import_button
	scripted_button = nesnd_north_trade_export_button
	scripted_button = nesnd_north_trade_refuse_button

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				if = {
					limit = {
						NOT = { has_global_variable = nesnd_north_trade_ended }
						"scripted_bar_progress(nesnd_north_trade_bar)" >= 100
					}
					nesnd_triangle_settle = { KEY = north }
				}
			}
		}
	}
	on_monthly_pulse = {
		effect = { scope:journal_entry = { nesnd_triangle_month = { KEY = north } } }
	}
	on_yearly_pulse = {
		effect = { nesnd_triangle_ai = { KEY = north } }
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = nesnd_north_trade_end_conclusion
		trigger = {
			custom_tooltip = {
				text = nesnd_north_trade_end_tt
				has_global_variable = nesnd_north_trade_ended
			}
		}
		effect = {
			custom_tooltip = nesnd_triangle_trade_end_effect_tt
			hidden_effect = { nesnd_triangle_end = { KEY = north END_EVENT = nesnd_triangle_trade.10 } }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = global_var:nesnd_north_trade_dest }
				desc = nesnd_north_trade_order_status
			}
			desc = nesnd_triangle_trade_no_order_status
		}
		first_valid = {
			triggered_desc = {
				trigger = { has_global_variable = nesnd_north_trade_last global_var:nesnd_north_trade_last = 1 }
				desc = nesnd_north_trade_last_done
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_north_trade_last global_var:nesnd_north_trade_last = 2 }
				desc = nesnd_triangle_trade_last_no_order
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_north_trade_last global_var:nesnd_north_trade_last = 3 }
				desc = nesnd_north_trade_last_no_slavery
			}
		}
	}
}
