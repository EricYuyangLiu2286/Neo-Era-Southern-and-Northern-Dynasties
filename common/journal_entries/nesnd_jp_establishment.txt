nesnd_jp_establish_toyotomi = {
    icon = "gfx/interface/icons/invention_icons/urban_planning.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_jp_establishment_needed = yes c:JAP ?= this }
    possible = { nesnd_jp_establishment_needed = yes c:JAP ?= this has_variable = nesnd_jp_statistics_ready }
    immediate = { set_variable = nesnd_jp_establishment_opened }
    outcome = {
    	header = je_standard_completion_header
    	effect_header = je_standard_on_completion_header
    	color = "gold_positive"
    	conclusion_desc = je_standard_completion_conclusion_desc
    	trigger = { nesnd_jp_establishment_ready = yes }
    	effect = { nesnd_jp_finish_establishment = yes }
    }
    outcome = {
    	flags = { hidden invalid_outcome }
    	conclusion_desc = je_standard_failure_conclusion_desc
    	trigger = { OR = { NOT = { nesnd_jp_country = yes } NOT = { c:JAP ?= this } has_variable = nesnd_jp_established } }
    }
    weight = 910
    should_be_pinned_by_default_uninvolved_or_context = yes
}
