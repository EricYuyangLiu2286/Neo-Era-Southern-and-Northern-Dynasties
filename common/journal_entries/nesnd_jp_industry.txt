nesnd_jp_mining = {
    icon = "gfx/interface/icons/invention_icons/mechanical_tools.dds"
    group = je_group_historical_content can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_jp_mining_available = yes }
    possible = { nesnd_jp_mining_available = yes }
    immediate = { nesnd_jp_initialize = yes set_variable = nesnd_jp_mining_opened  }
    complete = { hidden_trigger = { nesnd_jp_mining_available = yes } country_has_building_type_levels = { target = bt:building_iron_mine value >= nesnd_jp_goal_mine }
        country_has_building_type_levels = { target = bt:building_coal_mine value >= nesnd_jp_goal_coal }
        country_has_building_type_levels = { target = bt:building_logging_camp value >= nesnd_jp_goal_logging }
        any_scope_building = { is_building_type = building_iron_mine occupancy >= 0.6 weekly_profit > 0 } }
    on_complete = { if = { limit = { NOT = { has_variable = nesnd_jp_mining_done } }
            set_variable = nesnd_jp_mining_done
            capital ?= { set_variable = nesnd_jp_mining_done }
            add_modifier = { name = nesnd_jp_metal_foundries months = 60 }
            nesnd_jp_ensure_industry = yes
        }
    }
    invalid = { OR = { NOT = { nesnd_jp_country = yes } has_variable = nesnd_jp_mining_done } }
    on_invalid = { remove_variable = nesnd_jp_mining_opened }

    weight = 650 should_be_pinned_by_default_uninvolved_or_context = yes
}
