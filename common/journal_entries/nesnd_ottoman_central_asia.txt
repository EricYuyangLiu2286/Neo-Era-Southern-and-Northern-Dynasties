# Ottoman B13 foreign-affairs journal: "Central Asia" (both constitutional and sultanic routes).
# Added only by nesnd_ottoman_ca_start (events .5/.6 and monthly fallback nesnd_ottoman_ca_fallback),
# so it defines neither possible nor is_shown_when_inactive.
# Non-event outcomes omit event_outcome_desc so effects display natively.

nesnd_ottoman_central_asia = {
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    group = je_group_foreign_affairs
    activation_description = { always = yes }
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes

    scripted_button = nesnd_ottoman_fund_tribes
    scripted_button = nesnd_ottoman_ca_protect
    scripted_button = nesnd_ottoman_ca_compact

    immediate = {
        set_variable = {
            name = nesnd_ottoman_ca_deadline
            months = 240
        }
    }

    on_monthly_pulse = {
        effect = {
            nesnd_ottoman_ca_update_dependent_months = yes
            if = {
                limit = {
                    NOT = { has_global_variable = nesnd_ca_foe }
                    any_country = {
                        nesnd_ottoman_ca_foe_candidate = yes
                    }
                }
                nesnd_ottoman_ca_pick_foe = yes
            }
            if = {
                limit = {
                    NOT = { has_variable = nesnd_ottoman_ca_had_client }
                    any_country = {
                        nesnd_ottoman_ca_client = yes
                    }
                }
                set_variable = nesnd_ottoman_ca_had_client
            }
        }
    }

    # 1. Hidden invalidation (tag changed away from TUR).
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_ottoman_ca_invalid_conclusion
        trigger = {
            NOT = { c:TUR ?= this }
        }
        effect = {
            set_variable = nesnd_ottoman_ca_closed
            nesnd_ottoman_ca_cleanup = yes
        }
    }

    # 2. Lost autonomy (R13: unable to act abroad for 12 consecutive months; may reopen after 24-month cooldown).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_ca_dependent_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_ca_dependent_tt
                has_variable = nesnd_ottoman_ca_dependent_months
                var:nesnd_ottoman_ca_dependent_months >= 12
            }
        }
        effect = {
            if = {
                limit = {
                    NOT = { has_modifier = nesnd_ottoman_lost_autonomy }
                }
                add_modifier = {
                    name = nesnd_ottoman_lost_autonomy
                }
            }
            custom_tooltip = nesnd_ottoman_ca_reopen_cd_tt
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_ca_outcome
                    value = flag:dependent
                }
                set_variable = {
                    name = nesnd_ottoman_ca_reopen_cd
                    months = 24
                }
                nesnd_ottoman_ca_cleanup = yes
            }
        }
    }

    # 3. Local states crushed (R10/F1: TUR previously had a client, no native state remains, TUR holds no state in region, and a foe candidate holds it).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_ca_lost_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_ca_lost_tt
                has_variable = nesnd_ottoman_ca_had_client
                NOT = {
                    any_country = {
                        nesnd_ottoman_ca_native = yes
                    }
                }
                NOT = {
                    any_scope_state = {
                        nesnd_ottoman_ca_region_state = yes
                    }
                }
                any_country = {
                    nesnd_ottoman_ca_foe_candidate = yes
                }
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_ca_lost_mod
            }
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_ca_outcome
                    value = flag:lost
                }
                set_variable = nesnd_ottoman_ca_closed
                nesnd_ottoman_ca_cleanup = yes
            }
        }
    }

    # 4. Patron of Central Asia (R10: at least two local Central Asian states are Ottoman subjects).
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_ottoman_ca_won_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_ca_won_tt
                nesnd_ottoman_ca_clients_enough = yes
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_ca_patron_mod
            }
            if = {
                limit = {
                    has_journal_entry = nesnd_ottoman_recovery
                }
                je:nesnd_ottoman_recovery ?= {
                    add_progress = {
                        name = nesnd_ottoman_enthusiasm_bar
                        value = 20
                    }
                }
            }
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_ca_outcome
                    value = flag:won
                }
                set_variable = nesnd_ottoman_ca_closed
                nesnd_ottoman_ca_cleanup = yes
            }
        }
    }

    # 5. Situation changed (R12: stored foe became invalid and TUR is not at war with it; does not trigger when no foe is stored).
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_ca_changed_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_ca_changed_tt
                nesnd_ottoman_ca_situation_changed = yes
            }
        }
        effect = {
            custom_tooltip = nesnd_ottoman_ca_reopen_cd_tt
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_ca_outcome
                    value = flag:changed
                }
                set_variable = {
                    name = nesnd_ottoman_ca_reopen_cd
                    months = 24
                }
                nesnd_ottoman_ca_cleanup = yes
            }
        }
    }

    # 6. Shelved (240-month deadline elapsed while not in a play or war with the foe; when no foe is stored, checks deadline alone).
    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        color = "blue_negative"
        conclusion_desc = nesnd_ottoman_ca_shelved_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_ottoman_ca_timeout_tt
                NOT = { has_variable = nesnd_ottoman_ca_deadline }
                nesnd_ottoman_ca_play_in_progress = no
                nesnd_ottoman_ca_at_war_with_foe = no
            }
        }
        effect = {
            add_modifier = {
                name = nesnd_ottoman_ca_shelved_mod
            }
            hidden_effect = {
                set_variable = {
                    name = nesnd_ottoman_ca_outcome
                    value = flag:shelved
                }
                set_variable = nesnd_ottoman_ca_closed
                nesnd_ottoman_ca_cleanup = yes
            }
        }
    }

    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    nesnd_ottoman_ca_at_war_with_foe = yes
                }
                desc = nesnd_ottoman_ca_status_war
            }
            triggered_desc = {
                trigger = {
                    nesnd_ottoman_ca_play_in_progress = yes
                }
                desc = nesnd_ottoman_ca_status_play
            }
            triggered_desc = {
                trigger = {
                    any_country = {
                        nesnd_ottoman_ca_client = yes
                    }
                }
                desc = nesnd_ottoman_ca_status_one_client
            }
            triggered_desc = {
                trigger = {
                    any_country = {
                        nesnd_ottoman_ca_local = yes
                    }
                }
                desc = nesnd_ottoman_ca_status_locals
            }
            desc = nesnd_ottoman_ca_status_none
        }
    }
}
