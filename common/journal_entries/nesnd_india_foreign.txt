# ==== Overseas dealings (India batch 17) ====
# Held by an independent, recognised Indian imperial government (nesnd_india_overseas_holder).
# Added each month by nesnd_india_overseas_upkeep from nesnd_india_refresh; has neither possible nor
# is_shown_when_inactive, so no inactive instance is pre-created for other countries.
# Ongoing journal: R6 is met by the one-time milestone event "Indian Ocean Trade" (nesnd_india_overseas.7)
# and the failure outcome "Overseas Lost" below.
nesnd_india_overseas = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_foreign_affairs
	activation_description = { always = yes }
	can_revolution_inherit = no
	weight = 430
	should_be_pinned_by_default_uninvolved_or_context = yes

	status_desc = {
		first_valid = {
			triggered_desc = { trigger = { has_variable = nesnd_india_ocean_trade } desc = nesnd_india_overseas_status_done }
			desc = nesnd_india_overseas_status_ongoing
		}
	}

	scripted_progress_bar = nesnd_india_overseas_himalayas_bar
	scripted_progress_bar = nesnd_india_overseas_indonesia_bar
	scripted_progress_bar = nesnd_india_overseas_east_africa_bar
	scripted_progress_bar = nesnd_india_overseas_arabia_bar
	scripted_progress_bar = nesnd_india_overseas_persia_bar

	scripted_button = nesnd_india_overseas_fir_himalayas_button
	scripted_button = nesnd_india_overseas_claim_himalayas_button
	scripted_button = nesnd_india_overseas_fir_indonesia_button
	scripted_button = nesnd_india_overseas_claim_indonesia_button
	scripted_button = nesnd_india_overseas_fir_east_africa_button
	scripted_button = nesnd_india_overseas_claim_east_africa_button
	scripted_button = nesnd_india_overseas_fir_arabia_button
	scripted_button = nesnd_india_overseas_claim_arabia_button
	scripted_button = nesnd_india_overseas_fir_persia_button
	scripted_button = nesnd_india_overseas_claim_persia_button

	# Failure outcome: "Overseas Lost". Subjected or stripped of recognition.
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = nesnd_india_overseas_lost_conclusion
		trigger = {
			hidden_trigger = { nesnd_india_empire = yes }
			OR = {
				is_subject = yes
				is_country_type = unrecognized
			}
		}
		effect = {
			add_modifier = { name = nesnd_india_overseas_lost days = normal_modifier_time }
			custom_tooltip = nesnd_india_overseas_lost_effect_tt
			hidden_effect = { nesnd_india_overseas_clear_flags = yes }
		}
	}

	# Hidden invalid outcome: only when no longer an Indian imperial government (e.g. annexed).
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = nesnd_india_overseas_void_conclusion
		trigger = { NOT = { nesnd_india_empire = yes } }
		effect = { nesnd_india_overseas_clear_flags = yes }
	}

	on_monthly_pulse = {
		effect = {
			nesnd_india_overseas_monthly = yes
		}
	}
}
