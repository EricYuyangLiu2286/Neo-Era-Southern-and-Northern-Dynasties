REPLACE:je_unite_the_nations = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	scripted_button = je_unite_the_nations_proclaim_turtle_island_button
    scripted_button = je_unite_the_nations_organize_the_tribes_button
    scripted_button = je_unite_the_nations_offer_confederation_button

    is_shown_in_lobby = {
        NOT = { has_variable = nesnd_native_khanate } NOT = { has_variable = nesnd_hus_native_successor }
		NOT = { is_country_type = decentralized }
		is_in_geographic_region = geographic_region_north_america
		any_primary_culture = {
			has_discrimination_trait_group = heritage_group_indigenous_american
		}
		OR = {
			NOT = {
				exists = c:TIS
			}
			AND = {
				exists = c:TIS
				c:TIS = this
			}
		}
	}

	is_shown_when_inactive = {
        NOT = { has_variable = nesnd_native_khanate } NOT = { has_variable = nesnd_hus_native_successor }
		NOT = { is_country_type = decentralized }
		is_in_geographic_region = geographic_region_north_america
		any_primary_culture = {
			has_discrimination_trait_group = heritage_group_indigenous_american
		}
		OR = {
			NOT = {
				exists = c:TIS
			}
			AND = {
				exists = c:TIS
				c:TIS = this
			}
		}
	}

	possible = {
        NOT = { has_variable = nesnd_native_khanate } NOT = { has_variable = nesnd_hus_native_successor }
		is_subject = no
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_unite_the_nations_complete_tt
				NOT = {
					any_decentralized_country = {
						is_in_geographic_region = geographic_region_north_america
						any_primary_culture = {
							shares_heritage_trait_group_with_any_primary_culture = ROOT
						}
					}
				}
			}
	        any_scope_state = {
	            is_homeland_of_country_cultures = ROOT
	            count >= 5
	        }
		}
		effect = {
			if = {
				limit = { is_country_type = unrecognized }
				set_country_type = recognized
			}
			add_modifier = {
				name = turtle_island_state_of_nations
				months = -1 # FOREVER
			}
			je_unite_the_nations_cleanup = yes
		}
	}

	weight = 100

		
	should_be_pinned_by_default_uninvolved_or_context = yes

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { OR = { has_variable = nesnd_native_khanate has_variable = nesnd_hus_native_successor } }
	}
}
