    scripted_progress_bar = nesnd_jp_political_bar
    outcome = {
    	header = je_standard_completion_header
    	effect_header = je_standard_on_completion_header
    	color = "gold_positive"
    	conclusion_desc = je_standard_completion_conclusion_desc
    	trigger = { custom_tooltip = { text = nesnd_jp_reform_settled nesnd_jp_political_ready = yes nesnd_jp_political_progress >= 100 nesnd_jp_political_good_months >= 12 } hidden_trigger = { NOT = { nesnd_jp_political_failure = yes } } }
    	effect = { nesnd_jp_political_finish = yes }
    }
    outcome = {
    	header = je_standard_failure_header
    	effect_header = je_standard_on_failure_header
    	color = "blue_negative"
    	conclusion_desc = je_standard_failure_conclusion_desc
    	trigger = { nesnd_jp_political_failure = yes }
    	effect = { nesnd_jp_political_stop = yes }
    }
    outcome = {
    	flags = { hidden invalid_outcome }
    	conclusion_desc = je_standard_failure_conclusion_desc
    	trigger = { OR = { NOT = { nesnd_jp_politics_eligible = yes } has_variable = nesnd_jp_political_setback } }
    	effect = { nesnd_jp_political_clear_costs = yes }
    }
    on_monthly_pulse = { effect = { nesnd_jp_political_monthly = yes } }
    status_desc = { first_valid = {
            triggered_desc = { trigger = { nesnd_jp_route_court = yes } desc = nesnd_jp_court_policy }
            triggered_desc = { trigger = { nesnd_jp_route_military = yes } desc = nesnd_jp_military_policy }
            triggered_desc = { trigger = { nesnd_jp_route_representative = yes } desc = nesnd_jp_representative_policy }
            desc = nesnd_jp_unsettled_policy
        }
    }
    weight = 850 should_be_pinned_by_default_uninvolved_or_context = yes
}
