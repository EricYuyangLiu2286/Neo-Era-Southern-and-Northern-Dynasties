REPLACE:je_opium = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		any_market = {
			owner ?= root
		}
		NOT = { has_variable = opium_je_done }
		any_scope_state = {
			has_potential_resource = building_opium_plantation
		}
	}

	possible = {
		any_scope_building = {
			is_building_type = building_opium_plantation
		}
		NOT = {
			any_primary_culture = {
				has_cultural_obsession = opium
			}
		}
		any_market = {
			NOT = { owner ?= root }
			owner = {
				NOT = { has_variable = ce_war_to_opium_experienced }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_building = {
				is_building_type = building_opium_plantation
				cash_reserves_ratio > 0.5
				weekly_profit > 0
				occupancy >= 0.75
				level >= 10
			}
			any_market = { nesnd_opium_target_market = yes }
		}
		effect = {
			trigger_event = { id = ce_opium_events.1 }
			set_variable = {
				name = opium_je_done
			}
		}
	}

	transferable = no
	weight = 100
}

je_ce_war_to_opium = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_foreign_affairs
	activation_description = { always = yes }

	#progressbar = yes # you don't need progress bar = yes if you have scripted_progress_bar set
	scripted_progress_bar = ce_war_to_opium_progress_bar

	possible = {
		custom_tooltip = {
			text = possible_je_ce_war_to_opium
			has_variable = can_have_je_ce_war_to_opium_variable
			NOT = { has_variable = ce_war_to_opium_experienced }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = { 
				desc = je_ce_war_to_opium_status_high
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(ce_war_to_opium_progress_bar)" >= 75
					}
				}
			}
			triggered_desc = { 
				desc = je_ce_war_to_opium_status_low
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(ce_war_to_opium_progress_bar)" <= 25
					}
				}
			}
			triggered_desc = { 
				desc = je_ce_war_to_opium_status_mid
			}
		}
	}

	is_shown_when_inactive = {
		has_variable = can_have_je_ce_war_to_opium_variable
	}

	immediate = {
		set_variable = {
			name = ce_war_to_opium_progress
			value = 0
		}
		# add_modifier = {
		# 	name = ce_war_to_opium_active_modifier
		# }
	}

	modifiers_while_active = {
		ce_war_to_opium_active_modifier 
	}

	current_value = {
		value = var:ce_war_to_opium_progress
	}


	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
		text = je_ce_war_to_opium_success
		scope:journal_entry = {
		"scripted_bar_progress(ce_war_to_opium_progress_bar)" <= 0
		}
		}
		}
		effect = {
			remove_variable = ce_war_to_opium_progress
			remove_variable = can_have_je_ce_war_to_opium_variable
			#remove_modifier = ce_war_to_opium_active_modifier
			trigger_event = { id = ce_opium_events.3 }
		}
	}


	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
		text = je_ce_war_to_opium_fail
		scope:journal_entry = {
		"scripted_bar_progress(ce_war_to_opium_progress_bar)" >= 100
		}
		}
		}
		effect = {
			remove_variable = ce_war_to_opium_progress
			remove_variable = can_have_je_ce_war_to_opium_variable
			#remove_modifier = ce_war_to_opium_active_modifier
			trigger_event = { id = ce_opium_events.4 }
		}
	}


	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_market = {
					owner = root
				}
			}
		}
		effect = {
			remove_variable = ce_war_to_opium_progress
			remove_variable = can_have_je_ce_war_to_opium_variable
			#remove_modifier = ce_war_to_opium_active_modifier
			trigger_event = { id = ce_opium_events.5 }
		}
	}


	on_monthly_pulse = {
		random_events = {
			98 = 0
			2 = ce_opium_events.6
			2 = ce_opium_events.7
			2 = ce_opium_events.8
			2 = ce_opium_events.9
		}
	}

	is_progressing = {
		is_banning_goods = g:opium
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_nesnd_opium_recovery = {
	icon = "gfx/interface/icons/character_trait_icons/opium_addiction.dds"
	group = je_group_internal_affairs
	activation_description = { has_modifier = ce_war_to_opium_fail_modifier }
	is_shown_when_inactive = { has_modifier = ce_war_to_opium_fail_modifier }
	possible = { has_modifier = ce_war_to_opium_fail_modifier }

	scripted_button = nesnd_opium_register_dispensaries
	scripted_button = nesnd_opium_repeal_regulations
	scripted_progress_bar = nesnd_opium_recovery_bar

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOR = {
						institution_investment_level = { institution = institution_health_system value >= 2 }
						institution_investment_level = { institution = institution_police value >= 2 }
					}
				}
				desc = je_nesnd_opium_recovery_no_capacity
			}
			triggered_desc = {
				trigger = { NOT = { has_modifier = nesnd_opium_regulations } }
				desc = je_nesnd_opium_recovery_awaiting_order
			}
			triggered_desc = { desc = je_nesnd_opium_recovery_underway }
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_nesnd_opium_recovery_conclusion
		trigger = {
			has_modifier = ce_war_to_opium_fail_modifier
			custom_tooltip = {
				text = je_nesnd_opium_recovery_goal
				scope:journal_entry = { "scripted_bar_progress(nesnd_opium_recovery_bar)" <= 40 }
			}
		}
		effect = {
			remove_modifier = ce_war_to_opium_fail_modifier
			every_primary_culture = {
				limit = { has_cultural_obsession = opium }
				remove_cultural_obsession = opium
			}
			if = {
				limit = { has_modifier = nesnd_opium_regulations }
				remove_modifier = nesnd_opium_regulations
			}
			if = {
				limit = { has_variable = nesnd_opium_repeal_lock }
				remove_variable = nesnd_opium_repeal_lock
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { NOT = { has_modifier = ce_war_to_opium_fail_modifier } }
		effect = {
			if = {
				limit = { has_modifier = nesnd_opium_regulations }
				remove_modifier = nesnd_opium_regulations
			}
			if = {
				limit = { has_variable = nesnd_opium_repeal_lock }
				remove_variable = nesnd_opium_repeal_lock
			}
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
}
