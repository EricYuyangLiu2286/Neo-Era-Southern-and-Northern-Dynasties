ce_cne_land_reform = {
	icon = "gfx/interface/icons/invention_icons/egalitarianism.dds"
	group = je_group_historical_content can_revolution_inherit = no
	is_shown_when_inactive = { c:CNE ?= this has_variable = flavorCNE_land_reform_value NOT = { has_variable = ce_CNE_land_reform_done } }
	possible = { c:CNE ?= this has_variable = flavorCNE_land_reform_value NOT = { has_variable = ce_CNE_land_reform_done } }
	immediate = { nesnd_land_init = yes
		nesnd_refresh_land_display = yes
	}

	scripted_button = nesnd_land_review

	scripted_button = nesnd_land_suspend
	on_monthly_pulse = { effect = { nesnd_land_monthly = yes } }
	on_yearly_pulse = { random_events = { 50 = 0 10 = flavorCNE.4 10 = flavorCNE.5 10 = flavorCNE.6 10 = flavorCNE.7 10 = flavorCNE.8 } }

	complete = {
		num_states >= 4
		country_has_monarchy_law = yes
		hidden_trigger = { NOT = { nesnd_cne_reform_failed = yes } nesnd_reform_active = yes has_variable = nesnd_reform_v2 var:nesnd_reform_samples >= 3 var:nesnd_reform_workforce > 0 }
		nesnd_reform_land_rights = yes
		institution_investment_level = { institution = institution_schools value >= 2 }
		literacy_rate >= 0.48
		bureaucracy >= 0
		is_at_war = no
		NOT = { has_modifier = declared_bankruptcy }
		custom_tooltip = { text = nesnd_cne_employment_condition nesnd_reform_employment_ready = yes }
		custom_tooltip = { text = nesnd_cne_livelihood_condition var:nesnd_reform_starving <= 0.08 var:nesnd_reform_sol >= 11 var:nesnd_reform_access >= 0.85 var:nesnd_reform_neglected_regions < 1 }
		custom_tooltip = { text = nesnd_cne_completion_condition var:cne_landreform_completion_tracker >= 100 }
		custom_tooltip = { text = nesnd_cne_stability_condition var:nesnd_reform_good_months >= 12 }
		custom_tooltip = { text = nesnd_cne_disorder_condition var:cne_landreform_failure_tracker < 50 }
	}
	on_complete = { nesnd_land_success = yes }
	fail = { OR = {
			NOT = { country_has_monarchy_law = yes }
			custom_tooltip = { text = nesnd_cne_confidence_gone nesnd_cne_confidence_exhausted = yes }
			custom_tooltip = { text = nesnd_cne_coup_failure nesnd_cne_coup_has_succeeded = yes }
			custom_tooltip = { text = nesnd_cne_civil_failure nesnd_cne_native_civil_war = yes }
			custom_tooltip = { text = nesnd_cne_crisis_failure nesnd_reform_extreme_crisis = yes }
		} }
	on_fail = { nesnd_cne_end_failed_reform = yes }
	invalid = { OR = { num_states < 4 has_variable = ce_CNE_land_reform_done NOT = { c:CNE ?= this } NOT = { has_variable = flavorCNE_land_reform_value } } }
	on_invalid = { nesnd_cne_programs_cleanup = yes remove_variable = flavorCNE_land_reform_value if = { limit = { has_modifier = CNE_land_draft_debuff } remove_modifier = CNE_land_draft_debuff } nesnd_reform_clear_local = yes }
	weight = 1200 should_be_pinned_by_default_uninvolved_or_context = yes












	scripted_progress_bar = nesnd_land_completion_bar
	scripted_progress_bar = nesnd_land_disorder_bar
	on_weekly_pulse = { effect = { nesnd_land_init = yes nesnd_cne_check_coup = yes nesnd_cne_prune_lost_estates = yes nesnd_cne_confidence_watch = yes nesnd_refresh_land_display = yes } }
	status_desc = { first_valid = {
			triggered_desc = { trigger = { OR = { NOT = { has_variable = nesnd_reform_v2 } var:nesnd_reform_samples < 3 } } desc = nesnd_reform_status_warmup }
			triggered_desc = { trigger = { var:nesnd_reform_neglected_regions > 0 } desc = nesnd_reform_status_regions }
			triggered_desc = { trigger = { nesnd_reform_land_rights = no } desc = nesnd_reform_status_law }
			triggered_desc = { trigger = { bureaucracy < 0 } desc = nesnd_land_status_admin }
			triggered_desc = { trigger = { is_at_war = yes } desc = nesnd_land_status_war }
			triggered_desc = { trigger = { nesnd_reform_healthy = yes } desc = nesnd_reform_status_healthy }
			desc = nesnd_land_status_normal
		} }
}
