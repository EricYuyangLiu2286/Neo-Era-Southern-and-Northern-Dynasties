ce_cne_land_reform = {	

	# 地权革命：

	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_historical_content
	can_revolution_inherit = yes
	
	is_shown_when_inactive = {
		exists = c:CNE
		c:CNE ?= THIS
		has_variable = flavorCNE_land_reform_value
		NOT = {
			has_variable = ce_CNE_land_reform_done
		}
	}
	
	immediate = {
		set_variable = {
			name = cne_landreform_completion_tracker
			value = 50
		}
	}

	possible = {
		has_variable = flavorCNE_land_reform_value
		NOT = {
			has_variable = ce_CNE_land_reform_done
		}
	}
	
	#inheritable = no
	
	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					ig:ig_rural_folk = {
						is_powerful = no
						is_in_government = no
					}
				}
				change_variable = {
					name = cne_landreform_completion_tracker
					add = 3
				}
			}
			else = {
				change_variable = {
					name = cne_landreform_completion_tracker
					add = 1
				}
			}
		}
		random_events = {
			900 = 0
			20 = flavorCNE.4 # 士绅不满
			20 = flavorCNE.5 # 军官家族来信
			20 = flavorCNE.6 # 滥捕
			20 = flavorCNE.7 # 改革红利
			20 = flavorCNE.8 # 农民涌入城市
		}
	}
	
	progressbar = yes
	
	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		
		
		#$ig_landowners$不[concept_powerful]且不[concept_in_government]
		#$ig_devout$不[concept_powerful]且不[concept_in_government]
		#$ig_rural_folk$不[concept_powerful]且不[concept_in_government]
		
	}
	
	on_complete = {
		hidden_effect = {
		#	remove_modifier = CNE_land_draft_debuff
		}
		trigger_event = { id = flavorCNE.3 }
		set_variable = {
			name = ce_CNE_land_reform_done
			value = yes
		}
		
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_serfdom
					has_law_or_variant = law_type:law_tenant_farmers
				}
			}
			activate_law = law_type:law_homesteading
		}
	}

	fail = {
		custom_tooltip = {
			text = cne_land_reform_failed_variable_tt
			has_variable = cne_land_reform_failed_variable
		}
	}

	# effect which is executed when 'fail' trigger becomes true
	on_fail = {
		trigger_event = { id = flavorCNE.10 } # failed land draft
		hidden_effect = {
			remove_variable = cne_landreform_completion_tracker
			set_variable = {
				name = ce_CNE_land_reform_done
				value = yes
			}
		}
	}

	invalid = {
		OR = {	
			num_states < 4
			any_civil_war = {
				civil_war_progress >= 0.90
			}
		}
	}	

	# effect which is executed when 'invalid' trigger becomes true
	on_invalid = {
		trigger_event = { id = flavorCNE.9 } # missed opportunity
		hidden_effect = {
			remove_variable = cne_landreform_completion_tracker
			set_variable = {
				name = ce_CNE_land_reform_done
				value = yes
			}
		}
	}
	
	current_value = {
		value = root.var:cne_landreform_completion_tracker
	}
	
	goal_add_value = {
		value = 100
	}

	weight = 1200

}

ce_cne_land_draft_failing = {	
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	is_shown_when_inactive = {
		exists = c:CNE
		c:CNE ?= THIS
		has_variable = flavorCNE_land_reform_value
		NOT = {
			has_variable = ce_CNE_land_reform_done
		}
	}
	
	group = je_group_historical_content
	possible = {
		has_variable = flavorCNE_land_reform_value
		NOT = {
			has_variable = ce_CNE_land_reform_done
		}
	}

	immediate = {
		set_variable = {
			name = cne_landreform_failure_tracker
			value = 0
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					OR = {
						ig:ig_landowners = {
							OR = {
								is_powerful = yes
								is_in_government = yes
							}
						}
						ig:ig_devout = {
							OR = {
								is_powerful = yes
								is_in_government = yes
							}
						}

					}
				}
				change_variable = {
					name = cne_landreform_failure_tracker
					add = 3
				}
			}
		}
    }

	complete = {
		scope:journal_entry = {
		 	is_goal_complete = yes
		}
		trigger_if = {
			limit = {
				has_variable = cne_landreform_failure_tracker
			}
			custom_tooltip = {
				text = cne_landreform_failure_tracker_100
				root.var:cne_landreform_failure_tracker >= 100
			}
		}
	}

	on_complete = {
		custom_tooltip = {
			text = failure_of_the_land_reform_je
			set_variable = {
				name = cne_land_reform_failed_variable
				value = yes
			}
		}
	}

	invalid = {
		has_variable = ce_CNE_land_reform_done
	}

      on_invalid = {
    	#
    }

	current_value = {
		value = root.var:cne_landreform_failure_tracker
	}

	goal_add_value = {
		value = 100
	}

	progressbar = yes
    
    weight = 1000
    should_be_pinned_by_default_uninvolved_or_context = yes
}
