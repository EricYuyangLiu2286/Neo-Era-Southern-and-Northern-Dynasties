nesnd_small_livelihood_journal = {
    icon = "gfx/interface/icons/event_icons/nesnd_x_rice_jar.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = {
        nesnd_small_state_country = yes
        custom_tooltip = {
            text = nesnd_ui_not_small_economy_done_tt
            NOT = { has_variable = nesnd_small_economy_done }
        }
        custom_tooltip = {
            text = nesnd_ui_not_small_economy_lapse_tt
            NOT = { has_variable = nesnd_small_economy_lapse }
        }
    }
    possible = {
        nesnd_small_state_country = yes
        custom_tooltip = {
            text = nesnd_ui_not_small_economy_done_tt
            NOT = { has_variable = nesnd_small_economy_done }
        }
        custom_tooltip = {
            text = nesnd_ui_not_small_economy_lapse_tt
            NOT = { has_variable = nesnd_small_economy_lapse }
        }
    }
    immediate = { nesnd_small_initialize = yes }
    scripted_progress_bar = nesnd_small_economy_bar
    scripted_button = nesnd_small_public
    scripted_button = nesnd_small_specialise_button
    scripted_button = nesnd_small_own_market_button
    scripted_button = nesnd_small_request_market
    scripted_button = nesnd_small_invest_button
    scripted_button = nesnd_small_defence_button
    # Twelve months with a market connection, profitable production and no default: the economy stands on its own.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            hidden_trigger = { nesnd_small_state_country = yes }
            in_default = no
            custom_tooltip = {
                text = nesnd_small_access_tt
                nesnd_small_transport >= 0.75
            }
            custom_tooltip = {
                text = nesnd_small_production_tt
                nesnd_small_production >= nesnd_small_scale
            }
            custom_tooltip = {
                text = nesnd_small_months_tt
                scope:journal_entry = { "scripted_bar_progress(nesnd_small_economy_bar)" >= 12 }
            }
        }
        effect = {
            set_variable = nesnd_small_economy_done
            if = {
                limit = { nesnd_small_prince_line = yes }
                add_modifier = {
                    name = nesnd_small_livelihood
                    months = 36
                }
            }
            else = {
                add_modifier = {
                    name = nesnd_southern_workshops
                    months = 36
                }
            }
        }
    }
    # Default or bankruptcy: a setback now, and the journal stays closed for five years.
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = nesnd_small_economy_failed_conclusion
        trigger = {
            hidden_trigger = { nesnd_small_state_country = yes }
            nesnd_india_broke = yes
        }
        effect = {
            add_modifier = {
                name = nesnd_small_economy_setback
                days = normal_modifier_time
            }
            set_variable = {
                name = nesnd_small_economy_lapse
                months = 60
            }
            custom_tooltip = nesnd_small_economy_lapse_tt
        }
    }
    outcome = {
        flags = {
            hidden
            invalid_outcome
        }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            NOT = { nesnd_small_state_country = yes }
        }
        effect = { nesnd_small_cleanup = yes }
    }
    should_be_pinned_by_default_uninvolved_or_context = yes
    weight = 550
}
nesnd_small_dispute_journal = {
    icon = "gfx/interface/icons/event_icons/nesnd_x_straw_hat.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = {
        nesnd_small_state_country = yes
        has_variable = nesnd_small_episode_active
    }
    possible = {
        nesnd_small_state_country = yes
        custom_tooltip = {
            text = nesnd_ui_small_episode_active_tt
            has_variable = nesnd_small_episode_active
        }
    }
    immediate = { nesnd_small_dispute_sync = yes }
    on_weekly_pulse = {
        effect = { nesnd_small_dispute_sync = yes }
    }
    scripted_button = nesnd_small_review_dispute
    scripted_progress_bar = nesnd_small_dispute_time_bar
    weight = 700
    should_be_pinned_by_default_uninvolved_or_context = yes
    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 2 }
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 1
                }
                desc = nesnd_small_dispute_promised_1
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 2 }
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 2
                }
                desc = nesnd_small_dispute_promised_2
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 2 }
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 3
                }
                desc = nesnd_small_dispute_promised_3
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 2 }
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 4
                }
                desc = nesnd_small_dispute_promised_4
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 2 }
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 5
                }
                desc = nesnd_small_dispute_promised_5
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 1 }
                }
                desc = nesnd_small_dispute_organised
            }
            triggered_desc = {
                trigger = {
                    nesnd_small_episode_choice_is = { VALUE = 3 }
                }
                desc = nesnd_small_dispute_suppressed
            }
            desc = nesnd_small_dispute_unanswered
        }
    }
    # Result 1: the promised law is in force.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_small_dispute_fulfilled_conclusion
        trigger = {
            hidden_trigger = {
                nesnd_small_dispute_result_is = { VALUE = 1 }
            }
            custom_tooltip = {
                text = nesnd_small_dispute_law_promised_tt
                has_variable = nesnd_small_promise
            }
            trigger_if = {
                limit = {
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 1
                }
                OR = {
                    has_law_or_variant = law_type:law_peasant_proprietorship
                    has_law = law_type:law_commercialized_agriculture
                    has_law = law_type:law_collectivized_agriculture
                }
            }
            trigger_if = {
                limit = {
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 2
                }
                OR = { has_law = law_type:law_slavery_banned }
            }
            trigger_if = {
                limit = {
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 3
                }
                OR = {
                    has_law = law_type:law_regulatory_bodies
                    has_law = law_type:law_worker_protections
                }
            }
            trigger_if = {
                limit = {
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 4
                }
                OR = {
                    has_law = law_type:law_hindu_caste_not_enforced
                    has_law = law_type:law_affirmative_action
                }
            }
            trigger_if = {
                limit = {
                    has_variable = nesnd_small_promise
                    var:nesnd_small_promise = 5
                }
                OR = {
                    has_law_or_variant = law_type:law_wealth_voting
                    has_law = law_type:law_census_voting
                    has_law = law_type:law_universal_suffrage
                }
            }
        }
        effect = {
            if = {
                limit = { var:nesnd_small_episode_issue = 1 }
                add_loyalists = {
                    pop_type = peasants
                    value = 0.01
                }
                add_loyalists = {
                    pop_type = laborers
                    value = 0.01
                }
                add_radicals = {
                    pop_type = aristocrats
                    value = 0.005
                }
            }
            else_if = {
                limit = { var:nesnd_small_episode_issue = 2 }
                add_loyalists = {
                    pop_type = laborers
                    value = 0.01
                }
                add_radicals = {
                    pop_type = capitalists
                    value = 0.005
                }
            }
            else = {
                add_loyalists = {
                    strata = lower
                    value = 0.01
                }
            }
            trigger_event = {
                id = nesnd_small_states.11
                popup = yes
            }
            nesnd_small_close_episode = yes
            set_variable = {
                name = nesnd_small_episode_cooldown
                months = 24
            }
        }
    }
    # Result 2: the promise term ran out and the law is not in force.
    outcome = {
        header = nesnd_journal_closures_header
        effect_header = nesnd_journal_closures_effect_header
        color = "blue_negative"
        conclusion_desc = nesnd_small_dispute_expired_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_small_dispute_expired_tt
                nesnd_small_dispute_result_is = { VALUE = 2 }
            }
        }
        effect = {
            nesnd_small_affected_radicals = { VALUE = 0.01 }
            trigger_event = {
                id = nesnd_small_states.12
                popup = yes
            }
            nesnd_small_close_episode = yes
            set_variable = {
                name = nesnd_small_episode_cooldown
                months = 24
            }
        }
    }
    # Result 5: the month for a choice ran out with the matter still open.
    outcome = {
        header = nesnd_journal_closures_header
        effect_header = nesnd_journal_closures_effect_header
        color = "blue_negative"
        conclusion_desc = nesnd_small_dispute_ignored_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_small_dispute_ignored_tt
                nesnd_small_dispute_result_is = { VALUE = 5 }
            }
        }
        effect = {
            nesnd_small_affected_radicals = { VALUE = 0.01 }
            nesnd_small_close_episode = yes
            set_variable = {
                name = nesnd_small_episode_cooldown
                months = 24
            }
        }
    }
    # Result 3: the support or suppression term ended, or the matter is gone. No consequence of its own.
    outcome = {
        flags = { hidden }
        conclusion_desc = nesnd_small_dispute_finished_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_small_dispute_finished_tt
                nesnd_small_dispute_result_is = { VALUE = 3 }
            }
        }
        effect = {
            nesnd_small_close_episode = yes
            set_variable = {
                name = nesnd_small_episode_cooldown
                months = 24
            }
        }
    }
    # Result 4: revolution or a change of regime. The monthly step has already closed the episode.
    outcome = {
        flags = { hidden }
        conclusion_desc = nesnd_small_dispute_ended_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_small_dispute_ended_tt
                nesnd_small_dispute_result_is = { VALUE = 4 }
            }
        }
    }
    outcome = {
        flags = {
            hidden
            invalid_outcome
        }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            NOT = { has_variable = nesnd_small_episode_result }
            OR = {
                NOT = { nesnd_small_state_country = yes }
                NOT = { has_variable = nesnd_small_episode_active }
            }
        }
    }
}
