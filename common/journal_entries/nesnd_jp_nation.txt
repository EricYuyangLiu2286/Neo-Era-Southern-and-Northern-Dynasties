nesnd_jp_nation = {
    icon = "gfx/interface/icons/invention_icons/egalitarianism.dds"
    group = je_group_historical_content can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_jp_nation_available = yes }
    possible = { nesnd_jp_nation_available = yes }
    immediate = { nesnd_jp_initialize = yes set_variable = nesnd_jp_nation_opened nesnd_jp_nation_sync = yes }
    scripted_progress_bar = nesnd_jp_nation_bar
    complete = { is_subject = no is_at_war = no custom_tooltip = { text = nesnd_jp_nation_complete nesnd_jp_nation_score >= 3 nesnd_jp_sovereign_months >= 12 } }
    on_complete = { if = { limit = { NOT = { has_variable = nesnd_jp_nation_done } }
            set_variable = nesnd_jp_nation_done capital ?= { set_variable = nesnd_jp_nation_done }
            add_modifier = { name = nesnd_jp_independent_state months = 120 }
            trigger_event = { id = nesnd_jp_development.1 popup = yes }
        }
    }
    invalid = { OR = { NOT = { nesnd_jp_country = yes } has_variable = nesnd_jp_nation_done } }
    on_monthly_pulse = { effect = { nesnd_jp_nation_monthly = yes } }
    status_desc = { first_valid = { triggered_desc = { trigger = { is_subject = yes } desc = nesnd_jp_nation_subject } desc = nesnd_jp_nation_independent } }
    weight = 800 should_be_pinned_by_default_uninvolved_or_context = yes
}
