# 唐衣大食「三教消长」：三教影响力（份额条）、转化条、代表人物与立国教。任务65 第②批。
nesnd_per_faith_journal = {
    icon = "gfx/interface/icons/religion_icons/shiite.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_per_faith_regime = yes nesnd_per_third_is_mandate = no }
    # 第三教开局二选一后才激活（事件 nesnd_per_faith.2）。
    possible = { nesnd_per_faith_regime = yes has_variable = nesnd_per_third_chosen nesnd_per_third_is_mandate = no }
    should_be_pinned_by_default_uninvolved_or_context = yes
    weight = 1090

    widget = {
        gui = "gui/journal_entry_widgets/nesnd_per_faith.gui"
        name = "nesnd_per_faith_representatives"
        container = "custom_widget_container_2"
    }
    widget = {
        gui = "gui/journal_entry_widgets/nesnd_per_faith.gui"
        name = "nesnd_per_faith_rules"
        container = "custom_widget_container_5"
    }

    immediate = {
        nesnd_per_faith_init = yes
        nesnd_per_rep_fill_all = yes
        nesnd_per_faith_measure = yes
    }

    scripted_progress_bar = nesnd_per_faith_shia_bar
    scripted_progress_bar = nesnd_per_faith_sunni_bar
    scripted_progress_bar = nesnd_per_faith_buddhist_bar
    scripted_progress_bar = nesnd_per_faith_conversion_bar
    scripted_button = nesnd_per_faith_adopt_shia
    scripted_button = nesnd_per_faith_adopt_sunni
    scripted_button = nesnd_per_faith_adopt_buddhist

    scaling_modifiers_while_active = {
        modifier = nesnd_per_faith_dissent
        value = {
            value = 0
            if = {
                limit = { has_variable = nesnd_per_faith_penalty_v NOT = { nesnd_per_tension_frozen = yes } }
                value = var:nesnd_per_faith_penalty_v
            }
        }
    }

    on_weekly_pulse = { effect = { nesnd_per_rep_fill_all = yes } }
    on_yearly_pulse = {
        effect = {
            nesnd_per_rep_keep_faith = { SLOT = shia REL = shiite }
            nesnd_per_rep_keep_faith = { SLOT = sunni REL = sunni }
            nesnd_per_rep_keep_faith = { SLOT = buddhist REL = mahayana }
        }
    }
    on_monthly_pulse = {
        effect = {
            nesnd_per_sync_clergy_leader_faith = yes
            if = { limit = { NOT = { nesnd_per_tension_frozen = yes } }
                nesnd_per_faith_measure = yes
                if = { limit = { "scope:journal_entry.scripted_bar_progress(nesnd_per_faith_conversion_bar)" >= 100 }
                    nesnd_per_faith_convert = yes
                    scope:journal_entry = { set_bar_progress = { name = nesnd_per_faith_conversion_bar value = 0 } }
                }
            }
        }
    }

    # 状态行只写处境：哪一教的讲坛声音最大。代价倍数由国家修正自行显示。
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_per_tension_frozen = yes } desc = nesnd_per_faith_status_frozen }
            triggered_desc = { trigger = { has_variable = nesnd_per_faith_cd } desc = nesnd_per_faith_status_cooldown }
            triggered_desc = { trigger = { has_variable = nesnd_per_faith_penalty_v var:nesnd_per_faith_penalty_v = 0 } desc = nesnd_per_faith_status_state_leads }
            triggered_desc = { trigger = { nesnd_per_faith_is_top = { FAITH = shia } } desc = nesnd_per_faith_status_shia_leads }
            triggered_desc = { trigger = { nesnd_per_faith_is_top = { FAITH = sunni } } desc = nesnd_per_faith_status_sunni_leads }
            triggered_desc = { trigger = { nesnd_per_faith_is_top = { FAITH = buddhist } } desc = nesnd_per_faith_status_buddhist_leads }
            desc = nesnd_per_faith_status_none
        }
    }
    outcome = {
        header = nesnd_journal_closures_header
        effect_header = nesnd_journal_closures_effect_header
        conclusion_desc = nesnd_per_faith_invalid_conclusion
        trigger = { custom_tooltip = { text = nesnd_per_faith_invalid_tt NOT = { nesnd_per_faith_regime = yes } } }
    }
}

# 唐衣大食「唐人同心」：西域汉羌团结度。参照西秦文化团结度。
nesnd_per_unity_journal = {
    icon = "gfx/interface/icons/ideology_icons/confucian.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = { nesnd_per_country = yes }
    possible = { nesnd_per_country = yes any_scope_state = { any_scope_pop = { has_pop_culture = ce_xiyu } } }
    should_be_pinned_by_default_uninvolved_or_context = yes
    weight = 1080

    immediate = {
        if = { limit = { NOT = { has_variable = nesnd_per_unity } } set_variable = { name = nesnd_per_unity value = 60 } }
    }

    scripted_progress_bar = nesnd_per_unity_bar

    on_monthly_pulse = {
        effect = {
            set_variable = { name = nesnd_per_unity value = "scope:journal_entry.scripted_bar_progress(nesnd_per_unity_bar)" }
            set_variable = {
                name = nesnd_per_tang_share_v
                value = {
                    value = 0
                    every_scope_state = { every_scope_pop = { limit = { nesnd_per_tang_pop = yes } add = total_size } }
                    divide = { value = 0 every_scope_state = { add = state_population } min = 1 }
                }
            }
            if = { limit = { NOT = { nesnd_per_tension_frozen = yes } } nesnd_per_unity_effects = yes }
        }
    }

    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_per_tension_frozen = yes } desc = nesnd_per_tension_status_frozen }
            triggered_desc = {
                trigger = { nesnd_per_ruler_counts_as_chinese = no }
                desc = nesnd_per_unity_status_persian_rule
            }
            triggered_desc = {
                trigger = { trigger_if = { limit = { has_variable = nesnd_per_unity } var:nesnd_per_unity > 75 } trigger_else = { always = no } }
                desc = nesnd_per_unity_status_high
            }
            triggered_desc = {
                trigger = { trigger_if = { limit = { has_variable = nesnd_per_unity } var:nesnd_per_unity < 40 } trigger_else = { always = no } }
                desc = nesnd_per_unity_status_low
            }
            desc = nesnd_per_unity_status_mid
        }
    }

    outcome = {
        header = nesnd_journal_closures_header
        effect_header = nesnd_journal_closures_effect_header
        conclusion_desc = nesnd_per_unity_invalid_conclusion
        trigger = {
            custom_tooltip = {
                text = nesnd_per_unity_invalid_tt
                OR = {
                    NOT = { nesnd_per_country = yes }
                    NOT = { any_scope_state = { any_scope_pop = { has_pop_culture = ce_xiyu } } }
                }
            }
        }
        effect = { remove_variable = nesnd_per_unity }
    }
}
