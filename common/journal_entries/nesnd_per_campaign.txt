    activation_description = { always = yes }
    can_revolution_inherit = yes
    is_shown_when_inactive = { nesnd_per_future_eligible = yes }
    possible = { nesnd_per_future_eligible = yes }
    should_be_pinned_by_default_uninvolved_or_context = yes
    weight = 1000

    immediate = { nesnd_per_initialize = yes nesnd_per_future_sync = yes }
    scripted_progress_bar = nesnd_per_future_bar
    scripted_button = nesnd_per_seek_aid_button
    scripted_button = nesnd_per_subvert_button
    outcome = {
    	header = je_standard_completion_header
    	effect_header = je_standard_on_completion_header
    	color = "gold_positive"
    	conclusion_desc = je_standard_completion_conclusion_desc
    	trigger = {
    	    custom_tooltip = {
    	        text = nesnd_per_future_complete_tt
    	        trigger_if = {
    	            limit = { has_variable = nesnd_per_future_progress }
    	            var:nesnd_per_future_progress >= 100
    	        }
    	        trigger_else = { always = no }
    	    }
    	}
    	effect = { nesnd_per_settle_future = yes }
    }
    outcome = {
    	flags = { hidden invalid_outcome }
    	conclusion_desc = je_standard_failure_conclusion_desc
    	trigger = { NOT = { nesnd_per_future_eligible = yes } }
    }
    on_monthly_pulse = { effect = { nesnd_per_future_monthly = yes } }
    status_desc = { first_valid = {
            triggered_desc = { trigger = { has_variable = nesnd_per_route_aid } desc = nesnd_per_future_status_aid }
            triggered_desc = { trigger = { has_variable = nesnd_per_route_covert } desc = nesnd_per_future_status_covert }
            desc = nesnd_per_future_status_choice
        }
    }
