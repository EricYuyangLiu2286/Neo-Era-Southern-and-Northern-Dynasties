		trigger = {
			nesnd_french_empire_politics_live = yes
			has_law_or_variant = law_type:law_monarchy
			exists = ruler
			trigger_if = {
				limit = {
					exists = ruler

				}

				has_journal_entry = je_ce_unstable_imperium
				custom_tooltip = {
					text = ce_french_empire_oreleans_progress_bar_100_tt
					scope:journal_entry = {
						"scripted_bar_progress(ce_french_empire_oreleans_progress_bar)" >= 200
					}

				trigger_if = {
					limit = { has_journal_entry = je_ce_unstable_imperium }

						text = ce_ce_monarchists_bonapartist_stability_progress_bar_fail_alt_tt

							"scripted_bar_progress(ce_monarchists_bonapartist_stability_progress_bar)" <= 200
						}
					}
				}

				NOT = {
					custom_tooltip = {
						text = ce_je_auto_fail_tt
						has_variable = ce_french_empire_stable_fail
					}
				}
			}
			trigger_else = {
				ruler = { has_ideology = ideology:ideology_orleanist }
			}
		}
		effect = {
			if = {
				limit = {
					has_variable = bonapartist_restoration_var
				}
				remove_variable = bonapartist_restoration_var
			}

			if = {
				limit = {
					NOT = { has_variable = orleanist_restoration_var }
				}
				set_variable = orleanist_restoration_var
			}


			if = { 
				limit = {
					exists = ruler
				}
				hidden_effect = {
					if ={
						limit = {
							any_scope_character = {
								has_template = FRA_Jerome_Bonaparte
								is_heir_of_own_country = yes
							}
						}
						random_scope_character = {
							limit = {
								has_template = FRA_Jerome_Bonaparte
								is_heir_of_own_country = yes
							}
							set_ideology = ideology:ideology_bonapartist
							exile_character_with_martyrdom = yes
						}
					}
				}
				if = { # safest situation
					limit = { has_gov_regency = no ruler = { NOT = { has_ideology = ideology:ideology_orleanist } } }
					ruler = {
						save_scope_as = fra_old_ruler_scope
					}
					hidden_effect = {

						if = {
							limit = {
								any_scope_character = {
									is_heir_of_own_country = yes
								}
							}
							random_scope_character = {
								limit = {
									is_heir_of_own_country = yes
								}
								kill_character = yes
							}

						}
						if = {
							limit = {
								has_gov_regency = yes
							}
							remove_regency_modifier = yes
						}
					}
					ce_set_orleanist_ruler = yes
					scope:fra_old_ruler_scope = {
						exile_character_with_martyrdom = yes
					}
				}
				else_if = { # Somehow you trigger this in orleanist non orleanist ideology regency (???)
					limit = {
						has_gov_regency = yes
						any_scope_character = {
							is_heir_of_own_country = yes
							NOT = { has_ideology = ideology:ideology_orleanist }
							has_variable = house_orleans
						}
					}
					random_scope_character = {
						limit = {
							is_heir_of_own_country = yes
							NOT = { has_ideology = ideology:ideology_orleanist }
							has_variable = house_orleans
						}
						set_ideology = ideology:ideology_orleanist
						save_scope_as = french_monarch_scope
					}
					hidden_effect = {
						ce_find_bonaparte_in_exile_effect = yes
					}

				}
				else_if = { # Somehow you trigger this in non orleanist regency (???)
					limit = {
						has_gov_regency = yes
						any_scope_character = {
							is_heir_of_own_country = yes
							NOT = { has_ideology = ideology:ideology_orleanist }
							NOT = { has_variable = house_orleans }
						}
					}
					hidden_effect = {
						random_scope_character = {
							limit = {
								is_ruler_of_own_country = yes
							}
