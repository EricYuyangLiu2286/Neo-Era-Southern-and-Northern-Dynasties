    scripted_button = nesnd_india_seek_tur
    scripted_button = nesnd_india_seek_overlord
    # Three grants of aid accepted (nesnd_india_aid_count, donors in nesnd_india_aid_donors). Once. The
    # journal comes back the next month by nesnd_india_refresh while the country may still send missions.
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = nesnd_india_aid_bound_conclusion
        trigger = {
            hidden_trigger = { nesnd_india_mission_country = yes NOT = { has_variable = nesnd_india_aid_bound } }
            custom_tooltip = { text = nesnd_india_aid_three_tt nesnd_india_aid_count_value >= 3 }
        }
        effect = {
            set_variable = nesnd_india_aid_bound
            custom_tooltip = {
                text = nesnd_india_aid_bound_tt
                every_in_list = {
                    variable = nesnd_india_aid_donors
                    limit = { is_country_alive = yes NOT = { this = root } }
                    change_relations = { country = root value = 30 }
                    if = { limit = { exists = root.capital } add_involvement = { strategic_region = root.capital.region value = 1000 } }
                }
            }
            hidden_effect = { remove_variable = nesnd_india_aid_count clear_variable_list = nesnd_india_aid_donors }
        }
    }
    outcome = {
        header = nesnd_journal_closures_header
        effect_header = nesnd_journal_closures_effect_header
        color = "blue_negative"
        conclusion_desc = nesnd_journal_closures_india_france_conclusion
        trigger = { custom_tooltip = { text = nesnd_journal_closures_india_france_tt NOT = { nesnd_india_mission_country = yes } } }
        effect = { custom_tooltip = nesnd_journal_closures_india_foreign_effect_tt nesnd_india_clear_request = yes nesnd_india_remove_advisers = yes }
    }
    status_desc = { desc = nesnd_india_foreign_status } weight = 400
    should_be_pinned_by_default_uninvolved_or_context = yes

    on_monthly_pulse = {
        effect = {
            nesnd_india_mission_labels = yes
            if = {
                limit = { is_ai = yes nesnd_india_mission_ready = yes }
                random = { chance = 14.26 nesnd_india_choose_ai_partner = yes }
            }
        }
    }
}

nesnd_india_revolt_war = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_dhal.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    is_shown_when_inactive = {
        has_variable = nesnd_india_revolt_actor
        NOT = { has_variable = nesnd_india_revolt_resolved }
    }
    possible = {
        custom_tooltip = {
            text = nesnd_ui_india_revolt_actor_tt
            has_variable = nesnd_india_revolt_actor
        }
        custom_tooltip = {
            text = nesnd_ui_not_india_revolt_resolved_tt
            NOT = { has_variable = nesnd_india_revolt_resolved }
        }
    }
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = nesnd_india_revolt_resolved_tt
                nesnd_india_revolt_won = yes
            }
        }
        effect = {
            if = {
                limit = {
                    hidden_trigger = {
                        NOT = { has_variable = nesnd_india_revolt_rewarded }
                    }
                }
                set_variable = nesnd_india_revolt_rewarded
                add_modifier = { name = nesnd_india_revolt_survived }
                # The owner we rose against: a claim on our region, and the empire that paid for the
                # rising becomes its AI target (R4, R5; GAME events/opium_wars_events.txt:254).
                clear_saved_scope = india_revolt_former
                clear_saved_scope = india_revolt_sponsor
                var:nesnd_india_revolt_former ?= { save_scope_as = india_revolt_former }
                var:nesnd_india_revolt_sponsor ?= { save_scope_as = india_revolt_sponsor }
                if = { limit = { exists = scope:india_revolt_former scope:india_revolt_former = { is_country_alive = yes } }
                    custom_tooltip = {
                        text = nesnd_india_revolt_former_claim_tt
                        var:nesnd_india_revolt_origin ?= { add_claim = scope:india_revolt_former }
                    }
                    if = { limit = { exists = scope:india_revolt_sponsor scope:india_revolt_sponsor = { is_country_alive = yes } NOT = { scope:india_revolt_sponsor = scope:india_revolt_former } }
                        custom_tooltip = {
                            text = nesnd_india_revolt_former_goal_tt
                            scope:india_revolt_former = { set_secret_goal = { country = scope:india_revolt_sponsor secret_goal = antagonize } }
                        }
                    }
                }
                trigger_event = { id = nesnd_india_revolt.2 popup = yes }
            }
        }
        event_outcome_desc = { first_valid = { triggered_desc = { desc = nesnd_india_revolt.2_outcome_tt } } }
    }
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = nesnd_ui_india_revolt_resolved_tt
                has_variable = nesnd_india_revolt_resolved
            }
            NOT = { nesnd_india_revolt_won = yes }
        }
        # The radicals were added when the region went back (nesnd_india_case_failed).
        effect = {
            custom_tooltip = nesnd_india_revolt_failed_effect_tt
            hidden_effect = { remove_variable = nesnd_india_revolt_former remove_variable = nesnd_india_revolt_sponsor }
        }
    }
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = nesnd_ui_not_india_revolt_actor_tt
                NOT = { has_variable = nesnd_india_revolt_actor }
            }
        }
    }
    weight = 1000
    should_be_pinned_by_default_uninvolved_or_context = yes
}
