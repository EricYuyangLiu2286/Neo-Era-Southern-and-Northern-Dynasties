# 教士秀全的旅途：全局人物日志，开局由 common/history/global/nesnd_xiuquan.txt 加入。
# 无成败结局；唯一结局是他去世时隐藏失效（由 on_character_death 设 nesnd_xq_dead）。
nesnd_xiuquan_journey = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	group = je_group_global_international_situations

	activation_description = { always = yes }
	possible = { always = no }
	is_shown_when_inactive = { always = no }

	should_be_involved = { nesnd_xq_should_be_involved = yes }
	should_show_when_not_involved = { is_player = yes }
	should_be_pinned_by_default_involved = yes

	immediate = { scope:journal_entry = { nesnd_xq_bind_portrait = yes } }

	widget = {
		gui = "gui/journal_entry_widgets/nesnd_xiuquan.gui"
		name = "nesnd_xq_portrait"
		container = "custom_widget_container_je_icon"
	}

	scripted_button = nesnd_xq_expel
	scripted_progress_bar = nesnd_xq_ministry_bar

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = { nesnd_xq_sync = yes }
		}
	}

	on_monthly_pulse = {
		effect = {
			remove_global_variable = nesnd_xq_expel_result

			# 传教进度满：所在州0.1%人口皈依一种基督教，人们对他信仰的判断各不相同。
			scope:journal_entry = {
				if = {
					limit = { scripted_bar_progress = { name = nesnd_xq_ministry_bar value >= 100 } }
					global_var:nesnd_xq_state ?= {
						random_list = {
							1 = {
								convert_population = { target = rel:catholic value = 0.001 }
								set_global_variable = { name = nesnd_xq_faith value = 1 }
							}
							1 = {
								convert_population = { target = rel:protestant value = 0.001 }
								set_global_variable = { name = nesnd_xq_faith value = 2 }
							}
							1 = {
								convert_population = { target = rel:orthodox value = 0.001 }
								set_global_variable = { name = nesnd_xq_faith value = 3 }
							}
							1 = {
								convert_population = { target = rel:oriental_orthodox value = 0.001 }
								set_global_variable = { name = nesnd_xq_faith value = 4 }
							}
						}
						set_global_variable = { name = nesnd_xq_convert_state value = this }
					}
					set_bar_progress = { name = nesnd_xq_ministry_bar value = 0 }
				}
			}

			# 每月25%：随追随者去一个相邻的州，不分国界。
			random = {
				chance = 25
				global_var:nesnd_xq_state ?= {
					random_neighbouring_state = { save_scope_as = nesnd_xq_next_state }
				}
				scope:nesnd_xq_next_state ?= { nesnd_xq_arrive = yes }
			}

			# 无意外则享年八十：活到1894年1月1日时去世。此前仍可能自然死亡。
			if = {
				limit = { game_date >= 1894.1.1 }
				global_var:nesnd_xq_char ?= { kill_character = yes }
			}

			scope:journal_entry = { nesnd_xq_sync = yes }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_expel_result global_var:nesnd_xq_expel_result = 1 }
				desc = nesnd_xq_status_expelled
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_expel_result global_var:nesnd_xq_expel_result = 2 }
				desc = nesnd_xq_status_stayed
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_faith global_var:nesnd_xq_faith = 1 }
				desc = nesnd_xq_status_catholic
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_faith global_var:nesnd_xq_faith = 2 }
				desc = nesnd_xq_status_protestant
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_faith global_var:nesnd_xq_faith = 3 }
				desc = nesnd_xq_status_orthodox
			}
			triggered_desc = {
				trigger = { has_global_variable = nesnd_xq_faith global_var:nesnd_xq_faith = 4 }
				desc = nesnd_xq_status_nestorian
			}
			desc = nesnd_xq_status_location
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = nesnd_xiuquan_journey_ended
		trigger = { has_global_variable = nesnd_xq_dead }
		effect = {
			global_var:nesnd_xq_state ?= {
				if = {
					limit = { has_modifier = nesnd_xq_ministry }
					remove_modifier = nesnd_xq_ministry
				}
			}
			remove_global_variable = nesnd_xq_expel_result
			remove_global_variable = nesnd_xq_faith
			remove_global_variable = nesnd_xq_convert_state
		}
	}
}
