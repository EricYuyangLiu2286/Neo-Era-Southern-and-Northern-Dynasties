ce_usm_conflicts_within = {	

	# 东西之争：

	icon = "gfx/interface/icons/invention_icons/pan_nationalism.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_historical_content
	can_revolution_inherit = yes
	
	modifiers_while_active = {
		ce_usm_lunzhan_modifier 
	}

	immediate = {
        nesnd_south_initialize = yes

		if = {
			limit = { NOT = { has_variable = usm_conflicts_within_tracker } }
			set_variable = {
				name = usm_conflicts_within_tracker
				value = 0
			}
		}
	
    nesnd_refresh_south_display = yes
}

	#inheritable = no
	
	on_monthly_pulse = {
    effect = {
        if = { limit = { has_modifier = nesnd_federal_compact } change_variable = { name = usm_conflicts_within_tracker add = 0.4 } }
        else = { change_variable = { name = usm_conflicts_within_tracker add = 1 } }
        clamp_variable = { name = usm_conflicts_within_tracker min = 0 max = 100 }
    }
    random_events = { 900 = 0 20 = nesnd_chen.2 }
 }
	
	
	
	complete = {
		custom_tooltip = {
			text = nesnd_usm_conflicts_within_tracker_ge_100
			nesnd_south_pressure_value >= 100
		}
		custom_tooltip = {
			text = nesnd_south_unsettled_tt
			NOT = {
				has_global_variable = nesnd_south_settled
			}
		}
		custom_tooltip = {
			text = nesnd_south_qin_departed_tt
			NOT = {
				any_scope_character = {
					has_template = USM_zheng_qin
					is_character_alive = yes
				}
			}
		}
	}
	
	on_complete = {
		
		trigger_event = { id = flavorUSM.4 }
		set_variable = {
			name = ce_usm_conflict_evoked
			value = yes
		}
		hidden_effect = {
			remove_variable = flavor_USM_crisis_intro
		}
	}
	
	fail = {
		custom_tooltip = {
			text = nesnd_south_conflicts_failure_tt
			trigger_if = {
				limit = {
					num_states >= 2
				}
				OR = {
					NOT = {
						exists = c:SIH
					}
					NOT = {
						c:SIH ?= {
							is_direct_subject_of = root
						}
					}
				}
				OR = {
					NOT = {
						exists = c:HNA
					}
					NOT = {
						c:HNA ?= {
							is_direct_subject_of = root
						}
					}
				}
				OR = {
					NOT = {
						exists = c:PHI
					}
					NOT = {
						c:PHI ?= {
							is_direct_subject_of = root
						}
					}
				}
				OR = {
					NOT = {
						exists = c:DEI
					}
					NOT = {
						c:DEI ?= {
							is_direct_subject_of = root
						}
					}
				}
				OR = {
					NOT = {
						exists = c:YN1
					}
					NOT = {
						c:YN1 ?= {
							is_direct_subject_of = root
						}
					}
				}
				custom_tooltip = {
					text = ce_usm_conflicts_invalid_tt_1
					always = yes
				}
			}
			trigger_if = {
				limit = {
					num_states < 2
				}
				custom_tooltip = {
					text = ce_usm_conflicts_invalid_tt_2
					always = yes
				}
			}
		}
	}

	# effect which is executed when 'fail' trigger becomes true
	on_fail = {
		hidden_effect = {
			trigger_event = { id = flavorUSM.2 } # looming conflicts
			set_variable = {
				name = ce_usm_conflict_looming
				value = yes
			}
		}
		show_as_tooltip = {
			add_modifier = {
				name = USM_looming_conflicts
				years = 35
				is_decaying = yes
			}
		}
		hidden_effect = {
		
			remove_variable = flavor_USM_crisis_intro
		}
	}
	
	invalid = {
		OR = {
			custom_tooltip = {
				text = nesnd_south_settled_tt
				has_global_variable = nesnd_south_settled
			}
			custom_tooltip = {
				text = nesnd_south_wu_established_tt
				has_variable = tag_WUU_var
			}
			custom_tooltip = {
				text = nesnd_south_monarchy_ended_tt
				NOT = {
					has_law_or_variant = law_type:law_monarchy
				}
			}
		}
	}
	
	on_invalid = {
    remove_variable = flavor_USM_crisis_intro
    if = { limit = { NOT = { has_variable = tag_WUU_var } NOT = { has_variable = nesnd_ming_preserved } NOT = { has_variable = nesnd_south_inconclusive_settlement } }
        set_global_variable = nesnd_south_settled
        trigger_event = { id = flavorUSM.3 popup = yes }
    }
 }
	
	
	
	

	weight = 1200


scripted_button = nesnd_federal_compact_button
scripted_button = nesnd_protect_chen_button
scripted_button = nesnd_early_accession

    
    
    
    
    

    
    
    
    

    scripted_progress_bar = nesnd_south_pressure_bar
    scripted_progress_bar = nesnd_south_compact_bar
    on_weekly_pulse = { effect = { nesnd_refresh_south_display = yes } }
    status_desc = { first_valid = {
        triggered_desc = { trigger = { any_scope_character = { has_template = USM_zheng_qin is_character_alive = yes } } desc = nesnd_south_status_qin }
        desc = nesnd_south_status_succession
    } }
}

ce_usm_civilwar_journal = {
	icon = "gfx/interface/icons/infamy_icons/pariah.dds"
	
	can_revolution_inherit = yes
	group = je_group_historical_content
	
	possible = {
		has_variable = ce_usm_civilwar_journal_trigger
		NOT = { has_global_variable = usm_civilwar_occured }
	}
	
	immediate = { set_variable = { name = usm_journal_start_buffer months = 3 } }
	
	fail = { NOT = { has_variable = usm_journal_start_buffer } var:usm_opposition_country_choice <= 0 any_country = { has_variable = usm_opposition_country } }

	on_fail = { trigger_event = { id = usm_civilwar.2 popup = yes } }
	
	invalid = { NOT = { has_variable = usm_journal_start_buffer } var:usm_opposition_country_choice <= 0 NOT = { any_country = { has_variable = usm_opposition_country } } }
	
	on_invalid = { trigger_event = { id = usm_civilwar.3 popup = yes } }
	
	timeout = 180
	
	on_timeout = { if = { limit = { any_country = { has_variable = usm_opposition_country } } trigger_event = { id = usm_civilwar.2 popup = yes } }
    else = { trigger_event = { id = usm_civilwar.3 popup = yes } } }

	weight = 1
	should_be_pinned_by_default_uninvolved_or_context = yes


on_monthly_pulse = { effect = {
    set_variable = { name = usm_opposition_country_choice value = 0 }
    set_variable = { name = usm_opposition_country_count value = 0 }
    every_country = { limit = { has_variable = nesnd_invited_south NOT = { has_variable = nesnd_south_answered } }
        root = { change_variable = { name = usm_opposition_country_choice add = 1 } }
    }
    every_country = { limit = { has_variable = usm_opposition_country }
        root = { change_variable = { name = usm_opposition_country_count add = 1 } }
    }
 } }
}

 }
	
	fail = { always = no }
	
	on_fail = { }
	
	invalid = { always = no }
	
	on_invalid = { }

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
