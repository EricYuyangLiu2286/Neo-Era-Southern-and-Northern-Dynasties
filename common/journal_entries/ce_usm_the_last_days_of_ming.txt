		#$ig_devout$不[concept_powerful]且不[concept_in_government]
		#$ig_rural_folk$不[concept_powerful]且不[concept_in_government]
		
	}
	
	on_complete = {
		
		trigger_event = { id = flavorUSM.4 }
		set_variable = {
			name = ce_usm_conflict_evoked
			value = yes
		}
		hidden_effect = {
			remove_variable = flavor_USM_crisis_intro
		}
	}
	
	fail = {
		trigger_if = {
			limit = {
				num_states >= 2
			}
			OR = {
				NOT = { exists = c:SIH }
				NOT = { c:SIH ?= { is_direct_subject_of = root } }
			}

				NOT = { exists = c:HNA }

			}

				text = ce_usm_conflicts_invalid_tt_2
				always = yes
			}
		}
	}

	on_fail = {
		hidden_effect = {
			trigger_event = { id = flavorUSM.2 } # looming conflicts

				name = ce_usm_conflict_looming

			}

		}
	}
	
	current_value = {
		value = root.var:usm_conflicts_within_tracker
	}
	
	goal_add_value = {
		value = 100
	}

	weight = 1200

}
