# Mughal's own journals (common/scripted_effects/nesnd_india_mughal.txt). Only the country holding
# nesnd_india_mughal_court gets them. Both are added each month by nesnd_mug_upkeep from
# nesnd_india_refresh; they have neither possible nor is_shown_when_inactive, so no instance exists
# until then.

# The old agreement with the overlord: a loan with a guarantee, or redemption of the subject status.
# A loan still being repaid goes on after this journal ends (nesnd_mug_loan_tick).
nesnd_mug_france = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_slate.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 760
    should_be_pinned_by_default_uninvolved_or_context = yes

    reason = {
        first_valid = {
            triggered_desc = { trigger = { has_variable = nesnd_mug_loan_from has_variable = nesnd_mug_loan_by_script } desc = nesnd_mug_france_reason_loan_script }
            triggered_desc = { trigger = { has_variable = nesnd_mug_loan_from } desc = nesnd_mug_france_reason_loan }
            desc = nesnd_mug_france_reason_none
        }
    }
    status_desc = {
        first_valid = {
            triggered_desc = { trigger = { nesnd_india_matter_kind_value = 7 } desc = nesnd_mug_france_status_loan_asked }
            triggered_desc = { trigger = { nesnd_india_matter_kind_value = 8 } desc = nesnd_mug_france_status_redeem_asked }
        }
    }

    scripted_button = nesnd_mug_loan_button
    scripted_button = nesnd_mug_redeem_button

    # No longer a subject, or the overlord is now an Indian country.
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_mug_france_void_conclusion
        trigger = { NOT = { nesnd_mug_france_live = yes } }
    }

    on_monthly_pulse = { effect = { nesnd_mug_france_ai = yes } }
}

# The council of princes, after the reform: railways, famine relief or common defence.
nesnd_mug_council = {
    icon = "gfx/interface/icons/event_icons/nesnd_india_chhatri.dds"
    group = je_group_historical_content
    activation_description = { always = yes }
    can_revolution_inherit = no
    weight = 755
    should_be_pinned_by_default_uninvolved_or_context = yes

    reason = {
        first_valid = {
            triggered_desc = { trigger = { has_variable = nesnd_mug_council_topic } desc = nesnd_mug_council_reason_sitting }
            triggered_desc = { trigger = { has_variable = nesnd_mug_council_wait } desc = nesnd_mug_council_reason_rest }
            desc = nesnd_mug_council_reason_ready
        }
    }

    scripted_button = nesnd_mug_council_button

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_mug_council_void_conclusion
        trigger = { NOT = { nesnd_mug_council_live = yes } }
    }

    on_monthly_pulse = { effect = { nesnd_mug_council_ai = yes } }
}
