nesnd_rep_warning = {
    icon = "gfx/interface/icons/event_icons/event_map.dds"
    group = je_group_historical_content
    can_revolution_inherit = no
    transferable = no
    possible = { always = no }
    is_shown_when_inactive = { always = no }
    should_be_pinned_by_default_uninvolved_or_context = yes
    weight = 1200
    timeout = 90
    status_desc = { desc = nesnd_rep_warning_status }
    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = nesnd_rep_stand_down_desc
        trigger = { nesnd_rep_warning_valid = no }
        effect = { nesnd_rep_stand_down = yes }
    }
    outcome = {
        flags = { timeout }
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        color = "blue_negative"
        conclusion_desc = nesnd_rep_deadline_desc
        effect = {
            if = {
                limit = { nesnd_rep_warning_valid = yes nesnd_rep_confidence <= 500 }
                nesnd_cne_spawn_republicans = yes
                if = { limit = { has_variable = nesnd_cne_republican_rose } nesnd_rep_clear = yes }
                else = { nesnd_rep_stand_down = yes }
            }
            else = { nesnd_rep_stand_down = yes }
        }
    }
}
