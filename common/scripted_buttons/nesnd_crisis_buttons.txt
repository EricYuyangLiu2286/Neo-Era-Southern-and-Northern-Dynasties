nesnd_land_review = { name = nesnd_land_review desc = nesnd_land_review_desc
    visible = { always = yes }
    possible = { nesnd_reform_can_review = yes }
    # AI: only when disorder is worth paying for (2026-11-16). Below 25 the -5 buys little and the 12-month
    # office costs 5% bureaucracy, which can break the "bureaucracy >= 0" completion condition.
    ai_chance = {
        value = 0
        if = { limit = { has_variable = cne_landreform_failure_tracker var:cne_landreform_failure_tracker >= 25 } value = 20 }
    }
    effect = {
        custom_tooltip = nesnd_land_review_outcome_tt
        hidden_effect = { nesnd_reform_review = yes }
    }
}
