nesnd_phi_request_a_button = {
	name = nesnd_phi_request_a_button
	desc = nesnd_phi_request_a_button_desc
	visible = {
		NOT = {
			has_variable = nesnd_phi_b_transferred
		}
		OR = {
			has_dlc_feature = subject_and_bloc_actions
			c:USM ?= {
				has_diplomatic_pact = {
					who = c:PHI
					type = decrease_payments
					is_initiator = yes
				}
			}
		}
	}
	possible = {
		nesnd_phi_can_request_a = yes
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				net_fixed_income < 0
			}
			add = 60
		}
	}
	effect = {
		nesnd_phi_request_a = yes
	}
}

nesnd_phi_cancel_request_button = {
    name = nesnd_phi_cancel_request_button desc = nesnd_phi_cancel_request_button_desc
    visible = { has_variable = nesnd_phi_pending }
    possible = { has_variable = nesnd_phi_pending NOT = { has_variable = nesnd_phi_creating } }
    ai_chance = { value = 0 }
    effect = { nesnd_phi_clear_pending = yes set_variable = { name = nesnd_phi_cooldown days = 30 } }
}
