# 北洋 / 西洋三角贸易的按钮。双向国：买入奴隶 / 卖出奴隶 / 暂停交易；只买国：买入奴隶 / 暂停交易；只卖国：卖出奴隶 / 暂停交易。
# AI 不点按钮，由日志年脉冲按权重重设模式（nesnd_triangle_ai）。

nesnd_north_trade_buy_button = {
	name = "nesnd_triangle_buy"
	desc = "nesnd_triangle_buy_desc"

	visible = {
		nesnd_triangle_region_country = { KEY = north }
		nesnd_north_triangle_can_buy = yes
		nesnd_north_triangle_can_sell = yes
	}

	possible = {
		nesnd_triangle_mode_available = { KEY = north }
		custom_tooltip = {
			text = nesnd_triangle_not_buying_tt
			NOT = { has_variable = nesnd_north_trade_buying }
		}
	}

	selected = {
		has_variable = nesnd_north_trade_buying
	}

	effect = {
		custom_tooltip = nesnd_triangle_to_buying_tt
		hidden_effect = {
			nesnd_triangle_set_mode = { KEY = north MODE = buying }
		}
	}

	ai_chance = {
		value = 0
	}
}
