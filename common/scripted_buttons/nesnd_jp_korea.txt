nesnd_jp_korean_governor = {
    name = nesnd_jp_korean_governor
    desc = nesnd_jp_korean_governor_desc
    visible = { nesnd_jp_korea_available = yes }
    possible = {
        custom_tooltip = { text = nesnd_jp_korea_choose_governor_tt NOT = { nesnd_jp_korea_policy_value = 1 } }
        custom_tooltip = { text = nesnd_jp_korea_wait_tt NOT = { has_variable = nesnd_jp_korea_policy_cooldown } }
        gold_reserves >= nesnd_jp_negotiation_cost
    }
    effect = {
        if = { limit = { nesnd_jp_korea_policy_value = 1 } custom_tooltip = nesnd_jp_korea_current_tt }
        else = {
            show_as_tooltip = { add_treasury = { value = nesnd_jp_negotiation_cost multiply = -1 } }
            custom_tooltip = nesnd_jp_korea_switch_tt
        }
        custom_tooltip = nesnd_jp_korea_governor_terms
        if = { limit = { has_variable = nesnd_jp_korea_policy_cooldown } custom_tooltip = nesnd_jp_korea_remaining_tt }
        hidden_effect = {
            if = { limit = { nesnd_jp_can_change_korea = yes NOT = { nesnd_jp_korea_policy_value = 1 } }
                add_treasury = { value = nesnd_jp_negotiation_cost multiply = -1 }
                set_variable = { name = nesnd_jp_korea_policy value = 1 }
                set_variable = { name = nesnd_jp_korea_policy_cooldown months = 24 }
                remove_variable = nesnd_jp_korea_months
                nesnd_jp_apply_korean_governor = yes
            }
        }
    }
    ai_chance = { value = 10 if = { limit = { nesnd_jp_route_military = yes } add = 20 } }
}

nesnd_jp_korean_envoy = {
    name = nesnd_jp_korean_envoy desc = nesnd_jp_korean_envoy_desc
    visible = { nesnd_jp_owned_korea = yes nesnd_jp_korea_counterpart = yes NOT = { has_variable = nesnd_jp_korea_settled } }
    possible = { gold_reserves >= nesnd_jp_negotiation_cost hidden_trigger = { nesnd_jp_korean_envoy_available = yes }
        is_subject = no
        custom_tooltip = { text = nesnd_jp_korea_envoy_free_tt NOR = { has_variable = nesnd_jp_request_clock has_variable = nesnd_external_pending } }
        custom_tooltip = { text = nesnd_jp_korea_envoy_wait_tt NOT = { has_variable = nesnd_jp_request_cooldown } }
        custom_tooltip = { text = nesnd_jp_korea_home_tt any_scope_state = { NOT = { nesnd_jp_korea_southern_state = yes } } }
        is_at_war = no
        is_active_in_diplomatic_play = no
        custom_tooltip = { text = nesnd_jp_korea_control_tt NOT = { any_scope_state = { nesnd_jp_korea_southern_state = yes NOT = { controller = owner } } } }
        trigger_if = { limit = { exists = c:KOR }
            c:KOR = { NOT = { is_direct_subject_of = prev } }
            trigger_if = { limit = { c:KOR = { is_subject = yes exists = overlord } } NOT = { has_war_with = c:KOR.overlord } }
        }
    }
    effect = {
        custom_tooltip = nesnd_sweep_jp_request_terms
        show_as_tooltip = { add_treasury = { value = nesnd_jp_negotiation_cost multiply = -1 } }
        hidden_effect = { if = { limit = { nesnd_jp_korean_envoy_available = yes }
            c:KOR = {
                save_scope_as = jp_korea
                if = { limit = { is_subject = yes } overlord = { save_scope_as = jp_partner } }
                else = { save_scope_as = jp_partner }
            }
            nesnd_jp_send_request = { KIND = 5 }
        }
     }
    }
    ai_chance = { value = 1 if = { limit = { nesnd_jp_route_representative = yes } add = 5 } }
}
