# 「宫廷力量」(nesnd_palace_*) 与「执政者与派系」(nesnd_cabinet_*) 的按钮。仿原版 08_tsarpoints_buttons.txt：
# cost 只作卡片显示，实际扣费在 effect 中；AI 花费后至少留 20 点。
# 卡片底图在 gfx/interface/illustrations/journal/nesnd_power/。宫廷按钮带 tag nesnd_palace，卡片据此显示宫廷或共和的费用图标。

########## 宫廷力量 ##########

nesnd_palace_btn_law = {
	name = nesnd_palace_btn_law
	desc = nesnd_palace_btn_law_desc
	icon = "gfx/interface/illustrations/journal/nesnd_power/nesnd_power_card_palace_law.dds"
	tags = { nesnd_palace }
	cost = nesnd_power_cost_large
	cooldown = { days = short_modifier_time }
	visible = { always = yes }
	possible = {
		nesnd_power_can_afford = { COST = large }
		enacting_any_law = yes
		trigger_if = {
			limit = { enacting_any_law = yes }
			custom_tooltip = {
				text = nesnd_palace_cond_holder_backs_law
				nesnd_power_has_holder = yes
				var:nesnd_power_holder ?= { law_enactment_stance = { law = owner.currently_enacting_law.type value > neutral } }
			}
		}
	}
	effect = {
		nesnd_power_pay = { COST = large }
		add_enactment_phase = 1
		nesnd_power_add = { BAR = balance AMT = 8 }
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				nesnd_power_ai_can_spare = { COST = large }
				enacting_any_law = yes
				ruler ?= { law_enactment_stance = { law = owner.currently_enacting_law.type value > neutral } }
			}
			add = 60
		}
	}
}

			if = { limit = { var:nesnd_power_holder_ig ?= { ig_approval < 0 } } add = 20 }
		}
	}
}

	}
	effect = {
		nesnd_power_pay = { COST = placate }
		if = {
			limit = { nesnd_power_case_a = yes }
			var:nesnd_power_top_ig ?= { add_modifier = { name = nesnd_power_placated days = normal_modifier_time } }
		}
		else = {
			every_interest_group = {
				limit = { is_in_government = yes NOT = { root.var:nesnd_power_holder_ig ?= this } }
				add_modifier = { name = nesnd_power_placated_small days = normal_modifier_time }
			}
		}
		nesnd_power_center = { AMT = 15 }
	}
	ai_chance = {
		value = 0
		if = {
			limit = { nesnd_power_ai_can_spare = { COST = placate } }
			add = 5
			if = { limit = { OR = { nesnd_power_b >= 70 nesnd_power_b <= 30 } } add = 35 }
		}
	}
}
