# On the answering government's journal (nesnd_india_foreign_affairs). An AI government answers
# through the journal's weekly pulse, so these buttons are for the player.
nesnd_india_matter_permit_button = {
    name = nesnd_india_matter_permit_button
    desc = nesnd_india_matter_consent_desc
    visible = { nesnd_india_matter_consent_role = yes }
    possible = { custom_tooltip = { text = nesnd_india_matter_unanswered_tt NOT = { has_variable = nesnd_india_matter_answered } } }
    ai_chance = { value = 0 }
    effect = {
        custom_tooltip = nesnd_india_matter_permit_tt
        hidden_effect = { nesnd_india_matter_permit = yes }
    }
}

nesnd_india_matter_accept_button = {
    name = nesnd_india_matter_accept_button
    desc = nesnd_india_matter_answer_desc
    visible = { nesnd_india_matter_answer_role = yes }
    possible = {
        custom_tooltip = { text = nesnd_india_matter_unanswered_tt NOT = { has_variable = nesnd_india_matter_answered } }
        nesnd_india_matter_can_accept = yes
    }
    ai_chance = { value = 0 }
    effect = {
        if = { limit = { nesnd_india_matter_is = { KIND = 1 } }
            if = { limit = { nesnd_india_project_offer_survey = yes }
                custom_tooltip = nesnd_india_matter_fund_survey_tt
                show_as_tooltip = { add_treasury = nesnd_india_matter_quote }
            }
            else = {
                if = { limit = { nesnd_india_project_offer_grant = yes } custom_tooltip = nesnd_india_matter_fund_grant_tt }
                else = { custom_tooltip = nesnd_india_matter_fund_stake_tt }
                custom_tooltip = nesnd_india_matter_fund_terms_tt
                show_as_tooltip = { add_treasury = { value = nesnd_india_matter_quote multiply = -1 } }
            }
        }
        else_if = { limit = { OR = { nesnd_india_matter_is = { KIND = 7 } nesnd_india_matter_is = { KIND = 8 } nesnd_india_matter_is = { KIND = 9 } } } nesnd_mug_accept_tooltips = yes }
        else_if = { limit = { OR = { nesnd_india_matter_is = { KIND = 10 } nesnd_india_matter_is = { KIND = 11 } nesnd_india_matter_is = { KIND = 12 } nesnd_india_matter_is = { KIND = 13 } } } nesnd_india_overseas_accept_tooltips = yes }
        else = { nesnd_india_frontier_accept_tooltips = yes }
        hidden_effect = { nesnd_india_matter_accept = yes }
    }
}
