REPLACE:je_confederation_of_the_rhine_confederate_button = {
	name = "je_confederation_of_the_rhine_confederate_button"
	desc = "je_confederation_of_the_rhine_confederate_button_desc"

	visible = {
		NOT = {
			exists = c:RHN
		}
	}

	possible = {
		NOT = {
			exists = c:RHN
		}
		custom_tooltip = {
			text = nesnd_rhine_no_prussian_bloc_tt
			NOT = { any_country = { nesnd_prussian_identity = yes is_power_bloc_leader = yes } }
		}
		any_state_in_german_confederation = {
			state_region = {
				OR = {
					is_homeland = cu:south_german
					is_homeland = cu:north_german
				}
			}
			OR = {
				owner = {
					is_subject_of = ROOT
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
					capital = {
						is_in_geographic_region = geographic_region_german_confederation
					}
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
				}
				owner = root
			}
			count >= 4
		}
	}

	ai_chance = {
		value = 50
	}

	effect = {
		trigger_event = { id = rhine_confederation.1 }
	}
}
