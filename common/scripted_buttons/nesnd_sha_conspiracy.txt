    name = nesnd_sha_plot_investigate
    desc = nesnd_sha_plot_investigate_desc
    visible = { nesnd_sha_native = yes has_journal_entry = nesnd_sha_conspiracy }
    possible = { nesnd_sha_republican_government = yes NOT = { has_variable = nesnd_sha_plot_investigation_wait } }
    effect = {
        add_modifier = { name = nesnd_sha_plot_inquiry months = 12 }
        add_radicals = { value = 0.03 }
        scope:journal_entry = { add_progress = { name = nesnd_sha_plot_bar value = -15 } }
        hidden_effect = {
            set_variable = { name = nesnd_sha_plot_investigation_wait months = 36 }
            nesnd_sha_plot_refresh = yes
        }
    }
    ai_chance = { value = 0 if = { limit = { scope:journal_entry = { "scripted_bar_progress(nesnd_sha_plot_bar)" >= 45 } } value = 25 } }
}
