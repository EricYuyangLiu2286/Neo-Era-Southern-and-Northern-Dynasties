ce_polish_diplo_play_btn = {
	name = "ce_polish_diplo_play_btn_name"
	desc = "ce_polish_diplo_play_btn_desc"

	visible = {
		exists = c:KRA
		c:KRA ?= this
	}

	possible = {
		has_journal_entry = je_ce_polish_war_preparation
		custom_tooltip = {
			text = ce_polish_script_bar_tt_100
			je:je_ce_polish_war_preparation = { "scripted_bar_progress(ce_polish_reconquest_war_prep_bar)" >= 100 }
		}

		custom_tooltip = {
			text = ce_any_state_not_us_tt
			any_state = {
				has_claim_by = c:KRA
				exists = c:KRA
				owner = { NOT = { c:KRA ?= this } }
				OR = {
					state_region = s:STATE_BREST
					state_region = s:STATE_EAST_GALICIA
					state_region = s:STATE_GREATER_POLAND
					state_region = s:STATE_KAUNAS
					state_region = s:STATE_LESSER_POLAND
					state_region = s:STATE_POSEN
					state_region = s:STATE_UPPER_SILESIA
					state_region = s:STATE_VOLHYNIA
					state_region = s:STATE_WEST_GALICIA
					state_region = s:STATE_WEST_PRUSSIA
				}
			}
		}

	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				global_var:ce_polish_claim_state_count > 3

			}
			add = 1000
		}
	}

	effect = {
		je:je_ce_polish_war_preparation ?= {
			add_progress = {
				value = -50
				name = ce_polish_reconquest_war_prep_bar
			}
		}
		random_state = {
			limit = {
				OR = {
					state_region = s:STATE_BREST
					state_region = s:STATE_EAST_GALICIA
					state_region = s:STATE_GREATER_POLAND
					state_region = s:STATE_KAUNAS
					state_region = s:STATE_LESSER_POLAND
					state_region = s:STATE_POSEN
					state_region = s:STATE_UPPER_SILESIA
					state_region = s:STATE_VOLHYNIA
					state_region = s:STATE_WEST_GALICIA
					state_region = s:STATE_WEST_PRUSSIA
				}
				has_claim_by = c:KRA
				exists = c:KRA
				owner = { NOT = { c:KRA ?= this } }
			}
			save_scope_as = target_diplo_play_state
			owner = {
				trigger_event = { id = ce_poland_events.2 popup = yes }
			}
			if = {
				limit = { exists = c:KRA owner = { NOT = { c:KRA ?= this } } }
				c:KRA = { trigger_event = { id = ce_poland_events.8 popup = yes } }
			}
		}
	}
}

ce_polish_dp_state_west_galicia_btn = {
	name = ce_polish_dp_state_west_galicia_btn_name
	desc = ce_polish_dp_state_west_galicia_btn_desc
	visible = { nesnd_poland_active = yes OR = { has_journal_entry = je_ce_polish_war_preparation has_journal_entry = je_ce_polish_territory_claim } }
	possible = {
		nesnd_poland_can_issue_ultimatum = yes
		s:STATE_WEST_GALICIA = { any_scope_state = { nesnd_poland_ultimatum_state = yes NOT = { owner = root } } }
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				global_var:ce_polish_claim_state_count > 3
			}
			add = 30
		}
	}
	effect = {
		if = { limit = { nesnd_poland_can_issue_ultimatum = yes }
			clear_saved_scope = target_diplo_play_state
			save_scope_as = ce_from_country
			s:STATE_WEST_GALICIA = { random_scope_state = { limit = { nesnd_poland_ultimatum_state = yes NOT = { owner = root } } save_scope_as = target_diplo_play_state } }
			if = { limit = { exists = scope:target_diplo_play_state }
				nesnd_poland_pay_ultimatum = yes
				set_variable = { name = nesnd_polish_pending_state value = scope:target_diplo_play_state months = 4 }
				scope:target_diplo_play_state.owner = { trigger_event = { id = ce_poland_events.2 popup = yes } }
			}
		}
	}
}
