# 立国教：三选一（当前国教不显示）。共用 20 年冷却。代价按所立宗教的影响力排名。
nesnd_per_faith_adopt_shia = {
    name = nesnd_per_faith_adopt_shia
    desc = nesnd_per_faith_adopt_shia_desc
    visible = { nesnd_per_faith_regime = yes nesnd_per_has_faith_journal = yes NOT = { country_has_state_religion = rel:shiite } }
    possible = {
        custom_tooltip = { text = nesnd_per_tension_not_frozen_tt NOT = { nesnd_per_tension_frozen = yes } }
        custom_tooltip = { text = nesnd_per_faith_cooldown_tt NOT = { has_variable = nesnd_per_faith_cd } }
    }
    ai_chance = { value = 0 if = { limit = { nesnd_per_faith_is_top = { FAITH = shia } } value = 5 } }
    effect = {
        if = { limit = { nesnd_per_faith_is_top = { FAITH = shia } } custom_tooltip = nesnd_per_faith_adopt_top_tt }
        else_if = { limit = { nesnd_per_faith_is_bottom = { FAITH = shia } } custom_tooltip = nesnd_per_faith_adopt_bottom_tt }
        else = { custom_tooltip = nesnd_per_faith_adopt_middle_tt }
        show_as_tooltip = { set_state_religion = rel:shiite }
        hidden_effect = {
            if = { limit = { nesnd_per_faith_live = yes NOT = { has_variable = nesnd_per_faith_cd } }
                set_variable = { name = nesnd_per_faith_cd months = 240 }
                # 立的不是影响力最高的宗教：最高宗教的信众激进（第二名 5%，第三名 10%）。
                if = { limit = { NOT = { nesnd_per_faith_is_top = { FAITH = shia } } }
                    set_local_variable = { name = nesnd_per_radicals value = 0.05 }
                    if = { limit = { nesnd_per_faith_is_bottom = { FAITH = shia } } set_local_variable = { name = nesnd_per_radicals value = 0.10 } }
                    if = { limit = { nesnd_per_faith_is_top = { FAITH = sunni } } add_radicals = { value = local_var:nesnd_per_radicals religion = rel:sunni } }
                    if = { limit = { nesnd_per_faith_is_top = { FAITH = buddhist } } add_radicals = { value = local_var:nesnd_per_radicals religion = rel:mahayana } }
                    if = { limit = { nesnd_per_faith_is_top = { FAITH = mandate } } add_radicals = { value = local_var:nesnd_per_radicals religion = rel:kaimeng_confucianism } }
                }
                set_state_religion = rel:shiite
            }
        }
    }
}

# 设讲堂：请西秦经师来讲学。最多 6 座，2 年冷却；向东断交或与西秦交战时每年荒废一座。
nesnd_per_faith_lecture_hall = {
    name = nesnd_per_faith_lecture_hall
    desc = nesnd_per_faith_lecture_hall_desc
    visible = { nesnd_per_tang_regime = yes has_journal_entry = nesnd_per_faith_mandate_journal }
    possible = {
        custom_tooltip = { text = nesnd_per_tension_not_frozen_tt NOT = { nesnd_per_tension_frozen = yes } }
        custom_tooltip = { text = nesnd_per_faith_hall_cooldown_tt NOT = { has_variable = nesnd_per_faith_cd_hall } }
        custom_tooltip = {
            text = nesnd_per_faith_hall_max_tt
            trigger_if = { limit = { has_variable = nesnd_per_mandate_halls } var:nesnd_per_mandate_halls < 6 }
            trigger_else = { always = yes }
        }
        custom_tooltip = { text = nesnd_per_faith_hall_ties_tt nesnd_per_qin_ties = yes }
        custom_tooltip = { text = nesnd_per_faith_hall_money_tt gold_reserves >= nesnd_per_hall_cost }
    }
    ai_chance = { value = 0 if = { limit = { nesnd_per_faith_is_bottom = { FAITH = mandate } } value = 5 } }
    effect = {
        custom_tooltip = nesnd_per_faith_hall_effect_tt
        show_as_tooltip = { add_treasury = { value = nesnd_per_hall_cost multiply = -1 } }
        hidden_effect = {
            if = { limit = { nesnd_per_faith_live = yes NOT = { has_variable = nesnd_per_faith_cd_hall } }
                set_variable = { name = nesnd_per_faith_cd_hall months = 24 }
                add_treasury = { value = nesnd_per_hall_cost multiply = -1 }
                if = { limit = { has_variable = nesnd_per_mandate_halls } change_variable = { name = nesnd_per_mandate_halls add = 1 } }
                else = { set_variable = { name = nesnd_per_mandate_halls value = 1 } }
                if = { limit = { var:nesnd_per_mandate_halls > 6 } set_variable = { name = nesnd_per_mandate_halls value = 6 } }
            }
        }
    }
}
