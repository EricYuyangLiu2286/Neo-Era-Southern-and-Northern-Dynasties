nesnd_court_choose_paris = {
    name = nesnd_court_choose_paris
    desc = nesnd_court_choose_paris_desc
    visible = { has_journal_entry = je_nesnd_france_court }
    possible = { nesnd_court_can_choose_paris = yes }
    effect = {
        custom_tooltip = nesnd_court_paris_policy_tt
        hidden_effect = {
            nesnd_court_select_policy = { FOCUS = paris }
        }
    }
    ai_chance = {
        value = 0
        if = { limit = { NOR = { has_variable = nesnd_court_fund_paris has_variable = nesnd_court_fund_provinces } } value = 5 }
    }
}
