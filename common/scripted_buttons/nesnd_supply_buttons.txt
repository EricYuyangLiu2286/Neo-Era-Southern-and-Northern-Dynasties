nesnd_supply_invite_hna = {
    name = nesnd_supply_invite_hna
    desc = nesnd_supply_invite_desc
    visible = {
        NOT = {
            c:HUS ?= this
        }
        NOT = {
            c:HNA ?= this
        }
    }
    possible = {
        trigger_if = {
            limit = {
                exists = c:HNA
            }
            exists = c:HNA
            c:HNA = {
                is_country_alive = yes
            }
            is_at_war = no
            NOT = {
                has_war_with = c:HNA
            }
            influence >= 100
            NOT = {
                has_variable = nesnd_supply_pending
            }
            NOT = {
                has_variable = nesnd_supply_partner
            }
            NOT = {
                has_variable = nesnd_supply_cooldown
            }
            OR = {
                is_ai = no
                NOT = {
                    has_variable = nesnd_supply_self_reliance
                }
            }
        }
        trigger_else = {
            always = no
        }
    }
    ai_chance = {
        value = 1
    }
    effect = {
        if = {
            limit = {
                exists = c:HNA
            }
            remove_variable = nesnd_supply_self_reliance
            set_variable = {
                name = nesnd_supply_pending
                value = c:HNA
                months = 4
            }
            set_variable = {
                name = nesnd_supply_cooldown
                months = 6
            }
            add_modifier = {
                name = nesnd_supply_envoys
                months = 6
            }
            nesnd_supply_save_pending_sponsor = yes
            save_scope_as = nesnd_supply_requester
            c:HNA = {
                trigger_event = {
                    id = nesnd_supply.1
                    days = 7
                    popup = yes
                }
            }
            custom_tooltip = nesnd_supply_invitation_tt
        }
    }
}

                second_country = var:nesnd_supply_partner

                    years = 2

                    {
                        article = trade_privilege
                        source_country = root
                        target_country = var:nesnd_supply_partner

                        article = trade_privilege
                        source_country = var:nesnd_supply_partner
                        target_country = root

                }
            }
        }
        trigger_else = {
            always = no
        }
    }
    ai_chance = {
        value = 1
    }
    effect = {
        if = {
            limit = {
                exists = var:nesnd_supply_partner
            }

                name = nesnd_supply_trade_pending

                months = 4
            }
            set_variable = {
                name = nesnd_supply_trade_cooldown
                months = 6
            }
            save_scope_as = nesnd_supply_requester
            var:nesnd_supply_partner ?= {
                trigger_event = {
                    id = nesnd_supply.2
                    days = 7
                    popup = yes
                }

            custom_tooltip = nesnd_supply_trade_request_tt
        }
    }
