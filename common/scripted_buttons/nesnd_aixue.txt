REPLACE:je_sell_alaska_button = {
	name = "je_sell_alaska_button"
	desc = "je_sell_alaska_button_desc"

	visible = {
		c:ALK ?= {
			is_player = no
		}
	}

	cooldown = { days = 365 }

	ai_chance = {
		value = 0
		if = {
			limit = {
				scaled_debt > 0
			}
			add = 25
		}
		if = {
			limit = {
				has_modifier = declared_bankruptcy
			}
			add = 50
		}
		if = {
			limit = {
				any_country = {
					NOT = { this = root }
					country_rank >= root.country_rank
					has_interest_marker_in_region = sr:region_pacific_coast
					relations:root <= relations_threshold:poor
				}
			}
			add = 25
		}
		if = {
			limit = {
				any_country = {
					NOT = { this = root }
					country_rank >= root.country_rank
					has_interest_marker_in_region = sr:region_pacific_coast
					relations:root <= relations_threshold:cold
				}
			}
			add = 50
		}
	}

	possible = {
		OR = {
			any_scope_state = { nesnd_is_aixue_state = yes }
			any_subject_or_below = {
				any_scope_state = { nesnd_is_aixue_state = yes }
			}
		}
		custom_tooltip = {
			text = friendly_nation_with_interest_in_alaska_tt
			any_country_in_north_america = {
				NOT = { is_subject = yes }
				NOT = { this = root }
				relations:root >= 0
				country_rank >= rank_value:great_power
				any_scope_state = {
					region = sr:region_pacific_coast
					is_incorporated = yes
				}
			}
		}
		custom_tooltip = {
			text = alaska_transaction_ongoing_tt
			NOT = {
				has_global_variable = alaska_transaction_ongoing
			}
		}
	}

	effect = {
		set_global_variable = { name = alaska_transaction_ongoing days = 180 }
		save_scope_as = alaska_seller
		set_variable = { name = alaska_sale_cooldown days = 365 }
		random_country = {
			limit = {
				NOT = { is_subject = yes }
				NOT = { this = root }
				is_in_geographic_region = geographic_region_north_america
				relations:root >= 0
				country_rank >= rank_value:great_power
				any_scope_state = {
					region = sr:region_pacific_coast
					is_incorporated = yes
				}
			}
			trigger_event = { id = alaska.4 popup = yes }
		}
	}
}

REPLACE:je_buy_alaska_button = {
	name = "je_buy_alaska_button"
	desc = "je_buy_alaska_button_desc"

	visible = {
		OR = {
			NOT = {
				exists = c:ALK
			}
			c:ALK ?= {
				is_player = no
			}
		}
	}

	cooldown = { days = 365 }

	ai_chance = {
		value = 100
	}

	possible = {
		is_subject = no
		sr:region_pacific_coast = {
		    any_scope_state = { nesnd_aixue_offer_candidate = yes }
		}
		custom_tooltip = {
			text = alaska_transaction_ongoing_tt
			NOT = {
				has_global_variable = alaska_transaction_ongoing
			}
		}
	}

	effect = {
		set_global_variable = { name = alaska_transaction_ongoing days = 180 }
		trigger_event = {
			id = alaska.4
			popup = yes
		}
	}
}
