nesnd_india_request_4 = {
 name = nesnd_india_request_4 desc = nesnd_india_diplomatic_desc
 # A subject that is not a small state (MUG, BHT, DRA, IR1 to IR6) sends request kind 4,
 # which only a subject below the highest autonomy can use, as vanilla da_increase_autonomy_of_subject
 # (38_subjects_increase_autonomy.txt:12-15). The princes are small states and go through nesnd_small_send_request.
 visible = { nesnd_india_diplomacy_live = yes is_subject = yes trigger_if = { limit = { NOT = { nesnd_small_state_country = yes } } can_increase_autonomy = yes } }
 possible = {
     trigger_if = { limit = { nesnd_small_state_country = yes } nesnd_small_can_request = { KIND = 6 RECEIVER = overlord } }
     trigger_else = { nesnd_india_autonomy_available = yes can_increase_autonomy = yes }
 }
 # A refusal costs liberty desire: an AI asks often only once it wants out (liberty desire 25, the threshold the author set for Mughal's AI redemption, D7).
 ai_chance = { value = 1 if = { limit = { liberty_desire >= 25 } value = 5 } if = { limit = { nesnd_small_state_country = yes } value = nesnd_small_ai_autonomy } }
 effect = { custom_tooltip = nesnd_india_request_tt custom_tooltip = nesnd_india_request_liberty_tt hidden_effect = {
     if = { limit = { nesnd_small_state_country = yes }
         if = { limit = { nesnd_small_can_request = { KIND = 6 RECEIVER = overlord } } overlord = { save_scope_as = small_receiver } nesnd_small_send_request = { KIND = 6 } }
     }
     else = { if = { limit = { nesnd_india_autonomy_available = yes can_increase_autonomy = yes } clear_saved_scope = india_partner overlord = { save_scope_as = india_partner } nesnd_india_send_request = { KIND = 4 } } }
 } }
}
