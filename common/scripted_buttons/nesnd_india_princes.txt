# The colonial owner's suppression: one button, aimed at the state of ours whose region has the furthest-advanced
# organisation against us (nesnd_india_suppress_candidate, nesnd_india_case_state_progress).
nesnd_india_suppress_button = {
    name = nesnd_india_suppress_button
    desc = nesnd_india_suppression_desc
    visible = { nesnd_india_suppress_target_exists = yes }
    possible = {
        gold_reserves >= nesnd_india_suppress_price
        custom_tooltip = {
            text = nesnd_ui_not_india_suppression_cooldown_tt
            NOT = { has_variable = nesnd_india_suppression_cooldown }
        }
    }
    ai_chance = { value = 15 }
    effect = {
        custom_tooltip = nesnd_india_suppression_tt
        show_as_tooltip = {
            add_treasury = {
                value = nesnd_india_suppress_price
                multiply = -1
            }
            ordered_scope_state = {
                limit = { nesnd_india_suppress_candidate = yes }
                order_by = nesnd_india_case_state_progress
                position = 0
                check_range_bounds = no
                custom_tooltip = nesnd_india_suppress_target_tt
                nesnd_india_suppress_fervor = yes
            }
        }
        hidden_effect = {
            if = {
                limit = {
                    nesnd_india_suppress_target_exists = yes
                    gold_reserves >= nesnd_india_suppress_price
                    NOT = { has_variable = nesnd_india_suppression_cooldown }
                }
                add_treasury = {
                    value = nesnd_india_suppress_price
                    multiply = -1
                }
                set_variable = {
                    name = nesnd_india_suppression_cooldown
                    months = 6
                }
                ordered_scope_state = {
                    limit = { nesnd_india_suppress_candidate = yes }
                    order_by = nesnd_india_case_state_progress
                    position = 0
                    check_range_bounds = no
                    nesnd_india_suppress_fervor = yes
                    nesnd_india_suppress_hit = yes
                }
            }
        }
    }
}
