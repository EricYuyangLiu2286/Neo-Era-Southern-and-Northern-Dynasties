    name = nesnd_jp_independence_play desc = nesnd_jp_independence_play_desc
    visible = { is_subject = yes }
    possible = { is_at_war = no hidden_trigger = { nesnd_jp_independence_play_ready = yes }
        nesnd_jp_diplomatic_standing = yes
        is_active_in_diplomatic_play = no
        NOT = { has_law_or_variant = law_type:law_isolationism }
        custom_tooltip = { text = nesnd_jp_independence_cooldown_tt NOT = { has_variable = nesnd_jp_independence_play_lock } }
        trigger_if = { limit = { is_subject = yes exists = overlord }
            NOT = { has_truce_with = overlord }
            overlord = { is_active_in_diplomatic_play = no relations:root < relations_threshold:cordial }
            OR = {
                can_increase_autonomy = no
                any_diplomatic_catalyst = { is_diplomatic_catalyst_type = catalyst_autonomy_increase_denied target = root.overlord }
            }
            trigger_if = { limit = { overlord = { is_subject = yes } }
                overlord.overlord = { relations:root < relations_threshold:cordial }
            }
        }
    }
    effect = { if = { limit = { nesnd_jp_independence_play_ready = yes }
            overlord = { save_scope_as = jp_old_overlord }
            if = { limit = { exists = scope:jp_old_overlord }
                create_diplomatic_play = { target_country = scope:jp_old_overlord type = dp_independence }
                set_variable = { name = nesnd_jp_independence_play_lock months = 6 }
            }
        }
    }
    ai_chance = { value = 0 if = { limit = { liberty_desire >= 90 }
            if = { limit = { exists = overlord } if = { limit = { army_power_projection >= overlord.army_power_projection } value = 15 } }
        }
    }
