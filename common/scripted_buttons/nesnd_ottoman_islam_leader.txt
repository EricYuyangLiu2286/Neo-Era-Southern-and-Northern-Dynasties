    name = nesnd_ottoman_seek_qin_pact
    desc = nesnd_ottoman_seek_qin_pact_desc
    visible = { has_journal_entry = nesnd_ottoman_islam_leader }
    possible = {
        exists = c:QE1
        c:QE1 = { is_country_alive = yes is_subject = no }
        custom_tooltip = { text = nesnd_north_foe_valid_tt nesnd_north_foe_valid = yes }
        "relations(c:QE1)" >= relations_threshold:amicable
        nesnd_crimea_can_sign_defensive_pact_with = { PARTNER = c:QE1 }
        custom_tooltip = { text = nesnd_ottoman_qin_cd_tt NOT = { has_variable = nesnd_ottoman_qin_diplomacy_cd } }
    }
    effect = {
        custom_tooltip = nesnd_ottoman_seek_qin_pact_tt
        c:QE1 ?= { trigger_event = { id = nesnd_ottoman_qe1.2 days = 3 popup = yes } }
        custom_tooltip = nesnd_ottoman_qin_cd_set_tt
        hidden_effect = { set_variable = { name = nesnd_ottoman_qin_diplomacy_cd months = 24 } }
    }
    ai_chance = {
        value = 5
        if = { limit = { has_global_variable = nesnd_crimea_active } add = 35 }
        if = { limit = { has_variable = nesnd_ottoman_charter_outcome var:nesnd_ottoman_charter_outcome = flag:qin } add = 10 }
        min = 0
    }
}

# Launch the Crimean / Northern War diplomatic play against global_var:nesnd_north_foe (Rules 7, 20.5).
nesnd_ottoman_challenge_russia = {
    name = nesnd_ottoman_challenge_russia
    desc = nesnd_ottoman_challenge_russia_desc
    visible = {
        has_journal_entry = nesnd_ottoman_islam_leader
        NOT = { has_global_variable = nesnd_crimea_active }
        NOT = { has_variable = nesnd_crimea_tur_closed }
        OR = {
            NOT = { has_global_variable = nesnd_crimea_result }
            global_var:nesnd_crimea_result ?= flag:changed
        }
    }
    possible = {
        custom_tooltip = { text = nesnd_north_foe_valid_tt nesnd_north_foe_valid = yes }
        nesnd_ottoman_can_act_abroad = yes
        custom_tooltip = { text = nesnd_crimea_not_active_tt NOT = { has_global_variable = nesnd_crimea_active } }
        custom_tooltip = { text = nesnd_crimea_not_ended_tt nesnd_crimea_can_open = yes }
        trigger_if = {
            limit = { nesnd_north_foe_valid = yes }
            custom_tooltip = {
                text = nesnd_north_foe_peaceful_tt
                global_var:nesnd_north_foe ?= {
                    NOT = { has_war_with = root }
                    NOT = { is_diplomatic_play_enemy_of = root }
                    NOT = { is_diplomatic_play_ally_of = root }
                    NOT = { has_truce_with = root }
                }
            }
        }
        custom_tooltip = { text = nesnd_crimea_rus_holds_target_tt nesnd_crimea_rus_holds_target = yes }
    }
    effect = {
        custom_tooltip = nesnd_ottoman_challenge_russia_tt
        nesnd_crimea_tur_launch_play = yes
    }
    ai_chance = {
        value = 0
        if = {
            limit = {
                game_date >= 1850.1.1
                nesnd_ottoman_enthusiasm_value >= 50
            }
            add = 1
            if = { limit = { nesnd_ottoman_has_major_ally = yes } add = 4 }
            if = { limit = { nesnd_ottoman_enthusiasm_value >= 75 } add = 2 }
        }
        if = { limit = { is_at_war = yes } multiply = 0 }
        min = 0
    }
}
