# Buttons for "Leader of the Islamic World" (nesnd_ottoman_islam_leader) and the Ottoman side of the Crimean War (nesnd_crimea_tur).

# Node "Central Asia and Western Qin": fund frontier tribes in Western Qin (shares 24-month cooldown with seek_qin_pact).
nesnd_ottoman_fund_tribes = {
    name = nesnd_ottoman_fund_tribes
    desc = nesnd_ottoman_fund_tribes_desc
    visible = { has_journal_entry = nesnd_ottoman_islam_leader }
    possible = {
        exists = c:QE1
        NOT = { has_treaty_alliance_with = { TARGET = c:QE1 } }
        NOT = { has_treaty_defensive_pact_with = { TARGET = c:QE1 } }
        custom_tooltip = { text = nesnd_ottoman_qin_cd_tt NOT = { has_variable = nesnd_ottoman_qin_diplomacy_cd } }
    }
    effect = {
        add_treasury = { value = nesnd_ottoman_fund_tribes_cost multiply = -1 }
        c:QE1 ?= {
            add_modifier = { name = nesnd_ottoman_tribes_funded months = 36 }
            change_relations = { country = root value = -25 }
            trigger_event = { id = nesnd_ottoman_qe1.1 days = 3 popup = yes }
        }
        custom_tooltip = nesnd_ottoman_qin_cd_set_tt
        hidden_effect = { set_variable = { name = nesnd_ottoman_qin_diplomacy_cd months = 24 } }
    }
    ai_chance = {
        value = 5
        if = { limit = { c:QE1 ?= { has_variable = nesnd_qe1_unity var:nesnd_qe1_unity < 50 } } add = 20 }
        if = {
            limit = {
                c:QE1 ?= {
                    OR = {
                        nesnd_qe1_fr_bar_at_least = { WHO = afg V = 40 }
                        nesnd_qe1_fr_bar_at_least = { WHO = kaz V = 40 }
                    }
                }
            }
            add = 10
        }
        if = { limit = { has_variable = nesnd_ottoman_charter_outcome var:nesnd_ottoman_charter_outcome = flag:qin } multiply = 0.25 }
        if = { limit = { has_global_variable = nesnd_crimea_active } multiply = 0.2 }
        min = 0
    }
}

# Node "Central Asia and Western Qin": propose a defensive pact against Russia (shares 24-month cooldown with fund_tribes).

    }
    ai_chance = {
        value = 5

        min = 0
    }
}

# Launch the Crimean War diplomatic play against Russia (10a).
nesnd_ottoman_challenge_russia = {
    name = nesnd_ottoman_challenge_russia
    desc = nesnd_ottoman_challenge_russia_desc
    visible = {
        has_journal_entry = nesnd_ottoman_islam_leader
        NOT = { has_global_variable = nesnd_crimea_active }
        NOT = { has_global_variable = nesnd_crimea_result }
    }
    possible = {
        exists = c:RUS
        is_subject = no
        c:RUS = { is_subject = no }
        custom_tooltip = { text = nesnd_crimea_not_active_tt NOT = { has_global_variable = nesnd_crimea_active } }
        custom_tooltip = { text = nesnd_crimea_not_ended_tt NOT = { has_global_variable = nesnd_crimea_result } }
        NOT = { has_war_with = c:RUS }
        NOT = { is_diplomatic_play_enemy_of = c:RUS }
        NOT = { is_diplomatic_play_ally_of = c:RUS }
        NOT = { has_truce_with = c:RUS }
        NOT = { has_treaty_alliance_with = { TARGET = c:RUS } }
        NOT = { has_treaty_defensive_pact_with = { TARGET = c:RUS } }
        custom_tooltip = { text = nesnd_crimea_rus_holds_target_tt nesnd_crimea_rus_holds_target = yes }
    }
    effect = {
        custom_tooltip = nesnd_ottoman_challenge_russia_tt
        nesnd_crimea_tur_launch_play = yes
    }
    ai_chance = {
        value = 0
        if = {
            limit = {
                game_date >= 1850.1.1
                nesnd_ottoman_enthusiasm_value >= 50
            }
            add = 1
            if = { limit = { nesnd_ottoman_has_major_ally = yes } add = 4 }
            if = {
                limit = {
                    exists = c:QE1
                    OR = {
                        has_treaty_defensive_pact_with = { TARGET = c:QE1 }
                        has_treaty_alliance_with = { TARGET = c:QE1 }
                    }
                }
                add = 4
            }
            if = { limit = { nesnd_ottoman_enthusiasm_value >= 75 } add = 2 }
        }
        min = 0
    }
}
