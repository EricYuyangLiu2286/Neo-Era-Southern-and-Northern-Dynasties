nesnd_poland_university_btn = {
	name = nesnd_poland_university_name
	desc = nesnd_poland_university_desc
	visible = { c:KRA ?= this }
	possible = {
		exists = s:STATE_GREATER_POLAND.region_state:KRA
		# Claude audit 2026-11-17: NOT { levels >= 3 } rendered as "not more than or equal to 3". Vanilla
		# only uses ">=" inside this trigger, so the same test is kept and shown as one named line.
		custom_tooltip = {
			text = nesnd_poland_university_below_3_tt
			NOT = { has_building_levels_in_state_region = { region = STATE_GREATER_POLAND building = building_university levels >= 3 } }
		}
		in_default = no
	}
	ai_chance = { value = 1 }
	effect = {
		add_treasury = { value = fixed_income min = 0 multiply = -4 }
		s:STATE_GREATER_POLAND.region_state:KRA = { create_building = { building = building_university level = 3 } }
	}
}
