# 「统治者与内阁」的按钮（诏令卡片，界面见 gui/journal_entry_widgets/nesnd_premier.gui）。Root = country。
# 费用以首相人气支付：每个按钮给首相加一个 2.5 年的人物修正（nesnd_premier_cost_*），cost 字段只作卡片显示。
# 冷却均不短于 2.5 年，同一按钮的费用修正不会叠加。

nesnd_premier_btn_law = {
	name = nesnd_premier_btn_law
	desc = nesnd_premier_btn_law_desc
	icon = "gfx/interface/illustrations/journal/nesnd_power/nesnd_power_card_cabinet_law.dds"
	cost = nesnd_premier_cost_large
	cooldown = { days = short_modifier_time }
	visible = { always = yes }
	possible = {
		custom_tooltip = { text = nesnd_premier_cond_has_pm nesnd_premier_has_pm = yes }
		enacting_any_law = yes
		trigger_if = {
			limit = { enacting_any_law = yes nesnd_premier_has_pm = yes }
			custom_tooltip = {
				text = nesnd_premier_cond_pm_backs_law
				var:nesnd_premier_pm ?= { law_enactment_stance = { law = owner.currently_enacting_law.type value > neutral } }
			}
		}
	}
	effect = {
		nesnd_premier_pay = { BTN = law DELTA = -20 }
		add_enactment_phase = 1
		nesnd_premier_add = { BAR = balance AMT = 8 }
	}
	ai_chance = {
		value = 0
		if = { limit = { nesnd_premier_pop_v >= 0 } add = 60 }
	}
}

nesnd_premier_btn_placate = {
	name = nesnd_premier_btn_placate
	desc = nesnd_premier_btn_placate_desc
	icon = "gfx/interface/illustrations/journal/nesnd_power/nesnd_power_card_cabinet_placate.dds"
	cost = nesnd_premier_cost_regular
	cooldown = { days = normal_modifier_time }
	visible = { always = yes }
	possible = {
		custom_tooltip = { text = nesnd_premier_cond_has_pm nesnd_premier_has_pm = yes }
	}
	effect = {
		nesnd_premier_pay = { BTN = placate DELTA = -10 }
		if = {
			limit = { nesnd_premier_mode_rival = yes }
			var:nesnd_premier_rival_ig ?= { add_modifier = { name = nesnd_power_placated days = normal_modifier_time } }
		}
		else = {
			save_scope_as = nesnd_premier_country
			every_interest_group = {
				limit = { NOT = { scope:nesnd_premier_country.var:nesnd_premier_pm_ig ?= this } ig_clout >= 0.05 }
				add_modifier = { name = nesnd_power_placated_small days = normal_modifier_time }
			}
		}
		nesnd_premier_center = { AMT = 15 }
	}
	ai_chance = {
		value = 0
		if = {
			limit = { nesnd_premier_pop_v >= 0 }
			add = 5
			if = { limit = { OR = { nesnd_premier_b >= 70 nesnd_premier_b <= 30 } } add = 35 }
		}
	}
}

	effect = {

		add_modifier = { name = nesnd_power_restraint days = short_modifier_time }
		nesnd_premier_sub = { BAR = intrigue AMT = 20 }

		if = {
			limit = { ruler ?= { popularity >= 20 } }

		}

		}
	}
	ai_chance = {
		value = 0
		if = { limit = { nesnd_premier_i >= 60 } add = 20 }
	}
}
