REPLACE:je_the_eastern_border_treaty_of_beijing = {
	name = "je_the_eastern_border_treaty_of_beijing"
	desc = "je_the_eastern_border_treaty_of_beijing_desc"

	visible = {
		exists = c:RUS
		c:RUS = ROOT
		is_subject = no
		has_technology_researched = nationalism
		NOR = {
			AND = {
				owns_entire_state_region = STATE_OUTER_MANCHURIA
				owns_entire_state_region = STATE_BOLI
			}
			has_variable = rus_bought_outer_manchuria
		}
		any_neighbouring_state = {
			OR = {
				state_region = s:STATE_OUTER_MANCHURIA
				state_region = s:STATE_BOLI
				state_region = s:STATE_AMUR
			}
		}
		any_country_in_qing_empire = {
            c:CHI ?= this
			OR = {
				country_has_primary_culture = cu:manchu
				country_has_primary_culture = cu:han
			}
			any_scope_state = {
				state_region = s:STATE_OUTER_MANCHURIA
				is_capital = no
			}
			is_country_type = unrecognized
			is_player = no
		}
	}

	ai_chance = {
		value = 10
		if = {
			limit = {
				has_journal_entry = je_tp_tsar_gendarme_of_europe
			}
			add = 90
		}
	}

	possible = {
		c:RUS ?= {
			country_rank >= rank_value:great_power
		}
		any_neighbouring_state = {
			OR = { state_region = s:STATE_OUTER_MANCHURIA state_region = s:STATE_BOLI }
		}
		any_country_in_qing_empire = {
            c:CHI ?= this
			any_scope_state = {
				state_region = s:STATE_OUTER_MANCHURIA
				is_capital = no
			}
			c:CHI ?= this
                country_is_valid_qing_treaty_partner = yes
		}
	}

	effect = {
        clear_saved_scope = donor_scope
		random_country_in_qing_empire = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_OUTER_MANCHURIA
					is_capital = no
				}
				c:CHI ?= this
                country_is_valid_qing_treaty_partner = yes
			}
			save_scope_as = donor_scope
		}
        if = { limit = { nesnd_rus_chi_donor_live = yes }

		if = {
			limit = {
				scope:donor_scope = {
					owes_obligation_to = ROOT
				}
			}
			custom_tooltip = obligation_for_outer_manchuria_tt
		}
		else_if = {
			limit = {
				scope:donor_scope = {
					is_subject_of = ROOT
				}
			}
			custom_tooltip = transfer_of_outer_manchuria_tt
		}
		else = {
			if = {
				limit = {
					exists = scope:donor_scope
				}
				c:RUS ?= {
					create_incident = {
						country = scope:donor_scope
						value = 10
					}
				}
			}
			custom_tooltip = transfer_of_outer_manchuria_tt
		}
		set_variable = { name = rus_bought_outer_manchuria months = 4 }
		trigger_event = { id = russo_chinese.1 popup = yes }
	
        }
    }
}
