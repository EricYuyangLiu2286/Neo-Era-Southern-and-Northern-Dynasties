# Journal nesnd_ottoman_charter. Two levers that only move interest group strength (rule R2); shared 24-month cooldown.
nesnd_ottoman_charter_assembly = {
    name = nesnd_ottoman_charter_assembly desc = nesnd_ottoman_charter_assembly_desc
    visible = { has_journal_entry = nesnd_ottoman_charter }
    possible = {
        custom_tooltip = { text = nesnd_ottoman_charter_cooldown_tt NOT = { has_variable = nesnd_ottoman_charter_lever_cd } }
        ig:ig_intelligentsia ?= { is_marginal = no }
    }
    effect = {
        add_modifier = { name = nesnd_ottoman_assembly_cost months = 12 }
        ig:ig_intelligentsia ?= { add_modifier = { name = nesnd_ottoman_charter_rally months = 12 } }
        custom_tooltip = nesnd_ottoman_charter_cooldown_set_tt
        hidden_effect = { set_variable = { name = nesnd_ottoman_charter_lever_cd months = 24 } }
    }
    ai_chance = {
        value = 10
        subtract = { value = nesnd_ottoman_qinist_leader_count multiply = 5 }
        min = 0
    }
}
nesnd_ottoman_charter_discipline = {
    name = nesnd_ottoman_charter_discipline desc = nesnd_ottoman_charter_discipline_desc
    visible = { has_journal_entry = nesnd_ottoman_charter }
    possible = {
        custom_tooltip = { text = nesnd_ottoman_charter_cooldown_tt NOT = { has_variable = nesnd_ottoman_charter_lever_cd } }
        ig:ig_armed_forces ?= { is_marginal = no }
    }
    effect = {
        ig:ig_armed_forces ?= { add_modifier = { name = nesnd_ottoman_charter_rally months = 12 } }
        every_interest_group = {
            limit = { leader ?= { has_ideology = ideology:ideology_ce_qin_royalist_leader } }
            leader = { add_modifier = { name = nesnd_ottoman_qinist_acclaim months = 12 } }
        }
        custom_tooltip = nesnd_ottoman_charter_cooldown_set_tt
        hidden_effect = { set_variable = { name = nesnd_ottoman_charter_lever_cd months = 24 } }
    }
    ai_chance = {
        value = 5
        add = { value = nesnd_ottoman_qinist_leader_count multiply = 10 }
    }
}
