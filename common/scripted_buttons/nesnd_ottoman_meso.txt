# Scripted buttons for Ottoman B9 "Mesopotamia" journal (nesnd_ottoman_meso).

# Rule 11: Contact Mesopotamian tribes (36-month cooldown).
nesnd_ottoman_meso_tribes = {
    name = nesnd_ottoman_meso_tribes
    desc = nesnd_ottoman_meso_tribes_desc
    visible = {
        has_journal_entry = nesnd_ottoman_meso
    }
    possible = {
        custom_tooltip = {
            text = nesnd_ottoman_meso_tribes_cd_tt
            NOT = { has_variable = nesnd_ottoman_meso_tribes_cd }
        }
        custom_tooltip = {
            text = nesnd_ottoman_meso_foe_valid_tt
            nesnd_ottoman_meso_foe_valid = yes
        }
    }
    effect = {
        s:STATE_BAGHDAD = {
            every_scope_state = {
                limit = {
                    nesnd_ottoman_state_outside_ottoman_hands = yes
                }
                add_radicals_in_state = {
                    value = small_radicals
                }
            }
        }
        s:STATE_MOSUL = {
            every_scope_state = {
                limit = {
                    nesnd_ottoman_state_outside_ottoman_hands = yes
                }
                add_radicals_in_state = {
                    value = small_radicals
                }
            }
        }
        s:STATE_BASRA = {
            every_scope_state = {
                limit = {
                    nesnd_ottoman_state_outside_ottoman_hands = yes
                }
                add_radicals_in_state = {
                    value = small_radicals
                }
            }
        }
        if = {
            limit = {
                has_global_variable = nesnd_meso_foe
            }
            global_var:nesnd_meso_foe ?= {
                change_relations = {
                    country = root
                    value = -15
                }
            }
        }
        if = {
            limit = {
                any_country = {
                    nesnd_ottoman_meso_state_holder = yes
                }
            }
            ordered_country = {
                limit = {
                    nesnd_ottoman_meso_state_holder = yes
                }
                order_by = nesnd_ottoman_meso_held_states_count
                position = 0
                check_range_bounds = no
                if = {
                    limit = {
                        NOT = { has_variable = nesnd_meso_alarm_cd }
                    }
                    trigger_event = {
                        id = nesnd_ottoman_recovery.23
                        days = 2
                        popup = yes
                    }
                }
            }
        }
        custom_tooltip = nesnd_ottoman_meso_tribes_cd_set_tt
        hidden_effect = {
            set_variable = {
                name = nesnd_ottoman_meso_tribes_cd
                months = 36
            }
        }
    }
    ai_chance = {
        value = 5
        if = {
            limit = {
                nesnd_ottoman_meso_has_target_claim = yes
            }
            add = 10
        }
        if = {
            limit = {
                is_at_war = yes
            }
            multiply = 0
        }
    }
}

# Rule 12: Set eastern campaign plan (once per journal opening; requires a claim on at least one target region).
nesnd_ottoman_meso_plan = {
    name = nesnd_ottoman_meso_plan
    desc = nesnd_ottoman_meso_plan_desc
    visible = {
        has_journal_entry = nesnd_ottoman_meso
    }
    possible = {
        custom_tooltip = {
            text = nesnd_ottoman_meso_not_planned_tt
            NOT = { has_variable = nesnd_ottoman_meso_planned }
        }
        custom_tooltip = {
            text = nesnd_ottoman_meso_has_claim_tt
            nesnd_ottoman_meso_has_target_claim = yes
        }
        custom_tooltip = {
            text = nesnd_ottoman_meso_foe_valid_tt
            nesnd_ottoman_meso_foe_valid = yes
        }
    }
    effect = {
        s:STATE_BAGHDAD = {
            nesnd_ottoman_conquer_goal_on_holders = yes
        }
        s:STATE_MOSUL = {
            nesnd_ottoman_conquer_goal_on_holders = yes
        }
        s:STATE_BASRA = {
            nesnd_ottoman_conquer_goal_on_holders = yes
        }
        add_modifier = {
            name = nesnd_ottoman_meso_campaign_mod
            months = 60
        }
        hidden_effect = {
            set_variable = nesnd_ottoman_meso_planned
        }
    }
    ai_chance = {
        value = 20
        if = {
            limit = {
                is_at_war = yes
            }
            multiply = 0
        }
    }
}

# Rule 13: Propose border demarcation to the Mesopotamia foe (60-month cooldown).
nesnd_ottoman_meso_demarcate = {
    name = nesnd_ottoman_meso_demarcate
    desc = nesnd_ottoman_meso_demarcate_desc
    visible = {
        has_journal_entry = nesnd_ottoman_meso
    }
    possible = {
        custom_tooltip = {
            text = nesnd_ottoman_meso_foe_valid_tt
            nesnd_ottoman_meso_foe_valid = yes
        }
        custom_tooltip = {
            text = nesnd_ottoman_meso_peace_with_foe_tt
            nesnd_ottoman_meso_at_war_with_foe = no
            nesnd_ottoman_meso_play_in_progress = no
        }
        custom_tooltip = {
            text = nesnd_ottoman_meso_parley_cd_tt
            NOT = { has_variable = nesnd_ottoman_meso_parley_cd }
        }
    }
    effect = {
        if = {
            limit = {
                has_global_variable = nesnd_meso_foe
            }
            global_var:nesnd_meso_foe ?= {
                trigger_event = {
                    id = nesnd_ottoman_recovery.24
                    days = 3
                    popup = yes
                }
            }
        }
        custom_tooltip = nesnd_ottoman_meso_parley_cd_set_tt
        hidden_effect = {
            set_variable = {
                name = nesnd_ottoman_meso_parley_cd
                months = 60
            }
        }
    }
    ai_chance = {
        value = 0
        if = {
            limit = {
                has_variable = nesnd_ottoman_route
                var:nesnd_ottoman_route = flag:sultanic
                nesnd_ottoman_qinist_leader_count >= 1
            }
            add = 5
        }
        if = {
            limit = {
                is_at_war = yes
            }
            add = 10
        }
        if = {
            limit = {
                has_variable = nesnd_ottoman_meso_planned
            }
            multiply = 0
        }
    }
}
