nesnd_per_civil_grant_button = {
    name = nesnd_per_civil_grant_button desc = nesnd_per_civil_grant_button_desc
    visible = { nesnd_per_country = yes }
    possible = { custom_tooltip = { text = nesnd_per_cost_20_tt nesnd_per_governance_points >= 20 } NOT = { has_modifier = nesnd_per_civil_grant } }
    effect = {
        custom_tooltip = nesnd_per_spend_20_tt
        show_as_tooltip = { add_modifier = { name = nesnd_per_civil_grant months = 12 } }
        hidden_effect = { if = { limit = { nesnd_per_country = yes has_journal_entry = nesnd_per_governance nesnd_per_governance_points >= 20 NOT = { has_modifier = nesnd_per_civil_grant } }
            change_variable = { name = nesnd_per_governance_points add = -20 }
            add_modifier = { name = nesnd_per_civil_grant months = 12 }
            nesnd_per_governance_sync = yes
        } }
    }
    ai_chance = { value = 0 if = { limit = { government_legitimacy < 50 } add = 20 } else_if = { limit = { nesnd_per_governance_points >= 60 } add = 5 } }
}
