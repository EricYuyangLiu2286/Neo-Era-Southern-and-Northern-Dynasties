nesnd_ottoman_reform_army = {
    name = nesnd_ottoman_reform_army desc = nesnd_ottoman_reform_army_desc
    visible = { nesnd_ottoman_recovery_open = yes }
    possible = { custom_tooltip = { text = nesnd_ottoman_reform_not_active NOT = { nesnd_ottoman_reform_active = yes } NOT = { has_variable = nesnd_ottoman_reform_pending } } }
    effect = {
        if = { limit = { nesnd_ottoman_tanzimat_success = yes }
            show_as_tooltip = { add_modifier = { name = nesnd_ottoman_qin_reform_success months = 12 } }
        }
        else_if = { limit = { nesnd_ottoman_tanzimat_failed = yes }
            show_as_tooltip = { add_modifier = { name = nesnd_ottoman_qin_reform_failed months = 12 } }
        }
        else = {
            show_as_tooltip = { add_modifier = { name = nesnd_ottoman_qin_reform months = 12 } }
        }
        custom_tooltip = nesnd_unity_fixed_term_tt
        hidden_effect = { nesnd_ottoman_start_reform = yes }
    }
    ai_chance = { value = 20 if = { limit = { is_at_war = yes } add = 60 } }
}
