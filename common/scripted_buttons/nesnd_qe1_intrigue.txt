# 「伊犁的朝议」的朝争按钮。

nesnd_qe1_calm_court_button = {
	name = nesnd_qe1_calm_court_button
	desc = nesnd_qe1_calm_court_button_desc
	cooldown = { days = short_modifier_time }
	visible = { nesnd_qe1_court_live = yes }
	possible = {
		custom_tooltip = { text = nesnd_qe1_condition_intrigue_30 has_variable = nesnd_qe1_intrigue var:nesnd_qe1_intrigue >= 30 }
	}
	effect = {
		nesnd_qe1_intrigue_change = { DIR = subtract AMT = 40 }
		add_modifier = { name = nesnd_qe1_court_purge days = short_modifier_time }
	}
	ai_chance = {
		value = 0
		if = { limit = { has_variable = nesnd_qe1_intrigue var:nesnd_qe1_intrigue >= 70 } add = 15 }
	}
}

		}
	}
