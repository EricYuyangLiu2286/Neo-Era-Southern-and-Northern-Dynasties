nesnd_india_republican_settlement = { name = nesnd_india_republican_settlement desc = nesnd_india_settlement_desc
    visible = { nesnd_india_society_live = yes NOT = { has_variable = nesnd_india_national_settlement } }
    possible = { custom_tooltip = { text = nesnd_india_community_ready_tt nesnd_india_community >= 75 } has_technology_researched = egalitarianism }
    # A monarchy whose dynasty is still accepted keeps its crown. The AI takes this road only when the
    # dynasty's support is low, or when the country is no longer a monarchy.
    ai_chance = {
        value = 0
        if = { limit = { NOT = { country_has_monarchy_law = yes } } value = 10 }
        else_if = { limit = { nesnd_india_dynasty < 30 } value = 10 }
    }
    effect = {
        show_as_tooltip = { nesnd_india_settlement_republic = yes nesnd_india_settlement_power_law = yes add_modifier = { name = nesnd_india_national_settlement months = 120 } }
        hidden_effect = {
            if = { limit = { nesnd_india_society_live = yes custom_tooltip = { text = nesnd_ui_not_india_national_settlement_tt NOT = { has_variable = nesnd_india_national_settlement } } nesnd_india_community >= 75 has_technology_researched = egalitarianism }
                set_variable = nesnd_india_national_settlement
                nesnd_india_settlement_republic = yes
                nesnd_india_settlement_power_law = yes
                add_modifier = { name = nesnd_india_national_settlement months = 120 }
            }
        }
    }
}
