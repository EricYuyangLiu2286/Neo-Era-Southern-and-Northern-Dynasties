nesnd_board_choose_ming = {
	name = nesnd_board_choose_ming
	desc = nesnd_board_choose_ming_desc
	visible = {
		is_involved_in_journal_entry = nesnd_south_succession_board
		OR = {
			c:USM ?= this
			nesnd_south_political_member = yes
		}
		NOT = {
			nesnd_board_fixed_ming_allegiance = yes
		}
	}
	possible = {
		custom_tooltip = {
			text = nesnd_board_stance_available_tt
			nesnd_board_can_choose = yes
		}
		custom_tooltip = {
			text = nesnd_board_not_ming_tt
			NOT = {
				nesnd_board_stance_value = 1
			}
		}
	}
	ai_chance = {
		value = 0
	}
	effect = {
		if = {
			limit = {
				nesnd_board_can_choose = yes
			}
			custom_tooltip = nesnd_board_stance_effect_tt
			hidden_effect = {
				if = {
					limit = {
						has_variable = nesnd_south_accord_active
					}
					nesnd_south_withdraw_accord = yes
				}
				set_variable = {
					name = nesnd_south_stance
					value = 1
				}
				set_variable = {
					name = nesnd_board_stance_cooldown
					months = 6
				}
				nesnd_board_refresh_all = yes
			}
		}
	}
}
