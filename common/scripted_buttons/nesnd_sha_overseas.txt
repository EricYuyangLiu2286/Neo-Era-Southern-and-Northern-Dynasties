        }

}

nesnd_sha_expand_transport = {
    name = nesnd_sha_expand_transport
    desc = nesnd_sha_expand_transport_desc
    visible = { has_journal_entry = nesnd_sha_settlement }
    possible = { NOT = { has_modifier = declared_bankruptcy }
        OR = {
            AND = { var:nesnd_sha_capacity = 5000 capital = { any_scope_building = { is_building_type = building_port level >= 2 } } }
            AND = { var:nesnd_sha_capacity = 10000 capital = { any_scope_building = { is_building_type = building_port level >= 3 } } }
        } }
    effect = {
        if = { limit = { var:nesnd_sha_capacity = 5000 } add_treasury = nesnd_sha_medium_cost hidden_effect = { set_variable = { name = nesnd_sha_capacity value = 10000 } } }
        else = { add_treasury = nesnd_sha_capital_cost hidden_effect = { set_variable = { name = nesnd_sha_capacity value = 20000 } } }
    }
    ai_chance = { value = 10 if = { limit = { in_default = yes } value = 0 } }
}
