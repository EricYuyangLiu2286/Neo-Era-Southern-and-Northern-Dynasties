    effect = {
        custom_tooltip = nesnd_egypt_choose_cost
        show_as_tooltip = { add_modifier = { name = nesnd_egypt_investment } add_modifier = { name = nesnd_egypt_fra_industry } }
        hidden_effect = { nesnd_egypt_select_investor = { TAG = FRA } }

        if = { limit = { NOT = { nesnd_egypt_investor_cooperating = yes } nesnd_egypt_independence_value > 35 } add = 15 }
    }
}
