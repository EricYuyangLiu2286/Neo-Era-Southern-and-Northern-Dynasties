nesnd_japan_send_terms = {
	name = nesnd_japan_send_terms
	desc = nesnd_japan_send_terms_desc
	visible = {
		NOT = {
			has_global_variable = nesnd_japan_offer_sent
		}
	}
	possible = {
		custom_tooltip = {
			text = nesnd_japan_send_terms_requirements
			nesnd_japan_can_send = yes
		}
	}
	ai_chance = {
		value = 5
		if = {
			limit = {
				c:JAP ?= {
					has_variable = usm_opposition_country
				}
			}
			add = 15
		}
	}
	effect = {
		if = {
			limit = {
				nesnd_japan_can_send = yes
			}
			set_global_variable = nesnd_japan_offer_sent
			if = {
				limit = {
					any_scope_character = {
						has_template = USM_lin_hanhan
						is_character_alive = yes
					}
				}
				random_scope_character = {
					limit = {
						has_template = USM_lin_hanhan
						is_character_alive = yes
					}
					set_variable = nesnd_japan_expedition_commander
				}
			}
			c:JAP = {
				trigger_event = {
					id = nesnd_japan.1
					popup = yes
				}
			}
		}
	}
}
