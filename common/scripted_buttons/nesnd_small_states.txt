nesnd_small_public = {
    name = nesnd_small_public
    desc = nesnd_small_public_desc
    visible = { nesnd_small_state_country = yes }
    possible = {
        nesnd_small_state_country = yes
        custom_tooltip = {
            text = nesnd_small_no_emergency_tt
            in_default = no
        }
        net_fixed_income >= nesnd_small_weekly_quote
        NOT = { has_modifier = nesnd_small_public_investment }
    }
    effect = {
        custom_tooltip = nesnd_small_public_tt
        show_as_tooltip = {
            add_modifier = {
                name = nesnd_small_public_investment
                months = 12
                multiplier = nesnd_small_weekly_quote
            }
            add_modifier = {
                name = nesnd_small_public_services
                months = 12
            }
        }
        hidden_effect = {
            if = {
                limit = {
                    nesnd_small_state_country = yes
                    in_default = no
                    net_fixed_income >= nesnd_small_weekly_quote
                    NOT = { has_modifier = nesnd_small_public_investment }
                }
                nesnd_small_start_public = yes
            }
        }
    }
    ai_chance = {
        value = {
            value = 0
            if = {
                limit = {
                    net_fixed_income >= {
                        value = nesnd_small_weekly_quote
                        multiply = 2
                    }
                    bureaucracy >= 0
                    nesnd_small_safe = yes
                }
                value = 8
            }
        }
    }
}
