# On the princes' journal (nesnd_india_princes).
nesnd_india_project_start = {
    name = nesnd_india_project_start
    desc = nesnd_india_project_start_desc
    visible = { nesnd_india_prince = yes NOT = { has_variable = nesnd_india_project_type } }
    possible = { nesnd_india_project_can_start = yes }
    ai_chance = { value = 0 }
    effect = {
        custom_tooltip = nesnd_india_project_start_tt
        hidden_effect = { if = { limit = { nesnd_india_project_can_start = yes } trigger_event = { id = nesnd_india_project.1 popup = yes } } }
    }
}

nesnd_india_matter_fund_button = {
    name = nesnd_india_matter_fund_button
    desc = nesnd_india_matter_fund_desc
    visible = { nesnd_india_matter_fund_role = yes }
    possible = {
        custom_tooltip = { text = nesnd_india_matter_unanswered_tt NOT = { has_variable = nesnd_india_matter_answered } }
        nesnd_india_matter_can_fund = yes
    }
    ai_chance = { value = 0 }
    effect = {
        if = { limit = { nesnd_india_matter_is_survey = yes }
            custom_tooltip = nesnd_india_matter_fund_survey_tt
            show_as_tooltip = { add_treasury = nesnd_india_matter_quote }
        }
        else = {
            if = { limit = { nesnd_india_matter_is_grant = yes } custom_tooltip = nesnd_india_matter_fund_grant_tt }
            else = { custom_tooltip = nesnd_india_matter_fund_stake_tt }
            custom_tooltip = nesnd_india_matter_fund_terms_tt
            show_as_tooltip = { add_treasury = { value = nesnd_india_matter_quote multiply = -1 } }
        }
        hidden_effect = { nesnd_india_matter_fund = yes }
    }
}
