nesnd_external_choose_wu = {
    name = nesnd_external_choose_wu
    desc = nesnd_external_choose_desc
    visible = { exists = scope:nesnd_label_usm 
        NOR = {
            c:MUG ?= this
            c:SHA ?= this
        }
    }
    possible = { nesnd_external_can_contact = { PARTNER = c:USM PARTNER_TT = nesnd_external_wu_available_tt } }
    ai_chance = {
        value = 1
        if = {
            limit = {
                c:FRA ?= this
            }
            add = 1
        }
        if = {
            limit = {
                OR = {
                    c:SHA ?= this
                    c:QE1 ?= this
                }
            }
            multiply = 0
        }
        if = {
            limit = {
                c:HUS ?= this
                c:USM ?= {
                    relations:root < relations_threshold:cordial
                }
            }
            multiply = 0
        }
    }
    effect = {
        custom_tooltip = nesnd_external_contact_offer_tt
        hidden_effect = {
            if = {
                limit = { nesnd_external_can_contact = { PARTNER = c:USM PARTNER_TT = nesnd_external_wu_available_tt } }
                set_variable = { name = nesnd_external_target value = c:USM }
            }
        }
    }
}

nesnd_external_choose_opposition = {

    desc = nesnd_external_choose_opposition_desc
    visible = { NOR = { c:MUG ?= this c:SHA ?= this } }
    possible = { nesnd_external_can_contact_opposition = yes }
    ai_chance = { value = 1 if = { limit = { c:HUS ?= this } add = 5 } if = { limit = { c:QE1 ?= this } add = 2 } }
    effect = {
        custom_tooltip = nesnd_external_contact_offer_tt
        hidden_effect = {
            if = {
                limit = { nesnd_external_can_contact_opposition = yes }
                random_country = {
                limit = {
                    is_country_alive = yes
                    has_variable = usm_opposition_country_leader_var
                    NOT = { this = root }
                    NOT = { has_war_with = root }
                }
                root = { set_variable = { name = nesnd_external_target value = prev } }
            }
            }
        }
    }
}

nesnd_external_dismiss = {
    name = nesnd_external_dismiss
    desc = nesnd_external_dismiss_desc
    visible = {
        always = yes
    }
    possible = {
        always = yes
    }
    ai_chance = {
        value = 0
        if = {
            limit = {
                is_ai = yes
                NOT = {
                    has_variable = nesnd_external_cooldown
                }
                has_variable = nesnd_external_idle_weeks
                nesnd_external_idle_weeks_value >= 26
                NOT = {
                    has_variable = nesnd_external_pending
                }
                NOT = {
                    has_variable = nesnd_external_aid_committed
                }
                NOT = {
                    nesnd_external_any_legal_aid = yes
                }
                OR = {
                    has_variable = nesnd_external_ai_reviewed
                    NOT = {
                        nesnd_external_any_partner_available = yes
                    }
                }
            }
            add = 5
        }
    }
    effect = {
        custom_tooltip = nesnd_external_dismiss_tt
        hidden_effect = {
            set_variable = nesnd_external_dismissed
            nesnd_external_clear_request = yes
        }
    }
}
nesnd_external_reconsider = {
    name = nesnd_external_reconsider
    desc = nesnd_external_reconsider_desc
    visible = {
        has_variable = nesnd_external_target
    }
    possible = {
        custom_tooltip = { text = nesnd_external_no_pending_tt NOT = { has_variable = nesnd_external_pending } }
        custom_tooltip = { text = nesnd_external_no_commitment_tt NOT = { has_variable = nesnd_external_aid_committed } }
    }
    ai_chance = {
        value = 0
        if = {
            limit = {
                is_ai = yes
                NOT = {
                    has_variable = nesnd_external_cooldown
                }
                has_variable = nesnd_external_idle_weeks
                nesnd_external_idle_weeks_value >= 13
                NOT = {
                    has_variable = nesnd_external_ai_reviewed
                }
                NOT = {
                    nesnd_external_any_legal_aid = yes
                }
            }
            add = 5
        }
    }
    effect = {
        custom_tooltip = nesnd_external_reconsider_tt
        hidden_effect = {
            remove_variable = nesnd_external_refused
            remove_variable = nesnd_external_target
            if = {
                limit = { is_ai = yes }
                set_variable = nesnd_external_ai_reviewed
                set_variable = { name = nesnd_external_idle_weeks value = 0 }
            }
        }
    }
}
