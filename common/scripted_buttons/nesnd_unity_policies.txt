nesnd_unity_material_subsidy = {
    name = nesnd_unity_material_subsidy
    desc = nesnd_unity_material_subsidy_desc
    visible = { nesnd_unity_policy_live = yes }
    selected = { nesnd_unity_material_active = yes }
    possible = {
        hidden_trigger = { nesnd_unity_policy_live = yes }
        custom_tooltip = { text = nesnd_unity_material_available_tt NOT = { nesnd_unity_material_active = yes } }
    }
    effect = {
        show_as_tooltip = {
            add_modifier = { name = nesnd_unity_material_expense months = 12 multiplier = nesnd_unity_material_weekly_cost }
            add_modifier = { name = nesnd_unity_material_assistance months = 12 }
        }
        if = { limit = { nesnd_qe1_unity_live = yes } custom_tooltip = nesnd_qe1_subsidy_material_tt }
        else_if = { limit = { nesnd_jp_cohesion_live = yes } custom_tooltip = nesnd_jp_subsidy_material_tt }
        else = { custom_tooltip = nesnd_unity_material_monthly_tt }
        custom_tooltip = nesnd_unity_sentiment_tt
        custom_tooltip = nesnd_unity_fixed_term_tt
        if = { limit = { OR = { nesnd_jp_cohesion_live = yes nesnd_qe1_unity_live = yes } } custom_tooltip = nesnd_unity_ends_with_journal_tt }
        hidden_effect = { nesnd_unity_start_material = yes }
    }
    ai_chance = {
        value = 0
        if = { limit = { nesnd_jp_country = yes } value = nesnd_jp_unity_material_ai_weight }
        else = {
            # An Indian government buys until the community reaches 60, the level at which it can take in a
            # prince (nesnd_india_union_host). Every other government stops at 50.
            if = { limit = { nesnd_india_empire = yes nesnd_unity_policy_current_value < 60 net_fixed_income > nesnd_unity_material_weekly_cost in_default = no NOT = { has_modifier = declared_bankruptcy } } add = 25 }
            else_if = { limit = { NOT = { nesnd_india_empire = yes } nesnd_unity_policy_current_value < 50 net_fixed_income > nesnd_unity_material_weekly_cost in_default = no NOT = { has_modifier = declared_bankruptcy } } add = 25 }
        }
    }
}
nesnd_unity_political_subsidy = {
    name = nesnd_unity_political_subsidy
    desc = nesnd_unity_political_subsidy_desc
    visible = { nesnd_unity_policy_live = yes }
    selected = { nesnd_unity_political_active = yes }
    possible = {
        hidden_trigger = { nesnd_unity_policy_live = yes }
        custom_tooltip = { text = nesnd_unity_political_available_tt NOT = { nesnd_unity_political_active = yes } }
    }
    effect = {
        show_as_tooltip = { add_modifier = { name = nesnd_unity_political_assistance months = 12 } }
        if = { limit = { nesnd_qe1_unity_live = yes } custom_tooltip = nesnd_qe1_subsidy_political_tt }
        else_if = { limit = { nesnd_jp_cohesion_live = yes } custom_tooltip = nesnd_jp_subsidy_political_tt }
        else = { custom_tooltip = nesnd_unity_political_monthly_tt }
        custom_tooltip = nesnd_unity_sentiment_tt
        custom_tooltip = nesnd_unity_fixed_term_tt
        if = { limit = { OR = { nesnd_jp_cohesion_live = yes nesnd_qe1_unity_live = yes } } custom_tooltip = nesnd_unity_ends_with_journal_tt }
        hidden_effect = { nesnd_unity_start_political = yes }
    }
    ai_chance = {
        value = 0
        if = { limit = { nesnd_jp_country = yes } value = nesnd_jp_unity_political_ai_weight }
        else = {
            # The same stopping levels as the material subsidy: 60 for an Indian government, 50 for any other.
            if = { limit = { nesnd_india_empire = yes nesnd_unity_policy_current_value < 60 government_legitimacy >= 40 } add = 25 }
            else_if = { limit = { NOT = { nesnd_india_empire = yes } nesnd_unity_policy_current_value < 50 government_legitimacy >= 40 } add = 25 }
        }
    }
}
