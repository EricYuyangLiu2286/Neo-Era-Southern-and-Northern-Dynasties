nesnd_unity_material_subsidy = {
    name = nesnd_unity_material_subsidy
    desc = nesnd_unity_material_subsidy_desc
    visible = { nesnd_unity_policy_live = yes }
    selected = { nesnd_unity_material_active = yes }
    possible = {
        hidden_trigger = { nesnd_unity_policy_live = yes }
        custom_tooltip = { text = nesnd_unity_material_available_tt NOT = { nesnd_unity_material_active = yes } }
    }
    effect = {
        show_as_tooltip = {
            add_modifier = { name = nesnd_unity_material_expense months = 12 multiplier = nesnd_unity_material_weekly_cost }
            add_modifier = { name = nesnd_unity_material_assistance months = 12 }
        }
        custom_tooltip = nesnd_unity_material_monthly_tt
        custom_tooltip = nesnd_unity_fixed_term_tt
        if = { limit = { OR = { nesnd_jp_cohesion_live = yes nesnd_qe1_unity_live = yes } } custom_tooltip = nesnd_unity_ends_with_journal_tt }
        hidden_effect = { nesnd_unity_start_material = yes }
    }
    ai_chance = {
        value = 0
        if = { limit = { nesnd_jp_country = yes } value = nesnd_jp_unity_material_ai_weight }
        else = {
            if = { limit = { nesnd_unity_policy_current_value < 50 net_fixed_income > nesnd_unity_material_weekly_cost } add = 25 }
        }
    }
}
