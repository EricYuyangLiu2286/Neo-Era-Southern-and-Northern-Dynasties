nesnd_per_seek_aid_button = {
    name = nesnd_per_seek_aid_button
    desc = nesnd_per_seek_aid_button_desc
    visible = { nesnd_per_future_live = yes nesnd_per_route_unset = yes }
    possible = {
        custom_tooltip = { text = nesnd_per_route_choice_available_tt NOT = { has_variable = nesnd_per_route_choice_pending } }
    }
    ai_chance = { value = 0 }
    effect = {
        custom_tooltip = nesnd_feedback_per_aid_tt
        hidden_effect = {
            if = {
                limit = { NOT = { has_variable = nesnd_per_route_choice_pending } }
                nesnd_per_choose_route = { ROUTE = aid }
            }
        }
    }
}
