nesnd_supply_clear_partner = {
    remove_variable = nesnd_supply_partner
    remove_variable = nesnd_supply_partner_sponsor
    remove_variable = nesnd_supply_pending
    remove_variable = nesnd_supply_pending_sponsor
    remove_variable = nesnd_supply_trade_pending
}
nesnd_supply_save_sponsor = {
    if = { limit = { is_subject = yes } set_variable = { name = nesnd_supply_sponsor value = overlord } }
    else = { set_variable = { name = nesnd_supply_sponsor value = this } }
}
nesnd_supply_initialize = {
    nesnd_supply_clear_legacy = yes
    nesnd_supply_save_sponsor = yes
    set_variable = { name = nesnd_supply_market value = capital.market }
    set_variable = { name = nesnd_supply_score value = 0 }
    set_variable = { name = nesnd_supply_weeks value = 0 }
    nesnd_supply_refresh = yes
}
nesnd_supply_refresh = {
    nesnd_supply_measure_workforce = yes
    if = {
        limit = {
            NOT = {
                has_variable = nesnd_supply_pending
            }
        }
        remove_variable = nesnd_supply_pending_sponsor
    }
    if = {
        limit = {
            trigger_if = {
                limit = {
                    exists = var:nesnd_supply_sponsor
                }
                NOT = {
                    OR = {
                        AND = {
                            is_subject = yes
                            overlord = var:nesnd_supply_sponsor
                        }
                        AND = {
                            is_subject = no
                            this = var:nesnd_supply_sponsor
                        }
                    }
                }
            }
            trigger_else = {
                always = yes
            }
        }
        nesnd_supply_clear_partner = yes
    }
    if = {
        limit = {
            has_variable = nesnd_supply_partner
            NOT = {
                nesnd_supply_partner_valid = yes
            }
        }
        nesnd_supply_clear_partner = yes
    }
    if = {
        limit = {
            trigger_if = {
                limit = {
                    exists = var:nesnd_supply_market
                }
                NOT = {
                    capital.market = var:nesnd_supply_market
                }
            }
            trigger_else = {
                always = yes
            }
        }
        set_variable = {
            name = nesnd_supply_weeks
            value = 0
        }
        set_variable = {
            name = nesnd_supply_market
            value = capital.market
        }
    }
    nesnd_supply_save_sponsor = yes
    set_variable = {
        name = nesnd_supply_score
        value = 0
    }
    if = {
        limit = {
            nesnd_supply_inputs_ready = yes
        }
        change_variable = {
            name = nesnd_supply_score
            add = 1
        }
    }
    if = {
        limit = {
            nesnd_supply_jobs_ready = yes
        }
        change_variable = {
            name = nesnd_supply_score
            add = 1
        }
    }
    if = {
        limit = {
            nesnd_supply_livelihood_ready = yes
        }
        change_variable = {
            name = nesnd_supply_score
            add = 1
        }
    }
    if = {
        limit = {
            nesnd_supply_diplomacy_ready = yes
        }
        change_variable = {
            name = nesnd_supply_score
            add = 1
        }
    }
    if = {
        limit = {
            NOT = {
                nesnd_supply_all_ready = yes
            }
        }
        set_variable = {
            name = nesnd_supply_weeks
            value = 0
        }
    }
    nesnd_supply_update_display = yes
}
nesnd_supply_week = {
    nesnd_supply_refresh = yes
    if = { limit = { nesnd_supply_all_ready = yes }
        change_variable = { name = nesnd_supply_weeks add = 1 }
        clamp_variable = { name = nesnd_supply_weeks min = 0 max = 12 }
    }
    nesnd_supply_update_display = yes
}
nesnd_supply_reward_jingnan_muster = {
    if = {
        limit = {
            OR = { c:HNA ?= this c:SIH ?= this c:GNG ?= this }
            has_variable = nesnd_supply_done
            nesnd_jingnan_may_receive_rebel_journal = yes
        }
        add_modifier = { name = nesnd_jingnan_muster months = 12 }
    }
}

nesnd_supply_cleanup = {
    nesnd_supply_clear_partner = yes
    remove_variable = nesnd_supply_market
    remove_variable = nesnd_supply_sponsor
    remove_variable = nesnd_supply_self_reliance

    remove_variable = nesnd_supply_workforce_metrics_v1
    remove_variable = nesnd_supply_workforce
    remove_variable = nesnd_supply_unemployed
    remove_variable = nesnd_supply_unemployment
    remove_variable = nesnd_supply_critical_states
}

nesnd_supply_save_pending_sponsor = {
    if = {
        limit = {
            exists = var:nesnd_supply_pending
        }
        if = {
            limit = {
                var:nesnd_supply_pending ?= {
                    is_subject = yes
                }
            }
            set_variable = {
                name = nesnd_supply_pending_sponsor
                value = var:nesnd_supply_pending.overlord
            }
        }
        else = {
            set_variable = {
                name = nesnd_supply_pending_sponsor
                value = var:nesnd_supply_pending
            }
        }
    }
}

# First refresh migrates old-save timing once. Live completion gates do not trust these display caches.
nesnd_supply_measure_workforce = {
    if = { limit = { NOT = { has_variable = nesnd_supply_workforce_metrics_v1 } }
        set_variable = nesnd_supply_workforce_metrics_v1
        set_variable = { name = nesnd_supply_weeks value = 0 }
    }
    set_variable = { name = nesnd_supply_workforce value = nesnd_supply_workforce_sum }
    set_variable = { name = nesnd_supply_unemployed value = nesnd_supply_unemployed_sum }
    set_variable = { name = nesnd_supply_unemployment value = nesnd_supply_workforce_unemployment }
    set_variable = { name = nesnd_supply_critical_states value = nesnd_supply_critical_employment_states }
}
nesnd_supply_update_display = {
    save_scope_as = nesnd_supply_display
    if = {
        limit = {
            exists = scope:nesnd_supply_display
        }
        je:nesnd_supply_preparations ?= {
            set_bar_progress = {
                name = nesnd_supply_readiness_bar
                value = scope:nesnd_supply_display.nesnd_supply_score_value
            }
            set_bar_progress = {
                name = nesnd_supply_continuity_bar
                value = scope:nesnd_supply_display.nesnd_supply_weeks_value
            }
        }
        je:nesnd_supply_hus_preparations ?= {
            set_bar_progress = {
                name = nesnd_supply_readiness_bar
                value = scope:nesnd_supply_display.nesnd_supply_score_value
            }
            set_bar_progress = {
                name = nesnd_supply_continuity_bar
                value = scope:nesnd_supply_display.nesnd_supply_weeks_value
            }
        }
    }
}
