# 教士秀全到达一州。作用域：目的州。开局、每月漫游、驱赶、州对象失效后的补位共用。
nesnd_xq_arrive = {
	global_var:nesnd_xq_state ?= {
		if = {
			limit = { has_modifier = nesnd_xq_ministry }
			remove_modifier = nesnd_xq_ministry
		}
	}
	set_global_variable = { name = nesnd_xq_state value = this }
	state_region = { save_scope_as = nesnd_xq_arrival_region }
	set_global_variable = { name = nesnd_xq_region value = scope:nesnd_xq_arrival_region }
	if = {
		limit = { NOT = { has_modifier = nesnd_xq_ministry } }
		add_modifier = { name = nesnd_xq_ministry }
	}
	owner ?= { nesnd_xq_enter_host = yes }
}

# 国家作用域。仅在实际抵达或所在地易主的回调中随行。
nesnd_xq_enter_host = {
	set_global_variable = { name = nesnd_xq_host_country value = this }
	global_var:nesnd_xq_char ?= {
		if = {
			limit = { NOT = { owner ?= prev } }
			transfer_character = prev
		}
	}
}

# 日志作用域。头像列表仅保存已创建的秀全；空列表不生成 GUI 人物控件。
nesnd_xq_bind_portrait = {
	if = {
		limit = {
			NOT = { has_variable_list = nesnd_xq_portrait_characters }
			exists = global_var:nesnd_xq_char
		}
		add_to_variable_list = { name = nesnd_xq_portrait_characters target = global_var:nesnd_xq_char }
	}
}

# 日志作用域。恢复同一所属国内的州记录，并刷新参与国；人物失联由 outcome 收尾。
nesnd_xq_sync = {
	nesnd_xq_bind_portrait = yes
	if = {
		limit = { NOT = { exists = global_var:nesnd_xq_state } }
		global_var:nesnd_xq_host_country = {
			random_scope_state = {
				limit = { state_region = global_var:nesnd_xq_region }
				nesnd_xq_arrive = yes
			}
		}
	}
	global_var:nesnd_xq_state ?= {
		if = {
			limit = { NOT = { has_modifier = nesnd_xq_ministry } }
			add_modifier = { name = nesnd_xq_ministry }
		}
	}
	nesnd_xq_sync_participants = yes
}

# 日志作用域。驱逐成功后立即刷新，周同步也处理归属和国界变化。
nesnd_xq_sync_participants = {
	every_scope_je_involved = {
		limit = { NOT = { nesnd_xq_should_be_involved = yes } }
		scope:journal_entry = { remove_involved_country = prev }
	}
	every_country = {
		limit = {
			nesnd_xq_should_be_involved = yes
			NOT = { is_involved_in_journal_entry = nesnd_xiuquan_journey }
		}
		scope:journal_entry = { add_involved_country = prev }
	}
}

# 死亡回调和日志失联结局共用。旧档若已发过追认圣徒，保留其已发送标记。
nesnd_xq_commemorate = {
	if = {
		limit = { NOT = { has_global_variable = nesnd_xq_dead } }
		set_global_variable = nesnd_xq_dead
		every_country = {
			limit = { is_player = yes }
			trigger_event = { id = nesnd_xiuquan.1 popup = yes }
		}
	}
}
