nesnd_qe1_check_civil_conflict = {
	if = { limit = { nesnd_qe1_civil_danger = yes any_scope_state = { is_capital = no state_population > 0 } }
		set_variable = nesnd_qe1_civil_raised
		clear_saved_scope = nesnd_qe1_rebel_seed
			ordered_scope_state = { limit = { state_population > 0  is_capital = no } order_by = state_population position = 0 check_range_bounds = no save_scope_as = nesnd_qe1_rebel_seed }
		if = { limit = { exists = scope:nesnd_qe1_rebel_seed }
			if = { limit = { exists = scope:nesnd_qe1_rebel_seed }
				create_country = { tag = QRC origin = root state = scope:nesnd_qe1_rebel_seed on_created = {
					set_variable = nesnd_qe1_civil_challenger set_variable = { name = nesnd_qe1_war_launch_window days = 90 }
					if = { limit = { trigger_if = { limit = { exists = c:QE1 } c:QE1 = { trigger_if = { limit = { has_variable = nesnd_qe1_balance } var:nesnd_qe1_balance <= -35 } trigger_else = { always = no } } } trigger_else = { always = no } }
						activate_law = law_type:law_monarchy activate_law = law_type:law_autocracy
						set_ruling_interest_groups = { ig_armed_forces ig_landowners }
						set_variable = nesnd_qe1_rebel_military
					} else = {
						activate_law = law_type:law_presidential_republic activate_law = law_type:law_wealth_voting
						set_ruling_interest_groups = { ig_intelligentsia ig_industrialists }
						set_variable = nesnd_qe1_rebel_economic
					}
					nesnd_qe1_setup_civil_challenger = yes
				} }
			}
			if = { limit = { exists = c:QRC }
				trigger_event = { id = nesnd_qe1_fracture.1 popup = yes }
				c:QRC = { trigger_event = { id = nesnd_qe1_fracture.90 days = 1 popup = no } }
			}
		}
	}
}
nesnd_qe1_setup_civil_challenger = {
	add_technology_researched = line_infantry
	add_technology_researched = artillery
	if = { limit = { has_modifier = nesnd_qe1_law_push } remove_modifier = nesnd_qe1_law_push } if = { limit = { has_modifier = nesnd_qe1_economy_push } remove_modifier = nesnd_qe1_economy_push } if = { limit = { has_modifier = nesnd_qe1_angovernment } remove_modifier = nesnd_qe1_angovernment } if = { limit = { has_modifier = nesnd_qe1_military_government } remove_modifier = nesnd_qe1_military_government } if = { limit = { has_modifier = nesnd_qe1_stalled } remove_modifier = nesnd_qe1_stalled } if = { limit = { has_modifier = nesnd_qe1_patronage_crown } remove_modifier = nesnd_qe1_patronage_crown } if = { limit = { has_modifier = nesnd_qe1_patronage_literati } remove_modifier = nesnd_qe1_patronage_literati } if = { limit = { has_modifier = nesnd_qe1_patronage_army } remove_modifier = nesnd_qe1_patronage_army } if = { limit = { has_modifier = nesnd_qe1_recovered_military } remove_modifier = nesnd_qe1_recovered_military } if = { limit = { has_modifier = nesnd_qe1_recovered_civil } remove_modifier = nesnd_qe1_recovered_civil }
	save_scope_as = nesnd_qe1_new_challenger
	create_military_formation = { name = nesnd_qe1_rebel_army type = army hq_region = scope:nesnd_qe1_new_challenger.capital.region
		combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = scope:nesnd_qe1_new_challenger.capital.state_region count = 16 }
		combat_unit = { type = unit_type:combat_unit_type_cannon_artillery state_region = scope:nesnd_qe1_new_challenger.capital.state_region count = 4 }
	}
	create_character = { first_name = nesnd_qe1_commander_first last_name = nesnd_qe1_commander_last is_general = yes interest_group = ig_armed_forces ideology = ideology_jingoist_leader }
}
nesnd_qe1_start_civil_war = {
	if = { limit = { c:QRC ?= this exists = c:QE1 has_variable = nesnd_qe1_civil_challenger has_variable = nesnd_qe1_war_launch_window NOT = { has_variable = nesnd_qe1_civil_war_started } }
		if = { limit = { is_subject = no is_diplomatic_play_committed_participant = no c:QE1 = { is_subject = no } NOT = { has_war_with = c:QE1 } }
			end_truce = c:QE1
			create_diplomatic_play = { type = dp_nesnd_qe1_civil target_country = c:QE1 war = yes annex_as_civil_war = no }
		}
		if = { limit = { OR = { has_war_with = c:QE1 any_diplomatic_play = { initiator = root target = c:QE1 is_diplomatic_play_type = dp_nesnd_qe1_civil } } } set_variable = nesnd_qe1_civil_war_started }
	}
}

					type = army

					combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = scope:nesnd_qe1_new_frontier.capital.state_region count = 3 }

					type = army

			}
			else_if = {

				create_military_formation = {

					type = army

nesnd_qe1_start_frontier_war = {
	if = {
		limit = { has_variable = nesnd_qe1_breakaway_attacker has_variable = nesnd_qe1_war_launch_window exists = c:QE1 NOT = { has_variable = nesnd_qe1_frontier_war_started } }
		nesnd_qe1_prepare_frontier_army = yes
		save_scope_as = nesnd_qe1_frontier_actor
		clear_saved_scope = nesnd_qe1_frontier_play
		random_diplomatic_play = {
			limit = { target = c:QE1 is_diplomatic_play_type = dp_nesnd_qe1_frontier initiator = { has_variable = nesnd_qe1_breakaway_attacker } }
			save_scope_as = nesnd_qe1_frontier_play
		}
		if = {
			limit = { exists = scope:nesnd_qe1_frontier_play }
			if = {
				limit = { is_subject = no NOT = { has_war_with = scope:nesnd_qe1_frontier_play.initiator } }
				scope:nesnd_qe1_frontier_play = {
					if = {
						limit = { is_war = yes exists = war }
						war = { if = { limit = { NOT = { is_war_participant = scope:nesnd_qe1_frontier_actor } } join_war = { target = scope:nesnd_qe1_frontier_actor side = scope:nesnd_qe1_frontier_play.initiator } } }
					}
					else_if = { limit = { is_war = no NOT = { nesnd_play_has_committed_initiator = { COUNTRY = scope:nesnd_qe1_frontier_actor } } } add_initiator_backers = { scope:nesnd_qe1_frontier_actor } }
					if = {
						limit = {
							nesnd_play_has_committed_initiator = { COUNTRY = scope:nesnd_qe1_frontier_actor }
							c:QE1 = { NOR = {
								play_participant_has_war_goal_of_type_against = { type = annex_country target = scope:nesnd_qe1_frontier_actor }
								war_participant_has_war_goal_of_type_against = { type = annex_country target = scope:nesnd_qe1_frontier_actor }
							} }
						}
						add_war_goal = { holder = c:QE1 target_country = scope:nesnd_qe1_frontier_actor type = annex_country }
					}
				}
			}
		}
		else = {
			if = {
				limit = { is_subject = no is_diplomatic_play_committed_participant = no c:QE1 = { is_subject = no } NOT = { has_war_with = c:QE1 } }
				end_truce = c:QE1
				create_diplomatic_play = {
					type = dp_nesnd_qe1_frontier target_country = c:QE1 war = yes annex_as_civil_war = no
					add_war_goal = { holder = c:QE1 target_country = scope:nesnd_qe1_frontier_actor type = annex_country }
				}
			}
		}
		if = {
			limit = { OR = {
				has_war_with = c:QE1
				any_diplomatic_play = { target = c:QE1 is_diplomatic_play_type = dp_nesnd_qe1_frontier nesnd_play_has_committed_initiator = { COUNTRY = scope:nesnd_qe1_frontier_actor } }
			} }
			set_variable = nesnd_qe1_frontier_war_started
		}
	}
}
