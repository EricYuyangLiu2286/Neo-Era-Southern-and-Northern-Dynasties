nesnd_refresh_land_display = {
        save_scope_as = nesnd_land_display_country
        if = { limit = { has_journal_entry = ce_cne_land_reform }
            je:ce_cne_land_reform = {
                set_bar_progress = { name = nesnd_land_completion_bar value = scope:nesnd_land_display_country.var:cne_landreform_completion_tracker }
                set_bar_progress = { name = nesnd_land_disorder_bar value = scope:nesnd_land_display_country.var:cne_landreform_failure_tracker }
            }
        }
    }
    nesnd_refresh_south_display = {
        nesnd_south_initialize = yes

        save_scope_as = nesnd_south_display_country
        if = { limit = { has_journal_entry = ce_usm_conflicts_within }
            
            je:ce_usm_conflicts_within = {
                set_bar_progress = { name = nesnd_south_pressure_bar value = scope:nesnd_south_display_country.nesnd_south_pressure_value }
                set_bar_progress = { name = nesnd_south_compact_bar value = scope:nesnd_south_display_country.nesnd_south_compact_value }
            }
        }
    }
