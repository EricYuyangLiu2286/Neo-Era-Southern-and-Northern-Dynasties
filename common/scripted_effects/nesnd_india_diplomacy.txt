nesnd_india_clear_request = {
    if = { limit = { custom_tooltip = { text = nesnd_ui_exists_eq_var_nesnd_india_request_partner_tt exists = var:nesnd_india_request_partner } }
        save_scope_as = india_request_cleanup_owner
        if = { limit = { exists = scope:india_request_cleanup_owner }
            var:nesnd_india_request_partner = {
                if = { limit = { custom_tooltip = { text = nesnd_ui_exists_eq_var_nesnd_india_incoming_union_sender_tt exists = var:nesnd_india_incoming_union_sender } }
                    if = { limit = { hidden_trigger = { var:nesnd_india_incoming_union_sender = scope:india_request_cleanup_owner } } remove_variable = nesnd_india_incoming_union_sender remove_variable = nesnd_india_incoming_union_clock }
                }
            }

        }
    }
    remove_variable = nesnd_india_request_clock remove_variable = nesnd_india_request_partner
    remove_variable = nesnd_india_request_kind remove_variable = nesnd_india_request_quote
}
nesnd_india_send_request = {
    if = { limit = { nesnd_india_mission_country = yes nesnd_india_channel_free = yes exists = scope:india_partner }
        if = { limit = { scope:india_partner = { is_country_alive = yes } NOT = { this = scope:india_partner } NOT = { has_war_with = scope:india_partner } }
            save_scope_as = india_sender
            save_scope_value_as = { name = india_sender_regime value = nesnd_india_regime }
            if = { limit = { is_subject = yes } overlord = { save_scope_as = india_sender_patron } } else = { save_scope_as = india_sender_patron }
            if = { limit = { exists = scope:india_sender exists = scope:india_partner exists = scope:india_sender_patron exists = scope:india_sender_regime }
                scope:india_partner = {
                    save_scope_value_as = { name = india_partner_regime value = nesnd_india_regime }
                    if = { limit = { is_subject = yes } overlord = { save_scope_as = india_partner_patron } } else = { save_scope_as = india_partner_patron }
                }
                set_variable = { name = nesnd_india_request_serial value = { value = nesnd_india_request_serial add = 1 } }
                save_scope_value_as = { name = india_serial value = nesnd_india_request_serial }
                if = { limit = { exists = scope:india_serial exists = scope:india_partner_patron exists = scope:india_partner_regime }
                    set_variable = { name = nesnd_india_request_kind value = $KIND$ }
                    set_variable = { name = nesnd_india_request_partner value = scope:india_partner }
                    set_variable = { name = nesnd_india_request_quote value = nesnd_india_grant }
                    set_variable = { name = nesnd_india_request_clock days = 45 }
                    set_variable = { name = nesnd_india_request_cooldown months = 6 }
                    if = { limit = { nesnd_india_request_kind = 6 } scope:india_partner = { set_variable = { name = nesnd_india_incoming_union_sender value = scope:india_sender } set_variable = { name = nesnd_india_incoming_union_clock days = 45 } } }
                    scope:india_partner = { trigger_event = { id = nesnd_india_diplomacy.1 days = 1 popup = yes } }


                }
            }
        }
    }
}
nesnd_india_refuse_request = {
    if = { limit = { nesnd_india_reply_valid = yes }
        scope:india_sender = {
            if = { limit = { OR = { nesnd_india_request_kind = 4 nesnd_india_request_kind = 7 } } add_liberty_desire = 10 }
            trigger_event = { id = nesnd_india_diplomacy.3 popup = yes }
            nesnd_india_clear_request = yes
        }
    }
}
nesnd_india_accept_request = {
    if = { limit = { nesnd_india_reply_can_accept = yes }
        if = { limit = { nesnd_india_incoming_kind <= 3 }
            add_treasury = { value = scope:india_sender.nesnd_india_request_quote multiply = -1 }
            scope:india_sender = {
                add_treasury = nesnd_india_request_quote
                add_modifier = { name = nesnd_india_french_advisers months = 48 }
                set_variable = { name = nesnd_india_aid_lock months = 48 }
                set_variable = { name = nesnd_india_adviser_partner value = scope:india_partner }
                if = { limit = { is_subject = yes } set_variable = { name = nesnd_india_adviser_patron value = overlord } } else = { set_variable = { name = nesnd_india_adviser_patron value = this } }
                set_owes_obligation_to = { country = scope:india_partner setting = yes }
                if = {
                    limit = { nesnd_india_request_kind = 1 }
                    add_modifier = { name = nesnd_india_french_teachers months = 48 }
                    scope:india_partner = {
                        if = { limit = { has_modifier = nesnd_india_teaching_mission } remove_modifier = nesnd_india_teaching_mission }
                        add_modifier = { name = nesnd_india_teaching_mission months = 48 }
                    }
                }
                else_if = {
                    limit = { nesnd_india_request_kind = 2 }
                    add_modifier = { name = nesnd_india_french_engineers months = 48 }
                    if = {
                        limit = { capital = { nesnd_india_workshop_site = yes } }
                        capital = { save_scope_as = india_workshop_state }
                    }
                    else = {
                        ordered_scope_state = {
                            limit = { nesnd_india_workshop_site = yes }
                            order_by = state_population
                            position = 0
                            check_range_bounds = no
                            save_scope_as = india_workshop_state
                        }
                    }
                    if = {
                        limit = { exists = scope:india_workshop_state scope:india_workshop_state = { nesnd_india_workshop_stake_met = no } }
                        scope:india_workshop_state = {
                            if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 5 } nesnd_india_grant_workshop = { LEVELS = 5 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 4 } nesnd_india_grant_workshop = { LEVELS = 4 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 3 } nesnd_india_grant_workshop = { LEVELS = 3 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 2 } nesnd_india_grant_workshop = { LEVELS = 2 } }
                            else = { nesnd_india_grant_workshop = { LEVELS = 1 } }
                        }
                    }
                }
                else = {
                    add_modifier = { name = nesnd_india_french_officers months = 48 }
                    create_treaty = {
                        first_country = scope:india_sender
                        second_country = scope:india_partner
                        is_draft = no
                        binding_period = { years = 5 }
                        articles_to_create = {
                            { article = military_access source_country = scope:india_sender target_country = scope:india_partner }
                        }
                    }
                }
            }
        }
        else_if = { limit = { nesnd_india_incoming_kind = 4 } scope:india_sender = { increase_autonomy = yes add_liberty_desire = -20 } }
        else_if = { limit = { nesnd_india_incoming_kind = 7 } scope:india_sender = { make_independent = yes create_bidirectional_truce = { country = scope:india_partner months = 60 } } }
        else_if = { limit = { nesnd_india_incoming_kind = 5 }
            create_treaty = { first_country = scope:india_sender second_country = scope:india_partner is_draft = no binding_period = { years = 3 } articles_to_create = { { article = defensive_pact } } }
        }
        if = { limit = { nesnd_india_incoming_kind = 6 }
            scope:india_sender = {
                add_treasury = { value = nesnd_india_request_quote multiply = -1 }
                add_modifier = { name = nesnd_india_princely_pensions months = 120 }
                nesnd_india_inherit_partner = yes
                nesnd_india_clear_request = yes
                annex = scope:india_partner
                nesnd_india_refresh = yes
            }
        }
        else = { scope:india_sender = { change_relations = { country = scope:india_partner value = 25 } trigger_event = { id = nesnd_india_diplomacy.2 popup = yes } nesnd_india_clear_request = yes } }
    }
}
nesnd_india_remove_advisers = {
    if = { limit = { has_modifier = nesnd_india_french_advisers } remove_modifier = nesnd_india_french_advisers } if = { limit = { has_modifier = nesnd_india_french_teachers } remove_modifier = nesnd_india_french_teachers } if = { limit = { has_modifier = nesnd_india_french_engineers } remove_modifier = nesnd_india_french_engineers } if = { limit = { has_modifier = nesnd_india_french_officers } remove_modifier = nesnd_india_french_officers }
    remove_variable = nesnd_india_adviser_partner remove_variable = nesnd_india_adviser_patron
}
nesnd_india_diplomacy_maintenance = {
    if = { limit = { NOT = { nesnd_india_mission_country = yes } } nesnd_india_clear_request = yes nesnd_india_remove_advisers = yes }

    if = { limit = { has_variable = nesnd_india_request_partner NOT = { has_variable = nesnd_india_request_clock } } nesnd_india_clear_request = yes }
    if = { limit = { has_variable = nesnd_india_incoming_union_sender NOT = { has_variable = nesnd_india_incoming_union_clock } } remove_variable = nesnd_india_incoming_union_sender }
    if = { limit = { has_variable = nesnd_india_adviser_partner }
        if = { limit = { exists = var:nesnd_india_adviser_partner exists = var:nesnd_india_adviser_patron }
            if = { limit = { OR = {
                        NOT = { has_modifier = nesnd_india_french_advisers }
                        has_war_with = var:nesnd_india_adviser_partner
                        AND = { trigger_if = { limit = { is_subject = yes } NOT = { overlord = var:nesnd_india_adviser_patron } } trigger_else = { always = no } }
                        AND = { is_subject = no NOT = { this = var:nesnd_india_adviser_patron } }
                } } nesnd_india_remove_advisers = yes }
        } else = { nesnd_india_remove_advisers = yes }
    }
}
nesnd_india_inherit_partner = {
    if = { limit = { scope:india_partner = { has_variable = nesnd_india_schools_done } } set_variable = nesnd_india_schools_done }
    if = { limit = { scope:india_partner = { has_variable = nesnd_india_industry_done } } set_variable = nesnd_india_industry_done }
    if = { limit = { scope:india_partner = { has_variable = nesnd_india_reform_done } } set_variable = nesnd_india_reform_done }
    if = { limit = { scope:india_partner = { has_variable = nesnd_india_caste_done } } set_variable = nesnd_india_caste_done }
    if = { limit = { scope:india_partner = { has_variable = nesnd_india_unified_done } } set_variable = nesnd_india_unified_done }
    scope:india_partner = { nesnd_india_clear_request = yes nesnd_india_remove_advisers = yes }
}

# Pick one eligible country per mission. Independent senders give every candidate weight 1.
nesnd_india_choose_ai_partner = {
    save_scope_as = india_mission_requester
    clear_saved_scope = india_ai_partner
    random_country = {
        limit = {
            is_at_war = no
            nesnd_india_mission_partner = { SENDER = scope:india_mission_requester }
        }
        weight = {
            base = 1
            modifier = { factor = 6 c:YN1 ?= this scope:india_mission_requester = { is_direct_subject_of = c:USM } }
            modifier = { factor = 3 c:JAP ?= this scope:india_mission_requester = { is_direct_subject_of = c:USM } }
        }
        save_scope_as = india_ai_partner
    }
    if = { limit = { exists = scope:india_ai_partner } nesnd_india_open_mission = { PARTNER = scope:india_ai_partner } }
}
