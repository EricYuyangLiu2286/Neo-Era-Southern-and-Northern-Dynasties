ce_calculate_french_empire_stability_from_characters_factors = {
	if = {
		limit = { has_variable = ce_french_empire_stability_from_characters }
		every_scope_character = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_bonapartist
					has_ideology = ideology:ideology_reformer
					has_ideology = ideology:ideology_royalist
				}
			}
			ROOT = {
				change_variable = { name = ce_french_empire_stability_from_characters add = prev.popularity }
			}
		}
		change_variable = { name = ce_french_empire_stability_from_characters divide = 100 }
	}
}

	if = {
		limit = { has_variable = ce_french_empire_industry_count }
		every_scope_building = {
			limit = {
				is_building_group = bg_manufacturing

	}
}

ce_calculate_french_empire_sol_factors = {
	if = {
		limit = { has_variable = ce_french_empire_sol_count }
		set_variable = { name = ce_french_number_of_incorperated_state value = 0 }
		every_scope_state = {
			limit = {
				is_incorporated = yes
			}
			ROOT = {
				set_variable = { name = ce_french_incorporated_state_sol value = 0 }
				change_variable = { name = ce_french_number_of_incorperated_state add = 1 }
				change_variable = { name = ce_french_incorporated_state_sol add = prev.average_sol }
				change_variable = { name = ce_french_incorporated_state_sol subtract = 9 } # below 9 count as negative
				change_variable = { name = ce_french_empire_sol_count add = var:ce_french_incorporated_state_sol } # below 9 count as negative
				remove_variable = ce_french_incorporated_state_sol # save memory
				# it is possible to also bring-in population as a multipler, but that would degrate performances further, so I decide not to
			}
		}
		change_variable = { name = ce_french_empire_sol_count divide = var:ce_french_number_of_incorperated_state }
		change_variable = { name = ce_french_empire_sol_count multiply = 5 } # if every incor. state has 20 avg. sol, then the value would be ((10-9)*N/N)*20 = 20
		remove_variable = ce_french_number_of_incorperated_state	
	}
}

ce_calculate_french_empire_movement_factors = {
	if = {
		limit = { has_variable = ce_french_empire_movement_count }
		
		if = {
			limit = {
				any_political_movement = {
					OR = {
						is_political_movement_type = movement_cultural_majority
						is_political_movement_type = movement_royalist_constitutional
					}
				}
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_cultural_majority
						is_political_movement_type = movement_royalist_constitutional
					}
				}
				ROOT = {
					set_variable  = { name = ce_french_empire_pro_movement_local_variable value = prev.political_movement_support }
					change_variable = { name = ce_french_empire_pro_movement_local_variable multiply = 10 }
					#
					change_variable = { name = ce_french_empire_movement_count add = var:ce_french_empire_pro_movement_local_variable }
					remove_variable = ce_french_empire_pro_movement_local_variable
				}
			}
		}
		if = {
			limit = {
				any_political_movement = {
					is_political_movement_type = movement_bonapartist
				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_bonapartist
				}
				ROOT = {
					set_variable  = { name = ce_french_empire_pro_movement_local_variable value = prev.political_movement_support }
					change_variable = { name = ce_french_empire_pro_movement_local_variable multiply = 20 }
					#
					change_variable = { name = ce_french_empire_movement_count add = var:ce_french_empire_pro_movement_local_variable }
					remove_variable = ce_french_empire_pro_movement_local_variable
				}
			}
		}
	}
}

ce_set_orleanist_ruler = {
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = house_orleans_leader
				}
			}
		}
		ce_create_orleanist_ig_leader = yes
	}

	if = {
		limit = {
			has_variable = bonapartist_restoration_var
		}
		remove_variable = bonapartist_restoration_var
	}

	if = {
		limit = {
			NOT = { has_variable = orleanist_restoration_var }
		}
		remove_variable = orleanist_restoration_var
	}

	if = {
		limit = {
			any_scope_character = {
				has_variable = house_orleans_leader
			}
		}
		random_scope_character = {
			limit = {
				has_variable = house_orleans_leader
			}
			cleanup_ruler_character_roles = yes
		
			set_character_as_ruler = yes
			add_character_role = politician
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				has_template = FRA_louis_philippe
			}
		}
		random_scope_character = {
			limit = {
				has_template = FRA_louis_philippe
			}
			cleanup_ruler_character_roles = yes
		
			set_character_as_ruler = yes
			add_character_role = politician
		}
	}
	else_if = {
		limit = {
			year <= 1850
		}
		create_character = {
			template = FRA_ferdinand_philippe
		}
		random_scope_character = {
			limit = {
				has_template = FRA_ferdinand_philippe
			}
			
			cleanup_ruler_character_roles = yes
			set_character_as_ruler = yes
			add_character_role = politician
		}
	}
	else_if = {
		limit = {
			year >= 1850
			year < 1894
		}
		create_character = {
			template = FRA_philippe_de_orleans_of_paris
			on_created = {#
				cleanup_ruler_character_roles = yes
				set_character_as_ruler = yes
				add_character_role = politician
			}
		}
		if = {
			limit = {
				year >= 1870
				year < 1894
			}
			create_character = {
				template = FRA_philippe_de_orleans_of_orleans
				on_created = {
					cleanup_ruler_character_roles = yes
					set_character_as_ruler = yes
					add_character_role = politician
				}
			}
		}
	}
	else_if = {
		limit = {
			year >= 1894
			year < 1926
		}
		create_character = {
			template = FRA_philippe_de_orleans_of_orleans
			on_created = {
				cleanup_ruler_character_roles = yes
				set_character_as_ruler = yes
				add_character_role = politician
			}
		}
		create_character = {
			template = FRA_jean_de_orleans
		}
	}
	else = {
		create_character = {
			last_name = d_Orleans
			female = no
			culture = primary_culture
			ig_leader = yes
			interest_group = ig:ig_petty_bourgeoisie
			ideology = ideology_orleanist
			age = 21
			on_created = {
				cleanup_ruler_character_roles = yes
				set_variable = {
					name = is_married
				}
				add_character_role = politician
				set_variable = house_orleans set_variable = { name = dynasty_var value = flag:house_orleans }
				set_character_as_ruler = yes
				save_scope_as = FRA_ce_orl_leader_scope
			}
		}
	}
	random_scope_character = {
		limit = {
			is_ruler_of_own_country = yes
		}
		save_scope_as = french_monarch_scope
	}
}
