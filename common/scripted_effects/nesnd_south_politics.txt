nesnd_south_measure_domestic = {
	hidden_effect = {
		save_scope_as = nesnd_south_measured_country
		set_variable = {
			name = nesnd_south_gov_total
			value = 0
		}
		set_variable = {
			name = nesnd_south_gov_good
			value = 0
		}
		every_interest_group = {
			limit = {
				is_in_government = yes
			}
			scope:nesnd_south_measured_country ?= {
				change_variable = {
					name = nesnd_south_gov_total
					add = prev.ig_clout
				}
			}
			if = {
				limit = {
					ig_approval >= 0
				}
				scope:nesnd_south_measured_country ?= {
					change_variable = {
						name = nesnd_south_gov_good
						add = prev.ig_clout
					}
				}
			}
		}
	}
}
