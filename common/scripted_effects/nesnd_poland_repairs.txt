nesnd_poland_failure_secessions = {
	if = { limit = { c:KRA ?= this }
		if = { limit = { OR = { owns_entire_state_region = STATE_KAUNAS owns_entire_state_region = STATE_BREST } }

			if = { limit = { owns_entire_state_region = STATE_KAUNAS }
				s:STATE_KAUNAS = { add_homeland = cu:lithuanian add_claim = root }
				if = { limit = { NOT = { exists = c:LIT } }
					create_country = { tag = LIT origin = root state = s:STATE_KAUNAS.region_state:KRA on_created = { set_capital = STATE_KAUNAS } }
				} else = { s:STATE_KAUNAS.region_state:KRA = { set_state_owner = c:LIT } }
				if = { limit = { exists = c:LIT }
					if = { limit = { NOT = { has_war_with = c:LIT } c:LIT = { is_subject = no } is_diplomatic_play_committed_participant = no }
						create_diplomatic_play = { target_country = c:LIT type = dp_annex_war war = yes add_war_goal = { holder = c:LIT type = make_protectorate target_country = root } }
					}
				}
			} else = { every_scope_state = { add_radicals_in_state = { value = 0.05 } } }
			if = { limit = { owns_entire_state_region = STATE_BREST }
				s:STATE_BREST = { add_homeland = cu:byelorussian }
				if = { limit = { NOT = { exists = c:BYE } } create_country = { tag = BYE origin = root state = s:STATE_BREST.region_state:KRA } }
				else = { s:STATE_BREST.region_state:KRA = { set_state_owner = c:BYE } }
			} else = { every_scope_state = { add_radicals_in_state = { value = 0.05 } } }

		}
		else = { every_scope_state = { add_radicals_in_state = { value = 0.10 } } }
	}
}
# Called from je_ce_the_polish_problem (immediate and monthly pulse); scope:journal_entry is that JE.
# Writes the inputs read by its bars and status: guarantees still binding Krakow, how far Paris
# will go (also set on the France bar; this effect is that bar's only writer) and the phase.
nesnd_poland_root_month = {
	if = { limit = { exists = scope:journal_entry }
		if = { limit = { exists = c:KRA }
			c:KRA = { set_global_variable = { name = nesnd_poland_guarantees value = nesnd_poland_guarantee_count } }
		}
		else = { set_global_variable = { name = nesnd_poland_guarantees value = 0 } }
		set_global_variable = { name = nesnd_poland_france_will value = nesnd_poland_france_will_value }
		scope:journal_entry = { set_global_variable = { name = nesnd_poland_last_consensus value = "scripted_bar_progress(nesnd_poland_consensus_bar)" } }
		scope:journal_entry = { set_bar_progress = { name = nesnd_poland_france_will_bar value = global_var:nesnd_poland_france_will } }
		if = { limit = { always = $ADVANCE_WINDOWS$ } nesnd_poland_windows_month = yes }
		if = { limit = { has_global_variable = nesnd_poland_window } set_global_variable = { name = nesnd_poland_phase value = 3 } }
		else_if = { limit = { scope:journal_entry = { "scripted_bar_progress(nesnd_poland_consensus_bar)" >= 60 } } set_global_variable = { name = nesnd_poland_phase value = 1 } }
		else_if = { limit = { scope:journal_entry = { "scripted_bar_progress(nesnd_poland_consensus_bar)" >= 30 } } set_global_variable = { name = nesnd_poland_phase value = 2 } }
		else = { set_global_variable = { name = nesnd_poland_phase value = 3 } }
	}
}

# Country scope, called weekly by a journal. Keeps exactly one of three tiered modifiers that
# matches the journal's bar: KEY_30 at 30+, KEY_50 at 50+, KEY_TOP at TOP+.
nesnd_poland_tier_modifiers = {
	if = { limit = { je:$JE$ ?= { scripted_bar_progress = { name = $BAR$ value >= $TOP$ } } }
		if = { limit = { NOT = { has_modifier = $KEY_TOP$ } } add_modifier = $KEY_TOP$ }
		if = { limit = { has_modifier = $KEY_50$ } remove_modifier = $KEY_50$ }
		if = { limit = { has_modifier = $KEY_30$ } remove_modifier = $KEY_30$ }
	}
	else_if = { limit = { je:$JE$ ?= { scripted_bar_progress = { name = $BAR$ value >= 50 } } }
		if = { limit = { NOT = { has_modifier = $KEY_50$ } } add_modifier = $KEY_50$ }
		if = { limit = { has_modifier = $KEY_TOP$ } remove_modifier = $KEY_TOP$ }
		if = { limit = { has_modifier = $KEY_30$ } remove_modifier = $KEY_30$ }
	}
	else_if = { limit = { je:$JE$ ?= { scripted_bar_progress = { name = $BAR$ value >= 30 } } }
		if = { limit = { NOT = { has_modifier = $KEY_30$ } } add_modifier = $KEY_30$ }
		if = { limit = { has_modifier = $KEY_TOP$ } remove_modifier = $KEY_TOP$ }
		if = { limit = { has_modifier = $KEY_50$ } remove_modifier = $KEY_50$ }
	}
	else = {
		if = { limit = { has_modifier = $KEY_30$ } remove_modifier = $KEY_30$ }
		if = { limit = { has_modifier = $KEY_50$ } remove_modifier = $KEY_50$ }
		if = { limit = { has_modifier = $KEY_TOP$ } remove_modifier = $KEY_TOP$ }
	}
}
