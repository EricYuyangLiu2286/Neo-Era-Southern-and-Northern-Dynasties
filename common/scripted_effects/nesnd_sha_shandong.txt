nesnd_sha_enter_shandong = {
    if = {
        limit = { nesnd_sha_shandong_open = yes }
        if = {
            limit = { var:nesnd_sha_route ?= flag:nesnd_sha_war NOT = { has_journal_entry = nesnd_sha_crossing } }
            add_journal_entry = { type = nesnd_sha_crossing }
        }
        else_if = {
            limit = { var:nesnd_sha_route ?= flag:nesnd_sha_peace NOT = { has_journal_entry = nesnd_sha_qilu } }
            add_journal_entry = { type = nesnd_sha_qilu }
        }
    }
}

}

        limit = { NOT = { has_claim = s:STATE_SHANDONG } }
        s:STATE_SHANDONG = { add_claim = c:SHA }
        set_variable = nesnd_sha_granted_claim
    }
    set_variable = { name = nesnd_sha_claim_opponent value = p:x9D1474.state.owner }
}
