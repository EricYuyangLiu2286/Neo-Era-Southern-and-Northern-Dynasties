nesnd_india_schools_tick = {
    if = { limit = { nesnd_india_schools_available = yes }
        if = { limit = { NOT = { has_variable = nesnd_india_schools_months } } set_variable = { name = nesnd_india_schools_months value = 0 } }
        if = { limit = { nesnd_india_schools_ready = yes } change_variable = { name = nesnd_india_schools_months add = 1 } }
        clamp_variable = { name = nesnd_india_schools_months min = 0 max = 12 }
        nesnd_india_sync_schools = yes
    }
}

nesnd_india_initialize = {
    if = { limit = { nesnd_india_empire = yes }
        set_variable = nesnd_india_imperial_line
        if = { limit = { NOT = { has_variable = nesnd_india_dynasty } } set_variable = { name = nesnd_india_dynasty value = 50 } }
        if = { limit = { NOT = { has_variable = nesnd_india_community } } set_variable = { name = nesnd_india_community value = 25 } }
    }
}

    if = { limit = { nesnd_india_usm_live = yes NOT = { has_journal_entry = nesnd_india_usm } } add_journal_entry = { type = nesnd_india_usm } }

    nesnd_india_diplomacy_maintenance = yes

    save_scope_as = india_display_country
