# ==== Identity, monthly refresh and the reform journals (formerly nesnd_india_core.txt) ====
# The months counted by the two reform tracks, the national community and the dynasty's support are
# kept on the progress bars of their journals (common/scripted_progress_bars/nesnd_india_core.txt) and
# read from them (common/script_values/nesnd_india_core.txt). Nothing here counts them a second time.
nesnd_india_initialize = {
    # The government that began as Mughal keeps this through any change of name (BHT, DRA); the
    # Mughal-only rewards of nesnd_india_schools and nesnd_india_industry read it.
    if = { limit = { c:MUG ?= this NOT = { has_variable = nesnd_india_mughal_court } } set_variable = nesnd_india_mughal_court }
    if = { limit = { nesnd_india_empire = yes } set_variable = nesnd_india_imperial_line }
}
# Country scope, from the buttons nesnd_india_dynastic_constitution and nesnd_india_republican_settlement.
# Census voting replaces a distribution of power law less progressive than it (progressiveness below 50
# in GAME common/laws/00_distribution_of_power.txt). A law at or above it is kept.
nesnd_india_settlement_power_law = {
    if = {
        limit = {
            OR = {
                has_law_or_variant = law_type:law_autocracy
                has_law = law_type:law_technocracy
                has_law_or_variant = law_type:law_oligarchy
                has_law = law_type:law_elder_council
                has_law = law_type:law_landed_voting
                has_law = law_type:law_wealth_voting
            }
        }
        activate_law = law_type:law_census_voting
    }
    else = { custom_tooltip = nesnd_india_keep_power_law_tt }
}
# Country scope, from nesnd_india_republican_settlement. A monarchy, chiefdom, theocracy or colonial
# administration becomes a presidential republic. Any other form of government is kept.
nesnd_india_settlement_republic = {
    if = {
        limit = {
            OR = {
                country_has_monarchy_law = yes
                has_law = law_type:law_chiefdom
                has_law = law_type:law_theocracy
                has_law = law_type:law_colonial_administration
            }
        }
        activate_law = law_type:law_presidential_republic
    }
    else = { custom_tooltip = nesnd_india_keep_governance_tt }
}
nesnd_india_refresh = {
    nesnd_india_initialize = yes
    if = { limit = { nesnd_india_reform_live = yes NOT = { has_journal_entry = nesnd_india_reform } } add_journal_entry = { type = nesnd_india_reform } }
    if = { limit = { nesnd_india_track_visible = { T = schools } NOT = { has_journal_entry = nesnd_india_schools } } if = { limit = { NOR = { has_journal_entry = nesnd_india_schools has_variable = nesnd_jeq_nesnd_india_schools } } set_variable = { name = nesnd_jeq_nesnd_india_schools days = 3 } add_journal_entry = { type = nesnd_india_schools } } }
    if = { limit = { nesnd_india_track_visible = { T = industry } NOT = { has_journal_entry = nesnd_india_industry } } if = { limit = { NOR = { has_journal_entry = nesnd_india_industry has_variable = nesnd_jeq_nesnd_india_industry } } set_variable = { name = nesnd_jeq_nesnd_india_industry days = 3 } add_journal_entry = { type = nesnd_india_industry } } }
    if = { limit = { nesnd_india_society_open = yes NOT = { has_journal_entry = nesnd_india_society } } add_journal_entry = { type = nesnd_india_society } }
    if = { limit = { nesnd_india_caste_live = yes NOT = { has_journal_entry = nesnd_india_caste } } add_journal_entry = { type = nesnd_india_caste } }
    if = { limit = { nesnd_india_prince = yes NOT = { has_journal_entry = nesnd_india_princes } } add_journal_entry = { type = nesnd_india_princes } }
    if = { limit = { nesnd_india_usm_live = yes NOT = { has_journal_entry = nesnd_india_usm } } add_journal_entry = { type = nesnd_india_usm } }
    if = { limit = { nesnd_india_empire = yes OR = { c:BHT ?= this c:DRA ?= this } NOT = { has_variable = nesnd_india_unified_done } NOT = { has_journal_entry = nesnd_india_unified } } add_journal_entry = { type = nesnd_india_unified } }
    if = { limit = { nesnd_tianzhu_governance_live = yes NOT = { has_journal_entry = nesnd_tianzhu_governance } } add_journal_entry = { type = nesnd_tianzhu_governance } }
    if = { limit = { nesnd_yuanshan_commission_entry = yes NOT = { has_journal_entry = nesnd_yuanshan_commission } } add_journal_entry = { type = nesnd_yuanshan_commission } }
    if = { limit = { nesnd_yuanshan_governance_live = yes NOT = { has_journal_entry = nesnd_yuanshan_governance } } add_journal_entry = { type = nesnd_yuanshan_governance } }
    nesnd_india_diplomacy_maintenance = yes
    nesnd_india_matter_maintenance = yes
    nesnd_india_project_maintenance = yes
    nesnd_india_union_pick_host = yes
    nesnd_india_order_upkeep = yes
    nesnd_mug_upkeep = yes
    nesnd_india_overseas_upkeep = yes
    if = { limit = { nesnd_india_mission_country = yes NOT = { has_journal_entry = nesnd_india_foreign } } add_journal_entry = { type = nesnd_india_foreign } }
    if = { limit = { nesnd_india_uprising_available = yes NOT = { has_journal_entry = nesnd_india_uprising } } add_journal_entry = { type = nesnd_india_uprising } }
    if = { limit = { nesnd_india_dynasty_open = yes NOT = { has_journal_entry = nesnd_india_dynasty_journal } } add_journal_entry = { type = nesnd_india_dynasty_journal } }
    nesnd_india_unified_watch = yes
}

# Country scope, monthly. After the new order, the country becomes a subject or holds no state of the
# region of its capital at the time (nesnd_india_unified_seat): the event nesnd_india_core.6 takes back
# the claims of the new order (the list nesnd_india_unified_claims). Once only.
nesnd_india_unified_watch = {
    if = {
        limit = {
            has_variable = nesnd_india_unified_done
            NOT = { has_variable = nesnd_india_unified_fell }
            OR = {
                is_subject = yes
                AND = { exists = var:nesnd_india_unified_seat NOT = { any_scope_state = { state_region = root.var:nesnd_india_unified_seat } } }
            }
        }
        set_variable = nesnd_india_unified_fell
        trigger_event = { id = nesnd_india_core.6 popup = yes }
    }
}

# Country scope. Shows VALUE (a script value read on this country) on the progress bar BAR of the journal JE.
nesnd_india_sync_bar = {
    save_scope_as = india_display_country
    if = { limit = { exists = scope:india_display_country has_journal_entry = $JE$ }
        je:$JE$ = { set_bar_progress = { name = $BAR$ value = scope:india_display_country.$VALUE$ } }
    }
}

nesnd_india_indep_admit = {

    }
    overlord = {

# The button and the monthly AI share the same price and eligibility. The event spends only on selection.
nesnd_india_indep_choose_movement = {
    save_scope_as = indep_actor
    save_scope_value_as = { name = indep_track value = $INDEX$ }
    overlord = {
        save_scope_as = indep_lord
        clear_saved_scope = indep_choice_1
        ordered_political_movement = {
            order_by = political_movement_$SUPPORT$_support
            position = 0
            check_range_bounds = no
            save_scope_as = indep_choice_1
        }
        clear_saved_scope = indep_choice_2
        ordered_political_movement = {
            order_by = political_movement_$SUPPORT$_support
            position = 1
            check_range_bounds = no
            save_scope_as = indep_choice_2
        }
        clear_saved_scope = indep_choice_3
        ordered_political_movement = {
            order_by = political_movement_$SUPPORT$_support
            position = 2
            check_range_bounds = no
            save_scope_as = indep_choice_3
        }
    }
    if = { limit = { exists = scope:indep_choice_1 }
        if = { limit = { is_ai = yes }
            scope:indep_choice_1 = { save_scope_as = indep_movement }
            nesnd_india_indep_fund = { TRACK = $TRACK$ }
        }
        else = { trigger_event = { id = nesnd_india_indep.1 popup = yes } }
    }
}
nesnd_india_indep_fund = {
    if = {
        limit = {
            nesnd_india_indep_can_fund = { TRACK = $TRACK$ }
            overlord = scope:indep_lord
            scope:indep_movement.owner ?= scope:indep_lord
        }
        save_scope_value_as = { name = indep_cost value = { value = nesnd_india_indep_price multiply = -1 } }
        je:nesnd_india_indep = { add_progress = { name = nesnd_india_indep_$TRACK$_bar value = scope:indep_cost } }
        # A receipt is unique across donors and survives a change of overlord.
        if = { limit = { NOT = { has_global_variable = nesnd_india_indep_serial } } set_global_variable = { name = nesnd_india_indep_serial value = 0 } }
        change_global_variable = { name = nesnd_india_indep_serial add = 1 }
        save_scope_value_as = { name = indep_receipt value = global_var:nesnd_india_indep_serial }
        scope:indep_movement = {
            if = { limit = { has_modifier = nesnd_india_indep_funded } remove_modifier = nesnd_india_indep_funded }
            add_modifier = { name = nesnd_india_indep_funded months = 24 }
            set_variable = { name = nesnd_india_indep_last_funding value = scope:indep_receipt months = 24 }
        }
        save_scope_value_as = { name = indep_kind value = 1 }
        nesnd_india_indep_discovery = yes
    }
}


nesnd_india_indep_store = {
    save_scope_as = indep_actor
    overlord = { save_scope_as = indep_lord }
    clear_saved_scope = indep_movement
    save_scope_value_as = { name = indep_kind value = 2 }
    je:nesnd_india_indep = { add_progress = { name = nesnd_india_indep_military_bar value = -50 } }
    set_variable = { name = nesnd_india_indep_stock value = { value = nesnd_india_indep_stores add = 1 max = 3 } }
    nesnd_india_indep_discovery = yes
}
nesnd_india_indep_discovery = {
    scope:indep_lord = {
        random = {
            chance = nesnd_india_indep_exposure
            nesnd_india_indep_file_case = yes
            trigger_event = { id = nesnd_india_indep.2 popup = yes }
        }
    }
}
nesnd_india_indep_investigate = {
    change_relations = { country = scope:indep_actor value = -30 }
    if = { limit = { scope:indep_kind = 2 }
        scope:indep_actor = {
            if = { limit = { nesnd_india_indep_stores > 1 } change_variable = { name = nesnd_india_indep_stock subtract = 1 } }
            else = { remove_variable = nesnd_india_indep_stock }
        }
    }
    else = {
        scope:indep_movement ?= {
            if = { limit = { var:nesnd_india_indep_last_funding ?= scope:indep_receipt }
                if = { limit = { has_modifier = nesnd_india_indep_funded } remove_modifier = nesnd_india_indep_funded }
                remove_variable = nesnd_india_indep_last_funding
            }
        }
    }
}
nesnd_india_indep_muster = {
    if = { limit = { nesnd_india_indep_stores > 0 exists = capital }
        if = { limit = { nesnd_india_indep_stores = 1 } nesnd_india_indep_raise_army = { COUNT = 2 } }
        else_if = { limit = { nesnd_india_indep_stores = 2 } nesnd_india_indep_raise_army = { COUNT = 4 } }
        else = { nesnd_india_indep_raise_army = { COUNT = 6 } }
        remove_variable = nesnd_india_indep_stock
    }
}
nesnd_india_indep_subject_ai = {
    if = { limit = { liberty_desire >= 25 nesnd_india_indep_peaceful_contact = yes }
        if = { limit = { nesnd_india_indep_can_back_anti = yes
                NOT = { any_political_lobby = { is_political_lobby_type = lobby_anti_overlord target = root.overlord appeasement >= 4 } }
            } nesnd_india_indep_back_anti = yes }
        else_if = { limit = { nesnd_india_indep_can_invest_domestic = yes economic_dependence = { target = overlord value > 1 } }
            nesnd_india_indep_invest_domestic = yes
        }
        else_if = { limit = { nesnd_india_indep_can_fund = { TRACK = economy } }
            nesnd_india_indep_choose_movement = { TRACK = economy SUPPORT = wealth INDEX = 1 }
        }
        else_if = { limit = { nesnd_india_indep_can_fund = { TRACK = military } }
            nesnd_india_indep_choose_movement = { TRACK = military SUPPORT = military INDEX = 2 }
        }
        else_if = { limit = { nesnd_india_indep_can_store = yes } nesnd_india_indep_store = yes }
    }
}

nesnd_india_indep_approach = {
    if = { limit = { exists = scope:indep_target
            nesnd_india_indep_policy_payer_ready = yes
            scope:indep_target = { nesnd_india_indep_policy_target = yes }
        }
        if = { limit = { scope:indep_policy = 1 }
            add_treasury = -20000
            set_variable = { name = nesnd_india_indep_conciliate_wait months = 12 }
            scope:indep_target = {
                nesnd_india_indep_back_pro = yes
                trigger_event = { id = nesnd_india_indep.6 popup = yes }
            }
        }
        else_if = { limit = { scope:indep_policy = 2 }
            set_variable = { name = nesnd_india_indep_coerce_wait months = 12 }
            change_relations = { country = scope:indep_target value = -30 }
            scope:indep_target = {
                remove_variable = nesnd_india_indep_stock
                je:nesnd_india_indep = {
                    add_progress = { name = nesnd_india_indep_economy_bar value = -50 }
                    add_progress = { name = nesnd_india_indep_military_bar value = -50 }
                }
                trigger_event = { id = nesnd_india_indep.7 popup = yes }
            }
        }
        else_if = { limit = { scope:indep_policy = 4 }
            create_diplomatic_pact = { country = scope:indep_target type = da_support_regime }
            if = { limit = { has_diplomatic_pact = { who = scope:indep_target type = da_support_regime is_initiator = yes } }
                scope:indep_target = { trigger_event = { id = nesnd_india_indep.10 popup = yes } }
            }
        }
        nesnd_india_indep_rank_subjects = yes
    }
}
nesnd_india_indep_lord_ai = {
    if = { limit = { has_variable = nesnd_india_indep_recent_case gold_reserves >= 10000 nesnd_india_indep_policy_ready = { POLICY = watch } } nesnd_india_indep_watch = yes }
    if = { limit = { nesnd_india_indep_funded_count >= 2 gold_reserves >= 10000 nesnd_india_indep_policy_ready = { POLICY = purge } } nesnd_india_indep_purge = yes }
    if = { limit = { scope:indep_subject_1 ?= { liberty_desire >= 75 } }
        scope:indep_subject_1 = { save_scope_as = indep_target }
        clear_saved_scope = indep_policy
        # Military superiority fixes the policy: wait out coercion's cooldown without switching tracks.
        if = { limit = { army_size >= scope:indep_target.nesnd_india_indep_triple_army }
            if = { limit = { nesnd_india_indep_policy_payer_ready_2 = yes
                    scope:indep_target = { nesnd_india_indep_policy_target_2 = { LORD = scope:indep_lord } }
                } save_scope_value_as = { name = indep_policy value = 2 } }
        }
        else_if = { limit = { scope:indep_target = {
                has_variable = nesnd_india_indep_pro_backer
                var:nesnd_india_indep_pro_backer ?= scope:indep_lord
            } }
            if = { limit = { nesnd_india_indep_policy_payer_ready_4 = yes
                    scope:indep_target = { nesnd_india_indep_policy_target_4 = { LORD = scope:indep_lord } }
                } save_scope_value_as = { name = indep_policy value = 4 } }
        }
        else_if = { limit = { nesnd_india_indep_policy_payer_ready_1 = yes
                scope:indep_target = { nesnd_india_indep_policy_target_1 = { LORD = scope:indep_lord } }
            } save_scope_value_as = { name = indep_policy value = 1 } }
        if = { limit = { exists = scope:indep_policy } nesnd_india_indep_approach = yes }
    }
}

# New lobbies start with the native on_created appeasement. Existing lobbies gain four points.
nesnd_india_indep_pick_lobby_member = {
    clear_saved_scope = indep_lobby_member
    ordered_interest_group = {
        limit = { nesnd_india_indep_founding_lobby_member = yes }
        order_by = ig_clout position = 0 check_range_bounds = no save_scope_as = indep_lobby_member
    }
}
nesnd_india_indep_back_anti = {
    if = { limit = { nesnd_india_indep_can_back_anti = yes }
        save_scope_as = indep_actor
        overlord = { save_scope_as = indep_lord }
        je:nesnd_india_indep = { add_progress = { name = nesnd_india_indep_economy_bar value = -50 } }
        if = { limit = { nesnd_india_indep_has_anti_lobby = yes }
            clear_saved_scope = indep_lobby_member
            ordered_interest_group = {
                limit = { nesnd_india_indep_free_lobby_member = yes }
                order_by = ig_clout position = 0 check_range_bounds = no save_scope_as = indep_lobby_member
            }
            random_political_lobby = {
                limit = { is_political_lobby_type = lobby_anti_overlord target = scope:indep_lord }
                change_appeasement = { amount = 4 factor = appeasement_lobby_pleased }
                if = { limit = { exists = scope:indep_lobby_member } add_lobby_member = scope:indep_lobby_member }
            }
        }
        else = {
            nesnd_india_indep_pick_lobby_member = yes
            create_political_lobby = { type = lobby_anti_overlord target = scope:indep_lord add_interest_group = scope:indep_lobby_member lobby_formation_reason = diplomacy }
        }
    }
}
nesnd_india_indep_back_pro = {
    if = { limit = { nesnd_india_indep_has_pro_lobby = yes }
        random_political_lobby = {
            limit = { is_political_lobby_type = lobby_pro_overlord target = scope:indep_lord }
            change_appeasement = { amount = 4 factor = appeasement_lobby_pleased }
        }
    }
    else = {
        nesnd_india_indep_pick_lobby_member = yes
        create_political_lobby = { type = lobby_pro_overlord target = scope:indep_lord add_interest_group = scope:indep_lobby_member lobby_formation_reason = diplomacy }
    }
    # Only a completed action counts, and it belongs to the overlord who performed it.
    # A pre-existing lobby alone does not let a new overlord skip this action.
    if = { limit = { nesnd_india_indep_has_pro_lobby = yes }
        set_variable = { name = nesnd_india_indep_pro_backer value = scope:indep_lord }
    }
}
