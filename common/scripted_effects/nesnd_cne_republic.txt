# Choose the district and seed once, before publishing the warning.
nesnd_rep_choose_district = {
    random_list = {
        1 = {
            trigger = { nesnd_rep_district_available = { REGION_TEST = nesnd_cne_republic_west_region } }
            nesnd_rep_reserve_district = { REGION_TEST = nesnd_cne_republic_west_region }
        }
        1 = {
            trigger = { nesnd_rep_district_available = { REGION_TEST = nesnd_cne_republic_northeast_region } nesnd_rep_northeast_unclaimed = yes }
            nesnd_rep_reserve_district = { REGION_TEST = nesnd_cne_republic_northeast_region }
        }
        1 = {
            trigger = { nesnd_rep_district_available = { REGION_TEST = nesnd_cne_republic_north_region } }
            nesnd_rep_reserve_district = { REGION_TEST = nesnd_cne_republic_north_region }
        }
    }
}

nesnd_rep_reserve_district = {
    every_scope_state = {
        limit = { $REGION_TEST$ = yes }
        owner = { add_to_variable_list = { name = nesnd_rep_states target = prev } }
    }
    random_scope_state = {
        limit = { nesnd_rep_reserved_state = yes }
        owner = { set_variable = { name = nesnd_rep_seed value = prev } }
    }
    set_variable = { name = nesnd_rep_count value = nesnd_rep_state_count }
    add_journal_entry = { type = nesnd_rep_warning }
    trigger_event = { id = nesnd_rising.1 popup = yes }
}

nesnd_rep_clear = {
    clear_variable_list = nesnd_rep_states
    remove_variable = nesnd_rep_count
    remove_variable = nesnd_rep_seed
    remove_variable = nesnd_rep_interrupted
}

nesnd_rep_stand_down = {
    if = { limit = { nesnd_rep_confidence > 500 nesnd_rep_confidence <= 550 } set_variable = nesnd_rep_wait_for_drop }
    set_variable = { name = nesnd_rep_cooldown months = 3 }
    nesnd_rep_clear = yes
}

nesnd_cne_spawn_republicans = {
    if = {
        limit = { nesnd_rep_warning_valid = yes nesnd_rep_confidence <= 500 }
        create_country = {
            tag = NRP
            origin = c:CNE
            state = var:nesnd_rep_seed
            on_created = { set_variable = nesnd_cne_republican_challenger }
        }
        if = {
            limit = { exists = c:NRP }
            every_scope_state = { limit = { nesnd_rep_reserved_state = yes } set_state_owner = c:NRP }
            if = { limit = { c:NRP = { has_variable = nesnd_cne_republican_challenger } }
                set_variable = nesnd_cne_republican_rose
                # Removed regional programmes must stop charging CNE immediately after the split.
                nesnd_cne_programs_maintenance = yes
                c:NRP = { nesnd_cne_republic_setup = yes }
                nesnd_cne_start_republic_war = yes
                trigger_event = { id = nesnd_cne_confidence.1 popup = yes }
            }
        }
    }
}

nesnd_cne_republic_setup = {
    nesnd_rising_clear_inherited = yes
    # Local-only inherited cleanup, never global CNE cleanup from the new country.
    nesnd_land_cleanup = yes
    activate_law = law_type:law_presidential_republic
    activate_law = law_type:law_census_voting
    activate_law = law_type:law_slavery_banned
    set_ruling_interest_groups = { ig_intelligentsia ig_armed_forces ig_rural_folk }
    set_variable = nesnd_cne_republican_challenger
    set_variable = { name = nesnd_cne_republic_launch_grace days = 7 }
    save_scope_as = nesnd_cne_new_republic
    create_military_formation = {
        name = nesnd_cne_republic_army
        type = army
        hq_region = scope:nesnd_cne_new_republic.capital.region
        combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = scope:nesnd_cne_new_republic.capital.state_region count = 12 }
        combat_unit = { type = unit_type:combat_unit_type_cannon_artillery state_region = scope:nesnd_cne_new_republic.capital.state_region count = 3 }
    }
    create_character = { first_name = nesnd_republic_commander_first last_name = nesnd_republic_commander_last is_general = yes interest_group = ig_armed_forces ideology = ideology_republican_leader }
    if = { limit = { NOT = { has_journal_entry = nesnd_cne_republic_journal } } add_journal_entry = { type = nesnd_cne_republic_journal } }
    trigger_event = { id = nesnd_cne_confidence.2 popup = yes }
}
nesnd_cne_start_republic_war = {
    if = {
        limit = {
            c:CNE ?= this
            nesnd_cne_republic_target_available = yes
            NOT = { nesnd_cne_republic_has_play = yes }
        }
        end_truce = c:NRP
        create_diplomatic_play = {
            name = nesnd_cne_republic_war
            type = dp_nesnd_cne_republic
            target_country = c:NRP
            war = yes
            annex_as_civil_war = no
            # Called from NRP scope too, so name both tags instead of using root.
            add_war_goal = { holder = c:CNE target_country = c:NRP type = wg_nesnd_cne_suppress_uprising primary_demand = yes }
            add_war_goal = { holder = c:NRP target_country = c:CNE type = wg_nesnd_cne_republican_humiliation primary_demand = yes }
        }
        if = {
            limit = { nesnd_cne_republic_has_play = yes }
            if = { limit = { has_journal_entry = nesnd_yan_warning } set_variable = nesnd_yan_interrupted }
            c:NRP = { set_variable = nesnd_cne_republic_war_created }
        }
    }
    # Also observes an already-existing direct war, without recreating it.
    if = {
        limit = { exists = c:NRP }
        c:NRP = { nesnd_cne_republic_note_actual_war = yes }
    }
}
nesnd_cne_republic_watch = {
    if = {
        limit = { c:NRP ?= this has_variable = nesnd_cne_republican_challenger }
        nesnd_cne_republic_note_actual_war = yes
        if = {
            limit = {
                exists = c:CNE
                NOT = { has_variable = nesnd_cne_republic_war_seen }
                NOT = { nesnd_cne_republic_has_play = yes }
                NOT = { nesnd_cne_republic_at_war = yes }
            }
            c:CNE = { nesnd_cne_start_republic_war = yes }
        }
    }
}
nesnd_cne_republic_victory = {
    if = { limit = { c:NRP ?= this has_variable = nesnd_cne_republican_challenger }
        if = { limit = { exists = c:CNE }
            # End all imperial reform effects before changing identities.
            c:CNE = { nesnd_land_cleanup = yes }
            annex_with_incorporation = c:CNE
        }
        if = { limit = { NOT = { exists = c:CNE } }
            change_tag = CNE
            activate_law = law_type:law_presidential_republic
            set_variable = nesnd_cne_republic_restored
            nesnd_land_cleanup = yes
            remove_variable = nesnd_cne_republican_challenger
            nesnd_cne_republic_southern_claims = yes
            trigger_event = { id = nesnd_cne_confidence.3 popup = yes }
        }
    }
}
nesnd_cne_republic_defeat = {
    if = {
        limit = { exists = c:CNE exists = c:NRP }
        if = {
            limit = {
                c:NRP = this
                has_variable = nesnd_cne_republican_challenger
                has_variable = nesnd_cne_republic_war_seen
                NOT = { nesnd_cne_republic_at_war = yes }
                NOT = { nesnd_cne_republic_has_play = yes }
                NOT = { has_variable = nesnd_cne_republic_launch_grace }
                NOT = { has_variable = nesnd_cne_republic_humbled_empire }
            }
            c:CNE = {
                set_variable = nesnd_cne_republic_settled
                annex_with_incorporation = prev
                nesnd_cne_republicans_defeated = yes
            }
        }
    }
}

nesnd_cne_republic_note_actual_war = {
    if = {
        limit = { exists = c:CNE exists = c:NRP }
        if = {
            limit = {
                c:NRP = this
                has_variable = nesnd_cne_republican_challenger
                has_war_with = c:CNE
            }
            set_variable = nesnd_cne_republic_war_seen
            c:CNE = { set_variable = nesnd_cne_republic_war_confirmed }
        }
    }
}

# Country creation can copy variables. Never clear SHA or CNE from a spawned country.
nesnd_rising_clear_inherited = {
    if = { limit = { has_variable_list = nesnd_rep_states } clear_variable_list = nesnd_rep_states }
    if = { limit = { has_variable_list = nesnd_yan_states } clear_variable_list = nesnd_yan_states }
    remove_variable = nesnd_rep_seed
    remove_variable = nesnd_rep_count
    remove_variable = nesnd_rep_interrupted
    remove_variable = nesnd_yan_count
    remove_variable = nesnd_yan_interrupted
}
