# Effects for Ottoman B9 "Mesopotamia" journal (nesnd_ottoman_meso).
# Country scope (TUR) unless noted.

# Selects the highest-scoring Mesopotamia foe candidate and stores it in global_var:nesnd_meso_foe (Rule 4).
nesnd_ottoman_meso_pick_foe = {
    if = {
        limit = {
            has_global_variable = nesnd_meso_foe
        }
        remove_global_variable = nesnd_meso_foe
    }
    if = {
        limit = {
            any_country = {
                nesnd_ottoman_meso_foe_candidate = yes
            }
        }
        ordered_country = {
            limit = {
                nesnd_ottoman_meso_foe_candidate = yes
            }
            order_by = nesnd_ottoman_meso_foe_score
            position = 0
            check_range_bounds = no
            set_global_variable = {
                name = nesnd_meso_foe
                value = this
            }
        }
    }
}

# Snapshots STATE_DIYARBAKIR and STATE_DEIR_EZ_ZOR if wholly in Ottoman / Ottoman-subject hands at journal start (Rule 7).
nesnd_ottoman_meso_snapshot = {
    clear_variable_list = nesnd_ottoman_meso_home_regions
    s:STATE_DIYARBAKIR = {
        if = {
            limit = {
                any_scope_state = {
                    always = yes
                }
                NOT = {
                    any_scope_state = {
                        nesnd_ottoman_state_outside_ottoman_hands = yes
                    }
                }
            }
            c:TUR ?= {
                add_to_variable_list = {
                    name = nesnd_ottoman_meso_home_regions
                    target = prev
                }
            }
        }
    }
    s:STATE_DEIR_EZ_ZOR = {
        if = {
            limit = {
                any_scope_state = {
                    always = yes
                }
                NOT = {
                    any_scope_state = {
                        nesnd_ottoman_state_outside_ottoman_hands = yes
                    }
                }
            }
            c:TUR ?= {
                add_to_variable_list = {
                    name = nesnd_ottoman_meso_home_regions
                    target = prev
                }
            }
        }
    }
}

# Unified entry effect for opening nesnd_ottoman_meso (Rule 6).
nesnd_ottoman_meso_start = {
    hidden_effect = {
        if = {
            limit = {
                has_variable = nesnd_ottoman_meso_demarcated
            }
            remove_variable = nesnd_ottoman_meso_demarcated
        }
        nesnd_ottoman_meso_pick_foe = yes
        nesnd_ottoman_meso_snapshot = yes
    }
    add_journal_entry = {
        type = nesnd_ottoman_meso
    }
}

# Monthly counter of consecutive months unable to act abroad for R13 (Rule 8).
nesnd_ottoman_meso_update_dependent_months = {
    if = {
        limit = {
            nesnd_ottoman_can_act_abroad = yes
        }
        if = {
            limit = {
                has_variable = nesnd_ottoman_meso_dependent_months
            }
            remove_variable = nesnd_ottoman_meso_dependent_months
        }
    }
    else_if = {
        limit = {
            has_variable = nesnd_ottoman_meso_dependent_months
        }
        change_variable = {
            name = nesnd_ottoman_meso_dependent_months
            add = 1
        }
    }
    else = {
        set_variable = {
            name = nesnd_ottoman_meso_dependent_months
            value = 1
        }
    }
}

# Adds TUR claims on snapshotted home regions (Diyarbakir / Deir ez-Zor) that have a state outside Ottoman hands (Rule 9.1).
nesnd_ottoman_meso_claim_lost_home = {
    if = {
        limit = {
            has_variable_list = nesnd_ottoman_meso_home_regions
        }
        every_in_list = {
            variable = nesnd_ottoman_meso_home_regions
            if = {
                limit = {
                    any_scope_state = {
                        nesnd_ottoman_state_outside_ottoman_hands = yes
                    }
                    NOT = {
                        c:TUR ?= {
                            has_claim = prev
                        }
                    }
                }
                add_claim = c:TUR
            }
        }
    }
}

# Removes TUR claims on the three Mesopotamia state regions (Rule 13, event .24 option a).
nesnd_ottoman_meso_renounce_claims = {
    s:STATE_BAGHDAD = {
        if = {
            limit = {
                c:TUR ?= {
                    has_claim = s:STATE_BAGHDAD
                }
            }
            remove_claim = c:TUR
        }
    }
    s:STATE_MOSUL = {
        if = {
            limit = {
                c:TUR ?= {
                    has_claim = s:STATE_MOSUL
                }
            }
            remove_claim = c:TUR
        }
    }
    s:STATE_BASRA = {
        if = {
            limit = {
                c:TUR ?= {
                    has_claim = s:STATE_BASRA
                }
            }
            remove_claim = c:TUR
        }
    }
}

# Shared cleanup called at the end of every outcome of nesnd_ottoman_meso (Rule 9).
nesnd_ottoman_meso_cleanup = {
    if = {
        limit = {
            has_global_variable = nesnd_meso_foe
        }
        remove_global_variable = nesnd_meso_foe
    }
    clear_variable_list = nesnd_ottoman_meso_home_regions
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_dependent_months
        }
        remove_variable = nesnd_ottoman_meso_dependent_months
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_deadline
        }
        remove_variable = nesnd_ottoman_meso_deadline
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_tribes_cd
        }
        remove_variable = nesnd_ottoman_meso_tribes_cd
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_planned
        }
        remove_variable = nesnd_ottoman_meso_planned
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_parley_cd
        }
        remove_variable = nesnd_ottoman_meso_parley_cd
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_meso_demarcated
        }
        remove_variable = nesnd_ottoman_meso_demarcated
    }
}
