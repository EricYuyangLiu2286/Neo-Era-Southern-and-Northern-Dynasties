   clear_variable_list = nesnd_restoration_enemy_land
   var:nesnd_restoration_opponent = {
    set_variable = nesnd_restoration_external_end
    set_variable = { name = nesnd_restoration_war_terminal value = 4 }
    clear_variable_list = nesnd_restoration_enemy_land
   }
   nesnd_restoration_publish_notice = { MESSAGE = nesnd_restoration_dispatch_failed_notice REGIONAL_MESSAGE = nesnd_regional_native_dispatch_failed_notice }
  }
  clear_saved_scope = nesnd_restoration_play clear_saved_scope = nesnd_restoration_dispatch_host
 }
}
nesnd_restoration_settlement = {
 if = { limit = {
  trigger_if = {
   limit = { exists = scope:target_country exists = scope:diplomatic_play }
   nesnd_restoration_goal_pair = yes
   scope:diplomatic_play = { OR = {
    AND = { initiator = root target = scope:target_country }
    AND = { target = root initiator = scope:target_country }
   } }
   trigger_if = { limit = { exists = var:nesnd_restoration_play } var:nesnd_restoration_play = scope:diplomatic_play }
   scope:target_country = { any_scope_state = { always = yes } }
  }
  trigger_else = { always = no }
  NOR = { has_variable = nesnd_restoration_external_end has_variable = nesnd_restoration_war_terminal }
  NOT = { has_variable = nesnd_restoration_settling } NOT = { has_variable = nesnd_restoration_settlement_done }
 }
  set_variable = nesnd_restoration_settling
  if = { limit = { has_variable = nesnd_restoration_war_host }
   scope:target_country = { nesnd_restoration_exile_people = yes }
  }
  annex = scope:target_country
  if = { limit = { NOT = { scope:target_country ?= { any_scope_state = { always = yes } } } }
   nesnd_restoration_confirm_unification = yes
  }
  remove_variable = nesnd_restoration_settling
 }
}
nesnd_restoration_confirm_unification = {
 if = { limit = { NOT = { has_variable = nesnd_restoration_settlement_done } }
  set_variable = nesnd_restoration_settlement_done
  set_variable = { name = nesnd_restoration_war_terminal value = 1 }
  nesnd_restoration_mark_resolved = yes
  if = { limit = { has_variable = nesnd_restoration_war_host } nesnd_diaspora_register = yes }
  else = {
   set_variable = nesnd_restoration_victory
   if = { limit = { nesnd_regional_native_actor = yes } add_modifier = { name = nesnd_regional_native_restored_rule months = 120 } }
   else = { add_modifier = { name = nesnd_native_restored_rule months = 120 } }
  }
  nesnd_restoration_popup = { ORIGINAL_EVENT = nesnd_american_restoration.5 REGIONAL_EVENT = nesnd_regional_native.3 }
  nesnd_restoration_publish_notice = { MESSAGE = nesnd_restoration_settlement_notice REGIONAL_MESSAGE = nesnd_regional_native_settlement_notice }
  clear_variable_list = nesnd_restoration_enemy_land
 }
}
# Run this from the opposing country, only for a continent in which it actually holds land.
nesnd_restoration_collect_provinces_in = {
 if = { limit = { any_scope_state = { is_in_geographic_region = $AREA$ } }
  $PROVINCES$ = {
   limit = { owner ?= scope:nesnd_restoration_land_owner }
   scope:nesnd_restoration_map_reader ?= {
    add_to_variable_list = { name = nesnd_restoration_enemy_land target = prev }
   }
  }
 }
}
# Province identities survive an annexation that merges or deletes a split State.
nesnd_restoration_record_enemy_land = {
 if = { limit = { nesnd_restoration_opposed_pair = yes }
  if = { limit = {
   var:nesnd_restoration_opponent = { any_scope_state = { always = yes } }
   OR = {
    NOT = { nesnd_restoration_enemy_map_complete = yes }
    trigger_if = {
     limit = { has_variable = nesnd_restoration_land_count }
     NOT = { var:nesnd_restoration_land_count = var:nesnd_restoration_opponent.nesnd_restoration_province_count }
    }
    trigger_else = { always = no }
    any_in_list = { variable = nesnd_restoration_enemy_land NOT = { owner ?= prev.var:nesnd_restoration_opponent } }
   }
  }
  set_variable = { name = nesnd_restoration_land_count value = var:nesnd_restoration_opponent.nesnd_restoration_province_count }
  save_scope_as = nesnd_restoration_map_reader
  clear_variable_list = nesnd_restoration_enemy_land
  var:nesnd_restoration_opponent = {
   save_scope_as = nesnd_restoration_land_owner
   nesnd_restoration_collect_provinces_in = { AREA = geographic_region_americas PROVINCES = every_province_in_americas }
   nesnd_restoration_collect_provinces_in = { AREA = geographic_region_europe PROVINCES = every_province_in_europe }
   nesnd_restoration_collect_provinces_in = { AREA = geographic_region_africa PROVINCES = every_province_in_africa }
   # The effective Asian region includes Oceania; scanning Oceania again would duplicate provinces.
   nesnd_restoration_collect_provinces_in = { AREA = geographic_region_asia PROVINCES = every_province_in_asia }
  }
  clear_saved_scope = nesnd_restoration_land_owner
  clear_saved_scope = nesnd_restoration_map_reader
 }
 }
}
nesnd_restoration_monthly = {
 if = { limit = { nesnd_restoration_pressure_host = yes nesnd_restoration_has_pressure_entry = yes }
  if = { limit = { NOT = { has_variable = nesnd_restoration_mobilization_started } nesnd_restoration_pressure_value >= 100 }
   nesnd_restoration_select_seed = yes
   nesnd_restoration_start_mobilization = yes
   nesnd_restoration_clear_selected_seed = yes
  }
  else_if = { limit = { nesnd_restoration_mobilization_ready = yes nesnd_restoration_pressure_value >= 75 }
   if = { limit = { c:MEX ?= this } nesnd_restoration_revolt = { TAG = NAH } }
   else_if = { limit = { nesnd_andean_restoration_country = yes } nesnd_restoration_revolt = { TAG = TWT } }
   else_if = { limit = { nesnd_regional_native_actor = yes } nesnd_regional_native_revolt = yes }
  }
 }

nesnd_restoration_war_watch = {
 if = { limit = { nesnd_restoration_armed_case = yes
   NOT = { has_variable = nesnd_restoration_settlement_done }
   NOT = { has_variable = nesnd_restoration_external_end }
   any_scope_state = { always = yes }
  }
  if = { limit = { NOT = { var:nesnd_restoration_opponent ?= { any_scope_state = { always = yes } } } }
   if = { limit = { nesnd_restoration_owns_enemy_land = yes }
    nesnd_restoration_confirm_unification = yes
   }
   else = {
    set_variable = nesnd_restoration_external_end
    nesnd_restoration_missing_opponent_exit = yes
    clear_variable_list = nesnd_restoration_enemy_land
   }
  }
  else_if = { limit = { nesnd_restoration_opposed_pair = yes }
   # Reopen an inconclusive pre-upgrade peace only for the original armed pair.
   if = { limit = { has_variable = nesnd_restoration_war_terminal }
    remove_variable = nesnd_restoration_war_terminal
    remove_variable = nesnd_restoration_peace_shown
    remove_variable = nesnd_restoration_war_dispatched
    remove_variable = nesnd_restoration_play
    if = { limit = { nesnd_regional_native_actor = yes NOT = { has_journal_entry = nesnd_regional_native_war } }
     add_journal_entry = { type = nesnd_regional_native_war }
    }
    else_if = { limit = { NOT = { nesnd_regional_native_actor = yes } NOT = { has_journal_entry = nesnd_native_restoration_war } }
     add_journal_entry = { type = nesnd_native_restoration_war }
    }
   }
   nesnd_restoration_record_enemy_land = yes
   if = { limit = { has_variable = nesnd_restoration_war_host }
    if = { limit = { has_variable = nesnd_restoration_war_dispatched NOT = { nesnd_restoration_pair_at_war = yes } }
     remove_variable = nesnd_restoration_war_dispatched
     remove_variable = nesnd_restoration_play
    }
    nesnd_restoration_dispatch = yes
   }
  }
  else = {
   set_variable = nesnd_restoration_external_end
   nesnd_restoration_missing_opponent_exit = yes
   clear_variable_list = nesnd_restoration_enemy_land
  }
 }
 if = { limit = { has_variable = nesnd_restoration_refugees_pending } remove_variable = nesnd_restoration_refugees_pending }
}
