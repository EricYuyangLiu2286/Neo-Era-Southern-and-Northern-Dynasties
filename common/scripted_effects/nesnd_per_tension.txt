nesnd_per_tension_init = {
    if = { limit = { NOT = { has_variable = nesnd_per_tension } } set_variable = { name = nesnd_per_tension value = 20 } }
    if = { limit = { NOT = { has_variable = nesnd_per_tension_months } } set_variable = { name = nesnd_per_tension_months value = 0 } }
}
# 唐人统治的开局设置：唐人等级制与唐律成文。
nesnd_per_tang_setup = {
    if = { limit = { nesnd_per_tang_regime = yes }
        # 开局只设一次「唐律成文」：法律组启用时引擎会先自动给一条。
        if = { limit = { NOT = { has_variable = nesnd_per_tang_law_started } }
            set_variable = nesnd_per_tang_law_started
            if = { limit = { NOT = { has_law = law_type:law_nesnd_per_tang_codified } } activate_law = law_type:law_nesnd_per_tang_codified }
        }
        else_if = { limit = { game_date < 1836.3.1 NOT = { has_law = law_type:law_nesnd_per_tang_codified } }
            activate_law = law_type:law_nesnd_per_tang_codified
        }
        if = { limit = { NOT = { has_social_hierarchy = nesnd_per_tang_hierarchy } } set_social_hierarchy = nesnd_per_tang_hierarchy }
        # 第三教开局二选一（佛门或天命说）：开局问一次；中途接手或久未作答的按佛门。
        if = { limit = { NOT = { has_variable = nesnd_per_third_chosen } }
            if = { limit = { NOT = { has_variable = nesnd_per_third_asked } game_date < 1836.3.1 }
                set_variable = nesnd_per_third_asked
                trigger_event = { id = nesnd_per_faith.2 popup = yes }
            }
            else_if = { limit = { OR = { NOT = { has_variable = nesnd_per_third_asked } game_date >= 1836.6.1 } }
                set_variable = nesnd_per_third_chosen
            }
        }
    }
}
nesnd_per_tension_clear = {
    remove_variable = nesnd_per_tension
    remove_variable = nesnd_per_tension_months
    remove_variable = nesnd_per_tension_once_office
    remove_variable = nesnd_per_tension_once_shrine
    remove_variable = nesnd_per_tension_once_settle
    remove_variable = nesnd_per_tension_once_purge
    remove_variable = nesnd_per_tension_once_claim
}
