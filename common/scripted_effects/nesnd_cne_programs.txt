nesnd_cne_programs_init = {
 if = { limit = { nesnd_reform_active = yes NOT = { has_variable = nesnd_cne_programs_v3 } }
  set_variable = nesnd_cne_programs_v3
  set_variable = { name = nesnd_cne_camps_active value = 0 }
  if = { limit = { has_modifier = modifier_government_recently_couped } set_variable = nesnd_cne_old_coup_baseline }
 }
 # Repair a cleared running marker; never reset progress or resurrect a terminal JE.
 if = { limit = { nesnd_reform_active = yes has_journal_entry = ce_cne_land_reform NOT = { has_variable = nesnd_reform_terminal_pending } }
  set_variable = nesnd_cne_programs_running
  nesnd_cne_confidence_init = yes
  nesnd_cne_ensure_children = yes
 }
}
nesnd_cne_check_coup = {
	if = { limit = { nesnd_reform_active = yes has_journal_entry = ce_cne_land_reform }
		if = { limit = { NOT = { has_modifier = modifier_government_recently_couped } } remove_variable = nesnd_cne_old_coup_baseline }
		if = { limit = { has_modifier = modifier_government_recently_couped NOT = { has_variable = nesnd_cne_old_coup_baseline } }
			set_variable = nesnd_cne_coup_failure
			nesnd_cne_programs_cleanup = yes
		}
	}
}
