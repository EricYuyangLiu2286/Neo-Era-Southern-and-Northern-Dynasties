nesnd_c2_duan_monthly = {
    if = {
        limit = { nesnd_c2_duan_introduction = yes }
        if = {
            limit = {
                NOT = { is_template_used = nesnd_usm_duan }
                NOT = { has_global_variable = nesnd_seen_usm_duan }
            }
            create_character = {
                template = nesnd_usm_duan
                on_created = { set_global_variable = nesnd_seen_usm_duan }
            }
        }
        if = {
            limit = { any_scope_character = { nesnd_c2_duan_politician = yes } }
            trigger_event = { id = nesnd_civic_characters.14 popup = yes }
        }
    }
    else_if = {
        limit = { nesnd_c2_duan_commission = yes }
        trigger_event = { id = nesnd_civic_characters.200 popup = yes }
    }
}

nesnd_c2_seat_duan_at_academy = {
    random_scope_state = {
        limit = { nesnd_c2_sichuan_academy = yes }
        save_scope_as = c2_duan_school
    }
    random_scope_character = {
        limit = { nesnd_c2_duan_politician = yes }
        save_scope_as = c2_duan
        set_variable = { name = nesnd_story_civic_event days = 3 }
    }
}
