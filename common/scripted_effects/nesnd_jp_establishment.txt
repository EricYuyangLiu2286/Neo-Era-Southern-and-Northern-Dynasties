nesnd_jp_finish_establishment = {
    if = { limit = { nesnd_jp_establishment_needed = yes nesnd_jp_establishment_ready = yes }
        set_variable = nesnd_jp_established
        if = { limit = { NOT = { has_modifier = nesnd_jp_settled_administration } }
            add_modifier = { name = nesnd_jp_settled_administration months = 60 }
        }
    }
}
nesnd_jp_ensure_establishment = {
    if = { limit = { nesnd_jp_establishment_needed = yes has_variable = nesnd_jp_statistics_ready }
        if = { limit = { NOT = { has_variable = nesnd_jp_establishment_opened } nesnd_jp_establishment_ready = yes }
            set_variable = nesnd_jp_established
        }
        else = {
            if = { limit = { c:JAP ?= this NOT = { has_journal_entry = nesnd_jp_establish_toyotomi } } add_journal_entry = { type = nesnd_jp_establish_toyotomi } }
            if = { limit = { c:JTK ?= this NOT = { has_journal_entry = nesnd_jp_establish_edo } } add_journal_entry = { type = nesnd_jp_establish_edo } }
            if = { limit = { c:JIM ?= this NOT = { has_journal_entry = nesnd_jp_establish_kyoto } } add_journal_entry = { type = nesnd_jp_establish_kyoto } }
        }
    }
}
