# Remaining northern targets claimed upon Ottoman victory in the Crimean / Northern War (uses shared target region list).
nesnd_crimea_claim_remaining_targets = {
    every_state_region = {
        limit = {
            nesnd_north_is_north_target_region = yes
            any_scope_state = { NOT = { owner = root } }
            NOT = { root = { has_claim = prev } }
        }
        add_claim = root
    }
}

# Starts the Crimean / Northern War journals for TUR and the selected northern foe (Rule 7, 14, 20.5).
# Helper journals (nesnd_crimea_helper) are now added only when a foreign power accepts an appeal.
nesnd_crimea_start = {
    if = {
        limit = { has_global_variable = nesnd_crimea_result }
        remove_global_variable = nesnd_crimea_result
    }
    set_global_variable = nesnd_crimea_active
    c:TUR ?= {
        if = {
            limit = { is_country_alive = yes NOT = { has_journal_entry = nesnd_crimea_tur } }
            add_journal_entry = { type = nesnd_crimea_tur }
        }
    }
    if = {
        limit = { has_global_variable = nesnd_north_foe }
        global_var:nesnd_north_foe ?= {
            if = {
                limit = { is_country_alive = yes NOT = { has_journal_entry = nesnd_crimea_rus } }
                add_journal_entry = { type = nesnd_crimea_rus }
            }
        }
    }
}

# TUR launches the Crimean / Northern War diplomatic play against global_var:nesnd_north_foe (Rules 7, 10).
nesnd_crimea_tur_launch_play = {
    clear_saved_scope = nesnd_crimea_state
    clear_saved_scope = nesnd_foe_target
    if = {
        limit = { nesnd_north_foe_valid = yes }
        global_var:nesnd_north_foe ?= {
            save_scope_as = nesnd_foe_target
            if = {
                limit = { any_scope_state = { state_region = s:STATE_KARS } }
                random_scope_state = { limit = { state_region = s:STATE_KARS } save_scope_as = nesnd_crimea_state }
            }
            else_if = {
                limit = { any_scope_state = { state_region = s:STATE_CRIMEA } }
                random_scope_state = { limit = { state_region = s:STATE_CRIMEA } save_scope_as = nesnd_crimea_state }
            }
            else_if = {
                limit = { any_scope_state = { state_region = s:STATE_GREATER_CAUCASUS } }
                random_scope_state = { limit = { state_region = s:STATE_GREATER_CAUCASUS } save_scope_as = nesnd_crimea_state }
            }
            else_if = {
                limit = { any_scope_state = { state_region = s:STATE_ARMENIA } }
                random_scope_state = { limit = { state_region = s:STATE_ARMENIA } save_scope_as = nesnd_crimea_state }
            }
            else_if = {
                limit = { any_scope_state = { state_region = s:STATE_AZERBAIJAN } }
                random_scope_state = { limit = { state_region = s:STATE_AZERBAIJAN } save_scope_as = nesnd_crimea_state }
            }
            else_if = {
                limit = { any_scope_state = { state_region = s:STATE_ELIZAVETPOL } }
                random_scope_state = { limit = { state_region = s:STATE_ELIZAVETPOL } save_scope_as = nesnd_crimea_state }
            }
        }
    }
    if = {
        limit = { exists = scope:nesnd_foe_target exists = scope:nesnd_crimea_state }
        scope:nesnd_crimea_state.state_region = {
            if = { limit = { NOT = { root = { has_claim = prev } } } add_claim = root }
        }
        create_diplomatic_play = {
            name = nesnd_crimea_play
            type = dp_return_state
            target_country = scope:nesnd_foe_target
            target_state = scope:nesnd_crimea_state
            add_war_goal = {
                holder = scope:nesnd_foe_target
                target_country = root
                type = humiliation
            }
        }
        if = {
            limit = { nesnd_north_play_in_progress = yes }
            nesnd_north_snapshot = yes
            nesnd_crimea_start = yes
        }
    }
}

# Helper country backs TUR in the active pre-war diplomatic play between TUR and global_var:nesnd_north_foe (Rules 7, 13).
nesnd_crimea_join_tur_play = {
    if = {
        limit = { exists = c:TUR has_global_variable = nesnd_north_foe }
        global_var:nesnd_north_foe ?= {
            random_diplomatic_play = {
                limit = {
                    is_war = no
                    OR = {
                        AND = { initiator_is = c:TUR target_is = prev }
                        AND = { initiator_is = prev target_is = c:TUR }
                    }
                }
                if = {
                    limit = { initiator_is = c:TUR }
                    add_initiator_backers = { root }
                    if = {
                        limit = { has_global_variable = nesnd_crimea_active }
                        root = { set_variable = nesnd_crimea_joined_tur }
                    }
                }
                else_if = {
                    limit = { target_is = c:TUR }
                    add_target_backers = { root }
                    if = {
                        limit = { has_global_variable = nesnd_crimea_active }
                        root = { set_variable = nesnd_crimea_joined_tur }
                    }
                }
            }
        }
    }
}

# Signs a 5-year defensive pact treaty if legal and not already bound by an alliance or defensive pact.
nesnd_crimea_sign_defensive_pact = {
    if = {
        limit = { nesnd_crimea_can_sign_defensive_pact_with = { PARTNER = $PARTNER$ } }
        create_treaty = {
            first_country = root
            second_country = $PARTNER$
            is_draft = no
            binding_period = { years = 5 }
            articles_to_create = {
                { article = defensive_pact }
            }
        }
    }
}

# Yearly check on the northern foe: player chooses via nesnd_crimea.1; AI rolls by relation tier and year.
nesnd_crimea_rus_yearly_check = {
    if = {
        limit = { nesnd_crimea_rus_yearly_eligible = yes }
        if = {
            limit = { is_player = yes NOT = { has_variable = nesnd_crimea_rus_defer } }
            trigger_event = { id = nesnd_crimea.1 popup = yes }
        }
        else_if = {
            limit = { is_player = no }
            if = {
                limit = { game_date < 1851.1.1 }
                random_list = { 2 = { nesnd_crimea_rus_launch_play = yes } 98 = { } }
            }
            else_if = {
                limit = { "relations(c:TUR)" <= relations_threshold:hostile }
                random_list = { 25 = { nesnd_crimea_rus_launch_play = yes } 75 = { } }
            }
            else_if = {
                limit = { "relations(c:TUR)" <= relations_threshold:cold }
                random_list = { 18 = { nesnd_crimea_rus_launch_play = yes } 82 = { } }
            }
            else_if = {
                limit = { "relations(c:TUR)" <= relations_threshold:poor }
                random_list = { 13 = { nesnd_crimea_rus_launch_play = yes } 87 = { } }
            }
            else = {
                random_list = { 5 = { nesnd_crimea_rus_launch_play = yes } 95 = { } }
            }
        }
    }
}
