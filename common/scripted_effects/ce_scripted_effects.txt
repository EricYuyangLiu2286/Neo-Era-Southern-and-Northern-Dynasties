ce_change_culture_to_huaxia = {
	custom_tooltip = all_culture_china_transfer
	
	hidden_effect = {
		if = {
			limit = { NOT = { country_has_primary_culture = cu:ce_zhonghua } }
			add_primary_culture = cu:ce_zhonghua
		}

		if = {
			limit = {
				exists = c:CHI
				c:CHI ?= This
				country_has_primary_culture = cu:manchu
			}
			remove_primary_culture = cu:manchu
		}
		every_scope_state = {
			limit = {
				is_incorporated = yes
			}
			every_scope_pop = {
				limit = {
					OR = {
						culture = { has_discrimination_trait = heritage_han }
						has_pop_culture = manchu # because redux cloth has a very bad trigger for qing cloth sorry
					}
						
					NOT = { has_pop_culture = ce_zhonghua }
				}
				change_pop_culture = { target = cu:ce_zhonghua value = 1.00 }
			}
		}
		if = {
			limit = {
				any_scope_state = { 
					state_region = { is_ce_zhonghua_eligible_culture_homeland = yes }
				}
			}
			every_scope_state = {
				limit = {
					state_region = { is_ce_zhonghua_eligible_culture_homeland = yes }
				}
				state_region = {
					#ce_remove_ce_zhonghua_eligible_culture_homeland_effect = yes
					add_homeland = cu:ce_zhonghua
				}
			}
		}
	}
}

ce_change_all_chinese_character_to_huaxia = {
	custom_tooltip = all_culture_character_china_transfer
	hidden_effect = {
		every_scope_character = {
			limit = {
				OR = {
					culture = { has_discrimination_trait = heritage_han }
					has_culture = cu:manchu # because redux cloth has a very bad trigger for qing cloth sorry
				}
				NOT = { has_culture = cu:ce_zhonghua }
				is_character_alive = yes
			}
			change_character_culture = cu:ce_zhonghua
		}
	}
}

ce_unification_of_china_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = ce_event_culture_1_happened
			}
		}
		#custom_tooltip = all_culture_china_transfer
		#ce_change_culture_to_huaxia = yes
		#change_tag = CHI
		#
		add_modifier = {
			name = ce_culture_reunification
			months = 120
		}
		set_variable = {
			name = ce_event_culture_1_happened
			value = yes
		}
	}
}

		limit = {
			has_variable = ce_anti_polish_variable
