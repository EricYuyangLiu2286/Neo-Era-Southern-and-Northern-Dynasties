nesnd_qe1_unity_sync = {
	if = { limit = { has_journal_entry = nesnd_qe1_unity_journal } je:nesnd_qe1_unity_journal = { set_bar_progress = { name = nesnd_qe1_unity_bar value = root.var:nesnd_qe1_unity } } }
}
nesnd_qe1_unity_monthly = {
	if = { limit = { nesnd_qe1_unity_live = yes }
		nesnd_qe1_record_peak_states = yes
		nesnd_qe1_measure_heritages = yes

		set_variable = { name = nesnd_qe1_sol_current value = nesnd_qe1_average_sol }
		set_variable = { name = nesnd_qe1_sol_change value = 0 }
		if = { limit = { has_variable = nesnd_qe1_sol_previous } set_variable = { name = nesnd_qe1_sol_change value = { value = var:nesnd_qe1_sol_current subtract = var:nesnd_qe1_sol_previous } } }
		set_variable = { name = nesnd_qe1_sol_previous value = var:nesnd_qe1_sol_current }
		if = { limit = { is_at_war = yes } change_variable = { name = nesnd_qe1_war_months add = 1 } } else = { set_variable = { name = nesnd_qe1_war_months value = 0 } }
		change_variable = { name = nesnd_qe1_unity add = nesnd_qe1_unity_delta }
		clamp_variable = { name = nesnd_qe1_unity min = 0 max = 100 }
		if = { limit = { var:nesnd_qe1_unity > 75 has_variable = nesnd_qe1_next_all }
			every_scope_state = {
				limit = { nesnd_qe1_convert_state = yes }
				every_scope_pop = {
					limit = {
						NOT = {
							culture = { has_discrimination_trait = tradition_sinosphere }
						}
						NOT = { culture = cu:nesnd_qe1_conversion_cohort }
					}
					change_pop_culture = {
						target = cu:ce_xiyu
						value = 0.005
					}
				}
			}
		}
		else_if = { limit = { nesnd_qe1_forecast_state_valid = yes }
			var:nesnd_qe1_next_state = {
				save_scope_as = nesnd_qe1_declining_state
				state_region = { save_scope_as = nesnd_qe1_declining_region }
			}
			# Hand the identity drawn last month to the event under names that
			# nesnd_qe1_prepare_identity below does not overwrite.
			if = { limit = { exists = var:nesnd_qe1_next_culture exists = var:nesnd_qe1_next_religion }
				var:nesnd_qe1_next_culture = { save_scope_as = nesnd_qe1_settle_culture }
				var:nesnd_qe1_next_religion = { save_scope_as = nesnd_qe1_settle_religion }
			}
			trigger_event = { id = nesnd_qe1_identity.1 popup = yes }
		}
		nesnd_qe1_unity_sync = yes
		nesnd_qe1_prepare_identity = yes
	}
}
nesnd_qe1_clear_forecast = {
	remove_variable = nesnd_qe1_next_state
	remove_variable = nesnd_qe1_next_all
	remove_variable = nesnd_qe1_next_culture
	remove_variable = nesnd_qe1_next_religion
}
nesnd_qe1_prepare_identity = {
	nesnd_qe1_clear_forecast = yes
	if = {
		limit = { nesnd_qe1_unity_eligible = yes has_variable = nesnd_qe1_unity }
		if = {
			limit = {
				var:nesnd_qe1_unity > 75
				any_scope_state = {
					nesnd_qe1_convert_state = yes
					any_scope_pop = {
						NOT = {
							culture = { has_discrimination_trait = tradition_sinosphere }
						}
						NOT = { culture = cu:nesnd_qe1_conversion_cohort }
					}
				}
			}
			set_variable = nesnd_qe1_next_all
		}
		else_if = {
			limit = { var:nesnd_qe1_unity < 50 any_scope_state = { nesnd_qe1_decay_state = yes } }
			random_list = {
				80 = { }
				20 = {
					random_scope_state = {
						limit = { nesnd_qe1_decay_state = yes }
						owner = { set_variable = { name = nesnd_qe1_next_state value = prev } }
						# Draw the returning culture and faith now, so the journal can show them.
						nesnd_qe1_select_old_identity = yes
						if = { limit = { exists = scope:nesnd_qe1_return_culture exists = scope:nesnd_qe1_return_religion }
							owner = {
								set_variable = { name = nesnd_qe1_next_culture value = scope:nesnd_qe1_return_culture }
								set_variable = { name = nesnd_qe1_next_religion value = scope:nesnd_qe1_return_religion }
							}
						}
						clear_saved_scope = nesnd_qe1_return_culture
						clear_saved_scope = nesnd_qe1_return_religion
					}
				}
			}
		}
	}
}
