nesnd_land_init = {
    if = { limit = { NOT = { has_variable = cne_landreform_completion_tracker } } set_variable = { name = cne_landreform_completion_tracker value = 25 } }
    if = { limit = { NOT = { has_variable = cne_landreform_failure_tracker } } set_variable = { name = cne_landreform_failure_tracker value = 20 } }
    if = { limit = { NOT = { has_variable = nesnd_land_months } } set_variable = { name = nesnd_land_months value = 0 } }
}
nesnd_land_monthly = {
    nesnd_land_init = yes
    change_variable = { name = nesnd_land_months add = 1 }
    if = { limit = { bureaucracy >= 0 is_at_war = no } change_variable = { name = cne_landreform_completion_tracker add = 0.75 } }
    else = { change_variable = { name = cne_landreform_failure_tracker add = 0.75 } }
    if = { limit = { bureaucracy < 0 } change_variable = { name = cne_landreform_failure_tracker add = 0.5 } }
    if = { limit = { country_turmoil > 0.25 } change_variable = { name = cne_landreform_failure_tracker add = 0.5 } }
    if = { limit = { has_modifier = declared_bankruptcy } change_variable = { name = cne_landreform_failure_tracker add = 1 } }
    if = { limit = { has_variable = nesnd_north_intervention } change_variable = { name = cne_landreform_failure_tracker add = 1 } }
    if = { limit = { has_technology_researched = nationalism has_technology_researched = labor_movement } change_variable = { name = cne_landreform_completion_tracker add = 0.25 } }
    if = { limit = { NOT = { any_scope_state = { market_access < 0.85 } } } change_variable = { name = cne_landreform_completion_tracker add = 0.25 } }
    if = { limit = { has_law_or_variant = law_type:law_public_schools NOT = { has_law_or_variant = law_type:law_serfdom } } change_variable = { name = cne_landreform_completion_tracker add = 0.25 } }
    if = { limit = { ig:ig_landowners = { is_powerful = yes is_in_government = yes } } change_variable = { name = cne_landreform_failure_tracker add = 0.5 } }
    if = { limit = { has_modifier = nesnd_land_appeals } change_variable = { name = cne_landreform_failure_tracker add = -1 } }
    if = { limit = { var:cne_landreform_failure_tracker >= 55 NOT = { has_variable = nesnd_land_petition_seen } }
        set_variable = nesnd_land_petition_seen
        trigger_event = { id = nesnd_crisis.30 popup = yes }
    }
    clamp_variable = { name = cne_landreform_completion_tracker min = 0 max = 100 } clamp_variable = { name = cne_landreform_failure_tracker min = 0 max = 100 }

    nesnd_refresh_land_display = yes
}
nesnd_land_cleanup = {
    set_variable = ce_CNE_land_reform_done
    remove_variable = flavorCNE_land_reform_value
    remove_modifier = CNE_land_draft_debuff
    remove_variable = nesnd_north_intervention
}
nesnd_land_success = {
    nesnd_land_cleanup = yes
    set_variable = nesnd_land_succeeded
    if = { limit = { var:nesnd_land_months <= 60 var:cne_landreform_failure_tracker < 35 }
        set_variable = nesnd_land_rapid_success
        add_modifier = { name = nesnd_land_rapid months = 120 }
    }
    if = { limit = { OR = { has_law_or_variant = law_type:law_serfdom has_law_or_variant = law_type:law_tenant_farmers } }
        activate_law = law_type:law_peasant_proprietorship
    }
    trigger_event = { id = flavorCNE.3 popup = yes }
}

nesnd_release_nsx = {
        if = { limit = { NOT = { exists = c:NSX } num_states > 1
            exists = s:STATE_SHANXI.region_state:CNE
            s:STATE_SHANXI.region_state:CNE = { is_capital = no }
        }
            create_country = { tag = NSX origin = this state = s:STATE_SHANXI.region_state:CNE
                on_created = {
                    set_variable = nesnd_republican_rising
                    activate_law = law_type:law_presidential_republic
                    add_modifier = { name = nesnd_secession_disorder months = 36 }
                    add_change_relations_progress = { country = c:CNE value = -75 }
                }
            }
            s:STATE_SHANXI = { add_claim = root }
        }
    }

nesnd_join_opposition_play = {
    if = { limit = { NOT = { has_variable = nesnd_joined_south_play } is_subject = no exists = c:USM
        any_diplomatic_play = { has_variable = this_is_nanming_civilwar_var is_war = no }
    }
        random_diplomatic_play = {
            limit = { has_variable = this_is_nanming_civilwar_var is_war = no }
            add_target_backers = { root }
        }
        set_variable = nesnd_joined_south_play
    }
}

nesnd_open_southern_war = {
    if = { limit = { NOT = { has_global_variable = usm_civilwar_occured } }
        if = { limit = { c:SIH ?= { has_variable = usm_opposition_country } } c:SIH = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:HNA ?= { has_variable = usm_opposition_country } } c:HNA = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:GNG ?= { has_variable = usm_opposition_country } } c:GNG = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:PHI ?= { has_variable = usm_opposition_country } } c:PHI = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:DEI ?= { has_variable = usm_opposition_country } } c:DEI = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:YN1 ?= { has_variable = usm_opposition_country } } c:YN1 = { save_scope_as = nesnd_opposition_leader } }
else_if = { limit = { c:JAP ?= { has_variable = usm_opposition_country } } c:JAP = { save_scope_as = nesnd_opposition_leader } }

        if = { limit = { exists = scope:nesnd_opposition_leader }
            scope:nesnd_opposition_leader = { set_variable = usm_opposition_country_leader_var }
            every_country = { limit = { has_variable = usm_opposition_country is_subject_of = root } make_independent = yes }
            create_diplomatic_play = { name = usm_civilwar_play target_country = scope:nesnd_opposition_leader type = dp_usm_civil_war }
            random_diplomatic_play = {
                limit = { initiator = root target = scope:nesnd_opposition_leader }
                set_variable = this_is_nanming_civilwar_var
            }
            set_global_variable = usm_civilwar_occured
            set_variable = { name = nesnd_war_opening_grace months = 6 }
            every_country = { limit = { has_variable = usm_opposition_country NOT = { this = scope:nesnd_opposition_leader } is_subject = no }
                random_diplomatic_play = { limit = { has_variable = this_is_nanming_civilwar_var } add_target_backers = { prev } }
            }
            every_country = { limit = { OR = { has_variable = usm_opposition_country is_subject_of = root c:JAP ?= this } NOT = { c:USM ?= this } }
                trigger_event = { id = usm_civilwar.2 days = 1 popup = yes }
            }
            c:CNE ?= { if = { limit = { has_variable = nesnd_north_intervention } nesnd_join_opposition_play = yes } }
            c:JAP ?= { if = { limit = { has_variable = ce_usm_cw_ally_with_mng is_subject = no }
                random_diplomatic_play = { limit = { has_variable = this_is_nanming_civilwar_var } add_initiator_backers = { c:JAP } }
            } }
        } else = { trigger_event = { id = usm_civilwar.3 popup = yes } }
    }
}
