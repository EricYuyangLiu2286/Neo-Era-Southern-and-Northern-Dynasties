# Effects for Ottoman success line, block 3 ("Academies and the Press") and the labor event. Country scope.

# Shared timeout / ulema victory outcome across all four rings (300 months total from ring 1 activation).
nesnd_ottoman_academy_ulema_win = {
    if = {
        limit = {
            NOT = { has_law = law_type:law_religious_schools }
            any_law = {
                law_is_available = yes
                this.type = law_type:law_religious_schools
            }
        }
        activate_law = law_type:law_religious_schools
    }
    add_modifier = { name = nesnd_ottoman_ulema_schools }
    if = {
        limit = { has_journal_entry = nesnd_ottoman_charter }
        custom_tooltip = {
            text = nesnd_ottoman_charter_bonus_qin_gain_tt
            set_variable = { name = nesnd_ottoman_qin_bonus months = 60 }
        }
    }
    set_variable = { name = nesnd_ottoman_academy_outcome value = flag:ulema }
    set_variable = nesnd_ottoman_academy_closed
}

# Yearly 20% roll for "Unions and Factories" (event nesnd_ottoman_recovery.17).
nesnd_ottoman_labor_yearly_check = {
    if = {
        limit = { nesnd_ottoman_labor_event_eligible = yes }
        random_list = {
            20 = { trigger_event = { id = nesnd_ottoman_recovery.17 } }
            80 = { }
        }
    }
}
