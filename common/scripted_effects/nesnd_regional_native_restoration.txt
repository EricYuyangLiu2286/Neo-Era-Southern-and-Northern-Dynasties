  if = { limit = { nesnd_regional_native_host = yes nesnd_restoration_selected_seed_valid = yes NOT = { exists = c:$TAG$ } }

   nesnd_regional_native_transfer = { TAG = $TAG$ }
   c:$TAG$ ?= {
     if = { limit = { exists = scope:nesnd_restoration_host }
      if = { limit = { scope:nesnd_restoration_host = { has_variable = nesnd_regional_native_profile has_variable = nesnd_regional_native_kind has_variable = nesnd_regional_native_creation_mode } any_scope_state = { state_population > 0 } }
