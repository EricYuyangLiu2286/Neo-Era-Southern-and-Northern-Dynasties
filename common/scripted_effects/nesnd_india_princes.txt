# ==== The request channel: princes ask their overlord or a power for aid (formerly nesnd_india_diplomacy.txt) ====
nesnd_india_clear_request = {
    if = { limit = { exists = var:nesnd_india_request_partner }
        save_scope_as = india_request_cleanup_owner
        if = { limit = { exists = scope:india_request_cleanup_owner }
            var:nesnd_india_request_partner = {
                if = { limit = { exists = var:nesnd_india_incoming_union_sender }
                    if = { limit = { hidden_trigger = { var:nesnd_india_incoming_union_sender = scope:india_request_cleanup_owner } } remove_variable = nesnd_india_incoming_union_sender remove_variable = nesnd_india_incoming_union_clock }
                }
            }

        }
    }
    remove_variable = nesnd_india_request_clock remove_variable = nesnd_india_request_partner
    remove_variable = nesnd_india_request_kind remove_variable = nesnd_india_request_quote
    remove_variable = nesnd_india_union_mode
}
nesnd_india_send_request = {
    save_scope_value_as = { name = india_request_kind_to_send value = $KIND$ }
    if = { limit = { OR = { nesnd_india_mission_country = yes AND = { scope:india_request_kind_to_send = 4 has_journal_entry = nesnd_india_indep is_subject = yes can_increase_autonomy = yes } } nesnd_india_channel_free = yes exists = scope:india_partner }
        if = { limit = { scope:india_partner = { is_country_alive = yes } NOT = { this = scope:india_partner } NOT = { has_war_with = scope:india_partner } }
            save_scope_as = india_sender
            save_scope_value_as = { name = india_sender_regime value = nesnd_india_regime }
            if = { limit = { is_subject = yes } overlord = { save_scope_as = india_sender_patron } } else = { save_scope_as = india_sender_patron }
            if = { limit = { exists = scope:india_sender exists = scope:india_partner exists = scope:india_sender_patron exists = scope:india_sender_regime }
                scope:india_partner = {
                    save_scope_value_as = { name = india_partner_regime value = nesnd_india_regime }
                    if = { limit = { is_subject = yes } overlord = { save_scope_as = india_partner_patron } } else = { save_scope_as = india_partner_patron }
                }
                set_variable = { name = nesnd_india_request_serial value = { value = nesnd_india_request_serial add = 1 } }
                save_scope_value_as = { name = india_serial value = nesnd_india_request_serial }
                if = { limit = { exists = scope:india_serial exists = scope:india_partner_patron exists = scope:india_partner_regime }
                    set_variable = { name = nesnd_india_request_kind value = $KIND$ }
                    set_variable = { name = nesnd_india_request_partner value = scope:india_partner }
                    set_variable = { name = nesnd_india_request_quote value = nesnd_india_grant }
                    # A prince asking to join quotes what Mughal would pay.
                    if = { limit = { nesnd_india_request_kind = 8 } set_variable = { name = nesnd_india_request_quote value = scope:india_partner.nesnd_india_grant } }
                    set_variable = { name = nesnd_india_request_clock days = 45 }
                    set_variable = { name = nesnd_india_request_cooldown months = 6 }
                    if = { limit = { nesnd_india_request_kind = 6 } scope:india_partner = { set_variable = { name = nesnd_india_incoming_union_sender value = scope:india_sender } set_variable = { name = nesnd_india_incoming_union_clock days = 45 } } }
                    scope:india_partner = { trigger_event = { id = nesnd_india_diplomacy.1 days = 1 popup = yes } }


                }
            }
        }
    }
}
nesnd_india_refuse_request = {
    if = { limit = { nesnd_india_reply_valid = yes nesnd_india_union_reply = yes } nesnd_india_union_decline = yes }
    else_if = { limit = { nesnd_india_reply_valid = yes }
        scope:india_sender = {
            if = { limit = { nesnd_india_request_kind = 4 } add_liberty_desire = 10 }
            if = { limit = { nesnd_india_request_kind = 4 }
                change_relations = { country = scope:india_partner value = -30 }
                create_diplomatic_catalyst = { type = catalyst_autonomy_increase_denied target = scope:india_partner }
                scope:india_partner = { create_diplomatic_catalyst = { type = catalyst_autonomy_increase_requested target = scope:india_sender } }
            }
            trigger_event = { id = nesnd_india_diplomacy.3 popup = yes }
            nesnd_india_clear_request = yes
        }
    }
}
nesnd_india_accept_request = {
    if = { limit = { nesnd_india_reply_can_accept = yes }
        if = { limit = { nesnd_india_incoming_kind <= 3 }
            add_treasury = { value = scope:india_sender.nesnd_india_request_quote multiply = -1 }
            scope:india_sender = {
                add_treasury = nesnd_india_request_quote
                # Counted for the outcome "aid bound" of nesnd_india_foreign.
                set_variable = { name = nesnd_india_aid_count value = { value = nesnd_india_aid_count_value add = 1 } }
                if = { limit = { NOT = { is_target_in_variable_list = { name = nesnd_india_aid_donors target = scope:india_partner } } } add_to_variable_list = { name = nesnd_india_aid_donors target = scope:india_partner } }
                add_modifier = { name = nesnd_india_french_advisers months = 48 }
                set_variable = { name = nesnd_india_aid_lock months = 48 }
                set_variable = { name = nesnd_india_adviser_partner value = scope:india_partner }
                if = { limit = { is_subject = yes } set_variable = { name = nesnd_india_adviser_patron value = overlord } } else = { set_variable = { name = nesnd_india_adviser_patron value = this } }
                set_owes_obligation_to = { country = scope:india_partner setting = yes }
                if = {
                    limit = { nesnd_india_request_kind = 1 }
                    add_modifier = { name = nesnd_india_french_teachers months = 48 }
                    scope:india_partner = {
                        if = { limit = { has_modifier = nesnd_india_teaching_mission } remove_modifier = nesnd_india_teaching_mission }
                        add_modifier = { name = nesnd_india_teaching_mission months = 48 }
                    }
                }
                else_if = {
                    limit = { nesnd_india_request_kind = 2 }
                    add_modifier = { name = nesnd_india_french_engineers months = 48 }
                    if = {
                        limit = { capital = { nesnd_india_workshop_site = yes } }
                        capital = { save_scope_as = india_workshop_state }
                    }
                    else = {
                        ordered_scope_state = {
                            limit = { nesnd_india_workshop_site = yes }
                            order_by = state_population
                            position = 0
                            check_range_bounds = no
                            save_scope_as = india_workshop_state
                        }
                    }
                    if = {
                        limit = { exists = scope:india_workshop_state scope:india_workshop_state = { nesnd_india_workshop_stake_met = no } }
                        scope:india_workshop_state = {
                            if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 5 } nesnd_india_grant_workshop = { LEVELS = 5 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 4 } nesnd_india_grant_workshop = { LEVELS = 4 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 3 } nesnd_india_grant_workshop = { LEVELS = 3 } }
                            else_if = { limit = { scope:india_sender.nesnd_india_workshop_levels = 2 } nesnd_india_grant_workshop = { LEVELS = 2 } }
                            else = { nesnd_india_grant_workshop = { LEVELS = 1 } }
                        }
                    }
                }
                else = {
                    add_modifier = { name = nesnd_india_french_officers months = 48 }
                    create_treaty = {
                        first_country = scope:india_sender
                        second_country = scope:india_partner
                        is_draft = no
                        binding_period = { years = 5 }
                        articles_to_create = {
                            { article = military_access source_country = scope:india_sender target_country = scope:india_partner }
                        }
                    }
                }
            }
        }
        else_if = { limit = { nesnd_india_incoming_kind = 4 } scope:india_sender = { increase_autonomy = yes add_liberty_desire = -20 } }
        else_if = { limit = { nesnd_india_incoming_kind = 5 }
            create_treaty = { first_country = scope:india_sender second_country = scope:india_partner is_draft = no binding_period = { years = 3 } articles_to_create = { { article = defensive_pact } } }
        }
        # Request kind 6 is answered by the options of its own (nesnd_india_union_accept).
        if = { limit = { nesnd_india_incoming_kind = 8 }
            if = { limit = { scope:india_sender = { nesnd_india_union_mode_value = 2 } } nesnd_india_union_accept = { MODE = dominion } }
            else = { nesnd_india_union_accept = { MODE = merge } }
        }
        else_if = { limit = { NOT = { nesnd_india_incoming_kind = 6 } } scope:india_sender = { change_relations = { country = scope:india_partner value = 25 } trigger_event = { id = nesnd_india_diplomacy.2 popup = yes } nesnd_india_clear_request = yes } }
    }
}

# A prince that is another country's subject: its overlord is asked first (matter kind 6 of Indian
# affairs), and the invitation goes out once it consents (nesnd_india_union_handoff).
nesnd_india_invite_union = {
    if = { limit = { nesnd_india_can_invite_union = yes exists = $TARGET$ }
        if = { limit = { $TARGET$ = { nesnd_india_union_target = yes } }
            save_scope_as = india_union_inviter
            if = { limit = { $TARGET$ = { nesnd_india_consent_needed = { OTHER = scope:india_union_inviter GRANTS = yes } } }
                nesnd_india_matter_send = { KIND = 6 TO = $TARGET$ AMOUNT = nesnd_india_grant }
            }
            else = {
                clear_saved_scope = india_partner
                $TARGET$ = { save_scope_as = india_partner }
                nesnd_india_send_request = { KIND = 6 }
            }
        }
    }
}
# Empire scope. One more pension is owed; the ten years run again from today for all of them.
# The count starts over once the previous pensions have run out, and stops at nesnd_india_pension_cap:
# a prince taken in after that restarts the ten years without adding a pension.
nesnd_india_add_pension = {
    if = {
        limit = { OR = { NOT = { has_variable = nesnd_india_pension_count } NOT = { has_modifier = nesnd_india_princely_pensions } } }
        set_variable = { name = nesnd_india_pension_count value = 0 }
    }
    if = { limit = { nesnd_india_pension_count_value < nesnd_india_pension_cap } change_variable = { name = nesnd_india_pension_count add = 1 } }
    if = { limit = { has_modifier = nesnd_india_princely_pensions } remove_modifier = nesnd_india_princely_pensions }
    add_modifier = { name = nesnd_india_princely_pensions months = 120 multiplier = nesnd_india_pension_count_value }
}
# The one settlement for a prince joining Mughal. Empire scope, with scope:india_union_prince (the
# prince), scope:india_union_requester (whoever sent the invitation or the request) and the value
# scope:india_union_price (the quote). MODE is merge or dominion; the two ways share every step but
# the last. A prince that is another country's subject first leaves that overlord, which is then paid
# the quote as compensation.
nesnd_india_settle_union = {
    save_scope_as = india_union_empire
    save_scope_value_as = { name = india_union_mode value = nesnd_india_union_mode_$MODE$ }
    clear_saved_scope = india_union_former
    scope:india_union_prince = {
        if = { limit = { nesnd_india_consent_needed = { OTHER = scope:india_union_empire GRANTS = yes } }
            overlord = { save_scope_as = india_union_former }
            make_independent = yes
        }
    }
    add_treasury = { value = scope:india_union_price multiply = -1 }
    nesnd_india_add_pension = yes
    scope:india_union_requester = { nesnd_india_clear_request = yes }
    nesnd_india_union_remove_member = { WHO = scope:india_union_prince }
    nesnd_india_union_finish_$MODE$ = yes
    if = { limit = { exists = scope:india_union_former }
        add_treasury = { value = scope:india_union_price multiply = -1 }
        scope:india_union_former = { add_treasury = scope:india_union_price }
    }
    nesnd_india_refresh = yes
}
# The prince's state ceases to exist.
nesnd_india_union_finish_merge = {
    nesnd_india_inherit_partner = yes
    annex = scope:india_union_prince
}
# The prince keeps its states, government, laws, treasury and army, and becomes Mughal's dominion.
nesnd_india_union_finish_dominion = {
    scope:india_union_prince = { add_treasury = scope:india_union_price }
    save_scope_as = india_dominion_lord
    scope:india_union_prince = { nesnd_india_prince_become_dominion = yes }
}
# Prince scope, with scope:india_dominion_lord. The dominion pact, and the privileges the prince keeps
# (permanent). Also used when a risen country joins the empire that paid for it (nesnd_india_revolt.2).

    scope:india_dominion_lord = { create_diplomatic_pact = { country = prev type = dominion } }
    if = { limit = { NOT = { has_modifier = nesnd_india_prince_privileges } } add_modifier = { name = nesnd_india_prince_privileges } }
    set_variable = nesnd_india_prince_joined
}

nesnd_india_diplomacy_maintenance = {
    if = { limit = { NOT = { nesnd_india_mission_country = yes } }
        if = { limit = { NOT = { AND = { has_journal_entry = nesnd_india_indep nesnd_india_request_kind = 4 } } } nesnd_india_clear_request = yes }
        nesnd_india_remove_advisers = yes
    }

    if = { limit = { has_variable = nesnd_india_request_partner NOT = { has_variable = nesnd_india_request_clock } } nesnd_india_clear_request = yes }
    if = { limit = { has_variable = nesnd_india_incoming_union_sender NOT = { has_variable = nesnd_india_incoming_union_clock } } remove_variable = nesnd_india_incoming_union_sender }
    if = { limit = { has_variable = nesnd_india_union_mode NOR = { has_variable = nesnd_india_request_clock has_variable = nesnd_india_matter_clock } } remove_variable = nesnd_india_union_mode }
    if = { limit = { has_variable = nesnd_india_adviser_partner }
        if = { limit = { exists = var:nesnd_india_adviser_partner exists = var:nesnd_india_adviser_patron }
            if = { limit = { OR = {
                        NOT = { has_modifier = nesnd_india_french_advisers }
                        has_war_with = var:nesnd_india_adviser_partner
                        AND = { trigger_if = { limit = { is_subject = yes } NOT = { overlord = var:nesnd_india_adviser_patron } } trigger_else = { always = no } }
                        AND = { is_subject = no NOT = { this = var:nesnd_india_adviser_patron } }
                } } nesnd_india_remove_advisers = yes }
        } else = { nesnd_india_remove_advisers = yes }
    }
}

        }
        weight = {
            base = 1
            modifier = { factor = 6 c:YN1 ?= this scope:india_mission_requester = { is_direct_subject_of = c:USM } }

    }
    if = { limit = { exists = scope:india_ai_partner } nesnd_india_open_mission = { PARTNER = scope:india_ai_partner } }
}

# ==== Local projects (formerly nesnd_india_projects.txt) ====
# Local projects of the Indian princes. See common/scripted_triggers/nesnd_india_princes.txt.
#
# Order of a project paid for by another government:
#   the prince chooses project, state and funder (events .1 to .3)
#   -> nesnd_india_project_send        a matter of kind 1 (nesnd_india_matter_send): the overlord first
#                                      if it must agree, then the funder, both in the journal nesnd_india_foreign_affairs
#   -> the funder agrees: nesnd_india_project_accept
#   -> money moves, the building is ordered or created, the agreement is recorded on the state region.

# ---------------------------------------------------------------------------------------------
# The prince (requester scope)
# ---------------------------------------------------------------------------------------------
nesnd_india_project_clear = {
    remove_variable = nesnd_india_project_type remove_variable = nesnd_india_project_site
    remove_variable = nesnd_india_project_base remove_variable = nesnd_india_project_choice
    remove_variable = nesnd_india_project_build_clock
    nesnd_india_matter_clear = yes
}

# TYPE is 1 to 3, SITE the matching state trigger. Offer up to three states, largest first.
nesnd_india_project_pick = {
    nesnd_india_project_clear = yes
    set_variable = { name = nesnd_india_matter_kind value = 1 }
    set_variable = { name = nesnd_india_project_type value = $TYPE$ }
    set_variable = { name = nesnd_india_project_shown value = $TYPE$ }
    set_variable = { name = nesnd_india_project_choice days = 30 }
    clear_saved_scope = india_project_site_1
    clear_saved_scope = india_project_site_2
    clear_saved_scope = india_project_site_3
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 0 check_range_bounds = no save_scope_as = india_project_site_1 }
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 1 check_range_bounds = no save_scope_as = india_project_site_2 }
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 2 check_range_bounds = no save_scope_as = india_project_site_3 }
    if = { limit = { any_scope_state = { $SITE$ = yes count >= 2 } } trigger_event = { id = nesnd_india_project.2 popup = yes } }
    else_if = { limit = { exists = scope:india_project_site_1 } nesnd_india_project_set_site = { SITE = scope:india_project_site_1 } }
    else = { nesnd_india_project_clear = yes }
}

                    }
                }
            }
        }
        trigger_event = { id = nesnd_india_project.4 popup = yes }

        if = { limit = { nesnd_india_project_type_value = 1 } nesnd_india_project_build = yes }

# Answering scope in the reply event, request kinds 6 and 8.
# When a prince declines an empire's invitation (kind 6), the empire gains a claim on the prince's
# capital state (and on all remaining Indian states on a second refusal within ten years), an AI empire
# adopts a conquer secret goal, and any fellow Princely League members lose 10 relations with the empire.
nesnd_india_union_decline = {
    save_scope_as = india_union_decliner
    clear_saved_scope = india_union_claim_level
    clear_saved_scope = india_union_pact_backlash
    if = {
        limit = { scope:india_sender = { nesnd_india_empire = yes } }
        if = {
            limit = { var:nesnd_india_union_refused_empire ?= scope:india_sender }
            save_scope_value_as = { name = india_union_claim_level value = 2 }
            every_scope_state = {
                limit = { nesnd_india_state = yes NOT = { has_claim_by = scope:india_sender } }
                state_region = { add_claim = scope:india_sender }
            }
            set_variable = { name = nesnd_india_union_refused_empire value = scope:india_sender days = 3650 }
        }
        else = {
            save_scope_value_as = { name = india_union_claim_level value = 1 }
            capital ?= {
                if = {
                    limit = { NOT = { has_claim_by = scope:india_sender } }
                    state_region = { add_claim = scope:india_sender }
                }
            }
            set_variable = { name = nesnd_india_union_refused_empire value = scope:india_sender days = 3650 }
        }
        scope:india_sender = {
            if = {
                limit = { is_ai = yes }
                set_secret_goal = { country = scope:india_union_decliner secret_goal = conquer }
            }
        }
        if = {
            limit = { nesnd_india_union_is_member = yes }
            save_scope_value_as = { name = india_union_pact_backlash value = 1 }
            every_in_global_list = {
                variable = nesnd_india_union_members
                limit = {
                    is_country_alive = yes
                    is_subject = no
                    NOT = { this = scope:india_union_decliner }
                    NOT = { this = scope:india_sender }
                }
                change_relations = { country = scope:india_sender value = -10 }
            }
        }
    }
    trigger_event = { id = nesnd_india_union.2 popup = yes }
    scope:india_sender = {
        trigger_event = { id = nesnd_india_union.2 popup = yes }
        nesnd_india_clear_request = yes
    }
}

nesnd_india_case_maintain = {
    if = {
        limit = { has_variable = nesnd_india_case_stage }
        if = {
            limit = { nesnd_india_case_stage_value <= 3 }
            if = {
                limit = {
                    exists = var:nesnd_india_case_sponsor
                    exists = var:nesnd_india_case_owner
                    exists = var:nesnd_india_case_state

                            }
                            var:nesnd_india_case_owner = { nesnd_india_colonial_owner = yes }
                            var:nesnd_india_case_sponsor = { nesnd_india_empire = yes }
                        }
                        # The sponsoring JE advances preparation; this global pulse only checks ownership and lifecycle.
                    }
                    else = { nesnd_india_close_case = yes }
                }
            }
            else = { nesnd_india_close_case = yes }
        }
        else_if = {
            limit = { nesnd_india_case_stage_value = 4 }
            change_variable = { name = nesnd_india_case_retries add = 1 }
            if = {
                limit = { nesnd_india_case_retries_value <= 2 }
                nesnd_india_case_create_play = { R = $R$ STATE = $STATE$ TAG = $TAG$ }
            }
            else = {
                if = {
                    limit = {
                        exists = var:nesnd_india_case_owner
                        exists = c:$TAG$
                        exists = s:$STATE$.region_state:$TAG$
                    }
                    if = {
                        limit = { has_variable = nesnd_india_case_original_provinces }
                        if = {
                            limit = {
                                has_variable = nesnd_india_case_original_provinces
                                s:$STATE$.region_state:$TAG$ = {
                                    num_provinces = prev.var:nesnd_india_case_original_provinces
                                }
                                c:$TAG$ = {
                                    num_states = 1
                                    has_variable = nesnd_india_revolt_actor
                                    NOT = { has_variable = nesnd_india_revolt_territory_changed }
                                    is_at_war = no
                                    is_active_in_diplomatic_play = no
                                }
                            }
                            var:nesnd_india_case_owner = { save_scope_as = india_case_original_owner }
                            if = {
                                limit = { exists = scope:india_case_original_owner }
                                s:$STATE$.region_state:$TAG$ = {
                                    set_state_owner = scope:india_case_original_owner
                                }
                            }
                        }
                    }
                }
                c:$TAG$ ?= {
                    set_variable = nesnd_india_revolt_failed
                    set_variable = nesnd_india_revolt_resolved
                }
                nesnd_india_case_failed = yes
                nesnd_india_close_case = yes
            }
        }
        else = {
            if = {
                limit = { exists = var:nesnd_india_case_play }
                if = {
                    limit = {
                        var:nesnd_india_case_play = { is_war = yes }
                    }
                    set_variable = { name = nesnd_india_case_stage value = 6 }
                }
            }
            else = {
                # The play is over: the rising held if its country is independent and holds a state here.
                if = { limit = { c:$TAG$ ?= { is_subject = no any_scope_state = { state_region = s:$STATE$ } } } nesnd_india_case_won = yes }
                else = { nesnd_india_case_failed = yes }
                c:$TAG$ ?= {
                    set_variable = nesnd_india_revolt_negotiation_ended
                    set_variable = nesnd_india_revolt_resolved
                }
                nesnd_india_close_case = yes
            }
        }
    }
}

# Invited prince scope (from nesnd_india_union.5 option b). Relations -10 and notify inviter.
nesnd_india_union_decline_invite = {
    save_scope_as = india_league_target
    if = {
        limit = { exists = scope:india_league_inviter scope:india_league_inviter = { is_country_alive = yes } }
        change_relations = { country = scope:india_league_inviter value = -10 }
        save_scope_value_as = { name = india_league_reply value = 2 }

# Any scope. Removes WHO from nesnd_india_union_members and records it as the last departed member.
nesnd_india_union_remove_member = {
    if = {
        limit = {
            has_global_variable_list = nesnd_india_union_members
            is_target_in_global_variable_list = { name = nesnd_india_union_members target = $WHO$ }
        }
        $WHO$ = {
            if = {
                limit = { is_country_alive = yes }
                set_global_variable = { name = nesnd_india_union_last_departed value = this days = 3650 }
            }
        }
        remove_list_global_variable = { name = nesnd_india_union_members target = $WHO$ }
    }
}

# Remaining lone member scope. Applies the breakup consequences: -30 relations with the departed
# member if alive, 10-year nesnd_india_union_broken modifier, 10-year wait before reopening the
# journal, clears league state, and fires the breakup event nesnd_india_union.8.
nesnd_india_union_break_apply = {
    save_scope_as = india_league_breaker
    clear_saved_scope = india_league_departed
    if = {
        limit = { exists = global_var:nesnd_india_union_last_departed }
        global_var:nesnd_india_union_last_departed = {
            if = {
                limit = { is_country_alive = yes NOT = { this = scope:india_league_breaker } }
                save_scope_as = india_league_departed
            }
        }
    }
    if = {
        limit = { exists = scope:india_league_departed }
        change_relations = { country = scope:india_league_departed value = -30 }
    }
    add_modifier = { name = nesnd_india_union_broken days = long_modifier_time }
    set_variable = { name = nesnd_india_union_reopen_wait days = 3650 }
    remove_variable = nesnd_india_union_formed
    if = {
        limit = { nesnd_india_union_is_member = yes }
        remove_list_global_variable = { name = nesnd_india_union_members target = scope:india_league_breaker }
    }
    remove_global_variable = nesnd_india_union_had_two
    trigger_event = { id = nesnd_india_union.8 popup = yes }
}

# Global scope, monthly from on_monthly_pulse (nesnd_india_union_global_pulse).
# Prunes annexed, dead or subjugated members from nesnd_india_union_members in two passes via
# temporary global lists so nesnd_india_union_members is never mutated while being iterated.
# Existing treaties follow vanilla rules without script cancellation. If a previously 2+ league
# drops to 0 members, resets the flag; if 1 lone survivor remains whose journal already ended
# after pact formation, collects it and applies breakup outside the members loop.
nesnd_india_union_maintain = {
    if = {
        limit = { has_global_variable_list = nesnd_india_union_members }
        if = {
            limit = { has_global_variable_list = nesnd_india_union_leaving }
            clear_global_variable_list = nesnd_india_union_leaving
        }
        every_in_global_list = {
            variable = nesnd_india_union_members
            limit = { OR = { is_country_alive = no is_subject = yes NOT = { nesnd_india_prince = yes } } }
            add_to_global_variable_list = { name = nesnd_india_union_leaving target = this }
        }
        if = {
            limit = { has_global_variable_list = nesnd_india_union_leaving }
            every_in_global_list = {
                variable = nesnd_india_union_leaving
                if = {
                    limit = { is_country_alive = yes }
                    set_global_variable = { name = nesnd_india_union_last_departed value = this days = 3650 }
                }
                remove_list_global_variable = { name = nesnd_india_union_members target = this }
            }
            clear_global_variable_list = nesnd_india_union_leaving
        }
        if = {
            limit = { has_global_variable = nesnd_india_union_had_two }
            if = {
                limit = { NOT = { any_in_global_list = { variable = nesnd_india_union_members is_country_alive = yes is_subject = no nesnd_india_prince = yes } } }
                remove_global_variable = nesnd_india_union_had_two
            }
            else_if = {
                limit = { NOT = { any_in_global_list = { variable = nesnd_india_union_members count >= 2 is_country_alive = yes is_subject = no nesnd_india_prince = yes } } }
                if = {
                    limit = { has_global_variable_list = nesnd_india_union_breaking }
                    clear_global_variable_list = nesnd_india_union_breaking
                }
                every_in_global_list = {
                    variable = nesnd_india_union_members
                    limit = { is_country_alive = yes is_subject = no nesnd_india_prince = yes NOT = { has_journal_entry = nesnd_india_union } }
                    add_to_global_variable_list = { name = nesnd_india_union_breaking target = this }
                }
                if = {
                    limit = { has_global_variable_list = nesnd_india_union_breaking }
                    every_in_global_list = {
                        variable = nesnd_india_union_breaking
                        nesnd_india_union_break_apply = yes
                    }
                    clear_global_variable_list = nesnd_india_union_breaking
                }
            }
        }
    }
}
