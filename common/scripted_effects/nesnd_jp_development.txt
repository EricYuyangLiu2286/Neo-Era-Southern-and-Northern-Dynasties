nesnd_jp_initialize = {
    if = {
        limit = { OR = { has_variable = heir_heikatsu_spawned any_character = { nesnd_jp_is_heikatsu = yes } } }
        set_global_variable = nesnd_jp_heikatsu_ever_created
    }

    if = { limit = { nesnd_jp_country = yes }
        set_variable = nesnd_jp_managed
        if = { limit = { NOT = { has_variable = nesnd_jp_statistics_ready } NOT = { has_variable = nesnd_jp_statistics_pending } }
            set_variable = { name = nesnd_jp_statistics_pending days = 3 }
            trigger_event = { id = nesnd_jp_court.3 days = 2 }
        }
        if = { limit = { NOT = { has_variable = nesnd_jp_schema } }
            set_variable = { name = nesnd_jp_schema value = 1 }
            if = { limit = { c:JAP ?= this any_scope_state = { has_variable = nesnd_jp_political_receipt_JAP } } set_variable = nesnd_jp_political_done }
            if = { limit = { c:JTK ?= this any_scope_state = { has_variable = nesnd_jp_political_receipt_JTK } } set_variable = nesnd_jp_political_done }
            if = { limit = { c:JIM ?= this any_scope_state = { has_variable = nesnd_jp_political_receipt_JIM } } set_variable = nesnd_jp_political_done }
            if = { limit = { NOT = { has_variable = nesnd_jp_cohesion } } set_variable = { name = nesnd_jp_cohesion value = 70 } }
            set_variable = { name = nesnd_jp_war_months value = 0 }
            set_variable = { name = nesnd_jp_sol_previous value = average_sol }
            set_variable = { name = nesnd_jp_sol_change value = 0 }
        }
        if = { limit = { c:JIM ?= this }
            add_primary_culture = cu:japanese
            add_primary_culture = cu:ce_wuu
            set_variable = nesnd_jp_jiangnan_policy
        }
    }
}
nesnd_jp_sync_cohesion = {
    save_scope_as = jp_bar_owner
    if = { limit = { has_journal_entry = nesnd_jp_cohesion exists = scope:jp_bar_owner }
        je:nesnd_jp_cohesion = { set_bar_progress = { name = nesnd_jp_cohesion_bar value = scope:jp_bar_owner.nesnd_jp_cohesion_value } }
    }
}
nesnd_jp_cohesion_monthly = {
    if = { limit = { nesnd_jp_cohesion_live = yes has_variable = nesnd_jp_statistics_ready }
        nesnd_jp_initialize = yes
        set_variable = { name = nesnd_jp_sol_change value = { value = average_sol subtract = var:nesnd_jp_sol_previous } }
        set_variable = { name = nesnd_jp_sol_previous value = average_sol }
        if = { limit = { is_at_war = yes } change_variable = { name = nesnd_jp_war_months add = 1 } }
        else = { set_variable = { name = nesnd_jp_war_months value = 0 } }
        change_variable = { name = nesnd_jp_cohesion add = nesnd_jp_cohesion_delta }
        clamp_variable = { name = nesnd_jp_cohesion min = 0 max = 100 }
        if = { limit = { var:nesnd_jp_cohesion < 25 }
            if = { limit = { NOT = { has_modifier = nesnd_jp_divided_councils } } add_modifier = { name = nesnd_jp_divided_councils } }
        } else = { if = { limit = { has_modifier = nesnd_jp_divided_councils } remove_modifier = nesnd_jp_divided_councils } }
        nesnd_jp_settle_culture_plan = yes
        nesnd_jp_reset_culture_plan = yes
        nesnd_jp_prepare_culture_plan = yes
        nesnd_jp_sync_cohesion = yes
    } else = { nesnd_jp_reset_culture_plan = yes }
}

# Mode 0 is also a completed draw. Loading a save or refreshing the other
# Japanese journals must not replace a quiet month with another draw.
nesnd_jp_reset_culture_plan = {
    if = { limit = { has_variable = nesnd_jp_next_mode } remove_variable = nesnd_jp_next_mode }
    if = { limit = { has_variable = nesnd_jp_next_realm } remove_variable = nesnd_jp_next_realm }
    if = { limit = { has_variable = nesnd_jp_next_state } remove_variable = nesnd_jp_next_state }
    if = { limit = { has_variable = nesnd_jp_next_target } remove_variable = nesnd_jp_next_target }
    if = { limit = { has_variable = nesnd_jp_next_pop } remove_variable = nesnd_jp_next_pop }
    if = { limit = { has_variable = nesnd_jp_next_source_culture } remove_variable = nesnd_jp_next_source_culture }
}

nesnd_jp_prepare_culture_plan = {
    if = {
        limit = {
            nesnd_jp_cohesion_live = yes
            has_variable = nesnd_jp_statistics_ready
            NOT = { has_variable = nesnd_jp_next_mode }
        }
        set_variable = { name = nesnd_jp_next_mode value = 0 }
        set_variable = { name = nesnd_jp_next_realm value = country_definition }
        save_scope_as = nesnd_jp_plan_owner
        if = { limit = { nesnd_jp_cohesion_value < 25 }
            if = { limit = { nesnd_jp_drift_toward_japanese = yes }
                nesnd_jp_select_drift = { TARGET = cu:japanese }
            } else = {
                random_list = {
                    50 = { nesnd_jp_select_drift = { TARGET = cu:han } }
                    50 = { nesnd_jp_select_drift = { TARGET = cu:ce_wuu } }
                }
            }
        } else_if = { limit = { nesnd_jp_cohesion_value > 75 }
            if = { limit = { nesnd_jp_jiangnan_policy = yes }
                nesnd_jp_select_exchange = { TARGET = cu:ce_wuu }
            } else = { nesnd_jp_select_exchange = { TARGET = cu:japanese } }
        }
        clear_saved_scope = nesnd_jp_plan_owner
    }
}

nesnd_jp_select_drift = {
    random_scope_state = {
        limit = { any_scope_pop = {
            culture = { is_primary_culture_of = scope:nesnd_jp_plan_owner }
            NOT = { culture = $TARGET$ }
        } }
        scope:nesnd_jp_plan_owner = { set_variable = { name = nesnd_jp_next_state value = prev } }
        random_scope_pop = {
            limit = {
                culture = { is_primary_culture_of = scope:nesnd_jp_plan_owner }
                NOT = { culture = $TARGET$ }
            }
            scope:nesnd_jp_plan_owner = {
                set_variable = { name = nesnd_jp_next_mode value = 1 }
                set_variable = { name = nesnd_jp_next_pop value = prev }
                set_variable = { name = nesnd_jp_next_source_culture value = prev.culture }
                set_variable = { name = nesnd_jp_next_target value = $TARGET$ }
            }
        }
    }
}

nesnd_jp_select_exchange = {
    if = { limit = { any_scope_state = { state_population > 0 any_scope_pop = { NOT = { culture = $TARGET$ } } } }
        random_list = {
            70 = { }
            30 = { random_scope_state = {
                limit = { state_population > 0 any_scope_pop = { NOT = { culture = $TARGET$ } } }
                scope:nesnd_jp_plan_owner = {
                    set_variable = { name = nesnd_jp_next_state value = prev }
                    set_variable = { name = nesnd_jp_next_mode value = 2 }
                    set_variable = { name = nesnd_jp_next_target value = $TARGET$ }
                }
            } }
        }
    }
}

nesnd_jp_settle_culture_plan = {
    if = { limit = { nesnd_jp_culture_plan_valid = yes }
        var:nesnd_jp_next_target = { save_scope_as = nesnd_jp_settle_target }
        if = { limit = { var:nesnd_jp_next_mode = 1 }
            var:nesnd_jp_next_pop = { change_pop_culture = { target = scope:nesnd_jp_settle_target value = 0.25 } }
        } else = {
            var:nesnd_jp_next_state = {
                if = { limit = { scope:nesnd_jp_settle_target = cu:ce_wuu }
                    set_variable = { name = nesnd_jp_conversion_fraction value = nesnd_jp_conversion_fraction_jiangnan }
                } else = {
                    set_variable = { name = nesnd_jp_conversion_fraction value = nesnd_jp_conversion_fraction_japanese }
                }
                every_scope_pop = {
                    limit = { NOT = { culture = scope:nesnd_jp_settle_target } }
                    change_pop_culture = { target = scope:nesnd_jp_settle_target value = state.var:nesnd_jp_conversion_fraction }
                }
                remove_variable = nesnd_jp_conversion_fraction
            }
        }
        clear_saved_scope = nesnd_jp_settle_target
    }
}
nesnd_jp_bootstrap = {
    if = { limit = { nesnd_jp_country = yes }
        nesnd_jp_initialize = yes
        if = { limit = { NOT = { has_journal_entry = nesnd_jp_cohesion } } add_journal_entry = { type = nesnd_jp_cohesion } }
        nesnd_jp_prepare_culture_plan = yes
        nesnd_jp_ensure_establishment = yes

        nesnd_jp_ensure_industry = yes
        nesnd_jp_ensure_nation = yes
        nesnd_jp_ensure_politics = yes
        nesnd_jp_ensure_foreign = yes
        nesnd_jp_ensure_korea = yes
        nesnd_jp_sync_cohesion = yes
    }
}
