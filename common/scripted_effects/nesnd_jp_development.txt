nesnd_jp_initialize = {
    if = {
        limit = { OR = { hidden_trigger = { has_variable = heir_heikatsu_spawned } any_character = { nesnd_jp_is_heikatsu = yes } } }
        set_global_variable = nesnd_jp_heikatsu_ever_created
    }

    if = { limit = { nesnd_jp_country = yes }
        set_variable = nesnd_jp_managed
        if = { limit = { NOT = { has_variable = nesnd_jp_statistics_ready } hidden_trigger = { NOT = { has_variable = nesnd_jp_statistics_pending } } }
            set_variable = { name = nesnd_jp_statistics_pending days = 3 }
            trigger_event = { id = nesnd_jp_court.3 days = 2 }
        }
        if = { limit = { hidden_trigger = { NOT = { has_variable = nesnd_jp_schema } } }
            set_variable = { name = nesnd_jp_schema value = 1 }
            if = { limit = { c:JAP ?= this any_scope_state = { hidden_trigger = { has_variable = nesnd_jp_political_receipt_JAP } } } set_variable = nesnd_jp_political_done }
            if = { limit = { c:JTK ?= this any_scope_state = { hidden_trigger = { has_variable = nesnd_jp_political_receipt_JTK } } } set_variable = nesnd_jp_political_done }
            if = { limit = { c:JIM ?= this any_scope_state = { hidden_trigger = { has_variable = nesnd_jp_political_receipt_JIM } } } set_variable = nesnd_jp_political_done }
            set_variable = { name = nesnd_jp_war_months value = 0 }
            set_variable = { name = nesnd_jp_sol_previous value = average_sol }
        }
        if = { limit = { c:JIM ?= this }
            add_primary_culture = cu:japanese
            add_primary_culture = cu:ce_wuu
            set_variable = nesnd_jp_jiangnan_policy
        }
    }
}
# 国民团结高于 50 时各融合度人口的激进化减缓，低于 50 时加快。档位变化时才重挂修正。
nesnd_jp_clear_unity_mood = {
    if = { limit = { has_modifier = nesnd_jp_unity_concord } remove_modifier = nesnd_jp_unity_concord }
    if = { limit = { has_modifier = nesnd_jp_unity_discord } remove_modifier = nesnd_jp_unity_discord }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_mood_step } } remove_variable = nesnd_jp_mood_step }
}
nesnd_jp_update_unity_mood = {
    if = {
        limit = { OR = { NOT = { has_variable = nesnd_jp_mood_step } NOT = { var:nesnd_jp_mood_step = nesnd_jp_mood_step_value } } }
        nesnd_jp_clear_unity_mood = yes
        set_variable = { name = nesnd_jp_mood_step value = nesnd_jp_mood_step_value }
        if = { limit = { var:nesnd_jp_mood_step > 0 } add_modifier = { name = nesnd_jp_unity_concord multiplier = { value = var:nesnd_jp_mood_step divide = 10 } } }
        else_if = { limit = { var:nesnd_jp_mood_step < 0 } add_modifier = { name = nesnd_jp_unity_discord multiplier = { value = var:nesnd_jp_mood_step divide = -10 } } }
    }
}
nesnd_jp_cohesion_monthly = {
    if = { limit = { nesnd_jp_cohesion_live = yes has_variable = nesnd_jp_statistics_ready }
        nesnd_jp_initialize = yes
        # 本月的团结增减已由进度条 nesnd_jp_cohesion_bar 先行计入；这里记下本月的生活水平与战月，供下月进度条使用。
        set_variable = { name = nesnd_jp_sol_previous value = average_sol }
        if = { limit = { is_at_war = yes } change_variable = { name = nesnd_jp_war_months add = 1 } }
        else = { set_variable = { name = nesnd_jp_war_months value = 0 } }
        nesnd_jp_update_unity_mood = yes
        if = { limit = { nesnd_jp_cohesion_value < 25 }
            if = { limit = { NOT = { has_modifier = nesnd_jp_divided_councils } } add_modifier = { name = nesnd_jp_divided_councils } }
        } else = { if = { limit = { has_modifier = nesnd_jp_divided_councils } remove_modifier = nesnd_jp_divided_councils } }
        nesnd_jp_settle_culture_plan = yes
        nesnd_jp_reset_culture_plan = yes
        nesnd_jp_prepare_culture_plan = yes
    } else = { nesnd_jp_reset_culture_plan = yes }
}

# Mode 0 is also a completed draw. Loading a save or refreshing the other
# Japanese journals must not replace a quiet month with another draw.
nesnd_jp_reset_culture_plan = {
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_mode } } remove_variable = nesnd_jp_next_mode }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_realm } } remove_variable = nesnd_jp_next_realm }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_state } } remove_variable = nesnd_jp_next_state }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_target } } remove_variable = nesnd_jp_next_target }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_pop } } remove_variable = nesnd_jp_next_pop }
    if = { limit = { hidden_trigger = { has_variable = nesnd_jp_next_source_culture } } remove_variable = nesnd_jp_next_source_culture }
}

nesnd_jp_prepare_culture_plan = {
    if = {
        limit = {
            nesnd_jp_cohesion_live = yes
            has_variable = nesnd_jp_statistics_ready
            hidden_trigger = {
                NOT = { has_variable = nesnd_jp_next_mode }
            }
        }
        set_variable = {
            name = nesnd_jp_next_mode
            value = 0
        }
        set_variable = {
            name = nesnd_jp_next_realm
            value = country_definition
        }
        save_scope_as = nesnd_jp_plan_owner
        if = {
            limit = {
                nesnd_jp_cohesion_value < 25
            }
            if = {
                limit = {
                    nesnd_jp_drift_toward_japanese = yes
                }
                nesnd_jp_select_drift = {
                    SOURCE_CHECK = nesnd_han_heritage_culture
                    TARGET_HERITAGE = japanese
                }
            }
            else = {
                nesnd_jp_select_drift = {
                    SOURCE_CHECK = nesnd_jp_heritage_culture
                    TARGET_HERITAGE = han
                }
            }
        }
        else_if = {
            limit = {
                nesnd_jp_cohesion_value > 75
            }
            if = {
                limit = {
                    nesnd_jp_jiangnan_policy = yes
                }
                nesnd_jp_select_exchange = {
                    TARGET_CHECK = nesnd_han_heritage_culture
                    TARGET_HERITAGE = han
                }
            }
            else = {
                nesnd_jp_select_exchange = {
                    TARGET_CHECK = nesnd_jp_heritage_culture
                    TARGET_HERITAGE = japanese
                }
            }
        }
        clear_saved_scope = nesnd_jp_plan_owner
    }
}

# State scope, with scope:nesnd_jp_plan_owner saved.
nesnd_jp_pick_target = {
    if = {
        limit = {
            scope:nesnd_jp_plan_owner = {
                any_primary_culture = {
                    has_discrimination_trait = heritage_$HERITAGE$
                }
            }
        }
        scope:nesnd_jp_plan_owner = {
            random_primary_culture = {
                limit = {
                    has_discrimination_trait = heritage_$HERITAGE$
                }
                scope:nesnd_jp_plan_owner = {
                    set_variable = {
                        name = nesnd_jp_next_target
                        value = prev
                    }
                }
            }
        }
    }
    else_if = {
        limit = {
            any_scope_culture = {
                has_discrimination_trait = heritage_$HERITAGE$
            }
        }
        random_scope_culture = {
            limit = {
                has_discrimination_trait = heritage_$HERITAGE$
            }
            scope:nesnd_jp_plan_owner = {
                set_variable = {
                    name = nesnd_jp_next_target
                    value = prev
                }
            }
        }
    }
    else = {
        if = {
            limit = {
                cu:japanese = {
                    has_discrimination_trait = heritage_$HERITAGE$
                }
            }
            scope:nesnd_jp_plan_owner = {
                set_variable = {
                    name = nesnd_jp_next_target
                    value = cu:japanese
                }
            }
        }
        else = {
            random_list = {
                50 = {
                    scope:nesnd_jp_plan_owner = {
                        set_variable = {
                            name = nesnd_jp_next_target
                            value = cu:han
                        }
                    }
                }
                50 = {
                    scope:nesnd_jp_plan_owner = {
                        set_variable = {
                            name = nesnd_jp_next_target
                            value = cu:ce_wuu
                        }
                    }
                }
            }
        }
    }
}

nesnd_jp_select_drift = {
    random_scope_state = {
        limit = {
            state_population > 0
            any_scope_pop = {
                culture = {
                    is_primary_culture_of = scope:nesnd_jp_plan_owner
                    $SOURCE_CHECK$ = yes
                }
            }
        }
        scope:nesnd_jp_plan_owner = {
            set_variable = {
                name = nesnd_jp_next_state
                value = prev
            }
        }
        random_scope_pop = {
            limit = {
                culture = {
                    is_primary_culture_of = scope:nesnd_jp_plan_owner
                    $SOURCE_CHECK$ = yes
                }
            }
            scope:nesnd_jp_plan_owner = {
                set_variable = {
                    name = nesnd_jp_next_mode
                    value = 1
                }
                set_variable = {
                    name = nesnd_jp_next_pop
                    value = prev
                }
                set_variable = {
                    name = nesnd_jp_next_source_culture
                    value = prev.culture
                }
            }
        }
        nesnd_jp_pick_target = {
            HERITAGE = $TARGET_HERITAGE$
        }
    }
}

nesnd_jp_select_exchange = {
    if = {
        limit = {
            any_scope_state = {
                state_population > 0
                any_scope_pop = {
                    NOT = {
                        culture = {
                            $TARGET_CHECK$ = yes
                        }
                    }
                }
            }
        }
        random_list = {
            70 = { }
            30 = {
                random_scope_state = {
                    limit = {
                        state_population > 0
                        any_scope_pop = {
                            NOT = {
                                culture = {
                                    $TARGET_CHECK$ = yes
                                }
                            }
                        }
                    }
                    scope:nesnd_jp_plan_owner = {
                        set_variable = {
                            name = nesnd_jp_next_state
                            value = prev
                        }
                        set_variable = {
                            name = nesnd_jp_next_mode
                            value = 2
                        }
                    }
                    nesnd_jp_pick_target = {
                        HERITAGE = $TARGET_HERITAGE$
                    }
                }
            }
        }
    }
}

# Executes only the plan saved last month; the journal preview reads the same variables.
# Culture filters use heritage triggers; settlement passes the saved target scope to change_pop_culture.
nesnd_jp_settle_culture_plan = {
    if = {
        limit = {
            nesnd_jp_culture_plan_valid = yes
        }
        var:nesnd_jp_next_target = {
            save_scope_as = nesnd_jp_target_culture
        }
        if = {
            limit = {
                var:nesnd_jp_next_target = {
                    nesnd_jp_heritage_culture = yes
                }
            }
            nesnd_jp_settle_to = {
                TARGET_CHECK = nesnd_jp_heritage_culture
                FRACTION = nesnd_jp_conversion_fraction_jp_heritage
            }
        }
        else_if = {
            limit = {
                var:nesnd_jp_next_target = {
                    nesnd_han_heritage_culture = yes
                }
            }
            nesnd_jp_settle_to = {
                TARGET_CHECK = nesnd_han_heritage_culture
                FRACTION = nesnd_jp_conversion_fraction_han_heritage
            }
        }
        clear_saved_scope = nesnd_jp_target_culture
    }
}

nesnd_jp_settle_to = {
    if = {
        limit = {
            var:nesnd_jp_next_mode = 1
        }
        var:nesnd_jp_next_pop = {
            change_pop_culture = {
                target = scope:nesnd_jp_target_culture
                value = 0.25
            }
        }
    }
    else = {
        var:nesnd_jp_next_state = {
            set_variable = {
                name = nesnd_jp_conversion_fraction
                value = $FRACTION$
            }
            every_scope_pop = {
                limit = {
                    NOT = {
                        culture = {
                            $TARGET_CHECK$ = yes
                        }
                    }
                }
                change_pop_culture = {
                    target = scope:nesnd_jp_target_culture
                    value = state.var:nesnd_jp_conversion_fraction
                }
            }
            remove_variable = nesnd_jp_conversion_fraction
        }
    }
}

        nesnd_jp_initialize = yes
        if = { limit = { NOT = { has_journal_entry = nesnd_jp_cohesion } nesnd_cohesion_open = yes } add_journal_entry = { type = nesnd_jp_cohesion } }
        nesnd_jp_prepare_culture_plan = yes
        nesnd_jp_ensure_establishment = yes

        nesnd_jp_ensure_industry = yes
        nesnd_jp_ensure_nation = yes
        nesnd_jp_ensure_politics = yes
        nesnd_jp_ensure_foreign = yes
    }
}
