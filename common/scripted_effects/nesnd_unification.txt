# Global refresh of the unification trend (root = none; only called from the global journal entry, so scope:journal_entry exists): joiners, involvement, heads, counts, leader, phase, modifiers and the lead bar.
nesnd_uni_refresh = {
	# 1. AI countries owning at least 3 regions of the scope join on their own.
	if = {
		limit = {
			NOT = { has_global_variable = nesnd_uni_declared }
		}
		every_country = {
			limit = {
				is_ai = yes
				is_subject = no
				nesnd_uni_bloc_follower = no
				NOT = { is_country_type = decentralized }
				NOT = { has_variable = nesnd_uni_joined }
				nesnd_uni_capital_in_scope = no
				nesnd_uni_own_count >= 3
			}
			set_variable = nesnd_uni_joined
		}
	}
	# 2. Joiners that no longer own 3 regions and have no capital in the scope drop out.
	every_country = {
		limit = {
			has_variable = nesnd_uni_joined
			nesnd_uni_capital_in_scope = no
			nesnd_uni_own_count < 3
		}
		remove_variable = nesnd_uni_joined
	}
	# 3. Involvement follows the qualified trigger.
	every_country = {
		limit = {
			nesnd_uni_qualified = yes
			NOT = { is_involved_in_journal_entry = nesnd_uni_trend }
		}
		scope:journal_entry ?= {
			add_involved_country = prev
		}
	}
	scope:journal_entry ?= {
		every_scope_je_involved = {
			limit = {
				nesnd_uni_qualified = no
			}
			scope:journal_entry ?= {
				remove_involved_country = prev
			}
		}
	}
	# 4. Heads and roles: every contender heads its own subjects, and (as bloc leader) its power bloc members and their subjects.
	every_country = {
		limit = {
			OR = {
				has_variable = nesnd_uni_head
				has_variable = nesnd_uni_held
				has_variable = nesnd_uni_map_lead
				has_variable = nesnd_uni_map_second
				has_variable = nesnd_uni_map_other
				has_variable = nesnd_uni_map_subject
				has_variable = nesnd_uni_map_ally
			}
		}
		remove_variable = nesnd_uni_head
		remove_variable = nesnd_uni_held
		remove_variable = nesnd_uni_map_lead
		remove_variable = nesnd_uni_map_second
		remove_variable = nesnd_uni_map_other
		remove_variable = nesnd_uni_map_subject
		remove_variable = nesnd_uni_map_ally
	}
	every_country = {
		limit = {
			nesnd_uni_contender = yes
		}
		save_scope_as = nesnd_uni_head_scope
		set_variable = {
			name = nesnd_uni_head
			value = scope:nesnd_uni_head_scope
		}
		set_variable = {
			name = nesnd_uni_held
			value = 0
		}
		every_subject_or_below = {
			set_variable = {
				name = nesnd_uni_head
				value = scope:nesnd_uni_head_scope
			}
			set_variable = nesnd_uni_map_subject
		}
		if = {
			limit = {
				is_power_bloc_leader = yes
			}
			power_bloc ?= {
				every_power_bloc_member = {
					limit = {
						NOT = { this = scope:nesnd_uni_head_scope }
					}
					set_variable = {
						name = nesnd_uni_head
						value = scope:nesnd_uni_head_scope
					}
					set_variable = nesnd_uni_map_ally
					every_subject_or_below = {
						set_variable = {
							name = nesnd_uni_head
							value = scope:nesnd_uni_head_scope
						}
						set_variable = nesnd_uni_map_ally
					}
				}
			}
		}
	}
	# 5. Regions: count the regions whose every state belongs to one head's tree and mark the states.
	nesnd_uni_count_regions = yes
	# 6. Number of rivals, the leader and second place.
	set_global_variable = {
		name = nesnd_uni_rivals
		value = 0
	}
	every_country = {
		limit = {
			nesnd_uni_contender = yes
		}
		change_global_variable = {
			name = nesnd_uni_rivals
			add = 1
		}
	}
	ordered_country = {
		limit = {
			nesnd_uni_contender = yes
		}
		order_by = nesnd_uni_held_value
		position = 0
		save_scope_as = nesnd_uni_lead
	}
	if = {
		limit = {
			exists = scope:nesnd_uni_lead
			scope:nesnd_uni_lead.var:nesnd_uni_held > 0
		}
		set_global_variable = {
			name = nesnd_uni_leader
			value = scope:nesnd_uni_lead
		}
		set_global_variable = {
			name = nesnd_uni_lead_count
			value = scope:nesnd_uni_lead.var:nesnd_uni_held
		}
	}
	else = {
		remove_global_variable = nesnd_uni_leader
		remove_global_variable = nesnd_uni_lead_count
	}
	# Second place, only with at least two rivals (an ordered list at position 1 needs two entries).
	if = {
		limit = {
			global_var:nesnd_uni_rivals >= 2
		}
		ordered_country = {
			limit = {
				nesnd_uni_contender = yes
			}
			order_by = nesnd_uni_held_value
			position = 1
			save_scope_as = nesnd_uni_second_scope
		}
		if = {
			limit = {
				exists = scope:nesnd_uni_second_scope
				scope:nesnd_uni_second_scope.var:nesnd_uni_held > 0
			}
			set_global_variable = {
				name = nesnd_uni_second
				value = scope:nesnd_uni_second_scope
			}
		}
		else = {
			remove_global_variable = nesnd_uni_second
		}
	}
	else = {
		remove_global_variable = nesnd_uni_second
	}
	# 7. Count the contenders holding at least 32 regions.
	set_global_variable = {
		name = nesnd_uni_big
		value = 0
	}
	every_country = {
		limit = {
			nesnd_uni_contender = yes
			var:nesnd_uni_held >= 32
		}
		change_global_variable = {
			name = nesnd_uni_big
			add = 1
		}
	}
	# 8. Phase: exactly two big contenders mean the standoff (a third that big cancels it); ten or more rivals mean the many-kingdoms phase; otherwise the tripod.
	remove_global_variable = nesnd_uni_standoff
	remove_global_variable = nesnd_uni_many
	if = {
		limit = {
			global_var:nesnd_uni_big = 2
		}
		set_global_variable = nesnd_uni_standoff
	}
	else_if = {
		limit = {
			global_var:nesnd_uni_rivals >= 10
		}
		set_global_variable = nesnd_uni_many
	}
	# 9. Map roles on the contenders themselves.
	every_country = {
		limit = {
			nesnd_uni_contender = yes
		}
		if = {
			limit = {
				global_var:nesnd_uni_leader ?= this
			}
			set_variable = nesnd_uni_map_lead
		}
		else_if = {
			limit = {
				has_global_variable = nesnd_uni_standoff
				global_var:nesnd_uni_second ?= this
			}
			set_variable = nesnd_uni_map_second
		}
		else = {
			set_variable = nesnd_uni_map_other
		}
	}
	# 10. Phase modifiers on every contender; involved non-contenders (subjects, bloc followers) lose them.
	every_country = {
		limit = {
			nesnd_uni_qualified = yes
			nesnd_uni_contender = no
		}
		if = {
			limit = {
				has_modifier = nesnd_uni_order
			}
			remove_modifier = nesnd_uni_order
		}
		if = {
			limit = {
				has_modifier = nesnd_uni_tripod
			}
			remove_modifier = nesnd_uni_tripod
		}
	}
	every_country = {
		limit = {
			nesnd_uni_contender = yes
		}
		if = {
			limit = {
				has_global_variable = nesnd_uni_many
			}
			if = {
				limit = {
					has_modifier = nesnd_uni_tripod
				}
				remove_modifier = nesnd_uni_tripod
			}
			if = {
				limit = {
					NOT = { has_modifier = nesnd_uni_order }
				}
				add_modifier = {
					name = nesnd_uni_order
				}
			}
		}
		else = {
			if = {
				limit = {
					has_modifier = nesnd_uni_order
				}
				remove_modifier = nesnd_uni_order
			}
			if = {
				limit = {
					NOT = { has_modifier = nesnd_uni_tripod }
				}
				add_modifier = {
					name = nesnd_uni_tripod
				}
			}
		}
	}
	# The standoff modifier belongs to the leader and the second place while the standoff lasts.
	every_country = {
		limit = {
			has_modifier = nesnd_uni_standoff_power
			NOT = {
				AND = {
					has_global_variable = nesnd_uni_standoff
					OR = {
						global_var:nesnd_uni_leader ?= this
						global_var:nesnd_uni_second ?= this
					}
				}
			}
		}
		remove_modifier = nesnd_uni_standoff_power
	}
	if = {
		limit = {
			has_global_variable = nesnd_uni_standoff
		}
		every_country = {
			limit = {
				OR = {
					global_var:nesnd_uni_leader ?= this
					global_var:nesnd_uni_second ?= this
				}
				NOT = { has_modifier = nesnd_uni_standoff_power }
			}
			add_modifier = {
				name = nesnd_uni_standoff_power
			}
		}
	}
	# 11. Write the lead bar.
	nesnd_uni_sync_lead_bar = yes
}

# Global scope (root = none, needs scope:journal_entry): the lead bar shows the leader's region count, or 0 without a leader.
nesnd_uni_sync_lead_bar = {
	if = {
		limit = {
			has_global_variable = nesnd_uni_lead_count
		}
		scope:journal_entry ?= {
			set_bar_progress = {
				name = nesnd_uni_lead_bar
				value = global_var:nesnd_uni_lead_count
			}
		}
	}
	else = {
		scope:journal_entry ?= {
			set_bar_progress = {
				name = nesnd_uni_lead_bar
				value = 0
			}
		}
	}
}

# Country scope: remove the phase modifiers and the unification variables of one country.
nesnd_uni_clear_country = {
	if = {
		limit = {
			has_modifier = nesnd_uni_order
		}
		remove_modifier = nesnd_uni_order
	}
	if = {
		limit = {
			has_modifier = nesnd_uni_tripod
		}
		remove_modifier = nesnd_uni_tripod
	}
	if = {
		limit = {
			has_modifier = nesnd_uni_standoff_power
		}
		remove_modifier = nesnd_uni_standoff_power
	}
	remove_variable = nesnd_uni_calm
	remove_variable = nesnd_uni_mobilize
	remove_variable = nesnd_uni_held
	remove_variable = nesnd_uni_head
	remove_variable = nesnd_uni_map_lead
	remove_variable = nesnd_uni_map_second
	remove_variable = nesnd_uni_map_other
	remove_variable = nesnd_uni_map_subject
	remove_variable = nesnd_uni_map_ally
}
