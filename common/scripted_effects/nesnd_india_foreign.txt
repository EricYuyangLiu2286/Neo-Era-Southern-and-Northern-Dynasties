# ==== Shared treaty records (formerly nesnd_india_frontier.txt) ====
# The treaties of Bhutan and Nepal with Mughal (matter kinds 2 to 5) were removed; those treaties are left
# to the game's own diplomacy. What remains here serves the trade privileges of the overseas journal
# (further down in this file): the payment on signing (nesnd_india_frontier_settle), the records, the
# monthly tick, renewal (event nesnd_india_frontier.3) and expiry (event nesnd_india_frontier.4). The local
# government keeps the record (T overseas_trade) and the Indian state is asked to renew. A treaty port of
# the overseas journal has no term and keeps only who ceded it to whom.

# From event .3, in the scope of the government asked whether to renew. scope:india_frontier_small keeps
# the record, scope:india_frontier_empire is its partner. T is the kind of record (overseas_trade, KIND 12),
# AMOUNT the sum paid on renewal. The renewal is put to the other side of the treaty. While the side asked
# has another matter open, nothing is sent and the question is put again at a later monthly check.
nesnd_india_frontier_renew = {
    if = { limit = { exists = scope:india_frontier_small scope:india_frontier_small = { exists = var:nesnd_india_frontier_$T$_with } NOT = { has_variable = nesnd_india_matter_kind } }
        if = { limit = { this = scope:india_frontier_small } nesnd_india_matter_send = { KIND = $KIND$ TO = var:nesnd_india_frontier_$T$_with AMOUNT = $AMOUNT$ } }
        else = { nesnd_india_matter_send = { KIND = $KIND$ TO = scope:india_frontier_small AMOUNT = $AMOUNT$ } }
    }
    else_if = { limit = { exists = scope:india_frontier_small }
        scope:india_frontier_small = { remove_variable = nesnd_india_frontier_$T$_asked }
    }
}

# PAYER pays RECEIVER the sum AMOUNT once: on signing, or on renewal.
nesnd_india_frontier_pay = {
    $PAYER$ = { add_treasury = { value = $AMOUNT$ multiply = -1 } }
    $RECEIVER$ = { add_treasury = $AMOUNT$ }
}
# After a treaty is signed: PARTNER pays HOLDER the sum AMOUNT once, and HOLDER keeps the record T.
nesnd_india_frontier_settle = {
    nesnd_india_frontier_pay = { PAYER = $PARTNER$ RECEIVER = $HOLDER$ AMOUNT = $AMOUNT$ }
    $HOLDER$ = { nesnd_india_frontier_record = { T = $T$ PARTNER = $PARTNER$ } }
}

# ---------------------------------------------------------------------------------------------
# Records, kept by one side of the treaty: the local government of the overseas journal (T is
# overseas_trade). The term and the notice are the script values nesnd_india_frontier_<T>_days and
# nesnd_india_frontier_<T>_notice_days.
# ---------------------------------------------------------------------------------------------
nesnd_india_frontier_record = {
    set_variable = { name = nesnd_india_frontier_$T$_with value = $PARTNER$ }
    set_variable = { name = nesnd_india_frontier_$T$_term days = nesnd_india_frontier_$T$_days }
    set_variable = { name = nesnd_india_frontier_$T$_notice days = nesnd_india_frontier_$T$_notice_days }
    remove_variable = nesnd_india_frontier_$T$_asked
    remove_variable = nesnd_india_frontier_$T$_renewed
}
nesnd_india_frontier_forget = {
    remove_variable = nesnd_india_frontier_$T$_with
    remove_variable = nesnd_india_frontier_$T$_term
    remove_variable = nesnd_india_frontier_$T$_notice
    remove_variable = nesnd_india_frontier_$T$_asked
    remove_variable = nesnd_india_frontier_$T$_renewed
}
# The treaty with PARTNER that holds ARTICLE is left without the penalties of breaking it. SOURCE and
# TARGET are leaver (us) or other (PARTNER); BY is the side that withdraws, leaver or other.
nesnd_india_frontier_withdraw_article = {
    save_scope_as = india_frontier_leaver
    $PARTNER$ = { save_scope_as = india_frontier_other }
    every_scope_treaty = {
        limit = {
            binds = scope:india_frontier_other
            any_scope_article = { nesnd_canon_directed_article = { TYPE = $ARTICLE$ SOURCE = scope:india_frontier_$SOURCE$ TARGET = scope:india_frontier_$TARGET$ } }
        }
        withdraw = { country = scope:india_frontier_$BY$ apply_break_penalties = no }
    }
}
# One per kind of record (T): nesnd_india_frontier_<T>_withdraw.
nesnd_india_frontier_overseas_trade_withdraw = { nesnd_india_frontier_withdraw_article = { PARTNER = $PARTNER$ ARTICLE = trade_privilege SOURCE = leaver TARGET = other BY = leaver } }
# The term is over and the treaty was not renewed: we withdraw, and both governments are told.
nesnd_india_frontier_expire = {
    save_scope_as = india_frontier_small
    var:nesnd_india_frontier_$T$_with = { save_scope_as = india_frontier_empire }
    save_scope_value_as = { name = india_frontier_kind value = $KIND$ }
    nesnd_india_frontier_$T$_withdraw = { PARTNER = scope:india_frontier_empire }
    nesnd_india_frontier_forget = { T = $T$ }
    trigger_event = { id = nesnd_india_frontier.4 popup = yes }
    scope:india_frontier_empire = { trigger_event = { id = nesnd_india_frontier.4 popup = yes } }
}

# Monthly, in the scope of the country that keeps the record. KIND is the kind of matter that made the
# treaty (10, the overseas journal's trade privilege). ASK is the side asked whether to renew: empire (the
# partner, the overseas Indian state).
nesnd_india_frontier_tick = {
    if = { limit = { has_variable = nesnd_india_frontier_$T$_with }
        # The partner is gone: only the record is cleared.
        if = { limit = { OR = { NOT = { exists = var:nesnd_india_frontier_$T$_with } var:nesnd_india_frontier_$T$_with = { is_country_alive = no } } }
            nesnd_india_frontier_forget = { T = $T$ }
        }
        # The treaty has ended some other way.
        else_if = { limit = { NOT = { nesnd_india_frontier_$T$_treaty = { PARTNER = var:nesnd_india_frontier_$T$_with } } }
            nesnd_india_frontier_forget = { T = $T$ }
        }
        else_if = { limit = { NOT = { has_variable = nesnd_india_frontier_$T$_term } }
            if = { limit = { has_variable = nesnd_india_frontier_$T$_renewed } nesnd_india_frontier_record = { T = $T$ PARTNER = var:nesnd_india_frontier_$T$_with } }
            else = { nesnd_india_frontier_expire = { T = $T$ KIND = $KIND$ } }
        }
        # 180 days before expiry: renew or let it lapse. Put off while the side asked has another matter open.
        else_if = { limit = {
                NOT = { has_variable = nesnd_india_frontier_$T$_notice }
                NOT = { has_variable = nesnd_india_frontier_$T$_asked }
                NOT = { has_variable = nesnd_india_frontier_$T$_renewed }
                nesnd_india_frontier_free_$ASK$ = { T = $T$ }
            }
            set_variable = nesnd_india_frontier_$T$_asked
            save_scope_as = india_frontier_small
            var:nesnd_india_frontier_$T$_with = { save_scope_as = india_frontier_empire }
            save_scope_value_as = { name = india_frontier_kind value = $KIND$ }
            scope:india_frontier_$ASK$ = { trigger_event = { id = nesnd_india_frontier.3 popup = yes } }
        }
    }
}

# ==== Overseas trade (formerly nesnd_india_overseas.txt) ====
# The overseas journal (nesnd_india_overseas). See common/scripted_triggers/nesnd_india_foreign.txt.
#
# Order of a request:
#   a button of direction D (nesnd_india_overseas_open_choice) opens event nesnd_india_overseas.1, which
#   lists up to four local governments of D by their relations with us
#   -> nesnd_india_overseas_send: matter kind 10 (trade privilege) or 11 (treaty port) into Indian affairs
#      (nesnd_india_matters.txt); the local government's overlord is asked first where
#      nesnd_india_consent_needed says so
#   -> the local government answers: nesnd_india_overseas_accept signs the treaty. For a trade privilege
#      the shared record (nesnd_india_frontier_settle) is kept by the local government with us as partner;
#      for a treaty port the local government keeps only nesnd_india_frontier_overseas_port_with
#   -> a trade privilege: 180 days before expiry the shared tick (nesnd_india_frontier_tick, ASK = empire)
#      asks us whether to renew (event nesnd_india_frontier.3, matter kind 12, half the sum of signing paid
#      on renewal); one not renewed is withdrawn from by the local government in the month it expires
#      (event nesnd_india_frontier.4)
#   -> a treaty port has no term: no expiry, no renewal, no withdrawal by script. Under the article a
#      treaty port that ends becomes an ordinary state of its receiver (GAME concepts_l_english.yml:1330).

# ---------------------------------------------------------------------------------------------
# Monthly, from nesnd_india_refresh
# ---------------------------------------------------------------------------------------------
nesnd_india_overseas_upkeep = {
    if = { limit = { nesnd_india_overseas_holder = yes }
        if = { limit = { NOT = { has_journal_entry = nesnd_india_overseas } } add_journal_entry = { type = nesnd_india_overseas } }
        nesnd_india_overseas_show = yes
    }
    # A local government keeps the records of the treaties it granted.
    nesnd_india_overseas_trade_tick = yes
    nesnd_india_overseas_port_tick = yes
}

# Local government scope. A trade privilege runs its term through the shared tick (matter kind 10).
nesnd_india_overseas_trade_tick = {
    if = { limit = { has_variable = nesnd_india_frontier_overseas_trade_with }
        clear_saved_scope = india_overseas_partner
        if = { limit = { exists = var:nesnd_india_frontier_overseas_trade_with } var:nesnd_india_frontier_overseas_trade_with = { save_scope_as = india_overseas_partner } }
        nesnd_india_frontier_tick = { T = overseas_trade KIND = 10 ASK = empire }
        # The record has ended: the Indian state's journal is brought up to date at once.
        if = { limit = { NOT = { has_variable = nesnd_india_frontier_overseas_trade_with } }
            remove_variable = nesnd_india_overseas_trade_dir
            scope:india_overseas_partner ?= { if = { limit = { nesnd_india_overseas_holder = yes } nesnd_india_overseas_show = yes } }
        }
    }
    else = { remove_variable = nesnd_india_overseas_trade_dir }
}
# Local government scope. A treaty port has no term: its record is cleared only when the Indian state is
# gone or the treaty has ended some other way.
nesnd_india_overseas_port_tick = {
    if = { limit = { has_variable = nesnd_india_frontier_overseas_port_with }
        clear_saved_scope = india_overseas_partner
        if = { limit = { exists = var:nesnd_india_frontier_overseas_port_with } var:nesnd_india_frontier_overseas_port_with = { save_scope_as = india_overseas_partner } }
        if = { limit = { NOT = { exists = scope:india_overseas_partner } } nesnd_india_overseas_port_forget = yes }
        else_if = { limit = { scope:india_overseas_partner = { is_country_alive = no } } nesnd_india_overseas_port_forget = yes }
        else_if = { limit = { NOT = { nesnd_india_frontier_overseas_port_treaty = { PARTNER = scope:india_overseas_partner } } } nesnd_india_overseas_port_forget = yes }
    }
    else = { remove_variable = nesnd_india_overseas_port_dir }
}
# Local government scope: the record of a treaty port is cleared, and the Indian state's journal is brought
# up to date at once.
nesnd_india_overseas_port_forget = {
    remove_variable = nesnd_india_frontier_overseas_port_with
    remove_variable = nesnd_india_overseas_port_dir
    scope:india_overseas_partner ?= { if = { limit = { nesnd_india_overseas_holder = yes } nesnd_india_overseas_show = yes } }
}

# Indian state scope. For each direction and kind: how many treaties we hold, and one of their local
# governments to name in the journal.
nesnd_india_overseas_show = {
    save_scope_as = india_overseas_viewer
    nesnd_india_overseas_show_one = { D = 1 K = trade }
    nesnd_india_overseas_show_one = { D = 1 K = port }
    nesnd_india_overseas_show_one = { D = 2 K = trade }
    nesnd_india_overseas_show_one = { D = 2 K = port }
    nesnd_india_overseas_show_one = { D = 3 K = trade }
    nesnd_india_overseas_show_one = { D = 3 K = port }
}
nesnd_india_overseas_show_one = {
    remove_variable = nesnd_india_overseas_show_$D$_$K$
    set_variable = { name = nesnd_india_overseas_count_$D$_$K$ value = 0 }
    every_country = {
        limit = {
            var:nesnd_india_frontier_overseas_$K$_with ?= scope:india_overseas_viewer
            var:nesnd_india_overseas_$K$_dir ?= $D$
        }
        scope:india_overseas_viewer = { change_variable = { name = nesnd_india_overseas_count_$D$_$K$ add = 1 } }
    }
    clear_saved_scope = india_overseas_shown
    ordered_country = {
        limit = {
            var:nesnd_india_frontier_overseas_$K$_with ?= scope:india_overseas_viewer
            var:nesnd_india_overseas_$K$_dir ?= $D$
        }
        order_by = gdp
        position = 0
        check_range_bounds = no
        save_scope_as = india_overseas_shown
    }
    if = { limit = { exists = scope:india_overseas_shown } set_variable = { name = nesnd_india_overseas_show_$D$_$K$ value = scope:india_overseas_shown } }
}
# The journal has ended: what it showed is cleared. A matter still open lapses through Indian affairs.
nesnd_india_overseas_clear_show = {
    remove_variable = nesnd_india_overseas_show_1_trade remove_variable = nesnd_india_overseas_count_1_trade
    remove_variable = nesnd_india_overseas_show_1_port remove_variable = nesnd_india_overseas_count_1_port
    remove_variable = nesnd_india_overseas_show_2_trade remove_variable = nesnd_india_overseas_count_2_trade
    remove_variable = nesnd_india_overseas_show_2_port remove_variable = nesnd_india_overseas_count_2_port
    remove_variable = nesnd_india_overseas_show_3_trade remove_variable = nesnd_india_overseas_count_3_trade
    remove_variable = nesnd_india_overseas_show_3_port remove_variable = nesnd_india_overseas_count_3_port
}

# From event .1's immediate. The four local governments of DIR on the best terms with us.
nesnd_india_overseas_pick = {
    save_scope_as = india_overseas_asker
    ordered_country = { limit = { nesnd_india_overseas_local = { DIR = $DIR$ } NOT = { this = scope:india_overseas_asker } } order_by = relations:root position = 0 check_range_bounds = no save_scope_as = india_overseas_pick_1 }
    ordered_country = { limit = { nesnd_india_overseas_local = { DIR = $DIR$ } NOT = { this = scope:india_overseas_asker } } order_by = relations:root position = 1 check_range_bounds = no save_scope_as = india_overseas_pick_2 }
    ordered_country = { limit = { nesnd_india_overseas_local = { DIR = $DIR$ } NOT = { this = scope:india_overseas_asker } } order_by = relations:root position = 2 check_range_bounds = no save_scope_as = india_overseas_pick_3 }
    ordered_country = { limit = { nesnd_india_overseas_local = { DIR = $DIR$ } NOT = { this = scope:india_overseas_asker } } order_by = relations:root position = 3 check_range_bounds = no save_scope_as = india_overseas_pick_4 }
    # For a treaty port, the state each of them would cede.
    if = { limit = { scope:india_overseas_kind = 11 }
        scope:india_overseas_pick_1 ?= { nesnd_india_overseas_find_port = { DIR = $DIR$ SAVE = india_overseas_port_1 } }
        scope:india_overseas_pick_2 ?= { nesnd_india_overseas_find_port = { DIR = $DIR$ SAVE = india_overseas_port_2 } }
        scope:india_overseas_pick_3 ?= { nesnd_india_overseas_find_port = { DIR = $DIR$ SAVE = india_overseas_port_3 } }
        scope:india_overseas_pick_4 ?= { nesnd_india_overseas_find_port = { DIR = $DIR$ SAVE = india_overseas_port_4 } }
    }
}
# Local government scope; scope:india_overseas_asker is the Indian state. The most populous state of
# direction DIR it could cede to the Indian state as a treaty port, saved as SAVE.
nesnd_india_overseas_find_port = {
    save_scope_as = india_overseas_port_giver
    ordered_scope_state = {
        limit = {
            nesnd_india_overseas_port_state = { DIR = $DIR$ TAKER = scope:india_overseas_asker }
            save_temporary_scope_as = india_overseas_port_try
            scope:india_overseas_asker = {
                can_create_treaty = {
                    first_country = this
                    second_country = scope:india_overseas_port_giver
                    is_draft = no
                    binding_period = { years = 10 }
                    articles_to_create = {
                        { article = treaty_port source_country = scope:india_overseas_port_giver target_country = this inputs = { { state = scope:india_overseas_port_try } } }
                    }
                }
            }
        }
        order_by = state_population
        position = 0
        check_range_bounds = no
        save_scope_as = $SAVE$
    }
}

# From event .1 or the AI. PICK is the local government chosen; scope:india_overseas_dir and
# scope:india_overseas_kind say what is asked.
nesnd_india_overseas_send = {
    if = { limit = { exists = $PICK$ nesnd_india_overseas_option_ok = { PICK = $PICK$ } }
        set_variable = { name = nesnd_india_overseas_dir value = scope:india_overseas_dir }
        if = { limit = { scope:india_overseas_kind = 10 } nesnd_india_matter_send = { KIND = 10 TO = $PICK$ AMOUNT = nesnd_india_overseas_trade_payment } }
        else = { nesnd_india_matter_send = { KIND = 11 TO = $PICK$ AMOUNT = nesnd_india_overseas_port_payment } }
        # A member of the small states' system: the request takes up its turn there (C6).
        $PICK$ = { if = { limit = { nesnd_small_state_country = yes } set_variable = { name = nesnd_canon_request_cooldown months = 6 } } }
    }
}

# ---------------------------------------------------------------------------------------------
# AI (Indian state scope, monthly on the journal). Hard limits are in the triggers.

nesnd_india_overseas_ai_asks = {
    if = { limit = { is_ai = yes nesnd_india_overseas_holder = yes net_total_income >= 0 }
        nesnd_india_overseas_ai_dir = { DIR = 1 }
        nesnd_india_overseas_ai_dir = { DIR = 2 }

# One chance a month for each kind of treaty in direction DIR that we do not yet hold there:
# a trade privilege 1 per cent, a treaty port 0.5 per cent (1 per cent, then half of that).
nesnd_india_overseas_ai_dir = {
    save_scope_as = india_overseas_asker
    if = { limit = { nesnd_india_overseas_count_$DIR$_trade_value = 0 nesnd_india_overseas_can_ask = { DIR = $DIR$ KIND = trade } }
        save_temporary_scope_value_as = { name = india_overseas_chance value = nesnd_india_overseas_trade_ask_chance }
        random = {
            chance = scope:india_overseas_chance
            save_scope_value_as = { name = india_overseas_dir value = $DIR$ }
            save_scope_value_as = { name = india_overseas_kind value = 10 }
            nesnd_india_overseas_ai_pick = { DIR = $DIR$ KIND = trade }
        }
    }
    if = { limit = { nesnd_india_overseas_count_$DIR$_port_value = 0 nesnd_india_overseas_can_ask = { DIR = $DIR$ KIND = port } }
        random = {
            chance = 1
            random = {
                chance = 50
                save_scope_value_as = { name = india_overseas_dir value = $DIR$ }
                save_scope_value_as = { name = india_overseas_kind value = 11 }
                nesnd_india_overseas_ai_pick = { DIR = $DIR$ KIND = port }
            }
        }
    }
}
# The local government on the best terms with us that we can ask.
nesnd_india_overseas_ai_pick = {

        order_by = relations:root
        position = 0

        save_scope_as = india_overseas_ai_pick

    if = { limit = { exists = scope:india_overseas_ai_pick } nesnd_india_overseas_send = { PICK = scope:india_overseas_ai_pick } }
}

# ---------------------------------------------------------------------------------------------
# The local government that answers (answering scope, from nesnd_india_matter_accept)
# ---------------------------------------------------------------------------------------------
# scope:india_matter_answerer answers, scope:india_matter_holder (the Indian state) asked.
nesnd_india_overseas_accept = {
    clear_saved_scope = india_overseas_port_signed
    if = { limit = { nesnd_india_matter_is = { KIND = 11 } }
        scope:india_matter_holder = { save_scope_as = india_overseas_asker }
        if = { limit = { scope:india_matter_holder = { nesnd_india_overseas_dir_value = 1 } } nesnd_india_overseas_find_port = { DIR = 1 SAVE = india_overseas_port_signed } }
        else_if = { limit = { scope:india_matter_holder = { nesnd_india_overseas_dir_value = 2 } } nesnd_india_overseas_find_port = { DIR = 2 SAVE = india_overseas_port_signed } }
        else = { nesnd_india_overseas_find_port = { DIR = 3 SAVE = india_overseas_port_signed } }
    }
    # No port is left to cede (nesnd_india_overseas_can_grant was checked just before, so this should not
    # happen): the matter no longer stands.
    if = { limit = { nesnd_india_matter_is = { KIND = 11 } NOT = { exists = scope:india_overseas_port_signed } }
        scope:india_matter_holder = { nesnd_india_matter_fail = { RESULT = 5 } }
    }
    else = {
        if = { limit = { nesnd_india_matter_is = { KIND = 10 } } nesnd_india_overseas_sign_trade = { LOCAL = scope:india_matter_answerer TAKER = scope:india_matter_holder } }
        else_if = { limit = { nesnd_india_matter_is = { KIND = 11 } } nesnd_india_overseas_sign_port = { LOCAL = scope:india_matter_answerer TAKER = scope:india_matter_holder STATE = scope:india_overseas_port_signed } }
        # A renewal of the trade privilege: the Indian state pays half the sum of signing.
        else_if = { limit = { nesnd_india_matter_is = { KIND = 12 } }
            scope:india_matter_answerer = { set_variable = nesnd_india_frontier_overseas_trade_renewed }
            nesnd_india_frontier_pay = { PAYER = scope:india_matter_holder RECEIVER = scope:india_matter_answerer AMOUNT = nesnd_india_overseas_trade_renew_payment }
        }
        scope:india_matter_holder = {
            nesnd_india_matter_save_names = yes
            trigger_event = { id = nesnd_india_overseas.2 popup = yes }
            nesnd_india_matter_clear = yes
            remove_variable = nesnd_india_overseas_dir
            if = { limit = { nesnd_india_overseas_holder = yes } nesnd_india_overseas_show = yes }
        }
    }
}
# Signing. The local government grants, the Indian state receives and pays once. A treaty port keeps no
# term: the local government records only to whom it ceded the port, for the journal and so that it is
# not asked twice.
nesnd_india_overseas_sign_trade = {
    create_treaty = {
        first_country = $TAKER$
        second_country = $LOCAL$
        is_draft = no
        binding_period = { years = 10 }
        articles_to_create = {
            { article = trade_privilege source_country = $LOCAL$ target_country = $TAKER$ }
        }
    }
    nesnd_india_frontier_settle = { HOLDER = $LOCAL$ PARTNER = $TAKER$ T = overseas_trade AMOUNT = nesnd_india_overseas_trade_payment }
    $LOCAL$ = { set_variable = { name = nesnd_india_overseas_trade_dir value = $TAKER$.nesnd_india_overseas_dir_value } }
}
nesnd_india_overseas_sign_port = {
    create_treaty = {
        first_country = $TAKER$
        second_country = $LOCAL$
        is_draft = no
        binding_period = { years = 10 }
        articles_to_create = {
            { article = treaty_port source_country = $LOCAL$ target_country = $TAKER$ inputs = { { state = $STATE$ } } }
        }
    }
    nesnd_india_frontier_pay = { PAYER = $TAKER$ RECEIVER = $LOCAL$ AMOUNT = nesnd_india_overseas_port_payment }
    $LOCAL$ = {
        set_variable = { name = nesnd_india_frontier_overseas_port_with value = $TAKER$ }
        set_variable = { name = nesnd_india_overseas_port_dir value = $TAKER$.nesnd_india_overseas_dir_value }
    }
}
# What the answering government is shown on the accept button. Tooltips only.
nesnd_india_overseas_accept_tooltips = {
    if = { limit = { nesnd_india_matter_is = { KIND = 10 } }
        custom_tooltip = nesnd_india_overseas_accept_trade_tt
        show_as_tooltip = { add_treasury = nesnd_india_matter_quote }
        custom_tooltip = nesnd_india_treaty_upkeep_theirs_trade_tt
    }
    else_if = { limit = { nesnd_india_matter_is = { KIND = 11 } }
        custom_tooltip = nesnd_india_overseas_accept_port_tt
        show_as_tooltip = { add_treasury = nesnd_india_matter_quote }
        custom_tooltip = nesnd_india_treaty_upkeep_theirs_port_tt
    }
    else = {
        custom_tooltip = nesnd_india_overseas_accept_renew_tt
        show_as_tooltip = { add_treasury = nesnd_india_matter_quote }
    }
}

# From nesnd_india_matter_fail (Indian state scope): the asking country is told, and the matter ends.
nesnd_india_overseas_matter_fail = {
    trigger_event = { id = nesnd_india_overseas.3 popup = yes }
    nesnd_india_matter_clear = yes
    remove_variable = nesnd_india_overseas_dir
}
