# ==== Overseas dealings (India batch 17) ====
# Called monthly from nesnd_india_refresh (common/scripted_effects/nesnd_india_core.txt).
nesnd_india_overseas_upkeep = {
	if = {
		limit = {
			nesnd_india_overseas_holder = yes
			NOT = { has_journal_entry = nesnd_india_overseas }
		}
		add_journal_entry = { type = nesnd_india_overseas }
	}
}

# Cleared when nesnd_india_overseas ends.
nesnd_india_overseas_clear_flags = {
	remove_variable = nesnd_india_overseas_above50_himalayas
	remove_variable = nesnd_india_overseas_above50_indonesia
	remove_variable = nesnd_india_overseas_above50_east_africa
	remove_variable = nesnd_india_overseas_above50_arabia
	remove_variable = nesnd_india_overseas_above50_persia
}

# ---------------------------------------------------------------------------------------------
# Monthly pulse on nesnd_india_overseas
# ---------------------------------------------------------------------------------------------
nesnd_india_overseas_monthly = {
	nesnd_india_overseas_check_50_dir = { DIR = himalayas REGION = region_himalayas }
	nesnd_india_overseas_check_50_dir = { DIR = indonesia REGION = region_indonesia }
	nesnd_india_overseas_check_50_dir = { DIR = east_africa REGION = region_east_africa }
	nesnd_india_overseas_check_50_dir = { DIR = arabia REGION = region_arabia }
	nesnd_india_overseas_check_50_dir = { DIR = persia REGION = region_greater_persia }
	nesnd_india_overseas_check_milestone = yes
	if = {
		limit = { is_ai = yes }
		nesnd_india_overseas_ai_monthly = yes
	}
}

# When a direction's bar first crosses to >= 50, notify every major or great power (excluding us and our
# subjects) that holds a state in REGION directly or through its subjects. Five-year per-power cooldown.
nesnd_india_overseas_check_50_dir = {
	if = {
		limit = { nesnd_india_overseas_bar_$DIR$ >= 50 }
		if = {
			limit = { NOT = { has_variable = nesnd_india_overseas_above50_$DIR$ } }
			set_variable = nesnd_india_overseas_above50_$DIR$
			save_scope_as = india_overseas_actor
			sr:$REGION$ = { save_scope_as = india_overseas_region }
			every_country = {
				limit = {
					is_country_alive = yes
					NOT = { this = scope:india_overseas_actor }
					NOT = { is_subject_of = scope:india_overseas_actor }
					country_rank >= rank_value:major_power
					OR = {
						any_scope_state = { region = sr:$REGION$ }
						any_subject_or_below = { any_scope_state = { region = sr:$REGION$ } }
					}
					save_temporary_scope_as = india_overseas_power
					NOT = {
						scope:india_overseas_actor = {
							is_target_in_variable_list = {
								name = nesnd_india_overseas_warned_$DIR$
								target = scope:india_overseas_power
							}
						}
					}
				}
				save_temporary_scope_as = india_overseas_power
				scope:india_overseas_actor = {
					add_to_variable_list = {
						name = nesnd_india_overseas_warned_$DIR$
						target = scope:india_overseas_power
						days = 1825
					}
				}
				trigger_event = { id = nesnd_india_overseas.5 popup = yes }
			}
		}
	}
	else = {
		remove_variable = nesnd_india_overseas_above50_$DIR$
	}
}

# One-time milestone "Indian Ocean Trade": three distinct directions with active FIR and tier >= 2.
nesnd_india_overseas_check_milestone = {
	if = {
		limit = { nesnd_india_overseas_ocean_trade_ready = yes }
		set_variable = nesnd_india_ocean_trade
		trigger_event = { id = nesnd_india_overseas.7 popup = yes }
	}
}

# Up to three regional countries in REGION, highest relations with us first.
nesnd_india_overseas_pick_fir = {
	save_scope_as = india_overseas_self
	clear_saved_scope = india_overseas_pick_1
	clear_saved_scope = india_overseas_pick_2
	clear_saved_scope = india_overseas_pick_3
	ordered_country = {
		limit = { nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self } }
		order_by = relations:root
		position = 0
		check_range_bounds = no
		save_scope_as = india_overseas_pick_1
	}
	ordered_country = {
		limit = { nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self } }
		order_by = relations:root
		position = 1
		check_range_bounds = no
		save_scope_as = india_overseas_pick_2
	}
	ordered_country = {
		limit = { nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self } }
		order_by = relations:root
		position = 2
		check_range_bounds = no
		save_scope_as = india_overseas_pick_3
	}
}

# Deduct 50 dealings from the bar of scope:india_overseas_dir (1-5).
nesnd_india_overseas_deduct_50 = {
	custom_tooltip = {
		text = nesnd_india_overseas_spend_50_tt
		if = {
			limit = { exists = scope:india_overseas_dir }
			if = { limit = { scope:india_overseas_dir = 1 } je:nesnd_india_overseas ?= { add_progress = { name = nesnd_india_overseas_himalayas_bar value = -50 } } }
			else_if = { limit = { scope:india_overseas_dir = 2 } je:nesnd_india_overseas ?= { add_progress = { name = nesnd_india_overseas_indonesia_bar value = -50 } } }
			else_if = { limit = { scope:india_overseas_dir = 3 } je:nesnd_india_overseas ?= { add_progress = { name = nesnd_india_overseas_east_africa_bar value = -50 } } }
			else_if = { limit = { scope:india_overseas_dir = 4 } je:nesnd_india_overseas ?= { add_progress = { name = nesnd_india_overseas_arabia_bar value = -50 } } }
			else = { je:nesnd_india_overseas ?= { add_progress = { name = nesnd_india_overseas_persia_bar value = -50 } } }
		}
	}
}

# Sign 10-year foreign investment rights directly (TARGET = host/source_country, root = investor/target_country),
# pay nesnd_india_overseas_fir_cost, establish a cultural community in TARGET's capital, and notify TARGET.
nesnd_india_overseas_sign_fir = {
	if = {
		limit = { exists = $TARGET$ }
		nesnd_india_overseas_deduct_50 = yes
		add_treasury = { value = nesnd_india_overseas_fir_cost multiply = -1 }
		create_treaty = {
			first_country = root
			second_country = $TARGET$
			is_draft = no
			binding_period = { years = 10 }
			articles_to_create = {
				{
					article = foreign_investment_rights
					source_country = $TARGET$
					target_country = root
				}
			}
		}
		custom_tooltip = {
			text = nesnd_india_overseas_diaspora_tt
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 0
				check_range_bounds = no
				save_scope_as = india_overseas_culture
			}
			if = {
				limit = { exists = scope:india_overseas_culture }
				$TARGET$ = {
					capital ?= {
						if = {
							limit = { NOT = { has_cultural_community = scope:india_overseas_culture } }
							add_cultural_community = scope:india_overseas_culture
						}
					}
				}
			}
		}
		hidden_effect = {
			save_scope_as = india_overseas_investor
			$TARGET$ = { trigger_event = { id = nesnd_india_overseas.2 popup = yes } }
			nesnd_india_overseas_sync_above50 = yes
			nesnd_india_overseas_check_milestone = yes
		}
	}
}

# Sync above-50 flags after spending progress so a later climb back to 50 checks the 5-year cooldown cleanly.
nesnd_india_overseas_sync_above50 = {
	if = { limit = { nesnd_india_overseas_bar_himalayas < 50 } remove_variable = nesnd_india_overseas_above50_himalayas }
	if = { limit = { nesnd_india_overseas_bar_indonesia < 50 } remove_variable = nesnd_india_overseas_above50_indonesia }
	if = { limit = { nesnd_india_overseas_bar_east_africa < 50 } remove_variable = nesnd_india_overseas_above50_east_africa }
	if = { limit = { nesnd_india_overseas_bar_arabia < 50 } remove_variable = nesnd_india_overseas_above50_arabia }
	if = { limit = { nesnd_india_overseas_bar_persia < 50 } remove_variable = nesnd_india_overseas_above50_persia }
}

# Up to three regional countries in REGION that hold an unclaimed state there, lowest relations with us first.
nesnd_india_overseas_pick_claim = {
	save_scope_as = india_overseas_self
	clear_saved_scope = india_overseas_pick_1
	clear_saved_scope = india_overseas_pick_2
	clear_saved_scope = india_overseas_pick_3
	clear_saved_scope = india_overseas_claim_state_1
	clear_saved_scope = india_overseas_claim_state_2
	clear_saved_scope = india_overseas_claim_state_3
	ordered_country = {
		limit = {
			nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
			any_scope_state = { region = sr:$REGION$ NOT = { has_claim_by = scope:india_overseas_self } }
		}
		order_by = { value = relations:root multiply = -1 }
		position = 0
		check_range_bounds = no
		save_scope_as = india_overseas_pick_1
	}
	ordered_country = {
		limit = {
			nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
			any_scope_state = { region = sr:$REGION$ NOT = { has_claim_by = scope:india_overseas_self } }
		}
		order_by = { value = relations:root multiply = -1 }
		position = 1
		check_range_bounds = no
		save_scope_as = india_overseas_pick_2
	}
	ordered_country = {
		limit = {
			nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
			any_scope_state = { region = sr:$REGION$ NOT = { has_claim_by = scope:india_overseas_self } }
		}
		order_by = { value = relations:root multiply = -1 }
		position = 2
		check_range_bounds = no
		save_scope_as = india_overseas_pick_3
	}
	scope:india_overseas_pick_1 ?= { nesnd_india_overseas_find_claim_state = { REGION = $REGION$ SAVE = india_overseas_claim_state_1 } }
	scope:india_overseas_pick_2 ?= { nesnd_india_overseas_find_claim_state = { REGION = $REGION$ SAVE = india_overseas_claim_state_2 } }
	scope:india_overseas_pick_3 ?= { nesnd_india_overseas_find_claim_state = { REGION = $REGION$ SAVE = india_overseas_claim_state_3 } }
}

# Target country scope; scope:india_overseas_self is the Indian country. Most populous unclaimed state in REGION.
nesnd_india_overseas_find_claim_state = {
	ordered_scope_state = {
		limit = {
			region = sr:$REGION$
			NOT = { has_claim_by = scope:india_overseas_self }
		}
		order_by = state_population
		position = 0
		check_range_bounds = no
		save_scope_as = $SAVE$
	}
}

# Reset bar of scope:india_overseas_dir (1-5) to 0.
nesnd_india_overseas_reset_bar = {
	custom_tooltip = {
		text = nesnd_india_overseas_reset_bar_tt
		if = {
			limit = { exists = scope:india_overseas_dir }
			if = { limit = { scope:india_overseas_dir = 1 } je:nesnd_india_overseas ?= { set_bar_progress = { name = nesnd_india_overseas_himalayas_bar value = 0 } } }
			else_if = { limit = { scope:india_overseas_dir = 2 } je:nesnd_india_overseas ?= { set_bar_progress = { name = nesnd_india_overseas_indonesia_bar value = 0 } } }
			else_if = { limit = { scope:india_overseas_dir = 3 } je:nesnd_india_overseas ?= { set_bar_progress = { name = nesnd_india_overseas_east_africa_bar value = 0 } } }
			else_if = { limit = { scope:india_overseas_dir = 4 } je:nesnd_india_overseas ?= { set_bar_progress = { name = nesnd_india_overseas_arabia_bar value = 0 } } }
			else = { je:nesnd_india_overseas ?= { set_bar_progress = { name = nesnd_india_overseas_persia_bar value = 0 } } }

# Reset the direction bar to 0, claim STATE's state region, and notify TARGET.
nesnd_india_overseas_do_claim = {
	if = {
		limit = { exists = $TARGET$ exists = $STATE$ }
		nesnd_india_overseas_reset_bar = yes
		$STATE$.state_region = { add_claim = root }
		hidden_effect = {
			save_scope_as = india_overseas_claimer
			$STATE$ = { save_scope_as = india_overseas_claimed_state }
			$TARGET$ = { trigger_event = { id = nesnd_india_overseas.4 popup = yes } }
			nesnd_india_overseas_sync_above50 = yes
		}
	}
}

# ---------------------------------------------------------------------------------------------
# AI (Indian country scope, monthly on nesnd_india_overseas)
# ---------------------------------------------------------------------------------------------
nesnd_india_overseas_ai_monthly = {
	nesnd_india_overseas_ai_dir = { DIR = himalayas REGION = region_himalayas IDX = 1 }
	nesnd_india_overseas_ai_dir = { DIR = indonesia REGION = region_indonesia IDX = 2 }
	nesnd_india_overseas_ai_dir = { DIR = east_africa REGION = region_east_africa IDX = 3 }
	nesnd_india_overseas_ai_dir = { DIR = arabia REGION = region_arabia IDX = 4 }
	nesnd_india_overseas_ai_dir = { DIR = persia REGION = region_greater_persia IDX = 5 }
}

nesnd_india_overseas_ai_dir = {
	save_scope_as = india_overseas_self
	if = {
		limit = {
			nesnd_india_overseas_bar_$DIR$ >= 50
			gold_reserves >= nesnd_india_overseas_ai_reserve
		}
		clear_saved_scope = india_overseas_ai_target
		ordered_country = {
			limit = {
				nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
				save_temporary_scope_as = india_overseas_cand
				scope:india_overseas_self = { nesnd_india_overseas_can_sign_fir_raw = { TARGET = scope:india_overseas_cand } }
			}
			order_by = relations:root
			position = 0
			check_range_bounds = no
			save_scope_as = india_overseas_ai_target
		}
		if = {
			limit = { exists = scope:india_overseas_ai_target }
			save_scope_value_as = { name = india_overseas_dir value = $IDX$ }
			sr:$REGION$ = { save_scope_as = india_overseas_region }
			nesnd_india_overseas_sign_fir = { TARGET = scope:india_overseas_ai_target }
		}
	}
	if = {
		limit = { nesnd_india_overseas_bar_$DIR$ >= 100 }
		random = {
			chance = 20
			clear_saved_scope = india_overseas_ai_target
			clear_saved_scope = india_overseas_ai_state
			ordered_country = {
				limit = {
					nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
					any_scope_state = { region = sr:$REGION$ NOT = { has_claim_by = scope:india_overseas_self } }
				}
				order_by = { value = relations:root multiply = -1 }
				position = 0
				check_range_bounds = no
				save_scope_as = india_overseas_ai_target
			}
			scope:india_overseas_ai_target ?= { nesnd_india_overseas_find_claim_state = { REGION = $REGION$ SAVE = india_overseas_ai_state } }
			if = {
				limit = { exists = scope:india_overseas_ai_target exists = scope:india_overseas_ai_state }
				save_scope_value_as = { name = india_overseas_dir value = $IDX$ }
				sr:$REGION$ = { save_scope_as = india_overseas_region }
				nesnd_india_overseas_do_claim = { TARGET = scope:india_overseas_ai_target STATE = scope:india_overseas_ai_state }
			}
		}
	}
}
