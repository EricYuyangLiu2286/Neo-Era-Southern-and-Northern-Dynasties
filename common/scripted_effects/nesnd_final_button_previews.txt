nesnd_final_c2_project_preview = {
 if = { limit = { exists = $TARGET$ }
  if = { limit = { $TARGET$ = { owner = prev } }
   if = { limit = { $TARGET$ = { $CANDIDATE$ = yes } }
    custom_description = { text = nesnd_c2_project_target_tt subject = $TARGET$ }
    if = { limit = { has_variable = $QUOTE_VARIABLE$ OR = { has_variable = $ACTIVE_VARIABLE$ has_modifier = $EXPENSE$ } }
     custom_tooltip = $LOCKED_TT$
    }
    else = {
     show_as_tooltip = { add_modifier = { name = $EXPENSE$ months = 12 multiplier = $QUOTE_VALUE$ } }
    }
    show_as_tooltip = {
     $TARGET$ = { add_modifier = { name = $LOCAL$ months = 24 } }
     add_modifier = { name = $POLITICS$ months = 12 }
    }
    $TARGET$ = {
     if = { limit = { has_modifier = $LOCAL$ } custom_tooltip = $LOCAL_TT$ }
    }
    if = { limit = { has_modifier = $POLITICS$ } custom_tooltip = $POLITICS_TT$ }
   }
  }
 }
}
