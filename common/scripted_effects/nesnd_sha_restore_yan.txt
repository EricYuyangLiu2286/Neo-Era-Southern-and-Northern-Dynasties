# SHA scope. The CNE warning has verified each saved state piece before transfer.
nesnd_sha_raise_yan = {
    if = {
        limit = { nesnd_sha_yan_can_rise = yes }
        c:CNE = {
            if = { limit = { has_variable_list = nesnd_sha_yan_old_cores } clear_variable_list = nesnd_sha_yan_old_cores }
            every_scope_state = {
                limit = { nesnd_yan_announced_state = yes is_incorporated = yes }
                owner = { add_to_variable_list = { name = nesnd_sha_yan_old_cores target = prev.state_region } }
            }
            # Bind the five founding garrisons from the state pieces on the published notice.
            every_in_list = {
                variable = nesnd_yan_states
                if = { limit = { state_region = s:STATE_SONGJIANG } save_scope_as = nesnd_yan_garrison_songjiang }
                if = { limit = { state_region = s:STATE_OUTER_MANCHURIA } save_scope_as = nesnd_yan_garrison_outer_manchuria }
                if = { limit = { state_region = s:STATE_NORTHERN_MANCHURIA } save_scope_as = nesnd_yan_garrison_northern_manchuria }
                if = { limit = { state_region = s:STATE_AMUR } save_scope_as = nesnd_yan_garrison_amur }
                if = { limit = { state_region = s:STATE_BOLI } save_scope_as = nesnd_yan_garrison_boli }
            }
            create_country = {
                tag = YAN
                origin = c:CNE
                state = scope:nesnd_yan_garrison_songjiang
                state = scope:nesnd_yan_garrison_northern_manchuria
                state = scope:nesnd_yan_garrison_amur
                state = scope:nesnd_yan_garrison_outer_manchuria
                state = scope:nesnd_yan_garrison_boli
                on_created = {
                    nesnd_rising_clear_inherited = yes
                    set_variable = nesnd_sha_yan_restored_court
                    activate_law = law_type:law_monarchy
                    create_character = {
                        template = nesnd_sha_yan_xu_jingheng
                        on_created = { set_character_as_ruler = yes }
                    }
                    create_military_formation = {
                        name = nesnd_sha_yan_restoration_army
                        type = army
                        hq_region = s:STATE_SONGJIANG.region
                        combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_SONGJIANG count = 20 }
                        combat_unit = { type = unit_type:combat_unit_type_cannon_artillery state_region = s:STATE_SONGJIANG count = 5 }
                        combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_NORTHERN_MANCHURIA count = 20 }
                        combat_unit = { type = unit_type:combat_unit_type_cannon_artillery state_region = s:STATE_NORTHERN_MANCHURIA count = 5 }
                    }
                    create_character = { is_general = yes interest_group = ig_armed_forces }
                }
            }
        }
        if = {
            limit = { exists = c:YAN c:YAN ?= { has_variable = nesnd_sha_yan_restored_court } }
            # The five founding states have already transferred. Transfer only remaining signed garrisons.
            c:CNE = { every_scope_state = { limit = { nesnd_yan_announced_state = yes } set_state_owner = c:YAN } }
            set_variable = nesnd_sha_yan_rose
            remove_variable = nesnd_yan_notice
            if = {
                limit = { var:nesnd_sha_yan_partition = flag:nesnd_sha_yan_port }
                c:YAN = { nesnd_sha_yan_extra_infantry = { NAME = nesnd_sha_yan_local_army COUNT = 8 } }
            }
            if = {
                limit = { var:nesnd_sha_route = flag:nesnd_sha_peace }
                add_treasury = { value = gdp multiply = 0.05 min = 7500 multiply = -1 }
                c:YAN = { nesnd_sha_yan_extra_infantry = { NAME = nesnd_sha_yan_supplied_army COUNT = 6 } }
            }
            if = {
                limit = { has_variable = nesnd_sha_korean_compact exists = c:KOR }
                c:KOR = {
                    change_relations = { country = c:SHA value = 15 }
                    change_relations = { country = c:YAN value = 15 }
                }
            }
            c:CNE = {
                if = { limit = { has_journal_entry = nesnd_rep_warning } set_variable = nesnd_rep_interrupted }
                end_truce = c:YAN
                add_modifier = { name = nesnd_sha_yan_disorder months = 6 }
                if = {
                    limit = { c:SHA = { var:nesnd_sha_route = flag:nesnd_sha_war } }
                    create_diplomatic_play = {
                        name = nesnd_sha_yan_war
                        type = nesnd_sha_yan_suppression
                        target_country = c:YAN
                        add_target_backers = { c:SHA }
                        war = yes
                        annex_as_civil_war = no
                        add_war_goal = { holder = c:CNE target_country = c:YAN type = nesnd_sha_annex_yan primary_demand = yes }
                        add_war_goal = { holder = c:YAN target_country = c:CNE type = humiliation primary_demand = yes }
                    }
                }
                else = {
                    create_diplomatic_play = {
                        name = nesnd_sha_yan_war
                        type = nesnd_sha_yan_suppression
                        target_country = c:YAN
                        war = yes
                        annex_as_civil_war = no
                        add_war_goal = { holder = c:CNE target_country = c:YAN type = nesnd_sha_annex_yan primary_demand = yes }
                        add_war_goal = { holder = c:YAN target_country = c:CNE type = humiliation primary_demand = yes }
                    }
                }
            }
            if = {
                limit = { var:nesnd_sha_route = flag:nesnd_sha_war }
                create_diplomatic_pact = { country = c:YAN type = protectorate }
                nesnd_sha_yan_deliver_land = yes
            }
            trigger_event = { id = nesnd_sha_yan.1 popup = yes }
        }
    }
}

# YAN scope. Both increments are in addition to the 40 infantry and 10 artillery.
nesnd_sha_yan_extra_infantry = {
    create_military_formation = {
        name = $NAME$
        type = army
        hq_region = s:STATE_SONGJIANG.region
        combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_SONGJIANG count = $COUNT$ }
    }
    create_character = { is_general = yes interest_group = ig_armed_forces }
}

# SHA scope. Never take a province from a third country to fulfil Yan's promise.
nesnd_sha_yan_deliver_land = {
    s:STATE_AMUR.region_state:YAN ?= { set_state_owner = c:SHA }
    s:STATE_OUTER_MANCHURIA.region_state:YAN ?= { set_state_owner = c:SHA }
    if = {
        limit = { NOT = { var:nesnd_sha_yan_partition ?= flag:nesnd_sha_yan_port } }
        s:STATE_BOLI.region_state:YAN ?= { set_state_owner = c:SHA }
    }
}

nesnd_sha_yan_close = {
    if = { limit = { has_variable_list = nesnd_yan_contacts_by_state } clear_variable_list = nesnd_yan_contacts_by_state }
    remove_variable = nesnd_yan_network_started
    remove_variable = nesnd_yan_notice
    remove_variable = nesnd_yan_wait_for_full
    remove_variable = nesnd_yan_warning_entry
    set_variable = nesnd_sha_yan_ended
    nesnd_sha_enter_shandong = yes
    nesnd_sha_enter_overseas = yes
    remove_variable = nesnd_sha_yan_preparation_window
    remove_variable = nesnd_sha_yan_preparation_started
    remove_variable = nesnd_sha_yan_contacts
    remove_variable = nesnd_sha_yan_council_pending
    remove_variable = nesnd_sha_yan_rose
    remove_variable = nesnd_sha_yan_partition
    remove_variable = nesnd_sha_yan_war_over
    remove_variable = nesnd_sha_yan_terms_settled
    remove_variable = nesnd_sha_yan_korean_exiles
}

nesnd_sha_enter_yan = {
    if = {
        limit = { nesnd_sha_yan_open = yes NOT = { has_journal_entry = nesnd_sha_restore_yan } }
        add_journal_entry = { type = nesnd_sha_restore_yan }
    }
}

nesnd_yan_establish_network = {
    if = {
        limit = {
            nesnd_sha_yan_preparing = yes
            exists = s:STATE_SONGJIANG.region_state:CNE
            exists = s:STATE_OUTER_MANCHURIA.region_state:CNE
        }
        set_variable = nesnd_yan_network_started
        if = {
            limit = { NOT = { is_target_in_variable_list = { name = nesnd_yan_contacts_by_state target = s:STATE_SONGJIANG.region_state:CNE } } }
            add_to_variable_list = { name = nesnd_yan_contacts_by_state target = s:STATE_SONGJIANG.region_state:CNE }
        }
        if = {
            limit = { NOT = { is_target_in_variable_list = { name = nesnd_yan_contacts_by_state target = s:STATE_OUTER_MANCHURIA.region_state:CNE } } }
            add_to_variable_list = { name = nesnd_yan_contacts_by_state target = s:STATE_OUTER_MANCHURIA.region_state:CNE }
        }
    }
}
nesnd_yan_contact_attempt = {
    if = {
        limit = { nesnd_yan_can_contact = { REGION = $REGION$ } }
        je:nesnd_sha_restore_yan = { add_progress = { name = nesnd_sha_yan_banner value = -50 } }
        set_variable = { name = nesnd_yan_contact_cooldown months = 1 }
        s:$REGION$.region_state:CNE = { save_scope_as = nesnd_yan_contact_target }
        random = {
            chance = nesnd_yan_contact_success
            add_to_variable_list = { name = nesnd_yan_contacts_by_state target = scope:nesnd_yan_contact_target }
            nesnd_sha_yan_send_hint = yes
        }
        if = {
            limit = { is_target_in_variable_list = { name = nesnd_yan_contacts_by_state target = scope:nesnd_yan_contact_target } }
            trigger_event = { id = nesnd_rising.3 popup = yes }
        }
        else = { trigger_event = { id = nesnd_rising.4 popup = yes } }
    }
}
nesnd_yan_publish_warning = {
    if = {
        limit = { nesnd_yan_can_agree = yes }
        set_variable = { name = nesnd_sha_yan_partition value = flag:$TERMS$ }
        set_variable = nesnd_yan_notice
        c:CNE = {
            every_scope_state = {
                limit = { nesnd_yan_contact_state = yes }
                owner = { add_to_variable_list = { name = nesnd_yan_states target = prev } }
            }
            set_variable = { name = nesnd_yan_count value = nesnd_yan_valid_state_count }
            add_journal_entry = { type = nesnd_yan_warning }
            je:nesnd_yan_warning = { save_scope_as = nesnd_yan_new_entry }
            c:SHA = { set_variable = { name = nesnd_yan_warning_entry value = scope:nesnd_yan_new_entry } }
            trigger_event = { id = nesnd_rising.2 popup = yes }
        }
    }
}
nesnd_yan_warning_clear = {
    clear_variable_list = nesnd_yan_states
    remove_variable = nesnd_yan_count
    remove_variable = nesnd_yan_interrupted
    c:SHA ?= { remove_variable = nesnd_yan_warning_entry }
}
nesnd_yan_stand_down = {
    c:SHA ?= {
        set_variable = { name = nesnd_yan_cooldown months = 3 }
        if = {
            limit = { has_journal_entry = nesnd_sha_restore_yan
                je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 95 "scripted_bar_progress(nesnd_sha_yan_banner)" < 100 }
            }
            set_variable = nesnd_yan_wait_for_full
        }
        remove_variable = nesnd_yan_notice
        remove_variable = nesnd_sha_yan_partition
    }
    nesnd_yan_warning_clear = yes
}
