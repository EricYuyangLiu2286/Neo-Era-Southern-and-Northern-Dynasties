# Shared A2 capacity helpers. Building levels already inherited count toward the national target.
nesnd_a2_barrack_capacity = {
 save_scope_as = nesnd_a2_capacity_country
 set_variable = { name = nesnd_a2_capacity_total value = 0 }
 if = { limit = { exists = scope:nesnd_a2_capacity_country has_variable = nesnd_a2_capacity_total }
  every_scope_state = {
   save_scope_as = nesnd_a2_capacity_state
   set_variable = { name = nesnd_a2_spare value = nesnd_a2_barrack_level }
   set_variable = { name = nesnd_a2_added_levels value = 0 }
   every_scope_building = {
    limit = { is_building_type = building_barrack }
    scope:nesnd_a2_capacity_country = { change_variable = { name = nesnd_a2_capacity_total add = prev.level } }
    every_combat_unit = {
     if = { limit = { exists = scope:nesnd_a2_capacity_state }
      scope:nesnd_a2_capacity_state = {
       if = { limit = { has_variable = nesnd_a2_spare } change_variable = { name = nesnd_a2_spare add = -1 } }
      }
     }
    }
   }
   if = { limit = { has_variable = nesnd_a2_spare }
    set_variable = { name = nesnd_a2_spare value = { value = var:nesnd_a2_spare min = 0 } }
   }
  }
  while = {
   count = 40 limit = { var:nesnd_a2_capacity_total < $TARGET$ }
   ordered_scope_state = {
    limit = { state_population > 0 nesnd_a2_added_levels_value < owner.$PER_STATE$ }
    order_by = nesnd_a2_barrack_build_priority max = 1
    if = { limit = { has_variable = nesnd_a2_added_levels has_variable = nesnd_a2_spare }
    create_building = { building = building_barrack level = nesnd_a2_barrack_next reserves = 1 }
    change_variable = { name = nesnd_a2_added_levels add = 1 }
    change_variable = { name = nesnd_a2_spare add = 1 }
    scope:nesnd_a2_capacity_country = { change_variable = { name = nesnd_a2_capacity_total add = 1 } }
    }
   }
  }
  remove_variable = nesnd_a2_capacity_total
  clear_saved_scope = nesnd_a2_capacity_state clear_saved_scope = nesnd_a2_capacity_country
 }
}
# COUNTRY scope. A single full formation is raised per selected state, largest real spare first.
nesnd_a2_raise_army = {
 if = { limit = { army_size < $TARGET$ }
  save_scope_as = nesnd_a2_army_country
  set_variable = { name = nesnd_a2_missing value = { value = $TARGET$ subtract = army_size min = 0 } }
  if = { limit = { exists = scope:nesnd_a2_army_country has_variable = nesnd_a2_missing }
   set_variable = { name = nesnd_a2_artillery_remaining value = { value = var:nesnd_a2_missing multiply = $ARTILLERY$ floor = yes } }
   nesnd_a2_barrack_capacity = { TARGET = $TARGET$ PER_STATE = $PER_STATE$ }
   ordered_scope_state = {
    limit = { state_population > 0 nesnd_a2_spare_value > 0 }
    order_by = nesnd_a2_spare_value max = $STATES$
    save_scope_as = nesnd_a2_allocation_state
    if = { limit = { exists = scope:nesnd_a2_army_country }
     scope:nesnd_a2_army_country = {
      if = { limit = { has_variable = nesnd_a2_missing has_variable = nesnd_a2_artillery_remaining exists = scope:nesnd_a2_allocation_state }
       if = { limit = { var:nesnd_a2_missing > 0 }
        set_variable = { name = nesnd_a2_allocation_total value = { value = var:nesnd_a2_missing max = scope:nesnd_a2_allocation_state.nesnd_a2_spare_value } }
        if = { limit = { has_variable = nesnd_a2_allocation_total }
         set_variable = { name = nesnd_a2_allocation_artillery value = { value = var:nesnd_a2_artillery_remaining max = var:nesnd_a2_allocation_total } }
         if = { limit = { has_variable = nesnd_a2_allocation_artillery }
          clear_saved_scope = nesnd_a2_new_army
          nesnd_a2_literal_state_army = yes
          if = { limit = { exists = scope:nesnd_a2_new_army }
           change_variable = { name = nesnd_a2_missing subtract = var:nesnd_a2_allocation_total }
           change_variable = { name = nesnd_a2_artillery_remaining subtract = var:nesnd_a2_allocation_artillery }
           scope:nesnd_a2_allocation_state = {
            change_variable = { name = nesnd_a2_spare subtract = scope:nesnd_a2_army_country.var:nesnd_a2_allocation_total }
           }
           nesnd_a2_assign_new_army_general = yes
          }
         }
        }
       }
      }
     }
    }
   }
   every_scope_state = { remove_variable = nesnd_a2_spare remove_variable = nesnd_a2_added_levels }
   remove_variable = nesnd_a2_missing remove_variable = nesnd_a2_artillery_remaining
   remove_variable = nesnd_a2_allocation_total remove_variable = nesnd_a2_allocation_artillery
   clear_saved_scope = nesnd_a2_new_army clear_saved_scope = nesnd_a2_allocation_state clear_saved_scope = nesnd_a2_army_country
  }
 }
}
nesnd_a2_assign_new_army_general = {
 if = { limit = { exists = scope:nesnd_a2_new_army }
  clear_saved_scope = nesnd_a2_new_general
  random_scope_character = { limit = { has_role_of_type = general has_military_formation = no } save_scope_as = nesnd_a2_new_general }
  if = { limit = { NOT = { exists = scope:nesnd_a2_new_general } }
   create_character = { culture = primary_culture is_general = yes interest_group = ig_armed_forces ideology = ideology_jingoist_leader on_created = { save_scope_as = nesnd_a2_new_general } }
  }
  if = { limit = { exists = scope:nesnd_a2_new_general }
   scope:nesnd_a2_new_general = { transfer_to_formation = scope:nesnd_a2_new_army }
  }
  clear_saved_scope = nesnd_a2_new_general
 }
}
