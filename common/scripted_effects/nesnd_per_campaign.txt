nesnd_per_initialize = {
	if = { limit = { nesnd_per_country = yes NOT = { has_variable = nesnd_per_campaign_v1 } }
		set_variable = nesnd_per_campaign_v1
	}
}
nesnd_per_ensure_frontier = {
	if = { limit = { nesnd_per_west_eligible = yes NOT = { has_journal_entry = nesnd_per_west_journal } NOT = { has_inactive_journal_entry = nesnd_per_west_journal } } add_journal_entry = { type = nesnd_per_west_journal } }
	if = { limit = { nesnd_per_east_eligible = yes NOT = { has_journal_entry = nesnd_per_east_journal } NOT = { has_inactive_journal_entry = nesnd_per_east_journal } } add_journal_entry = { type = nesnd_per_east_journal } }
}
nesnd_per_attempt_claim = {
    if = { limit = { nesnd_per_country = yes nesnd_per_claim_points >= 25 NOT = { has_variable = nesnd_per_claim_cooldown }
            trigger_if = { limit = { exists = scope:nesnd_per_claim_state exists = scope:nesnd_per_claim_region }
                scope:nesnd_per_claim_state = { nesnd_per_border_claim_state = yes }
            } trigger_else = { always = no }
        }
        change_variable = { name = nesnd_per_claim_points add = -25 }
        nesnd_per_claim_sync = yes
		set_variable = { name = nesnd_per_claim_cooldown months = 1 }
		add_modifier = { name = nesnd_per_claim_cost months = 1 }
		clear_saved_scope = nesnd_per_obstructor
		every_country = {
			limit = { OR = { AND = { exists = c:FRA c:FRA ?= this } AND = { exists = c:MUG c:MUG ?= this } AND = { exists = c:USM c:USM ?= this } AND = { exists = c:RUS c:RUS ?= this } AND = { exists = c:QE1 c:QE1 ?= this } country_rank >= rank_value:major_power country_rank = rank_value:unrecognized_major_power } }
			if = { limit = { NOT = { exists = scope:nesnd_per_obstructor } }
				save_scope_as = nesnd_per_potential_obstructor
				if = { limit = { nesnd_per_claim_obstructor = yes } save_scope_as = nesnd_per_obstructor }
			}
		}
		if = { limit = { exists = scope:nesnd_per_obstructor } trigger_event = { id = nesnd_per.7 popup = yes } }
		else = { scope:nesnd_per_claim_region = { add_claim = root } nesnd_per_claim_tension = yes trigger_event = { id = nesnd_per.6 popup = yes } }
	}
}
# 宣称牵动「波斯之忧」。向西（伊斯兰诸国）宣称 +2，向东（西秦方向）−1，一次性。
nesnd_per_claim_tension = {
    if = { limit = { nesnd_per_tension_live = yes }
        if = { limit = { nesnd_per_west_eligible = yes } set_variable = { name = nesnd_per_tension_once_claim value = 2 } }
        else_if = { limit = { nesnd_per_east_eligible = yes } set_variable = { name = nesnd_per_tension_once_claim value = -1 } }
    }
}

# 定方略（由「将军府的忠心」调用，只定一次）。aid：留在西秦一边，开「重启远西战争」。covert：与西秦分手，开「近西战争重启」。
nesnd_per_settle_route = {
	if = {
		limit = { nesnd_per_country = yes custom_tooltip = { text = nesnd_ui_not_per_future_done_tt NOT = { has_variable = nesnd_per_future_done } } }
		set_variable = nesnd_per_route_$ROUTE$
		set_variable = nesnd_per_future_done
		if = { limit = { custom_tooltip = { text = nesnd_ui_per_route_aid_tt has_variable = nesnd_per_route_aid } } trigger_event = { id = nesnd_per.4 popup = yes } }
		else = { trigger_event = { id = nesnd_per.5 popup = yes } }
		nesnd_per_ensure_frontier = yes
	}
}
