nesnd_per_resources_initialize = {
    if = { limit = { NOT = { has_variable = nesnd_per_governance_points } } set_variable = { name = nesnd_per_governance_points value = 0 } }
    if = { limit = { NOT = { has_variable = nesnd_per_claim_points } } set_variable = { name = nesnd_per_claim_points value = 0 } }
}
nesnd_per_governance_sync = {
    if = { limit = { has_journal_entry = nesnd_per_governance }
        je:nesnd_per_governance = { set_bar_progress = { name = nesnd_per_governance_bar value = owner.nesnd_per_governance_points } }
    }
}
nesnd_per_claim_sync = {
    if = { limit = { has_journal_entry = nesnd_per_west_journal } je:nesnd_per_west_journal = { set_bar_progress = { name = nesnd_per_claim_resource_bar value = owner.nesnd_per_claim_points } } }
    if = { limit = { has_journal_entry = nesnd_per_east_journal } je:nesnd_per_east_journal = { set_bar_progress = { name = nesnd_per_claim_resource_bar value = owner.nesnd_per_claim_points } } }
}
nesnd_per_governance_monthly = {
    if = { limit = { nesnd_per_country = yes has_journal_entry = nesnd_per_governance }
        nesnd_per_resources_initialize = yes
        change_variable = { name = nesnd_per_governance_points add = nesnd_per_governance_delta }
        clamp_variable = { name = nesnd_per_governance_points min = 0 max = 100 }
        nesnd_per_governance_sync = yes
    }
}
nesnd_per_claim_monthly = {
    if = { limit = { nesnd_per_country = yes OR = { nesnd_per_west_eligible = yes nesnd_per_east_eligible = yes } }
        nesnd_per_resources_initialize = yes
        change_variable = { name = nesnd_per_claim_points add = nesnd_per_claim_delta }
        clamp_variable = { name = nesnd_per_claim_points min = 0 max = 100 }
        nesnd_per_claim_sync = yes
    }
}
nesnd_per_governance_clear = {
    if = { limit = { has_modifier = nesnd_per_civil_grant } remove_modifier = nesnd_per_civil_grant }
    if = { limit = { has_modifier = nesnd_per_industry_grant } remove_modifier = nesnd_per_industry_grant }
    if = { limit = { has_modifier = nesnd_per_education_grant } remove_modifier = nesnd_per_education_grant }
    if = { limit = { has_modifier = nesnd_per_legislation_grant } remove_modifier = nesnd_per_legislation_grant }
}
