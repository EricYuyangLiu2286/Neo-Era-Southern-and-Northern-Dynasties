 nesnd_kr_find_center = yes
 if = { limit = { NOT = { exists = global_var:nesnd_kr_center } scope:kr_seed ?= { owner = prev nesnd_kr_inhabited = yes controller = owner } }
 create_country = { tag = KOR origin = this $SEED$ = scope:$SCOPE$
 on_created = { make_independent = yes add_primary_culture = cu:korean
 save_scope_as = kr_new_country every_primary_culture = { limit = { NOT = { has_discrimination_trait = heritage_korean } } scope:kr_new_country = { remove_primary_culture = prev } }
 if = { limit = { NOT = { exists = ruler } } create_character = { ruler = yes culture = cu:korean } } set_global_variable = { name = nesnd_kr_center value = this } trigger_event = { id = nesnd_kr.5 days = 1 popup = yes } } }
 }
}

nesnd_kr_launch = {
 if = { limit = { nesnd_kr_seed_live = yes }
 global_var:nesnd_kr_seed = { save_scope_as = kr_seed }
 global_var:nesnd_kr_target = {
 nesnd_kr_find_center = yes
 if = { limit = { NOT = { exists = global_var:nesnd_kr_center } }
 # Leave the defender a real country. If this is its final state use a province.
 if = { limit = { any_scope_state = { count >= 2 } } nesnd_kr_create_center = { SEED = state SCOPE = kr_seed } }
 else_if = { limit = { scope:kr_seed = { any_province = { count >= 2 } } }
 scope:kr_seed = { random_province = { save_scope_as = kr_province } }
 nesnd_kr_create_center = { SEED = province SCOPE = kr_province } }
 global_var:nesnd_kr_center ?= {
 every_scope_state = { set_variable = nesnd_kr_war_seed }
 if = { limit = { any_scope_state = { nesnd_kr_inhabited = yes } } nesnd_restoration_army = yes }
 else = { every_scope_state = { limit = { nesnd_kr_land = yes has_variable = nesnd_kr_war_seed } set_state_owner = global_var:nesnd_kr_target } }
 } }
 if = { limit = { global_var:nesnd_kr_center ?= { is_subject = no is_at_war = no is_active_in_diplomatic_play = no } }
 set_global_variable = { name = nesnd_kr_actor value = global_var:nesnd_kr_center }
 set_global_variable = { name = nesnd_kr_phase value = 3 }
 global_var:nesnd_kr_actor = { create_diplomatic_play = { name = dp_nesnd_kr_restoration type = dp_nesnd_kr_restoration target_country = global_var:nesnd_kr_target escalation = 50 } }
 random_diplomatic_play = { limit = { is_diplomatic_play_type = dp_nesnd_kr_restoration initiator ?= global_var:nesnd_kr_actor target ?= global_var:nesnd_kr_target }
 set_global_variable = { name = nesnd_kr_play value = this } }
 if = { limit = { NOT = { exists = global_var:nesnd_kr_play } }
 global_var:nesnd_kr_actor ?= { every_scope_state = { limit = { nesnd_kr_land = yes has_variable = nesnd_kr_war_seed } set_state_owner = global_var:nesnd_kr_target } }
 nesnd_kr_end_case = yes }
 }
 else_if = { limit = { NOT = { exists = global_var:nesnd_kr_center } } nesnd_kr_end_case = yes }
 } }
}
