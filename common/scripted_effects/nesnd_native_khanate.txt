nesnd_khanate_clear_selection = {
 every_scope_state = {
  if = { limit = { has_variable = nesnd_khanate_selected } remove_variable = nesnd_khanate_selected }
  if = { limit = { has_variable = nesnd_khanate_excluded } remove_variable = nesnd_khanate_excluded }
 }
 clear_saved_scope = nesnd_khanate_1 clear_saved_scope = nesnd_khanate_2 clear_saved_scope = nesnd_khanate_3 clear_saved_scope = nesnd_khanate_4
 clear_saved_scope = nesnd_khanate_capital
}
nesnd_khanate_choose_next = {
 random_scope_state = {
  limit = { nesnd_khanate_selected_neighbour = yes }
  set_variable = nesnd_khanate_selected save_scope_as = nesnd_khanate_$SLOT$
 }
}
nesnd_khanate_create = {
 if = { limit = {
  nesnd_hus_original_country = yes nesnd_khanate_trigger_ready = yes has_variable = nesnd_khanate_campaign_started NOT = { has_global_variable = nesnd_khanate_created }
  exists = c:USA NOT = { exists = c:NKH } NOT = { has_variable = nesnd_khanate_selection_wait }
 }
  set_variable = { name = nesnd_khanate_selection_wait months = 1 }
  save_scope_as = nesnd_khanate_original_hus
  c:USA = {
   nesnd_khanate_clear_selection = yes
   if = { limit = { any_scope_state = { count >= 5 state_population > 0 } }
    while = {
     count = 15 limit = { NOT = { exists = scope:nesnd_khanate_4 } any_scope_state = { nesnd_khanate_eligible_state = yes NOT = { has_variable = nesnd_khanate_excluded } } }
     random_scope_state = { limit = { nesnd_khanate_eligible_state = yes NOT = { has_variable = nesnd_khanate_excluded } } set_variable = nesnd_khanate_selected save_scope_as = nesnd_khanate_1 }
     nesnd_khanate_choose_next = { SLOT = 2 } nesnd_khanate_choose_next = { SLOT = 3 } nesnd_khanate_choose_next = { SLOT = 4 }
     if = { limit = { NOT = { exists = scope:nesnd_khanate_4 } }
      every_scope_state = { limit = { has_variable = nesnd_khanate_selected } set_variable = nesnd_khanate_excluded remove_variable = nesnd_khanate_selected }
      clear_saved_scope = nesnd_khanate_1 clear_saved_scope = nesnd_khanate_2 clear_saved_scope = nesnd_khanate_3
     }
    }
   }
   if = { limit = {
    exists = scope:nesnd_khanate_1 exists = scope:nesnd_khanate_2 exists = scope:nesnd_khanate_3 exists = scope:nesnd_khanate_4
    any_scope_state = { count = 4 has_variable = nesnd_khanate_selected nesnd_khanate_eligible_state = yes }
    any_scope_state = { state_population > 0 NOT = { has_variable = nesnd_khanate_selected } }
   }
    ordered_scope_state = { limit = { NOT = { has_variable = nesnd_khanate_selected } state_population > 0 } order_by = state_population max = 1 state_region = { save_scope_as = nesnd_khanate_usa_capital } }
    if = { limit = { capital = { has_variable = nesnd_khanate_selected } exists = scope:nesnd_khanate_usa_capital } nesnd_a2_set_capital = { TARGET = nesnd_khanate_usa_capital } }
    create_country = {
     tag = NKH origin = this
     state = scope:nesnd_khanate_1 state = scope:nesnd_khanate_2 state = scope:nesnd_khanate_3 state = scope:nesnd_khanate_4
     on_created = {
      set_variable = nesnd_native_khanate set_variable = nesnd_a2_khanate_adopted make_independent = yes
      set_variable = { name = nesnd_khanate_usa_epoch value = 1 } set_variable = { name = nesnd_khanate_hus_epoch value = 1 }
      set_variable = { name = nesnd_khanate_original_usa value = c:USA }
      set_variable = { name = nesnd_khanate_original_hus value = scope:nesnd_khanate_original_hus }
      ordered_scope_state = { order_by = state_population max = 1 state_region = { save_scope_as = nesnd_khanate_capital } }
      if = { limit = { exists = scope:nesnd_khanate_capital } nesnd_a2_set_capital = { TARGET = nesnd_khanate_capital } }
     }
    }
    if = { limit = { nesnd_khanate_exists = yes c:NKH = { any_scope_state = { count = 4 state_population > 0 } } }
     set_global_variable = nesnd_khanate_created
     c:NKH = {
      every_scope_state = { state_region = { add_claim = c:USA add_claim = scope:nesnd_khanate_original_hus } remove_variable = nesnd_khanate_selected }
      set_variable = { name = nesnd_khanate_usa_outcome value = 0 } set_variable = { name = nesnd_khanate_hus_outcome value = 0 }
      trigger_event = { id = nesnd_native_khanate.1 days = 1 }
     }
     nesnd_khanate_transfer_native_half = yes
     c:NKH = { save_scope_as = nesnd_a2_khanate }
     scope:nesnd_khanate_original_hus = { nesnd_khanate_transfer_native_half = yes trigger_event = { id = nesnd_hus_registration.1 popup = yes } }
    }
   }
   nesnd_khanate_clear_selection = yes clear_saved_scope = nesnd_khanate_usa_capital
  }
  clear_saved_scope = nesnd_khanate_original_hus
 }
}
nesnd_khanate_transfer_native_half = {
 if = { limit = { nesnd_khanate_exists = yes is_country_alive = yes NOT = { c:NKH ?= this } NOT = { has_variable = nesnd_khanate_population_sent } }
  set_variable = nesnd_khanate_population_sent
  every_scope_state = { every_scope_pop = {
   limit = { culture = { nesnd_hus_native_culture = yes } }
   clear_saved_scope = nesnd_khanate_destination
   c:NKH = { random_scope_state = { limit = { state_population > 0 } save_scope_as = nesnd_khanate_destination } }
   if = { limit = { exists = scope:nesnd_khanate_destination scope:nesnd_khanate_destination = { owner = c:NKH } }
    move_partial_pop = { state = scope:nesnd_khanate_destination population_ratio = 0.5 }
   }
  } }
  clear_saved_scope = nesnd_khanate_destination
 }
}

 every_scope_state = { every_scope_building = {
  limit = { is_building_type = $BUILDING$ }
  scope:nesnd_a2_facility_country = { change_variable = { name = nesnd_a2_facility_total add = prev.level } }
 } }
 while = {
  count = 4 limit = { var:nesnd_a2_facility_total < $CAP$ }
  random_scope_state = {
   limit = { NOT = { any_scope_building = { is_building_type = $BUILDING$ } } }
   create_building = { building = $BUILDING$ level = 1 reserves = 1 }
   scope:nesnd_a2_facility_country = { change_variable = { name = nesnd_a2_facility_total add = 1 } }
  }
 }
 remove_variable = nesnd_a2_facility_total clear_saved_scope = nesnd_a2_facility_country
 }
}

nesnd_native_primary_cultures_only = {
    if = { limit = { OR = { c:NKH ?= this has_variable = nesnd_hus_native_successor } }
        save_scope_as = nesnd_native_culture_country
        # Add a fallback before removing anything: a country must retain a primary culture.
        if = { limit = { NOT = { any_primary_culture = { nesnd_hus_native_culture = yes } } } add_primary_culture = cu:siouan }
        every_scope_culture = {
            limit = { nesnd_hus_native_culture = yes }
            scope:nesnd_native_culture_country = { add_primary_culture = prev }
        }
        every_primary_culture = {
            limit = { NOT = { nesnd_hus_native_culture = yes } }
            scope:nesnd_native_culture_country = { remove_primary_culture = prev }
        }
        clear_saved_scope = nesnd_native_culture_country
    }
}
nesnd_khanate_start_war = {
 if = { limit = { nesnd_khanate_exists = yes }
 if = { limit = {
  nesnd_khanate_original_opponent = yes is_country_alive = yes nesnd_khanate_exists = yes c:NKH = { has_variable = nesnd_khanate_prepared has_variable = nesnd_khanate_usa_epoch has_variable = nesnd_khanate_hus_epoch NOT = { has_variable = nesnd_khanate_legacy_invalid } }
  NOT = { has_variable = nesnd_khanate_war_outcome } NOT = { has_variable = nesnd_khanate_war_dispatched } NOT = { has_variable = nesnd_khanate_retry_wait }
 }
  if = { limit = { NOT = { has_variable = nesnd_khanate_war_attempts } } set_variable = { name = nesnd_khanate_war_attempts value = 0 } }
  change_variable = { name = nesnd_khanate_war_attempts add = 1 }
  set_variable = { name = nesnd_khanate_retry_wait months = 1 }
  save_scope_as = nesnd_khanate_opponent
  if = { limit = { NOT = { has_war_with = c:NKH } NOT = { nesnd_khanate_has_play = yes } }
   end_truce = c:NKH
   every_scope_treaty = {
    limit = { binds = scope:nesnd_khanate_opponent binds = c:NKH any_scope_article = { OR = { has_type = alliance has_type = defensive_pact } } }
    withdraw = { country = scope:nesnd_khanate_opponent apply_break_penalties = yes }
    scope:nesnd_khanate_opponent = { set_variable = nesnd_khanate_withdrew_treaty }
   }
   if = { limit = { has_variable = nesnd_khanate_withdrew_treaty }
    remove_variable = nesnd_khanate_withdrew_treaty
    trigger_event = { id = nesnd_native_khanate.4 popup = yes }
    c:NKH = { trigger_event = { id = nesnd_native_khanate.4 popup = yes } }
   }
   set_variable = nesnd_khanate_dispatch_created
   set_variable = nesnd_khanate_creation_open
   set_variable = { name = nesnd_khanate_creation_case value = c:NKH }
   if = { limit = { c:USA ?= this } set_variable = { name = nesnd_khanate_creation_epoch value = c:NKH.var:nesnd_khanate_usa_epoch } }
   else = { set_variable = { name = nesnd_khanate_creation_epoch value = c:NKH.var:nesnd_khanate_hus_epoch } }
   create_diplomatic_play = {
    name = nesnd_khanate_war type = dp_nesnd_khanate_suppression target_country = c:NKH war = yes handle_annexation_as_civil_war = no
    add_war_goal = { holder = c:NKH type = wg_nesnd_khanate_humiliation target_country = scope:nesnd_khanate_opponent }
   }
   remove_variable = nesnd_khanate_creation_open remove_variable = nesnd_khanate_creation_case remove_variable = nesnd_khanate_creation_epoch
  }
  if = { limit = { has_variable = nesnd_khanate_dispatch_created }
  if = { limit = { any_diplomatic_play = { count = 1 is_diplomatic_play_type = dp_nesnd_khanate_suppression initiator = scope:nesnd_khanate_opponent target = c:NKH } }
  random_diplomatic_play = {
   limit = { is_diplomatic_play_type = dp_nesnd_khanate_suppression initiator = scope:nesnd_khanate_opponent target = c:NKH
    initiator = { play_participant_has_war_goal_of_type_against = { type = wg_nesnd_khanate_annex target = c:NKH } }
    target = { play_participant_has_war_goal_of_type_against = { type = wg_nesnd_khanate_humiliation target = scope:nesnd_khanate_opponent } }
   }
   save_scope_as = nesnd_khanate_play
   nesnd_khanate_bind_play = yes
  }
  }
   remove_variable = nesnd_khanate_dispatch_created
  }
  if = { limit = { NOT = { has_variable = nesnd_khanate_war_dispatched } var:nesnd_khanate_war_attempts >= 12 }
   set_variable = { name = nesnd_khanate_war_outcome value = 4 }
  }
  clear_saved_scope = nesnd_khanate_play clear_saved_scope = nesnd_khanate_opponent
 }
 }
}
nesnd_khanate_war_watch = {
 if = { limit = { has_variable = nesnd_khanate_war_dispatched NOT = { has_variable = nesnd_khanate_war_outcome } NOT = { nesnd_khanate_has_play = yes } }
  if = { limit = { nesnd_khanate_exists = yes } set_variable = { name = nesnd_khanate_war_outcome value = 3 } }
  else = { set_variable = { name = nesnd_khanate_war_outcome value = 5 } }
 }
}
nesnd_khanate_check_survival = {
 if = { limit = { c:NKH ?= this has_variable = nesnd_native_khanate NOT = { has_variable = nesnd_khanate_survived_wars } }
  if = { limit = { NOT = { var:nesnd_khanate_original_usa ?= { c:USA ?= this is_country_alive = yes } } } set_variable = { name = nesnd_khanate_usa_outcome value = 5 } }
  else_if = { limit = { var:nesnd_khanate_original_usa = { has_variable = nesnd_khanate_war_outcome } } set_variable = { name = nesnd_khanate_usa_outcome value = var:nesnd_khanate_original_usa.var:nesnd_khanate_war_outcome } }
  if = { limit = { NOT = { var:nesnd_khanate_original_hus ?= { nesnd_hus_original_country = yes } } } set_variable = { name = nesnd_khanate_hus_outcome value = 5 } }
  else_if = { limit = { var:nesnd_khanate_original_hus = { has_variable = nesnd_khanate_war_outcome } } set_variable = { name = nesnd_khanate_hus_outcome value = var:nesnd_khanate_original_hus.var:nesnd_khanate_war_outcome } }
  if = { limit = { var:nesnd_khanate_usa_outcome > 0 var:nesnd_khanate_hus_outcome > 0 }
   set_variable = nesnd_khanate_survived_wars trigger_event = { id = nesnd_native_khanate.2 popup = yes }
  }
 }
}
