nesnd_qe1_initialize = {
	if = { limit = { nesnd_qe1_country = yes }
		if = { limit = { NOT = { has_variable = nesnd_qe1_campaign_v1 } }
			set_variable = nesnd_qe1_campaign_v1
			set_variable = { name = nesnd_qe1_barracks_goal value = 107 }
			set_variable = nesnd_qe1_barracks_snapshot_pending
			trigger_event = { id = nesnd_qe1.90 days = 1 popup = no }
			set_variable = { name = nesnd_qe1_unity value = 70 }
			set_variable = { name = nesnd_qe1_war_months value = 0 }
			set_variable = nesnd_qe1_ruler_economic
			if = { limit = { exists = ruler } ruler = { nesnd_qe1_assign_ruler = yes } }
		}
	}
}
nesnd_qe1_assign_ruler = {
	if = { limit = { exists = owner owner = { nesnd_qe1_country = yes } }
		if = { limit = { owner = { country_has_monarchy_law = yes } }

			if = { limit = { NOT = { has_variable = nesnd_qe1_ruler_registered } }
				set_variable = nesnd_qe1_ruler_registered
				if = { limit = { has_template = QE1_chen_wuxing }
					set_variable = nesnd_qe1_person_economic
				} else = { random_list = { 60 = { set_variable = nesnd_qe1_person_economic } 40 = { set_variable = nesnd_qe1_person_military } } }
			}
			owner = {
				remove_variable = nesnd_qe1_ruler_economic remove_variable = nesnd_qe1_ruler_military
				if = { limit = { prev = { has_variable = nesnd_qe1_person_economic } } set_variable = nesnd_qe1_ruler_economic }
				else = { set_variable = nesnd_qe1_ruler_military }
			}

		} else = { owner = { remove_variable = nesnd_qe1_ruler_economic remove_variable = nesnd_qe1_ruler_military } }
	}
}
nesnd_qe1_ensure_agenda_children = {
	if = { limit = { nesnd_qe1_agenda_live = yes }
		if = { limit = { NOT = { has_journal_entry = nesnd_qe1_economy_journal } } if = { limit = { NOR = { has_journal_entry = nesnd_qe1_economy_journal has_variable = nesnd_jeq_nesnd_qe1_economy_journal } } set_variable = { name = nesnd_jeq_nesnd_qe1_economy_journal days = 3 } add_journal_entry = { type = nesnd_qe1_economy_journal } } }
		if = { limit = { NOT = { has_journal_entry = nesnd_qe1_army_journal } } if = { limit = { NOR = { has_journal_entry = nesnd_qe1_army_journal has_variable = nesnd_jeq_nesnd_qe1_army_journal } } set_variable = { name = nesnd_jeq_nesnd_qe1_army_journal days = 3 } add_journal_entry = { type = nesnd_qe1_army_journal } } }
	}
}
# 势力平衡点就是进度条 nesnd_qe1_balance_bar 的读数，由条的 monthly_progress 逐月累加，按钮用 add_progress 推动；读它用 nesnd_qe1_balance_at_least / _at_most。
nesnd_qe1_watch_endpoints = {
	if = { limit = { nesnd_qe1_agenda_live = yes }
		if = { limit = { nesnd_qe1_balance_at_most = { V = -50 } }
			if = { limit = { NOT = { has_variable = nesnd_qe1_econ_edge } } set_variable = nesnd_qe1_econ_edge set_variable = { name = nesnd_qe1_econ_wait days = 29 } }
		} else = { remove_variable = nesnd_qe1_econ_edge remove_variable = nesnd_qe1_econ_wait }
		if = { limit = { nesnd_qe1_balance_at_least = { V = 50 } }
			if = { limit = { NOT = { has_variable = nesnd_qe1_army_edge } } set_variable = nesnd_qe1_army_edge set_variable = { name = nesnd_qe1_army_wait days = 29 } }
		} else = { remove_variable = nesnd_qe1_army_edge remove_variable = nesnd_qe1_army_wait }
	}
}
nesnd_qe1_agenda_monthly = {
	if = { limit = { nesnd_qe1_agenda_live = yes }
		nesnd_qe1_watch_endpoints = yes
		nesnd_qe1_check_civil_conflict = yes
	}
}
nesnd_qe1_close_agenda = {
	set_variable = nesnd_qe1_agenda_done
	nesnd_qe1_unlock_later_journals = yes
	remove_variable = nesnd_qe1_econ_edge remove_variable = nesnd_qe1_econ_wait
	remove_variable = nesnd_qe1_army_edge remove_variable = nesnd_qe1_army_wait
	if = { limit = { has_modifier = nesnd_qe1_law_push } remove_modifier = nesnd_qe1_law_push } if = { limit = { has_modifier = nesnd_qe1_economy_push } remove_modifier = nesnd_qe1_economy_push } if = { limit = { has_modifier = nesnd_qe1_military_preparations } remove_modifier = nesnd_qe1_military_preparations }
}
nesnd_qe1_settle_agenda = {
	if = { limit = { nesnd_qe1_economic_victory = yes }
		add_modifier = { name = nesnd_qe1_angovernment years = 10 }
		nesnd_qe1_start_sol_observation = yes
		set_variable = nesnd_qe1_economic_winner
		trigger_event = { id = nesnd_qe1.1 popup = yes }
	} else_if = { limit = { nesnd_qe1_military_victory = yes }
		add_modifier = { name = nesnd_qe1_military_government years = 10 }
		set_variable = nesnd_qe1_military_winner
		trigger_event = { id = nesnd_qe1.2 popup = yes }
	} else = { custom_tooltip = nesnd_journal_effect_revealed_tt }
	if = { limit = { OR = { nesnd_qe1_economic_victory = yes nesnd_qe1_military_victory = yes } } nesnd_qe1_close_agenda = yes }
}
nesnd_qe1_agenda_timeout = {
	if = { limit = { num_states < 4 } nesnd_qe1_close_agenda = yes }
	else_if = { limit = { OR = { nesnd_qe1_economic_victory = yes nesnd_qe1_military_victory = yes } } nesnd_qe1_settle_agenda = yes }
	else = {
		nesnd_qe1_close_agenda = yes
		add_modifier = { name = nesnd_qe1_stalled years = 10 }
		nesnd_qe1_start_sol_observation = yes
		trigger_event = { id = nesnd_qe1.3 popup = yes }
	}
}


nesnd_qe1_capture_barracks = {
	if = { limit = { nesnd_qe1_country = yes has_variable = nesnd_qe1_barracks_snapshot_pending NOT = { has_variable = nesnd_qe1_barracks_snapshot_done } OR = { game_date > 1836.1.1 has_variable = nesnd_qe1_runtime_ready } }
		set_variable = { name = nesnd_qe1_barracks_base value = nesnd_qe1_barracks_total }
		set_variable = { name = nesnd_qe1_barracks_goal value = { value = var:nesnd_qe1_barracks_base add = 25 } }
		set_variable = nesnd_qe1_barracks_snapshot_done
		remove_variable = nesnd_qe1_barracks_snapshot_pending
	}
}

nesnd_qe1_start_agenda_clock = {
	if = { limit = { NOT = { has_variable = nesnd_qe1_agenda_started } NOT = { has_variable = nesnd_qe1_agenda_done } }
		set_variable = nesnd_qe1_agenda_started set_variable = { name = nesnd_qe1_agenda_clock years = 10 }
	}
}
