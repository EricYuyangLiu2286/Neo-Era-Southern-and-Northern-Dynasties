# 州作用域：移除本日志挂在州上的全部修正。州换到非日本国家手里时调用。
nesnd_jp_clear_korean_state = {
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_samurai_rule
        }
        remove_modifier = nesnd_jp_korea_samurai_rule
    }
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_governor
        }
        remove_modifier = nesnd_jp_korea_governor
    }
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_councils
        }
        remove_modifier = nesnd_jp_korea_councils
    }
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_model_zone
        }
        remove_modifier = nesnd_jp_korea_model_zone
    }
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_daikan_autonomy
        }
        remove_modifier = nesnd_jp_korea_daikan_autonomy
    }
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_independence_activity
        }
        remove_modifier = nesnd_jp_korea_independence_activity
    }
}

# 国家作用域：任何结局与交还时收尾。移除国家修正 administration，以及本国各州上的 samurai_rule、governor、councils。
nesnd_jp_close_korean_governance = {
    if = {
        limit = {
            has_modifier = nesnd_jp_korea_administration
        }
        remove_modifier = nesnd_jp_korea_administration
    }
    every_scope_state = {
        limit = {
            OR = {
                has_modifier = nesnd_jp_korea_samurai_rule
                has_modifier = nesnd_jp_korea_governor
                has_modifier = nesnd_jp_korea_councils
            }
        }
        if = {
            limit = {
                has_modifier = nesnd_jp_korea_samurai_rule
            }
            remove_modifier = nesnd_jp_korea_samurai_rule
        }
        if = {
            limit = {
                has_modifier = nesnd_jp_korea_governor
            }
            remove_modifier = nesnd_jp_korea_governor
        }
        if = {
            limit = {
                has_modifier = nesnd_jp_korea_councils
            }
            remove_modifier = nesnd_jp_korea_councils
        }
    }
}

nesnd_jp_korean_settlement = {
    if = {
        limit = {
            NOT = {
                has_global_variable = nesnd_kr_open
            }
            exists = scope:jp_korea
            exists = scope:jp_sender
        }
        if = {
            limit = {
                scope:jp_korea = {
                    is_country_alive = yes
                }
                is_subject = no
            }
            set_variable = nesnd_jp_korea_settled
            nesnd_jp_close_korean_governance = yes
            every_scope_state = {
                limit = {
                    nesnd_jp_korea_southern_state = yes
                }
                set_state_owner = scope:jp_korea
            }
            s:STATE_BUSAN = {
                remove_claim = scope:jp_sender
            }
            s:STATE_YANGHO = {
                remove_claim = scope:jp_sender
            }
            if = {
                limit = {
                    NOT = {
                        scope:jp_korea = scope:jp_partner
                    }
                }
                change_relations = {
                    country = scope:jp_korea
                    value = 25
                }
            }
        }
    }
}
