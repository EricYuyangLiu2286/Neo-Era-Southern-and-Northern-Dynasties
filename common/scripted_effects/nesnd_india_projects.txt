# Local projects of the Indian princes. See common/scripted_triggers/nesnd_india_projects.txt.
#
# Order of a project paid for by another government:
#   the prince chooses project, state and funder (events .1 to .3)
#   -> nesnd_india_project_send        a matter of kind 1 (nesnd_india_matter_send): the overlord first
#                                      if it must agree, then the funder, both in the journal nesnd_india_foreign_affairs
#   -> the funder agrees: nesnd_india_project_accept
#   -> money moves, the building is ordered or created, the agreement is recorded on the state region.

# ---------------------------------------------------------------------------------------------
# The prince (requester scope)
# ---------------------------------------------------------------------------------------------
nesnd_india_project_clear = {
    remove_variable = nesnd_india_project_type remove_variable = nesnd_india_project_site
    remove_variable = nesnd_india_project_base remove_variable = nesnd_india_project_choice
    nesnd_india_matter_clear = yes
}

# TYPE is 1 to 3, SITE the matching state trigger. Offer up to three states, largest first.
nesnd_india_project_pick = {
    nesnd_india_project_clear = yes
    set_variable = { name = nesnd_india_matter_kind value = 1 }
    set_variable = { name = nesnd_india_project_type value = $TYPE$ }
    set_variable = { name = nesnd_india_project_shown value = $TYPE$ }
    set_variable = { name = nesnd_india_project_choice days = 30 }
    clear_saved_scope = india_project_site_1
    clear_saved_scope = india_project_site_2
    clear_saved_scope = india_project_site_3
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 0 check_range_bounds = no save_scope_as = india_project_site_1 }
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 1 check_range_bounds = no save_scope_as = india_project_site_2 }
    ordered_scope_state = { limit = { $SITE$ = yes } order_by = state_population position = 2 check_range_bounds = no save_scope_as = india_project_site_3 }
    if = { limit = { any_scope_state = { $SITE$ = yes count >= 2 } } trigger_event = { id = nesnd_india_project.2 popup = yes } }
    else_if = { limit = { exists = scope:india_project_site_1 } nesnd_india_project_set_site = { SITE = scope:india_project_site_1 } }
    else = { nesnd_india_project_clear = yes }
}
nesnd_india_project_pick_survey = {
    nesnd_india_project_clear = yes
    set_variable = { name = nesnd_india_matter_kind value = 1 }
    set_variable = { name = nesnd_india_project_type value = 4 }
    set_variable = { name = nesnd_india_project_shown value = 4 }
    set_variable = { name = nesnd_india_project_choice days = 30 }
    nesnd_india_project_open_funding = yes
}
nesnd_india_project_set_site = {
    if = { limit = { has_variable = nesnd_india_project_type exists = $SITE$ }
        set_variable = { name = nesnd_india_project_site value = $SITE$ }
        nesnd_india_project_open_funding = yes
    }
}
# Find the richest Indian government and the richest foreign government that may be asked.
# A subject's only foreign partner is its overlord, as with missions.
nesnd_india_project_open_funding = {
    save_scope_as = india_project_owner
    clear_saved_scope = india_project_neighbour
    clear_saved_scope = india_project_foreign
    clear_saved_scope = india_project_overlord
    if = { limit = { is_subject = yes exists = overlord } overlord = { save_scope_as = india_project_overlord } }
    ordered_country = { limit = { nesnd_india_project_neighbour = { OWNER = scope:india_project_owner } } order_by = gdp position = 0 check_range_bounds = no save_scope_as = india_project_neighbour }
    ordered_country = { limit = { nesnd_india_project_foreigner = { OWNER = scope:india_project_owner } } order_by = gdp position = 0 check_range_bounds = no save_scope_as = india_project_foreign }
    trigger_event = { id = nesnd_india_project.3 popup = yes }
}

# Order one level at the chosen state through the government's own construction queue.
nesnd_india_project_build = {
    if = { limit = { nesnd_india_project_site_valid = yes exists = var:nesnd_india_project_site }
        if = { limit = { nesnd_india_project_type_value = 1 }
            set_variable = { name = nesnd_india_project_base value = var:nesnd_india_project_site.nesnd_india_project_level_1 }
            var:nesnd_india_project_site = { start_building_construction = building_university }
        }
        else_if = { limit = { nesnd_india_project_type_value = 2 }
            set_variable = { name = nesnd_india_project_base value = var:nesnd_india_project_site.nesnd_india_project_level_2 }
            var:nesnd_india_project_site = { start_building_construction = building_tooling_workshop }
        }
        else_if = { limit = { nesnd_india_project_type_value = 3 }
            set_variable = { name = nesnd_india_project_base value = var:nesnd_india_project_site.nesnd_india_project_level_3 }
            var:nesnd_india_project_site = { start_building_construction = building_railway }
        }
        set_variable = { name = nesnd_india_matter_kind value = 1 }
        set_variable = { name = nesnd_india_matter_stage value = 4 }
        remove_variable = nesnd_india_matter_clock
        remove_variable = nesnd_india_project_choice
    }
    else = { nesnd_india_project_clear = yes }
}

# FUNDER is asked. Where the overlord must agree, the overlord is asked first.
nesnd_india_project_send = {
    if = { limit = { has_variable = nesnd_india_project_type nesnd_india_project_can_ask = { FUNDER = $FUNDER$ } }
        remove_variable = nesnd_india_project_choice
        nesnd_india_matter_send = { KIND = 1 TO = $FUNDER$ AMOUNT = nesnd_india_project_ask_amount }
    }
}

# The site, kept for the report before the project is cleared.
nesnd_india_project_save_names = {
    clear_saved_scope = india_project_site
    if = { limit = { exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { save_scope_as = india_project_site } }
}
nesnd_india_project_withdraw = {
    if = { limit = { nesnd_india_matter_waiting = yes } nesnd_india_project_clear = yes }
}

# Monthly, from nesnd_india_refresh. A request waiting for an answer is checked by nesnd_india_matter_maintenance.
nesnd_india_project_maintenance = {
    if = { limit = { has_variable = nesnd_india_project_type }
        # a choice left open
        if = { limit = { nesnd_india_matter_stage_value = 0 NOT = { has_variable = nesnd_india_project_choice } } nesnd_india_project_clear = yes }
        else_if = { limit = { nesnd_india_matter_stage_value = 4 }
            if = { limit = { nesnd_india_project_built = yes }
                nesnd_india_project_save_names = yes
                trigger_event = { id = nesnd_india_project.6 popup = yes }
                nesnd_india_project_clear = yes
            }
            else_if = { limit = { NOT = { nesnd_india_project_site_owned = yes } } nesnd_india_project_clear = yes }
        }
    }
    # five-year agreements that have run out
    if = { limit = { any_scope_state = { nesnd_india_deal_lapsed = yes } }
        save_scope_as = india_project_owner
        every_scope_state = {
            limit = { nesnd_india_deal_lapsed = yes }
            save_scope_as = india_project_site
            state_region = {
                clear_saved_scope = india_matter_to
                if = { limit = { exists = var:nesnd_india_deal_funder } var:nesnd_india_deal_funder = { save_scope_as = india_matter_to } }
                if = { limit = { exists = var:nesnd_india_deal_host exists = scope:india_matter_to }
                    if = { limit = { var:nesnd_india_deal_host = scope:india_project_owner } scope:india_project_owner = { trigger_event = { id = nesnd_india_project.7 popup = yes } } }
                }
                nesnd_india_deal_forget = yes
            }
        }
    }
}

# State region scope.
nesnd_india_deal_forget = {
    remove_variable = nesnd_india_deal_host remove_variable = nesnd_india_deal_funder
    remove_variable = nesnd_india_deal_permit remove_variable = nesnd_india_deal_quote
    remove_variable = nesnd_india_deal_kind remove_variable = nesnd_india_deal_term
}
# State scope, with scope:india_project_owner and scope:india_project_payer saved.
# Who pays, who benefits, who agreed, how much, and five years during which the site takes no other agreement.
nesnd_india_deal_record = {
    state_region = {
        nesnd_india_deal_forget = yes
        set_variable = { name = nesnd_india_deal_host value = scope:india_project_owner }
        set_variable = { name = nesnd_india_deal_funder value = scope:india_project_payer }
        set_variable = { name = nesnd_india_deal_quote value = scope:india_project_owner.nesnd_india_matter_amount_value }
        set_variable = { name = nesnd_india_deal_kind value = scope:india_project_owner.nesnd_india_project_type_value }
        if = { limit = { exists = scope:india_project_owner.var:nesnd_india_matter_ok_own } set_variable = { name = nesnd_india_deal_permit value = scope:india_project_owner.var:nesnd_india_matter_ok_own } }
        set_variable = { name = nesnd_india_deal_term days = 1825 }
    }
}

# ---------------------------------------------------------------------------------------------
# The funder agrees (answering scope, from nesnd_india_matter_accept)
# ---------------------------------------------------------------------------------------------
# For a survey the prince pays the fee; otherwise the funder pays the grant, the prince owes an
# obligation, and the agreement is recorded on the state region.
nesnd_india_project_accept = {
    save_scope_as = india_project_payer
    var:nesnd_india_matter_from = {
        save_scope_as = india_project_owner
        nesnd_india_matter_save_names = yes
        if = { limit = { nesnd_india_project_type_value = 4 }
            add_treasury = { value = nesnd_india_matter_amount_value multiply = -1 }
            scope:india_project_payer = { add_treasury = scope:india_project_owner.nesnd_india_matter_amount_value }
            if = { limit = { has_modifier = nesnd_india_survey } remove_modifier = nesnd_india_survey }
            add_modifier = { name = nesnd_india_survey months = 24 }
        }
        else = {
            add_treasury = nesnd_india_matter_amount_value
            scope:india_project_payer = { add_treasury = { value = scope:india_project_owner.nesnd_india_matter_amount_value multiply = -1 } }
            set_owes_obligation_to = { country = scope:india_project_payer setting = yes }
            change_relations = { country = scope:india_project_payer value = 25 }
            if = { limit = { exists = scope:india_project_site }
                scope:india_project_site = {
                    nesnd_india_deal_record = yes
                    if = { limit = { scope:india_project_owner.nesnd_india_project_type_value = 2 }
                        create_building = { building = building_tooling_workshop add_ownership = { country = { country = scope:india_project_payer levels = 1 } } }
                    }
                    else_if = { limit = { scope:india_project_owner.nesnd_india_project_type_value = 3 }
                        create_building = { building = building_railway add_ownership = { country = { country = scope:india_project_payer levels = 1 } } }
                    }
                }
            }
        }
        trigger_event = { id = nesnd_india_project.4 popup = yes }
        # A university is built by the prince with the grant; a workshop or railway already stands.
        if = { limit = { nesnd_india_project_type_value = 1 } nesnd_india_project_build = yes }
        else = { nesnd_india_project_clear = yes }
    }
}

# What the prince is shown before asking FUNDER. Tooltips only.
nesnd_india_project_ask_tooltips = {
    if = { limit = { nesnd_india_project_type_value = 4 } custom_tooltip = nesnd_india_project_ask_survey_tt show_as_tooltip = { add_modifier = { name = nesnd_india_survey months = 24 } } }
    else_if = { limit = { nesnd_india_project_type_value = 1 } custom_tooltip = nesnd_india_project_ask_grant_tt }
    else = { custom_tooltip = nesnd_india_project_ask_stake_tt }
    if = { limit = { nesnd_india_project_consent_needed = { FUNDER = $FUNDER$ } } custom_tooltip = nesnd_india_project_consent_tt }
    custom_tooltip = nesnd_india_project_wait_tt
}
