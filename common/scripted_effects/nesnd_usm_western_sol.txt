# Monthly occupation-cohort estimate, not a hook into each native SoL change.
# The native impact function and cohort sums are already used by the Qin campaign.
nesnd_usm_western_sample_cohort = {
    set_variable = { name = nesnd_west_sol_n value = $N$ }
    if = { limit = { var:nesnd_west_sol_n > 0 }
        set_variable = { name = nesnd_west_sol_current value = { value = $S$ divide = var:nesnd_west_sol_n } }
        set_variable = { name = nesnd_west_sol_factor value = { value = $F$ divide = var:nesnd_west_sol_n } }
        if = { limit = { has_variable = $PREVIOUS$ }
            if = { limit = { var:nesnd_west_sol_current < var:$PREVIOUS$ }
                add_radicals_in_state = {
                    pop_type = pop_type:$TYPE$
                    value = { value = var:$PREVIOUS$ subtract = var:nesnd_west_sol_current multiply = var:nesnd_west_sol_factor multiply = 0.10 min = 0 max = 0.05 }
                }
            }
        }
        set_variable = { name = $PREVIOUS$ value = var:nesnd_west_sol_current days = 40 }
    }
    else = { remove_variable = $PREVIOUS$ }
}
