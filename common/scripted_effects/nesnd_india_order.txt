# The journal nesnd_india_order. See common/scripted_triggers/nesnd_india_order.txt for the three
# matters and common/script_values/nesnd_india_order.txt for the variables.
#
# Order of things:
#   nesnd_india_order_choose     a button or the AI settles a matter one way; what this journal built
#                                for the other way is cleared first
#   nesnd_india_order_apply      monthly and after each choice: Mughal's own modifiers, then each Indian
#                                subject gets the treaties and modifiers it does not have yet
#   nesnd_india_order_leave      a subject that left (or whose Mughal no longer holds the journal) is
#                                cleared from its own monthly refresh
#   nesnd_india_order_end        Mughal no longer holds the journal: everything this journal set is cleared
# Only treaties marked on the subject (nesnd_india_order_*_pact) are ever removed; treaties of the same
# kind made by the player or by other scripts are left alone.

# Any country, monthly from nesnd_india_refresh.
nesnd_india_order_upkeep = {
    if = { limit = { nesnd_india_order_left = yes } nesnd_india_order_leave = yes }
    if = { limit = { nesnd_india_order_live = yes }
        if = { limit = { NOT = { has_journal_entry = nesnd_india_order } } add_journal_entry = { type = nesnd_india_order } }
        nesnd_india_order_apply = yes
    }
    else_if = {
        limit = { OR = { has_variable = nesnd_india_order_tax has_variable = nesnd_india_order_army has_variable = nesnd_india_order_school } }
        nesnd_india_order_end = yes
    }
}

# Mughal scope.
nesnd_india_order_apply = {
    save_scope_as = india_order_lord
    nesnd_india_order_home = yes
    every_subject_or_below = {
        limit = { nesnd_india_order_member = { LORD = scope:india_order_lord } }
        save_scope_as = india_order_member
        # Marks left by another Mughal are cleared before this one's are set.
        if = { limit = { exists = var:nesnd_india_order_lord NOT = { var:nesnd_india_order_lord = scope:india_order_lord } } nesnd_india_order_leave = yes }
        set_variable = { name = nesnd_india_order_lord value = scope:india_order_lord }
        nesnd_india_order_build = yes
    }
}

# Mughal scope. Its own modifiers follow its choices.
nesnd_india_order_home = {
    if = { limit = { nesnd_india_order_is = { ISSUE = tax END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_tax_central } } add_modifier = { name = nesnd_india_order_tax_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_tax_central } remove_modifier = nesnd_india_order_tax_central }
    if = { limit = { nesnd_india_order_is = { ISSUE = army END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_army_central } } add_modifier = { name = nesnd_india_order_army_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_army_central } remove_modifier = nesnd_india_order_army_central }
    if = { limit = { nesnd_india_order_is = { ISSUE = school END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_school_central } } add_modifier = { name = nesnd_india_order_school_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_school_central } remove_modifier = nesnd_india_order_school_central }
}

# Subject scope, with scope:india_order_lord and scope:india_order_member saved. Builds what is missing.
# A treaty is marked only once it stands; one the engine would not create is tried again next month.
nesnd_india_order_build = {
    # tax
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = tax END = 1 } }
            NOT = { has_variable = nesnd_india_order_tax_pact }
            nesnd_india_order_can_raise = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = raise_payments } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = raise_payments is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_tax_pact value = 1 }
        }
    }
    else_if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = tax END = 2 } }
            NOT = { has_variable = nesnd_india_order_tax_pact }
            nesnd_india_order_can_decrease = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = decrease_payments } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = decrease_payments is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_tax_pact value = 2 }
        }
    }
    # army
    if = { limit = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = army END = 1 } } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_army_subject } } add_modifier = { name = nesnd_india_order_army_subject } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_army_subject } remove_modifier = nesnd_india_order_army_subject }
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = army END = 2 } }
            NOT = { has_variable = nesnd_india_order_army_pact }
            nesnd_india_order_can_exempt = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = exempt_from_service } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = exempt_from_service is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_army_pact value = 2 }
        }
    }
    # school
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = school END = 1 } }
            NOT = { has_variable = nesnd_india_order_school_pact }
            nesnd_india_order_can_share = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = da_knowledge_sharing } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = da_knowledge_sharing is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_school_pact value = 1 }
        }
    }
    if = { limit = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = school END = 2 } } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_school_subject } } add_modifier = { name = nesnd_india_order_school_subject } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_school_subject } remove_modifier = nesnd_india_order_school_subject }
}

# Subject scope, with scope:india_order_member saved. LORD is the Mughal whose marks these are.
# Removes this journal's treaty in one matter (if it still stands), its marker and its modifier.
nesnd_india_order_clear_tax = {
    if = {
        limit = { exists = $LORD$ nesnd_india_order_tax_pact_value = 1 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = raise_payments is_initiator = yes } } }
        $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = raise_payments } }
    }
    if = {
        limit = { exists = $LORD$ nesnd_india_order_tax_pact_value = 2 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = decrease_payments is_initiator = yes } } }
        $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = decrease_payments } }
    }
    remove_variable = nesnd_india_order_tax_pact
}
nesnd_india_order_clear_army = {
    if = {
        limit = { exists = $LORD$ has_variable = nesnd_india_order_army_pact $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = exempt_from_service is_initiator = yes } } }
        $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = exempt_from_service } }
    }
    remove_variable = nesnd_india_order_army_pact
    if = { limit = { has_modifier = nesnd_india_order_army_subject } remove_modifier = nesnd_india_order_army_subject }
}
nesnd_india_order_clear_school = {
    if = {
        limit = { exists = $LORD$ has_variable = nesnd_india_order_school_pact $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = da_knowledge_sharing is_initiator = yes } } }
        $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = da_knowledge_sharing } }
    }
    remove_variable = nesnd_india_order_school_pact
    if = { limit = { has_modifier = nesnd_india_order_school_subject } remove_modifier = nesnd_india_order_school_subject }
}

# Subject scope. Everything this journal gave it, cleared; the treaties go if the engine has not already
# broken them.
nesnd_india_order_leave = {
    save_scope_as = india_order_member
    clear_saved_scope = india_order_old_lord
    if = { limit = { exists = var:nesnd_india_order_lord } var:nesnd_india_order_lord = { save_scope_as = india_order_old_lord } }
    nesnd_india_order_clear_tax = { LORD = scope:india_order_old_lord }
    nesnd_india_order_clear_army = { LORD = scope:india_order_old_lord }
    nesnd_india_order_clear_school = { LORD = scope:india_order_old_lord }
    remove_variable = nesnd_india_order_lord
}

# Mughal scope. It no longer holds the journal.
nesnd_india_order_end = {
    save_scope_as = india_order_lord
    every_subject_or_below = {
        limit = { exists = var:nesnd_india_order_lord var:nesnd_india_order_lord = scope:india_order_lord }
        nesnd_india_order_leave = yes
    }
    remove_variable = nesnd_india_order_tax remove_variable = nesnd_india_order_tax_wait
    remove_variable = nesnd_india_order_army remove_variable = nesnd_india_order_army_wait
    remove_variable = nesnd_india_order_school remove_variable = nesnd_india_order_school_wait
    if = { limit = { has_modifier = nesnd_india_order_tax_central } remove_modifier = nesnd_india_order_tax_central }
    if = { limit = { has_modifier = nesnd_india_order_army_central } remove_modifier = nesnd_india_order_army_central }
    if = { limit = { has_modifier = nesnd_india_order_school_central } remove_modifier = nesnd_india_order_school_central }
}

# Mughal scope. ISSUE is tax, army or school; END 1 central, 2 local.
nesnd_india_order_choose = {
    if = { limit = { nesnd_india_order_can_choose = { ISSUE = $ISSUE$ END = $END$ } }
        save_scope_as = india_order_lord
        every_subject_or_below = {
            limit = { nesnd_india_order_member = { LORD = scope:india_order_lord } }
            save_scope_as = india_order_member
            nesnd_india_order_clear_$ISSUE$ = { LORD = scope:india_order_lord }
        }
        set_variable = { name = nesnd_india_order_$ISSUE$ value = $END$ }
        set_variable = { name = nesnd_india_order_$ISSUE$_wait months = 60 }
        nesnd_india_order_apply = yes
    }
}

# Mughal scope, monthly from the journal. An AI settles an open matter now and then, and later turns to
# the local arrangement when most of its Indian subjects are restless.
nesnd_india_order_ai = {
    if = { limit = { is_ai = yes nesnd_india_order_live = yes }
        # tax
        if = { limit = { NOT = { has_variable = nesnd_india_order_tax } nesnd_india_order_free = { ISSUE = tax } }
            random = { chance = 10
                if = { limit = { net_fixed_income < 0 } random_list = { 70 = { nesnd_india_order_choose = { ISSUE = tax END = 1 } } 30 = { nesnd_india_order_choose = { ISSUE = tax END = 2 } } } }
                else = { random_list = { 50 = { nesnd_india_order_choose = { ISSUE = tax END = 1 } } 50 = { nesnd_india_order_choose = { ISSUE = tax END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = tax END = 1 } nesnd_india_order_free = { ISSUE = tax } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = tax END = 2 } }
        }
        # army
        if = { limit = { NOT = { has_variable = nesnd_india_order_army } nesnd_india_order_free = { ISSUE = army } }
            random = { chance = 10
                if = { limit = { is_at_war = yes } random_list = { 70 = { nesnd_india_order_choose = { ISSUE = army END = 1 } } 30 = { nesnd_india_order_choose = { ISSUE = army END = 2 } } } }
                else = { random_list = { 40 = { nesnd_india_order_choose = { ISSUE = army END = 1 } } 60 = { nesnd_india_order_choose = { ISSUE = army END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = army END = 1 } nesnd_india_order_free = { ISSUE = army } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = army END = 2 } }
        }
        # school
        if = { limit = { NOT = { has_variable = nesnd_india_order_school } nesnd_india_order_free = { ISSUE = school } }
            random = { chance = 10
                if = { limit = { literacy_rate < 0.3 } random_list = { 60 = { nesnd_india_order_choose = { ISSUE = school END = 1 } } 40 = { nesnd_india_order_choose = { ISSUE = school END = 2 } } } }
                else = { random_list = { 50 = { nesnd_india_order_choose = { ISSUE = school END = 1 } } 50 = { nesnd_india_order_choose = { ISSUE = school END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = school END = 1 } nesnd_india_order_free = { ISSUE = school } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = school END = 2 } }
        }
    }
}
