# Superseded routes. Existing earned modifiers remain; obsolete calls cannot issue new rewards.
nesnd_rg_finish = {
    set_variable = nesnd_rg_closed remove_variable = nesnd_rg_pending
    if = { limit = { always = no } trigger_event = { id = $EVENT$ } }
}
nesnd_rg_end_without_reward = { set_variable = nesnd_rg_closed remove_variable = nesnd_rg_pending }
nesnd_rg_archive = {
    set_variable = nesnd_rg_closed remove_variable = nesnd_rg_pending
    if = { limit = { $IDENTITY$ = no } set_variable = nesnd_rg_closed }
}
nesnd_rg_reward = {
    remove_variable = nesnd_rg_pending
    if = { limit = { always = no $IDENTITY$ = yes } add_modifier = { name = $MODIFIER$ months = 60 } }
}
