					if = { limit = { owner = { has_modifier = nesnd_qe1_stalled } var:nesnd_qe1_sol_s < var:nesnd_qe1_sol_previous_slaves }
						add_radicals_in_state = { pop_type = pop_type:slaves value = { value = var:nesnd_qe1_sol_previous_slaves subtract = var:nesnd_qe1_sol_s multiply = var:nesnd_qe1_sol_f multiply = 0.10 min = 0 max = 0.05 } }
					}
				}
				set_variable = { name = nesnd_qe1_sol_previous_slaves value = var:nesnd_qe1_sol_s }
			} else = { remove_variable = nesnd_qe1_sol_previous_slaves }

			set_variable = { name = nesnd_qe1_sol_n value = nesnd_qe1_sol_soldiers_n }
			if = { limit = { var:nesnd_qe1_sol_n > 0 }
				set_variable = { name = nesnd_qe1_sol_s value = { value = nesnd_qe1_sol_soldiers_s divide = var:nesnd_qe1_sol_n } }
				set_variable = { name = nesnd_qe1_sol_f value = { value = nesnd_qe1_sol_soldiers_f divide = var:nesnd_qe1_sol_n } }
				if = { limit = { has_variable = nesnd_qe1_sol_previous_soldiers }
					if = { limit = { owner = { has_modifier = nesnd_qe1_angovernment } var:nesnd_qe1_sol_s > var:nesnd_qe1_sol_previous_soldiers }
						add_loyalists_in_state = { pop_type = pop_type:soldiers value = { value = var:nesnd_qe1_sol_s subtract = var:nesnd_qe1_sol_previous_soldiers multiply = var:nesnd_qe1_sol_f multiply = 0.15 min = 0 max = 0.05 } }
					}
					if = { limit = { owner = { has_modifier = nesnd_qe1_stalled } var:nesnd_qe1_sol_s < var:nesnd_qe1_sol_previous_soldiers }
						add_radicals_in_state = { pop_type = pop_type:soldiers value = { value = var:nesnd_qe1_sol_previous_soldiers subtract = var:nesnd_qe1_sol_s multiply = var:nesnd_qe1_sol_f multiply = 0.10 min = 0 max = 0.05 } }
					}
				}
				set_variable = { name = nesnd_qe1_sol_previous_soldiers value = var:nesnd_qe1_sol_s }
			} else = { remove_variable = nesnd_qe1_sol_previous_soldiers }

			remove_variable = nesnd_qe1_sol_n remove_variable = nesnd_qe1_sol_s remove_variable = nesnd_qe1_sol_f
		}
	} else_if = { limit = { has_variable = nesnd_qe1_sol_bonus_tracking }
		every_scope_state = { nesnd_qe1_clear_sol_state = yes }
		remove_variable = nesnd_qe1_sol_bonus_tracking
	}
}
