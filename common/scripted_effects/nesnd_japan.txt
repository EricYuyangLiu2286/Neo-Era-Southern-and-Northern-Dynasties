# Must be called in JAP. Existing foreign-owned state intersections are never selected.
nesnd_partition_japan = {
    set_variable = nesnd_jp_managed
    if = {
        limit = {
            c:JAP ?= this
            nesnd_japan_partition_possible = yes
        }
        nesnd_jp_new_realm = yes
        clear_saved_scope = nesnd_partition_tokugawa_incumbent
        clear_saved_scope = nesnd_partition_yamato_incumbent
        if = { limit = { ruler ?= { nesnd_character_of_dynasty = { HOUSE = house_tokugawa } } }
            ruler = { save_scope_as = nesnd_partition_tokugawa_incumbent }
        }
        else_if = { limit = { ruler ?= { nesnd_character_of_dynasty = { HOUSE = house_yamato } } }
            ruler = { save_scope_as = nesnd_partition_yamato_incumbent }
        }
        clear_saved_scope = nesnd_tokugawa_successor
        clear_saved_scope = nesnd_japanese_emperor
        if = {
            limit = {
                any_scope_character = {
                    has_template = JAP_ieyoshi_tokugawa
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
            }
            random_scope_character = {
                limit = {
                    has_template = JAP_ieyoshi_tokugawa
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
                save_scope_as = nesnd_tokugawa_successor
            }
        }
        else_if = {
            limit = {
                any_scope_character = {
                    nesnd_character_of_dynasty = { HOUSE = house_tokugawa }
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
            }
            random_scope_character = {
                limit = {
                    nesnd_character_of_dynasty = { HOUSE = house_tokugawa }
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
                save_scope_as = nesnd_tokugawa_successor
            }
        }
        if = {
            limit = {
                trigger_if = {
                    limit = { exists = var:emperor_var }
                    var:emperor_var = {
                        is_character_alive = yes
                        owner = c:JAP
                        is_ruler_of_own_country = no
                    }
                }
                trigger_else = { always = no }
            }
            var:emperor_var = {
                save_scope_as = nesnd_japanese_emperor
            }
        }
        else_if = {
            limit = {
                any_scope_character = {
                    nesnd_character_of_dynasty = { HOUSE = house_yamato }
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
            }
            random_scope_character = {
                limit = {
                    nesnd_character_of_dynasty = { HOUSE = house_yamato }
                    is_character_alive = yes
                    is_ruler_of_own_country = no
                }
                save_scope_as = nesnd_japanese_emperor
            }
        }
        if = {
            limit = {
                NOT = {
                    c:JTK ?= {
                        is_country_alive = yes
                    }
                }
            }
            create_country = {
                tag = JTK
                origin = c:JAP
                state = s:STATE_KANTO.region_state:JAP
                on_created = {
                    nesnd_jp_new_realm = yes
                    if = {
                        limit = {
                            c:JAP ?= {
                                has_variable = nesnd_japan_recovery_created
                            }
                        }
                        set_variable = nesnd_japan_recovery_created
                    }
                    remove_variable = nesnd_japan_unity_entered
                    remove_variable = nesnd_japan_unified
                    remove_variable = emperor_var
                    activate_law = law_type:law_monarchy
                    activate_law = law_type:law_autocracy
                    set_capital = STATE_KANTO
                    if = {
                        limit = { NOT = { exists = scope:nesnd_partition_tokugawa_incumbent } }
                        if = {
                            limit = {
                                exists = scope:nesnd_tokugawa_successor
                                scope:nesnd_tokugawa_successor = {
                                    is_character_alive = yes
                                    OR = {
                                        AND = {
                                            owner = c:JAP
                                            is_ruler_of_own_country = no
                                        }
                                        owner = c:JTK
                                    }
                                }
                            }
                            scope:nesnd_tokugawa_successor = {
                                if = {
                                    limit = {
                                        NOT = {
                                            owner = c:JTK
                                        }
                                    }
                                    transfer_character = c:JTK
                                }
                                set_character_as_ruler = yes
                            }
                        }
                        else = {
                            create_character = {
                                first_name = "nesnd_japan_tokugawa_successor_name"
                                last_name = "Tokugawa"
                                culture = cu:japanese
                                age = 30
                                ruler = yes
                                interest_group = ig_landowners
                                ideology = ideology_moderate
                                on_created = {
                                    set_variable = house_tokugawa set_variable = { name = dynasty_var value = flag:house_tokugawa } set_variable = { name = dynasty_var value = flag:house_tokugawa }
                                }
                            }
                        }
                    }
                    set_variable = nesnd_japan_partition_state
                }
            }
        }
        if = {
            limit = {
                NOT = {
                    c:JIM ?= {
                        is_country_alive = yes
                    }
                }
            }
            create_country = {
                tag = JIM
                origin = c:JAP
                state = s:STATE_KYOTO.region_state:JAP
                on_created = {
                    nesnd_jp_new_realm = yes
                    set_variable = nesnd_jp_managed
                    add_primary_culture = cu:ce_wuu
                    if = {
                        limit = {
                            c:JAP ?= {
                                has_variable = nesnd_japan_recovery_created
                            }
                        }
                        set_variable = nesnd_japan_recovery_created
                    }
                    remove_variable = nesnd_japan_unity_entered
                    remove_variable = nesnd_japan_unified
                    activate_law = law_type:law_monarchy
                    activate_law = law_type:law_autocracy
                    set_capital = STATE_KYOTO
                    if = {
                        limit = { NOT = { exists = scope:nesnd_partition_yamato_incumbent } }
                        if = {
                            limit = {
                                exists = scope:nesnd_japanese_emperor
                                scope:nesnd_japanese_emperor = {
                                    is_character_alive = yes
                                    OR = {
                                        AND = {
                                            owner = c:JAP
                                            is_ruler_of_own_country = no
                                        }
                                        owner = c:JIM
                                    }
                                }
                            }
                            scope:nesnd_japanese_emperor = {
                                if = {
                                    limit = {
                                        NOT = {
                                            owner = c:JIM
                                        }
                                    }
                                    transfer_character = c:JIM
                                }
                                set_character_as_ruler = yes
                            }
                        }
                        else = {
                            create_character = {
                                first_name = "nesnd_japan_yamato_successor_name"
                                last_name = "Yamato"
                                culture = cu:japanese
                                age = 30
                                ruler = yes
                                interest_group = ig_landowners
                                ideology = ideology_moderate
                                on_created = {
                                    set_variable = house_yamato set_variable = { name = dynasty_var value = flag:house_yamato } set_variable = { name = dynasty_var value = flag:house_yamato }
                                }
                            }
                        }
                        ruler = {
                            save_scope_as = nesnd_japan_current_emperor
                        }
                        set_variable = {
                            name = emperor_var
                            value = scope:nesnd_japan_current_emperor
                        }
                    }
                    set_variable = nesnd_japan_partition_state
                }
            }
        }
        if = {
            limit = {
                c:JTK ?= {
                    is_country_alive = yes
                }
                c:JIM ?= {
                    is_country_alive = yes
                }
            }
            if = { limit = { c:USM = { has_variable = nesnd_eastern_submission_accepted } NOT = { has_law = law_type:law_monarchy } }
                activate_law = law_type:law_monarchy
            }
            if = { limit = { OR = {
                    exists = scope:nesnd_partition_tokugawa_incumbent
                    exists = scope:nesnd_partition_yamato_incumbent
                    AND = { c:USM = { has_variable = nesnd_eastern_submission_accepted } NOT = { ruler ?= { nesnd_character_of_dynasty = { HOUSE = house_toyotomi } } } }
                } }
                nesnd_japan_seat_partition_incumbent = yes
            }
            s:STATE_KANTO = {
                every_scope_state = {
                    limit = {
                        owner = c:JAP
                        OR = {
                            NOT = {
                                c:JAP ?= {
                                    has_variable = nesnd_japan_recovery_created
                                }
                            }
                            controller = owner
                        }
                    }
                    set_state_owner = c:JTK
                }
            }
            s:STATE_KYOTO = {
                every_scope_state = {
                    limit = {
                        owner = c:JAP
                        OR = {
                            NOT = {
                                c:JAP ?= {
                                    has_variable = nesnd_japan_recovery_created
                                }
                            }
                            controller = owner
                        }
                    }
                    set_state_owner = c:JIM
                }
            }
            if = {
                limit = {
                    OR = {
                        NOT = {
                            c:JAP ?= {
                                has_variable = nesnd_japan_recovery_created
                            }
                        }
                        p:xB03EF0 = {
                            state = {
                                owner = c:JAP
                                controller = owner
                            }
                        }
                    }
                }
                transfer_province = {
                    GIVER = c:JAP
                    TAKER = c:JIM
                    PROVINCE = xB03EF0
                }
            }
            s:STATE_TOHOKU = {
                every_scope_state = {

                        OR = {
                            NOT = {

# Both successor countries already exist. Preserve the real incumbent and move that
# character to the appropriate eastern court only after seating Toyotomi in JAP.
nesnd_japan_seat_partition_incumbent = {
    clear_saved_scope = nesnd_partition_western_successor
    if = { limit = { heir ?= { is_character_alive = yes owner = c:JAP nesnd_character_of_dynasty = { HOUSE = house_toyotomi } is_in_exile_pool = no } }
        heir = { save_scope_as = nesnd_partition_western_successor }
    }
    else_if = { limit = { any_scope_character = { is_character_alive = yes is_in_exile_pool = no is_ruler_of_own_country = no nesnd_character_of_dynasty = { HOUSE = house_toyotomi } } }
        random_scope_character = {
            limit = { is_character_alive = yes is_in_exile_pool = no is_ruler_of_own_country = no nesnd_character_of_dynasty = { HOUSE = house_toyotomi } }
            save_scope_as = nesnd_partition_western_successor
        }
    }
    if = { limit = { exists = scope:nesnd_partition_western_successor }
        scope:nesnd_partition_western_successor = { set_character_as_ruler = yes }
    }
    else = {
        create_character = {
            first_name = "nesnd_jp_toyotomi_court_name" last_name = Toyotomi culture = cu:japanese
            age = 30 interest_group = ig_landowners ideology = ideology_moderate ruler = yes
            on_created = { set_variable = house_toyotomi set_variable = { name = dynasty_var value = flag:house_toyotomi } set_variable = { name = dynasty_var value = flag:house_toyotomi } }
        }
    }
    if = { limit = { exists = scope:nesnd_partition_tokugawa_incumbent }
        scope:nesnd_partition_tokugawa_incumbent = {
            if = { limit = { owner = c:JAP is_character_alive = yes is_ruler_of_own_country = no }
                transfer_character = c:JTK
                if = { limit = { owner = c:JTK } set_character_as_ruler = yes }
            }
        }
    }
    else_if = { limit = { exists = scope:nesnd_partition_yamato_incumbent }
        scope:nesnd_partition_yamato_incumbent = {
            if = { limit = { owner = c:JAP is_character_alive = yes is_ruler_of_own_country = no }
                transfer_character = c:JIM
                if = { limit = { owner = c:JIM }
                    set_character_as_ruler = yes
                    if = { limit = { is_ruler_of_own_country = yes }
                        c:JIM = { set_variable = { name = emperor_var value = scope:nesnd_partition_yamato_incumbent } }
                    }
                }
            }
        }
    }
    clear_saved_scope = nesnd_partition_tokugawa_incumbent
    clear_saved_scope = nesnd_partition_yamato_incumbent
    clear_saved_scope = nesnd_partition_western_successor
}
nesnd_japan_grant_partition_claims = {
    if = {
        limit = { has_global_variable = nesnd_japan_partitioned NOT = { has_global_variable = nesnd_japan_partition_claims_considered } }
        set_global_variable = nesnd_japan_partition_claims_considered
        if = { limit = { exists = c:JAP exists = c:JTK exists = c:JIM }
            if = { limit = { c:JTK = { owns_entire_state_region = STATE_KANTO } }
                s:STATE_KANTO = { add_claim = c:JAP add_claim = c:JIM }
            }
            if = { limit = { c:JIM = { owns_entire_state_region = STATE_KYOTO } }
                s:STATE_KYOTO = { add_claim = c:JAP add_claim = c:JTK }
            }
            if = { limit = { c:JAP = { owns_entire_state_region = STATE_CHUGOKU } }
                s:STATE_CHUGOKU = { add_claim = c:JTK add_claim = c:JIM }
            }
        }
    }
}

nesnd_japan_consider_terms_once = {
    if = {
        limit = {
            exists = c:USM
            c:USM ?= this
            has_variable = tag_WUU_var
            has_variable = nesnd_japan_dossier_entered
            has_global_variable = nesnd_south_settled
            NOT = { has_variable = nesnd_japan_consultation_considered }
        }
        set_variable = nesnd_japan_consultation_considered
        if = {
            limit = {
                exists = c:JAP
                NOR = {
                    has_variable = nesnd_japan_consultation_period
                    has_variable = nesnd_japan_recovery_pending
                    has_global_variable = nesnd_japan_offer_sent
                    has_global_variable = nesnd_japan_unity_offer
                    has_global_variable = nesnd_japan_war_started
                    has_global_variable = nesnd_japan_dossier_closed
                    has_global_variable = nesnd_japan_partitioned
                }
            }
            set_variable = { name = nesnd_japan_consultation_period months = 24 }
        }
    }
}
# Review a legacy partition once, including cases that already settled or partly lost land.
nesnd_japan_review_old_split = {
    if = { limit = { has_global_variable = nesnd_japan_partitioned NOT = { has_global_variable = nesnd_japan_partition_claims_considered } }
        if = { limit = { nesnd_japan_legacy_claims_available = yes }
            nesnd_japan_grant_partition_claims = yes
        }
        else = { set_global_variable = nesnd_japan_partition_claims_considered }
    }
}

nesnd_japan_observe_other_house = {
    if = { limit = {
        nesnd_japan_other_house_in_power = yes
        NOR = { has_global_variable = nesnd_japan_dossier_closed has_global_variable = nesnd_japan_offer_sent has_global_variable = nesnd_japan_partitioned }
    }
        c:JAP = { ruler = { save_scope_as = nesnd_eastern_other_ruler } }
        if = { limit = {
            trigger_if = { limit = { exists = var:nesnd_japan_other_house_ruler }
                NOT = { var:nesnd_japan_other_house_ruler = scope:nesnd_eastern_other_ruler }
            }
            trigger_else = { always = yes }
        }
            set_variable = { name = nesnd_japan_other_house_ruler value = scope:nesnd_eastern_other_ruler }
            set_variable = { name = nesnd_japan_other_house_observation months = 12 }
        }
        clear_saved_scope = nesnd_eastern_other_ruler
    }
    else = { remove_variable = nesnd_japan_other_house_ruler remove_variable = nesnd_japan_other_house_observation }
}
nesnd_japan_create_eastern_intervention = {
    create_diplomatic_play = {
        escalation = 80
        name = dp_nesnd_eastern_intervention target_country = c:JAP type = dp_nesnd_eastern_intervention
        add_war_goal = { holder = c:USM target_country = c:JAP type = wg_nesnd_japan_reorganization primary_demand = yes }
        add_war_goal = { holder = c:JAP target_country = c:USM type = humiliation primary_demand = yes }
    }
    if = { limit = { nesnd_japan_matching_intervention = yes }
        set_global_variable = nesnd_japan_war_started
    }
}
# Only a verified previously existing pact is eligible for restoration.
nesnd_japan_open_subject_intervention = {
    if = { limit = { nesnd_japan_restorable_pact = { PACT = $PACT$ } }
        save_scope_as = nesnd_eastern_original_patron
        c:JAP = { save_scope_as = nesnd_eastern_original_subject make_independent = yes }
        if = { limit = { nesnd_japan_original_intervention_pair = yes }
            nesnd_japan_create_eastern_intervention = yes
        }
        if = { limit = { nesnd_japan_original_intervention_pair = yes nesnd_japan_matching_intervention = no }
            # This restores the pre-play pact after a failed launch. It is not a Japanese surrender.
            set_global_variable = nesnd_japan_dossier_closed
            create_diplomatic_pact = { country = scope:nesnd_eastern_original_subject type = $PACT$ }
        }
        clear_saved_scope = nesnd_eastern_original_subject
        clear_saved_scope = nesnd_eastern_original_patron
    }
}
nesnd_japan_launch_intervention = {
    if = { limit = { c:JAP = { is_subject = no } }
        nesnd_japan_create_eastern_intervention = yes
    }
    else_if = { limit = { c:JAP = { is_direct_subject_of = c:USM } }
        if = { limit = { nesnd_japan_restorable_pact = { PACT = protectorate } }
            nesnd_japan_open_subject_intervention = { PACT = protectorate }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = puppet } }
            nesnd_japan_open_subject_intervention = { PACT = puppet }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = tributary } }
            nesnd_japan_open_subject_intervention = { PACT = tributary }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = vassal } }
            nesnd_japan_open_subject_intervention = { PACT = vassal }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = dominion } }
            nesnd_japan_open_subject_intervention = { PACT = dominion }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = colony } }
            nesnd_japan_open_subject_intervention = { PACT = colony }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = personal_union } }
            nesnd_japan_open_subject_intervention = { PACT = personal_union }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = crown_land } }
            nesnd_japan_open_subject_intervention = { PACT = crown_land }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = chartered_company } }
            nesnd_japan_open_subject_intervention = { PACT = chartered_company }
        }
        else_if = { limit = { nesnd_japan_restorable_pact = { PACT = nesnd_egypt_protectorate } }
            nesnd_japan_open_subject_intervention = { PACT = nesnd_egypt_protectorate }
        }
    }

# Both the military and timeout outcomes settle here. The receipt precedes all awards.
nesnd_japan_eastern_failure = {
    if = {
        limit = {
            has_variable = tag_WUU_var
            NOT = { has_global_variable = nesnd_japan_dossier_closed }
            nesnd_eastern_target_valid = yes
        }
        save_scope_as = eastern_wu
        set_global_variable = nesnd_japan_dossier_closed
        c:JAP = {
            save_scope_as = eastern_japan
            add_modifier = { name = nesnd_eastern_vindication years = 30 }
            every_interest_group = { add_modifier = { name = nesnd_eastern_ig_confidence years = 30 } }
            nesnd_jp_initialize = yes
            change_variable = { name = nesnd_jp_cohesion add = 25 }
            clamp_variable = { name = nesnd_jp_cohesion min = 0 max = 100 }
            nesnd_jp_sync_cohesion = yes

            # move_pop transfers whole Pops. Count people first, then fit whole communities to the quota.
            set_local_variable = { name = eastern_refugees_remaining value = 0 }
            every_scope_pop = {
                limit = { nesnd_eastern_refugee_pop = yes }
                change_local_variable = { name = eastern_refugees_remaining add = total_size }
            }
            change_local_variable = { name = eastern_refugees_remaining multiply = 0.1 }
            ordered_scope_pop = {
                limit = { nesnd_eastern_refugee_pop = yes }
                order_by = total_size
                if = {
                    limit = { total_size <= local_var:eastern_refugees_remaining }
                    change_local_variable = { name = eastern_refugees_remaining subtract = total_size }
                    c:USM.capital = { save_scope_as = eastern_refuge }
                    move_pop = scope:eastern_refuge
                }
                else = { save_temporary_scope_as = eastern_smallest_remaining_pop }
            }
            scope:eastern_smallest_remaining_pop ?= {
                if = {
                    limit = { total_size < { value = local_var:eastern_refugees_remaining multiply = 2 } }
                    c:USM.capital = { save_scope_as = eastern_refuge }
                    move_pop = scope:eastern_refuge
                }
            }
            trigger_event = { id = nesnd_japan.7 popup = yes }
        }
        trigger_event = { id = nesnd_japan.6 popup = yes }
    }
}
