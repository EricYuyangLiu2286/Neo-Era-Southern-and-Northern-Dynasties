REPLACE:country_generate_japanese_shogun = {
    if = { limit = { NOT = { has_variable = nesnd_jp_managed } }

        if = {
            limit = {
                year < 1840
                exists = c:JAP
                c:JAP ?= This
            }
            if = {
                limit = {
                    NOT = {
                        any_scope_character = {
                            has_template = JAP_toyotomi_hideharu
                        }
                    }
                }
                if = {
                    limit = {
                        NOT = { any_character = { has_template = JAP_toyotomi_hideharu } }
                    }
                    create_character = {
                        template = JAP_toyotomi_hideharu
                    }
                }
                else_if = {
                    limit = {
                        any_character_in_exile_pool = {
                            has_template = JAP_toyotomi_hideharu
                        }
                    }
                    random_character_in_exile_pool =  {
                        limit = {
                            has_template = JAP_toyotomi_hideharu
                        }
                        transfer_character = root
                        hidden_effect = {
                            free_from_exile_effect = yes
                        }

                        if = {
                            limit = {
                                has_role_of_type = agitator
                            }
                            remove_character_role = agitator
                        }
                        cleanup_ruler_character_roles = yes
                        set_character_as_ruler = yes
                        if = {
                            limit = {
                                NOT = { is_interest_group_type = ig_landowners }
                            }
                            set_interest_group = owner.ig:ig_landowners
                        }
                        if = {
                            limit = {
                                NOT = { has_ideology = ideology:ideology_authoritarian }
                            }
                            set_ideology = ideology:ideology_authoritarian
                        }
                        if = {
                            limit = {
                                NOT = { has_variable = ce_historical_ig_leader }
                            }
                            set_as_interest_group_leader = yes
                        }
                        if = {
                            limit = {
                                NOT = { has_variable = house_toyotomi }
                            }
                            set_variable = house_toyotomi
                        }

                    }
                }
                else = {
                    create_character = {
                        template = JAP_toyotomi_hideharu
                    }
                }


            }
            else = {
                random_scope_character = {
                    limit = {
                        has_template = JAP_toyotomi_hideharu
                    }
                    hidden_effect = {
                        free_from_exile_effect = yes
                    }

                    if = {
                        limit = {
                            has_role_of_type = agitator
                        }
                        remove_character_role = agitator
                    }
                    cleanup_ruler_character_roles = yes
                    set_character_as_ruler = yes
                    if = {
                        limit = {
                            NOT = { is_interest_group_type = ig_landowners }
                        }
                        set_interest_group = owner.ig:ig_landowners
                    }
                    if = {
                        limit = {
                            NOT = { has_ideology = ideology:ideology_authoritarian }
                        }
                        set_ideology = ideology:ideology_authoritarian
                    }
                    if = {
                        limit = {
                            NOT = { has_variable = ce_historical_ig_leader }
                        }
                        set_as_interest_group_leader = yes
                    }
                    if = {
                        limit = {
                            NOT = { has_variable = house_toyotomi }
                        }
                        set_variable = house_toyotomi
                    }
                }
            }
            # His son
            if = {
                limit = {
                    NOT = { any_character = { has_template = JAP_toyotomi_shimei } }
                }
                create_character = {
                    template = JAP_toyotomi_shimei
                    on_created = {
                        add_character_role = heir
                    }
                }
                set_variable = heir_shimei_spawned
            }
            else_if = {
                limit = {
                    any_character_in_exile_pool = {
                        has_template = JAP_toyotomi_shimei
                    }
                }
                random_character_in_exile_pool =  {
                    limit = {
                        has_template = JAP_toyotomi_shimei
                    }
                    transfer_character = root
                    hidden_effect = {
                        free_from_exile_effect = yes
                    }

                    if = {
                        limit = {
                            has_role_of_type = agitator
                        }
                        remove_character_role = agitator
                    }
                    cleanup_ruler_character_roles = yes

                    if = {
                        limit = {
                            NOT = { is_interest_group_type = ig_armed_forces }
                        }
                        set_interest_group = owner.ig:ig_armed_forces
                    }
                    if = {
                        limit = {
                            NOT = { has_ideology = ideology:ideology_modernizer_leader }
                        }
                        set_ideology = ideology:ideology_modernizer_leader
                    }
                    if = {
                        limit = {
                            NOT = { has_variable = ce_historical_ig_leader }
                        }
                        set_as_interest_group_leader = yes
                    }
                    if = {
                        limit = {
                            NOT = { has_variable = house_toyotomi }
                        }
                        set_variable = house_toyotomi
                    }
                    save_scope_as = japanes_heir_scope
                }
                save_scope_as = this_shogunate
                scope:this_shogunate = {
                    set_heir = scope:japanes_heir_scope
                    set_variable = heir_shimei_spawned
                }

            }
            else = {
                create_character = {
                    template = JAP_toyotomi_shimei
                    on_created = {
                        add_character_role = heir
                    }
                }
                set_variable = heir_shimei_spawned
            }


        }
        else_if = {
            limit = {
                year >= 1840
                year < 1846
                exists = c:JAP
                c:JAP ?= This
            }
            create_character = {
                template = JAP_toyotomi_shimei
                on_created = {
                    set_character_as_ruler = yes
                }
            }
            create_character = {

                    add_character_role = heir

                    set_character_as_ruler = yes
                }
            }
            create_character = {
                template = JAP_toyotomi_heikatsu
                on_created = {
                    set_global_variable = nesnd_jp_heikatsu_ever_created

        else_if = {
            limit = {
                year >= 1861

        else_if = {
            limit = {
                year >= 1892
                year < 1911
                exists = c:JAP
                c:JAP ?= This
            }
            create_character = {
                template = JAP_toyotomi_chokei

                    set_character_as_ruler = yes
                }

                    add_character_role = heir

            }
            set_variable = heir_tocho_spawned
        }
        else_if = {
            limit = {
                year >= 1911

                c:JAP ?= This
            }
            create_character = {
                template = JAP_toyotomi_tocho
                on_created = {
                    set_character_as_ruler = yes
                }
            }
        }
        else_if = {
            limit = {
                # year >= 1911
                OR = {
                    year >= 1940
                    year <= 1836
                }
                exists = c:JAP
                c:JAP ?= This
            }
            create_character = { # random toyotomi shogun
                last_name = Toyotomi
                female = no
                culture = primary_culture
                ig_leader = yes
                interest_group = ig:ig_landowners
                ideology = ideology_modernizer_leader

                        limit = {
                            NOT = { is_interest_group_type = ig_landowners }
                        }

                    }
                    if = {
                        limit = {
                            NOT = { has_ideology = ideology:ideology_modernizer_leader }
                        }
                        set_ideology = ideology:ideology_modernizer_leader
                    }
                    if = {
                        limit = {
                            NOT = { has_variable = ce_historical_ig_leader }
                        }
                        set_as_interest_group_leader = yes
                    }

                    if = {
                        limit = {
                            NOT = { has_variable = house_toyotomi }
                        }
                        set_variable = house_toyotomi
                    }
                }
            }
        }

    }
}
