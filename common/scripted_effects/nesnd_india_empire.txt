# ==== The centre and the states (formerly nesnd_india_order.txt) ====
# The journal nesnd_india_order. See common/scripted_triggers/nesnd_india_empire.txt for the three
# matters and common/script_values/nesnd_india_empire.txt for the variables.
#
# Order of things:
#   nesnd_india_order_choose     a button or the AI settles a matter one way; what this journal built
#                                for the other way is cleared first
#   nesnd_india_order_apply      monthly and after each choice: Mughal's own modifiers, then each Indian
#                                subject gets the treaties and modifiers it does not have yet
#   nesnd_india_order_leave      a subject that left (or whose Mughal no longer holds the journal) is
#                                cleared from its own monthly refresh
#   nesnd_india_order_end        Mughal no longer holds the journal: everything this journal set is cleared
# Only treaties marked on the subject (nesnd_india_order_*_pact) are ever removed; treaties of the same
# kind made by the player or by other scripts are left alone.

# Any country, monthly from nesnd_india_refresh.
nesnd_india_order_upkeep = {
    if = { limit = { nesnd_india_order_left = yes } nesnd_india_order_leave = yes }
    if = { limit = { nesnd_india_order_live = yes }
        # Once settled (nesnd_india_order_settled), the arrangement stays and the journal does not come back.
        if = { limit = { NOT = { has_journal_entry = nesnd_india_order } NOT = { has_variable = nesnd_india_order_settled } } add_journal_entry = { type = nesnd_india_order } }
        nesnd_india_order_default = yes
        nesnd_india_order_apply = yes
    }
    else_if = {
        limit = { OR = { has_variable = nesnd_india_order_tax has_variable = nesnd_india_order_army has_variable = nesnd_india_order_school } }
        nesnd_india_order_end = yes
    }
}

# Mughal scope.
nesnd_india_order_apply = {
    save_scope_as = india_order_lord
    nesnd_india_order_home = yes
    every_subject_or_below = {
        limit = { nesnd_india_order_member = { LORD = scope:india_order_lord } }
        save_scope_as = india_order_member
        # Marks left by another Mughal are cleared before this one's are set. While a treaty of the other
        # Mughal outlives its removal, the subject keeps that Mughal's name and gets nothing from this one;
        # the removal is tried again next month.
        if = { limit = { exists = var:nesnd_india_order_lord }
            if = { limit = { NOT = { var:nesnd_india_order_lord = scope:india_order_lord } } nesnd_india_order_leave = yes }
        }
        if = { limit = { NOT = { exists = var:nesnd_india_order_lord } } set_variable = { name = nesnd_india_order_lord value = scope:india_order_lord } }
        if = { limit = { exists = var:nesnd_india_order_lord }
            if = { limit = { var:nesnd_india_order_lord = scope:india_order_lord } nesnd_india_order_build = yes }
        }
    }
}

# Mughal scope. Its own modifiers follow its choices.
nesnd_india_order_home = {
    if = { limit = { nesnd_india_order_is = { ISSUE = tax END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_tax_central } } add_modifier = { name = nesnd_india_order_tax_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_tax_central } remove_modifier = nesnd_india_order_tax_central }
    if = { limit = { nesnd_india_order_is = { ISSUE = army END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_army_central } } add_modifier = { name = nesnd_india_order_army_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_army_central } remove_modifier = nesnd_india_order_army_central }
    if = { limit = { nesnd_india_order_is = { ISSUE = school END = 1 } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_school_central } } add_modifier = { name = nesnd_india_order_school_central } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_school_central } remove_modifier = nesnd_india_order_school_central }
}

# Subject scope, with scope:india_order_lord and scope:india_order_member saved. Builds what is missing.
# A treaty is marked only once it stands; one the engine would not create is tried again next month.
nesnd_india_order_build = {
    # A marker for the end not chosen now belongs to a treaty that outlived its removal when the matter
    # was changed (a treaty in its forced duration). Its removal is tried again each month, and until it
    # is gone the marker stays and the treaty of the new end is not built.
    if = { limit = { has_variable = nesnd_india_order_tax_pact NOT = { nesnd_india_order_tax_marked = { LORD = scope:india_order_lord } } } nesnd_india_order_clear_tax = { LORD = scope:india_order_lord } }
    if = { limit = { has_variable = nesnd_india_order_army_pact NOT = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = army END = 2 } } } } nesnd_india_order_clear_army = { LORD = scope:india_order_lord } }
    if = { limit = { has_variable = nesnd_india_order_school_pact NOT = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = school END = 1 } } } } nesnd_india_order_clear_school = { LORD = scope:india_order_lord } }
    # tax
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = tax END = 1 } }
            NOT = { has_variable = nesnd_india_order_tax_pact }
            nesnd_india_order_can_raise = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = raise_payments } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = raise_payments is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_tax_pact value = 1 }
        }
    }
    else_if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = tax END = 2 } }
            NOT = { has_variable = nesnd_india_order_tax_pact }
            nesnd_india_order_can_decrease = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = decrease_payments } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = decrease_payments is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_tax_pact value = 2 }
        }
    }
    # army
    if = { limit = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = army END = 1 } } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_army_subject } } add_modifier = { name = nesnd_india_order_army_subject } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_army_subject } remove_modifier = nesnd_india_order_army_subject }
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = army END = 2 } }
            NOT = { has_variable = nesnd_india_order_army_pact }
            nesnd_india_order_can_exempt = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = exempt_from_service } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = exempt_from_service is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_army_pact value = 2 }
        }
    }
    # school
    if = {
        limit = {
            scope:india_order_lord = { nesnd_india_order_is = { ISSUE = school END = 1 } }
            NOT = { has_variable = nesnd_india_order_school_pact }
            nesnd_india_order_can_share = yes
        }
        scope:india_order_lord = { create_diplomatic_pact = { country = scope:india_order_member type = da_knowledge_sharing } }
        if = { limit = { scope:india_order_lord = { has_diplomatic_pact = { who = scope:india_order_member type = da_knowledge_sharing is_initiator = yes } } }
            set_variable = { name = nesnd_india_order_school_pact value = 1 }
        }
    }
    if = { limit = { scope:india_order_lord = { nesnd_india_order_is = { ISSUE = school END = 2 } } }
        if = { limit = { NOT = { has_modifier = nesnd_india_order_school_subject } } add_modifier = { name = nesnd_india_order_school_subject } }
    }
    else_if = { limit = { has_modifier = nesnd_india_order_school_subject } remove_modifier = nesnd_india_order_school_subject }
}

# Subject scope, with scope:india_order_member saved. LORD is the Mughal whose marks these are.
# Removes this journal's treaty in one matter (if it still stands), its marker and its modifier.
# The marker goes only once the treaty is gone: a treaty still in its forced duration may survive the
# removal, and keeping the marker lets the next monthly pass try again.
nesnd_india_order_clear_tax = {
    if = { limit = { exists = $LORD$ }
        if = {
            limit = { nesnd_india_order_tax_pact_value = 1 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = raise_payments is_initiator = yes } } }
            $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = raise_payments } }
        }
        if = {
            limit = { nesnd_india_order_tax_pact_value = 2 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = decrease_payments is_initiator = yes } } }
            $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = decrease_payments } }
        }
        if = {
            limit = {
                NOR = {
                    AND = { nesnd_india_order_tax_pact_value = 1 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = raise_payments is_initiator = yes } } }
                    AND = { nesnd_india_order_tax_pact_value = 2 $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = decrease_payments is_initiator = yes } } }
                }
            }
            remove_variable = nesnd_india_order_tax_pact
        }
    }
    else = { remove_variable = nesnd_india_order_tax_pact }
}
nesnd_india_order_clear_army = {
    if = { limit = { exists = $LORD$ }
        if = {
            limit = { has_variable = nesnd_india_order_army_pact $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = exempt_from_service is_initiator = yes } } }
            $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = exempt_from_service } }
        }
        if = {
            limit = { NOT = { $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = exempt_from_service is_initiator = yes } } } }
            remove_variable = nesnd_india_order_army_pact
        }
    }
    else = { remove_variable = nesnd_india_order_army_pact }
    if = { limit = { has_modifier = nesnd_india_order_army_subject } remove_modifier = nesnd_india_order_army_subject }
}
nesnd_india_order_clear_school = {
    if = { limit = { exists = $LORD$ }
        if = {
            limit = { has_variable = nesnd_india_order_school_pact $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = da_knowledge_sharing is_initiator = yes } } }
            $LORD$ = { remove_diplomatic_pact = { country = scope:india_order_member type = da_knowledge_sharing } }
        }
        if = {
            limit = { NOT = { $LORD$ = { has_diplomatic_pact = { who = scope:india_order_member type = da_knowledge_sharing is_initiator = yes } } } }
            remove_variable = nesnd_india_order_school_pact
        }
    }
    else = { remove_variable = nesnd_india_order_school_pact }
    if = { limit = { has_modifier = nesnd_india_order_school_subject } remove_modifier = nesnd_india_order_school_subject }
}

# Subject scope. Everything this journal gave it, cleared; the treaties go if the engine has not already
# broken them.
nesnd_india_order_leave = {
    save_scope_as = india_order_member
    clear_saved_scope = india_order_old_lord
    if = { limit = { exists = var:nesnd_india_order_lord } var:nesnd_india_order_lord = { save_scope_as = india_order_old_lord } }
    # A member that left as an independent country is remembered by its old Mughal for ten years; a
    # second one within them ends the journal (outcome "the union scatters").
    if = { limit = { is_subject = no exists = scope:india_order_old_lord }
        scope:india_order_old_lord = {
            if = { limit = { exists = var:nesnd_india_order_first_departed NOT = { var:nesnd_india_order_first_departed = scope:india_order_member } }
                set_variable = { name = nesnd_india_order_second_departed value = scope:india_order_member }
            }
            else_if = { limit = { NOT = { exists = var:nesnd_india_order_first_departed } }
                set_variable = { name = nesnd_india_order_first_departed value = scope:india_order_member days = 3650 }
            }
        }
    }
    nesnd_india_order_clear_tax = { LORD = scope:india_order_old_lord }
    nesnd_india_order_clear_army = { LORD = scope:india_order_old_lord }
    nesnd_india_order_clear_school = { LORD = scope:india_order_old_lord }
    # A treaty that outlived its removal keeps the subject on this list, and the removal is tried again next month.
    if = { limit = { NOR = { has_variable = nesnd_india_order_tax_pact has_variable = nesnd_india_order_army_pact has_variable = nesnd_india_order_school_pact } } remove_variable = nesnd_india_order_lord }
}

# Mughal scope. It no longer holds the journal. The sixty months since each choice keep running, so a
# Mughal that holds the journal again soon cannot change a matter earlier than it could have before.
nesnd_india_order_end = {
    save_scope_as = india_order_lord
    every_subject_or_below = {
        limit = { has_variable = nesnd_india_order_lord var:nesnd_india_order_lord ?= scope:india_order_lord }
        nesnd_india_order_leave = yes
    }
    remove_variable = nesnd_india_order_tax
    remove_variable = nesnd_india_order_army
    remove_variable = nesnd_india_order_school
    remove_variable = nesnd_india_order_central_since
    remove_variable = nesnd_india_order_local_since
    remove_variable = nesnd_india_order_clock_wait
    if = { limit = { has_modifier = nesnd_india_order_tax_central } remove_modifier = nesnd_india_order_tax_central }
    if = { limit = { has_modifier = nesnd_india_order_army_central } remove_modifier = nesnd_india_order_army_central }
    if = { limit = { has_modifier = nesnd_india_order_school_central } remove_modifier = nesnd_india_order_school_central }
}

# Mughal scope. ISSUE is tax, army or school; END 1 central, 2 local.
nesnd_india_order_choose = {
    if = { limit = { nesnd_india_order_can_choose = { ISSUE = $ISSUE$ END = $END$ } }
        save_scope_as = india_order_lord
        every_subject_or_below = {
            limit = { nesnd_india_order_member = { LORD = scope:india_order_lord } }
            save_scope_as = india_order_member
            nesnd_india_order_clear_$ISSUE$ = { LORD = scope:india_order_lord }
        }
        set_variable = { name = nesnd_india_order_$ISSUE$ value = $END$ }
        set_variable = { name = nesnd_india_order_$ISSUE$_wait months = 60 }
        nesnd_india_order_apply = yes
    }
}

# Mughal scope, monthly from the journal. An AI settles an open matter now and then, and later turns to
# the local arrangement when most of its Indian subjects are restless. The central end of tax and of
# schools signs treaties that take our influence; with too little influence left for them, the AI
# mostly takes the local end.
nesnd_india_order_ai = {
    if = { limit = { is_ai = yes nesnd_india_order_live = yes }
        # tax
        if = { limit = { NOT = { has_variable = nesnd_india_order_tax } nesnd_india_order_free = { ISSUE = tax } }
            random = { chance = 10
                if = { limit = { influence < nesnd_india_order_tax_influence } random_list = { 10 = { nesnd_india_order_choose = { ISSUE = tax END = 1 } } 90 = { nesnd_india_order_choose = { ISSUE = tax END = 2 } } } }
                else_if = { limit = { net_fixed_income < 0 } random_list = { 70 = { nesnd_india_order_choose = { ISSUE = tax END = 1 } } 30 = { nesnd_india_order_choose = { ISSUE = tax END = 2 } } } }
                else = { random_list = { 50 = { nesnd_india_order_choose = { ISSUE = tax END = 1 } } 50 = { nesnd_india_order_choose = { ISSUE = tax END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = tax END = 1 } nesnd_india_order_free = { ISSUE = tax } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = tax END = 2 } }
        }
        # army
        if = { limit = { NOT = { has_variable = nesnd_india_order_army } nesnd_india_order_free = { ISSUE = army } }
            random = { chance = 10
                if = { limit = { is_at_war = yes } random_list = { 70 = { nesnd_india_order_choose = { ISSUE = army END = 1 } } 30 = { nesnd_india_order_choose = { ISSUE = army END = 2 } } } }
                else = { random_list = { 40 = { nesnd_india_order_choose = { ISSUE = army END = 1 } } 60 = { nesnd_india_order_choose = { ISSUE = army END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = army END = 1 } nesnd_india_order_free = { ISSUE = army } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = army END = 2 } }
        }
        # school
        if = { limit = { NOT = { has_variable = nesnd_india_order_school } nesnd_india_order_free = { ISSUE = school } }
            random = { chance = 10
                if = { limit = { influence < nesnd_india_order_school_influence } random_list = { 10 = { nesnd_india_order_choose = { ISSUE = school END = 1 } } 90 = { nesnd_india_order_choose = { ISSUE = school END = 2 } } } }
                else_if = { limit = { literacy_rate < 0.3 } random_list = { 60 = { nesnd_india_order_choose = { ISSUE = school END = 1 } } 40 = { nesnd_india_order_choose = { ISSUE = school END = 2 } } } }
                else = { random_list = { 50 = { nesnd_india_order_choose = { ISSUE = school END = 1 } } 50 = { nesnd_india_order_choose = { ISSUE = school END = 2 } } } }
            }
        }
        else_if = { limit = { nesnd_india_order_is = { ISSUE = school END = 1 } nesnd_india_order_free = { ISSUE = school } nesnd_india_order_ai_restless = yes }
            random = { chance = 5 nesnd_india_order_choose = { ISSUE = school END = 2 } }
        }
    }
}

# ==== Mughal's old agreement, the council of princes, famine relief (formerly nesnd_india_mughal.txt) ====
# Mughal's own affairs. "Mughal" is only the country holding nesnd_india_mughal_court; a prince that
# forms BHT or DRA does not get them. Variables and sums: common/script_values/nesnd_india_empire.txt.
#
#   The old agreement with the overlord (journal nesnd_mug_france):
#     a sale        matter kind 7 to the overlord; on agreement the overlord pays and one level of a
#                   building we hold becomes its own (nesnd_mug_deal).
#     a buyback     matter kind 8 to the overlord; on agreement we pay and one level it holds here
#                   becomes ours. No independence is bought: liberty desire stays vanilla's.
#   The council of princes (journal nesnd_mug_council, events nesnd_mug.11 to .13).
#   Famine (events nesnd_mug.21 to .24): Mughal or a prince relieves its struck states, or a prince asks
#     Mughal or its overlord (matter kind 9).

# ---------------------------------------------------------------------------------------------
# Monthly, from nesnd_india_refresh
# ---------------------------------------------------------------------------------------------
nesnd_mug_upkeep = {
    if = { limit = { nesnd_mug_france_live = yes NOT = { has_journal_entry = nesnd_mug_france } } add_journal_entry = { type = nesnd_mug_france } }
    # The overlord of the old agreement, for the outcome "the old agreement ends".
    if = { limit = { nesnd_mug_france_live = yes } set_variable = { name = nesnd_mug_france_lord value = overlord } }
    if = { limit = { nesnd_mug_council_live = yes NOT = { has_journal_entry = nesnd_mug_council } } add_journal_entry = { type = nesnd_mug_council } }
    # A plea for famine aid still on its way (it waits while the government asked is busy with another
    # matter, at most 90 days from sending): the struck states stay marked, so that the aid reaches them
    # when it is granted. Once the 90 days are over the marks run out on their own.
    if = { limit = { nesnd_india_matter_kind_value = 9 has_variable = nesnd_india_matter_queue }
        every_scope_state = {
            limit = { has_variable = nesnd_mug_famine_state }
            remove_variable = nesnd_mug_famine_state
            set_variable = { name = nesnd_mug_famine_state days = 120 }
        }
    }
    if = { limit = { nesnd_mug_famine_due = yes } nesnd_mug_famine_start = yes }
    if = { limit = { nesnd_mug_treaty_dispute_due = yes } nesnd_mug_treaty_dispute_start_event = yes }
}

# ---------------------------------------------------------------------------------------------
# The property deals (Mughal scope)
# ---------------------------------------------------------------------------------------------
# The building is chosen when the price is named: its state and type are kept on Mughal
# (nesnd_mug_deal_state, nesnd_mug_deal_kind; cleared with the matter) and nesnd_mug_deal looks for it.
nesnd_mug_ask = {
    if = { limit = { nesnd_mug_can_ask_$WHAT$ = yes exists = overlord }
        save_scope_value_as = { name = mug_ask_kind value = $KIND$ }
        clear_saved_scope = mug_ask_owner
        if = { limit = { scope:mug_ask_kind = 7 } save_scope_as = mug_ask_owner }
        else = { overlord = { save_scope_as = mug_ask_owner } }
        clear_saved_scope = mug_ask_building
        ordered_scope_building = {
            limit = { nesnd_mug_deal_site = { OWNER = scope:mug_ask_owner } }
            order_by = nesnd_mug_deal_order
            position = 0
            check_range_bounds = no
            save_scope_as = mug_ask_building
        }
        if = { limit = { exists = scope:mug_ask_building }
            set_variable = { name = nesnd_mug_deal_state value = scope:mug_ask_building.state }
            set_variable = { name = nesnd_mug_deal_kind value = scope:mug_ask_building.nesnd_mug_deal_type_index }
            nesnd_india_matter_send = { KIND = $KIND$ TO = overlord AMOUNT = $AMOUNT$ }
        }
    }
}

# Any scope. One level of the smallest building in HOLDER's states that is held wholly by OLD passes to
# NEW. There is no effect that moves ownership, so the building is removed and rebuilt at once with the
# new owners and its production methods switched back on (method: workshop mod 3032185644,
# more_unique_companies_events_deutsche_reichsbahn.txt, studied, not copied). scope:mug_deal_site is
# the state, for the reply event; scope:mug_deal_done is set when a building was found.
nesnd_mug_deal = {
    $NEW$ = { save_scope_as = mug_deal_new }
    $OLD$ = { save_scope_as = mug_deal_old }
    clear_saved_scope = mug_deal_building
    $HOLDER$ = {
        ordered_scope_building = {
            limit = { nesnd_mug_deal_site = { OWNER = scope:mug_deal_old } nesnd_mug_deal_chosen = { HOLDER = $HOLDER$ } }

# Country scope (asking country, root = this, from nesnd_mug.24 immediate). When famine aid is
# refused, unanswered or queued out (results 1, 2, 3, 6): Indian-culture pops in each struck state
# gain +10% radicals. If at least 3 states are struck and the 10-year cooldown is clear, our
# largest Indian primary culture begins a mass migration to the foreign country (capital outside
# India) with the highest relations that hosts a cultural community of that culture.
nesnd_mug_famine_refused = {
    every_scope_state = {
        limit = { nesnd_mug_famine_affected_state = yes }
        save_scope_as = india_famine_state
        every_scope_culture = {
            limit = { nesnd_india_culture = yes }
            scope:india_famine_state = {
                add_radicals_in_state = {
                    culture = prev
                    value = 0.10
                }
            }
        }
    }
    if = {
        limit = {
            NOT = { has_variable = nesnd_mug_famine_migration_cooldown }
            any_scope_state = {
                nesnd_mug_famine_affected_state = yes
                count >= 3
            }
        }
        clear_saved_scope = mug_famine_mig_culture
        ordered_primary_culture = {
            limit = { nesnd_india_culture = yes }
            order_by = "root.culture_percent_country(this)"
            position = 0
            check_range_bounds = no
            save_scope_as = mug_famine_mig_culture
        }
        if = {
            limit = { exists = scope:mug_famine_mig_culture }
            clear_saved_scope = mug_famine_mig_host
            clear_saved_scope = mug_famine_mig_state
            ordered_country = {
                limit = {
                    is_country_alive = yes
                    NOT = { this = root }
                    NOT = { is_country_type = decentralized }
                    capital ?= { NOT = { nesnd_india_state = yes } }
                    any_scope_state = {
                        NOT = { nesnd_india_state = yes }
                        has_cultural_community = scope:mug_famine_mig_culture
                        state_is_eligible_as_mass_migration_target = yes
                    }
                }
                order_by = relations:root
                position = 0
                check_range_bounds = no
                save_scope_as = mug_famine_mig_host
            }
            if = {
                limit = { exists = scope:mug_famine_mig_host }
                scope:mug_famine_mig_host = {
                    ordered_scope_state = {
                        limit = {
                            NOT = { nesnd_india_state = yes }
                            has_cultural_community = scope:mug_famine_mig_culture
                            state_is_eligible_as_mass_migration_target = yes
                        }
                        order_by = migration_pull
                        position = 0
                        check_range_bounds = no
                        save_scope_as = mug_famine_mig_state
                    }
                }
                if = {
                    limit = { exists = scope:mug_famine_mig_state }
                    set_variable = { name = nesnd_mug_famine_migration_cooldown days = 3650 }
                    scope:mug_famine_mig_state = {
                        create_mass_migration = {
                            origin = root
                            culture = scope:mug_famine_mig_culture
                        }
                    }
                    trigger_event = { id = nesnd_mug.25 popup = yes }
                }
            }
        }
    }
}

# any matter is settled, nothing changes. Also from the event nesnd_india_core.1.
nesnd_india_order_default = {
    if = {
        limit = {
            exists = var:nesnd_india_reform_path
            NOT = { has_variable = nesnd_india_order_defaulted }
            NOR = { has_variable = nesnd_india_order_tax has_variable = nesnd_india_order_army has_variable = nesnd_india_order_school }
        }
        set_variable = nesnd_india_order_defaulted
        set_variable = { name = nesnd_india_order_tax value = var:nesnd_india_reform_path }
        set_variable = { name = nesnd_india_order_army value = var:nesnd_india_order_tax }
        set_variable = { name = nesnd_india_order_school value = var:nesnd_india_order_tax }
    }
}
# Mughal scope, monthly from the journal. Ten years with all three matters central (or all local) settle
# the journal. nesnd_india_order_clock_wait runs the ten years; any other arrangement stops the clock.
nesnd_india_order_clock = {
    if = {
        limit = { nesnd_india_order_is = { ISSUE = tax END = 1 } nesnd_india_order_is = { ISSUE = army END = 1 } nesnd_india_order_is = { ISSUE = school END = 1 } }
        remove_variable = nesnd_india_order_local_since
        if = { limit = { NOT = { has_variable = nesnd_india_order_central_since } }
            set_variable = nesnd_india_order_central_since
            set_variable = { name = nesnd_india_order_clock_wait days = 3650 }
        }
    }
    else_if = {
        limit = { nesnd_india_order_is = { ISSUE = tax END = 2 } nesnd_india_order_is = { ISSUE = army END = 2 } nesnd_india_order_is = { ISSUE = school END = 2 } }
        remove_variable = nesnd_india_order_central_since
        if = { limit = { NOT = { has_variable = nesnd_india_order_local_since } }
            set_variable = nesnd_india_order_local_since
            set_variable = { name = nesnd_india_order_clock_wait days = 3650 }
        }
    }
    else = {
        remove_variable = nesnd_india_order_central_since
        remove_variable = nesnd_india_order_local_since
        remove_variable = nesnd_india_order_clock_wait
    }
}
# Subject scope, with scope:india_lobby_lord saved (its overlord). A lobby for the overlord: an existing
# one is pleased, otherwise the strongest free interest group founds one (as nesnd_india_indep_back_pro,
# without the marks of the independence journal).
nesnd_india_back_empire_lobby = {
    if = { limit = { nesnd_india_indep_lobby_country_ready = yes }
        if = { limit = { any_political_lobby = { is_political_lobby_type = lobby_pro_overlord target = scope:india_lobby_lord } }
            random_political_lobby = {
                limit = { is_political_lobby_type = lobby_pro_overlord target = scope:india_lobby_lord }
                change_appeasement = { amount = 4 factor = appeasement_lobby_pleased }
            }
        }
        else = {
            nesnd_india_indep_pick_lobby_member = yes
            if = { limit = { exists = scope:indep_lobby_member }
                create_political_lobby = { type = lobby_pro_overlord target = scope:india_lobby_lord add_interest_group = scope:indep_lobby_member lobby_formation_reason = diplomacy }
            }
        }
    }
}

# ==== Old treaty dispute (India batch 15) ====
# Mughal scope, monthly from nesnd_mug_upkeep. Sends the entry event nesnd_mug.30 to France.
nesnd_mug_treaty_dispute_start_event = {
    if = {
        limit = { nesnd_mug_treaty_dispute_due = yes }
        save_scope_as = mug_dispute_subject
        set_variable = { name = nesnd_india_treaty_dispute_pending days = 90 }
        c:FRA = { trigger_event = { id = nesnd_mug.30 popup = yes } }
    }
}
# Country scope (Mughal or France). Shifts the negotiation bar on both sides by VALUE.
nesnd_india_treaty_dispute_shift = {
    if = {
        limit = { has_journal_entry = nesnd_india_treaty_dispute }
        je:nesnd_india_treaty_dispute = {
            add_progress = { name = nesnd_india_treaty_dispute_bar value = $VALUE$ }
        }
        var:nesnd_india_dispute_overlord ?= {
            if = {
                limit = { has_journal_entry = nesnd_india_treaty_dispute_overlord }
                je:nesnd_india_treaty_dispute_overlord = {
                    add_progress = { name = nesnd_india_treaty_dispute_bar value = $VALUE$ }
                }
            }
        }
    }
    else_if = {
        limit = { has_journal_entry = nesnd_india_treaty_dispute_overlord }
        je:nesnd_india_treaty_dispute_overlord = {
            add_progress = { name = nesnd_india_treaty_dispute_bar value = $VALUE$ }
        }
        var:nesnd_india_dispute_subject ?= {
            if = {
                limit = { has_journal_entry = nesnd_india_treaty_dispute }
                je:nesnd_india_treaty_dispute = {
                    add_progress = { name = nesnd_india_treaty_dispute_bar value = $VALUE$ }
                }
            }
        }
    }
}
# Mughal scope, monthly. Keeps France's dispute bar locked to Mughal's value.
nesnd_india_dispute_sync_overlord = {
    if = {
        limit = {
            has_journal_entry = nesnd_india_treaty_dispute
            exists = var:nesnd_india_dispute_overlord
        }
        je:nesnd_india_treaty_dispute = {
            save_scope_value_as = {
                name = mug_dispute_bar_val
                value = "scripted_bar_progress(nesnd_india_treaty_dispute_bar)"
            }
        }
        var:nesnd_india_dispute_overlord = {
            if = {
                limit = { has_journal_entry = nesnd_india_treaty_dispute_overlord }
                je:nesnd_india_treaty_dispute_overlord = {
                    set_bar_progress = { name = nesnd_india_treaty_dispute_bar value = scope:mug_dispute_bar_val }
                }
            }
        }
    }
}
# Mughal button action: send envoys to petition (cost 4x weekly income, bar +10, 12m cooldown).
nesnd_india_dispute_petition_do = {
    if = {
        limit = {
            has_journal_entry = nesnd_india_treaty_dispute
            gold_reserves >= nesnd_india_treaty_dispute_price
            in_default = no
            NOT = { has_modifier = declared_bankruptcy }
            NOT = { has_variable = nesnd_india_dispute_petition_wait }
        }
        add_treasury = { value = nesnd_india_treaty_dispute_price multiply = -1 }
        set_variable = { name = nesnd_india_dispute_petition_wait months = 12 }
        nesnd_india_treaty_dispute_shift = { VALUE = 10 }
    }
}
# Mughal button action: withhold tribute (bar -15, primary cultures fervor target +5 for 5y, 12m cooldown).
nesnd_india_dispute_withhold_do = {
    if = {
        limit = {
            has_journal_entry = nesnd_india_treaty_dispute
            NOT = { has_variable = nesnd_india_dispute_withhold_wait }
        }
        set_variable = { name = nesnd_india_dispute_withhold_wait months = 12 }
        nesnd_india_treaty_dispute_shift = { VALUE = -15 }
        save_scope_as = mug_dispute_fervor_country
        every_primary_culture = {
            scope:mug_dispute_fervor_country = {
                add_fervor_target_modifier = { culture = prev days = 1825 multiplier = 5 }
            }
        }
    }
}
# France button action: reaffirm the old treaty (bar -10, relations -10, 12m cooldown).
nesnd_india_dispute_reaffirm_do = {
    if = {
        limit = {
            has_journal_entry = nesnd_india_treaty_dispute_overlord
            NOT = { has_variable = nesnd_india_dispute_reaffirm_wait }
        }
        set_variable = { name = nesnd_india_dispute_reaffirm_wait months = 12 }
        nesnd_india_treaty_dispute_shift = { VALUE = -10 }
        clear_saved_scope = mug_dispute_rel_target
        var:nesnd_india_dispute_subject ?= { save_scope_as = mug_dispute_rel_target }
        if = {
            limit = { exists = scope:mug_dispute_rel_target }
            change_relations = { country = scope:mug_dispute_rel_target value = -10 }
        }
    }
}
# France button action: grant revised terms (cost 4x weekly income, bar +10, 12m cooldown).
nesnd_india_dispute_concede_do = {
    if = {
        limit = {
            has_journal_entry = nesnd_india_treaty_dispute_overlord
            gold_reserves >= nesnd_india_treaty_dispute_price
            in_default = no
            NOT = { has_modifier = declared_bankruptcy }
            NOT = { has_variable = nesnd_india_dispute_concede_wait }
        }
        add_treasury = { value = nesnd_india_treaty_dispute_price multiply = -1 }
        set_variable = { name = nesnd_india_dispute_concede_wait months = 12 }
        nesnd_india_treaty_dispute_shift = { VALUE = 10 }
    }
}
# Mughal AI: 5% monthly chance; withhold if liberty_desire >= 50, else petition.
nesnd_india_dispute_subject_ai = {
    if = {
        limit = { is_ai = yes has_journal_entry = nesnd_india_treaty_dispute }
        random = {
            chance = 5
            if = {
                limit = { liberty_desire >= 50 }
                nesnd_india_dispute_withhold_do = yes
            }
            else = {
                nesnd_india_dispute_petition_do = yes
            }
        }
    }
}
# France AI: 5% monthly chance; reaffirm if relations < 0, else concede.
nesnd_india_dispute_overlord_ai = {
    if = {
        limit = { is_ai = yes has_journal_entry = nesnd_india_treaty_dispute_overlord }
        random = {
            chance = 5
            if = {
                limit = { nesnd_india_dispute_relations_below_zero = yes }
                nesnd_india_dispute_reaffirm_do = yes
            }
            else = {
                nesnd_india_dispute_concede_do = yes
            }
        }
    }
}
# Country scope, from a concluding dispute outcome. Pushes the partner's bar to the same end, so that its
# journal concludes with the same outcome on its next check instead of lingering with a bar a point short.
nesnd_india_dispute_push_partner = {
    scope:mug_dispute_partner ?= {
        if = {
            limit = { has_journal_entry = $JE$ }
            je:$JE$ = { set_bar_progress = { name = nesnd_india_treaty_dispute_bar value = $END$ } }
        }
    }
}

nesnd_india_dispute_cleanup_overlord = {

}
