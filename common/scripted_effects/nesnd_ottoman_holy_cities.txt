# Effects for Ottoman B11 "The Holy Cities" journal (nesnd_ottoman_holy_cities) and Two Caliphs events (.32-.35).
# Country scope (TUR) unless noted.

# Selects the highest-scoring Holy Cities foe candidate and stores it in global_var:nesnd_holy_foe (Rule 5).
nesnd_ottoman_holy_pick_foe = {
    if = {
        limit = {
            has_global_variable = nesnd_holy_foe
        }
        remove_global_variable = nesnd_holy_foe
    }
    if = {
        limit = {
            any_country = {
                nesnd_ottoman_holy_foe_candidate = yes
            }
        }
        ordered_country = {
            limit = {
                nesnd_ottoman_holy_foe_candidate = yes
            }
            order_by = nesnd_ottoman_holy_foe_score
            position = 0
            check_range_bounds = no
            set_global_variable = {
                name = nesnd_holy_foe
                value = this
            }
        }
    }
}

# Unified entry effect for opening nesnd_ottoman_holy_cities (Rule 7).
nesnd_ottoman_holy_start = {
    hidden_effect = {
        nesnd_ottoman_holy_pick_foe = yes
    }
    add_journal_entry = {
        type = nesnd_ottoman_holy_cities
    }
}

# Monthly counter of consecutive months unable to act abroad for R13 (Rule 8).
nesnd_ottoman_holy_update_dependent_months = {
    if = {
        limit = {
            nesnd_ottoman_can_act_abroad = yes
        }
        if = {
            limit = {
                has_variable = nesnd_ottoman_holy_dependent_months
            }
            remove_variable = nesnd_ottoman_holy_dependent_months
        }
    }
    else_if = {
        limit = {
            has_variable = nesnd_ottoman_holy_dependent_months
        }
        change_variable = {
            name = nesnd_ottoman_holy_dependent_months
            add = 1
        }
    }
    else = {
        set_variable = {
            name = nesnd_ottoman_holy_dependent_months
            value = 1
        }
    }
}

# Picks the independent small holder with the most states in STATE_HEDJAZ and dispatches protectorate offer event .32 (Rule 12).
nesnd_ottoman_holy_dispatch_offer = {
    set_variable = {
        name = nesnd_ottoman_holy_offer_cd
        months = 60
    }
    if = {
        limit = {
            any_country = {
                nesnd_ottoman_holy_offer_target = yes
            }
        }
        ordered_country = {
            limit = {
                nesnd_ottoman_holy_offer_target = yes
            }
            order_by = nesnd_ottoman_holy_held_states_count
            position = 0
            check_range_bounds = no
            trigger_event = {
                id = nesnd_ottoman_recovery.32
                days = 3
                popup = yes
            }
        }
    }
}

# Dispatches the one-time Two Caliphs event .34 on TUR (Rule 13).
nesnd_ottoman_two_caliphs_dispatch = {
    if = {
        limit = {
            nesnd_ottoman_two_caliphs_due = yes
        }
        trigger_event = {
            id = nesnd_ottoman_recovery.34
            popup = yes
        }
    }
}

# Shared cleanup called at the end of every outcome of nesnd_ottoman_holy_cities (Rule 9).
# Preserves nesnd_ottoman_caliph_claimed so dependent/changed outcomes can reopen after the 24-month cooldown.
nesnd_ottoman_holy_cleanup = {
    if = {
        limit = {
            has_global_variable = nesnd_holy_foe
        }
        remove_global_variable = nesnd_holy_foe
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_holy_dependent_months
        }
        remove_variable = nesnd_ottoman_holy_dependent_months
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_holy_deadline
        }
        remove_variable = nesnd_ottoman_holy_deadline
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_holy_planned
        }
        remove_variable = nesnd_ottoman_holy_planned
    }
    if = {
        limit = {
            has_variable = nesnd_ottoman_holy_offer_cd
        }
        remove_variable = nesnd_ottoman_holy_offer_cd
    }
}
