# COUNTRY scope (USA), during the secession crisis. Gives the seceding movement's
# supporters radicals so the engine starts its civil war; var:nesnd_acw_crisis_side
# 1 = southern secession (CSA), 2 = northern secession (FSA).
nesnd_acw_push_secession = {
    set_variable = { name = nesnd_acw_push_wait months = 6 }
    if = {
        limit = { has_variable = nesnd_acw_crisis_side var:nesnd_acw_crisis_side = 2 }
        if = {
            limit = { NOT = { any_political_movement = { is_political_movement_type = movement_anti_slavery } } }
            create_political_movement = { type = movement_anti_slavery }
        }
        every_scope_state = {
            limit = { is_slave_state = no }
            add_radicals_in_state = { value = 0.25 strata = middle }
            add_radicals_in_state = { value = 0.25 strata = upper }
        }
    }
    else = {
        if = {
            limit = { NOT = { any_political_movement = { is_political_movement_type = movement_pro_slavery } } }
            create_political_movement = { type = movement_pro_slavery }
        }
        # Same border-state exceptions as the vanilla secession weights.
        every_scope_state = {
            limit = {
                nesnd_acw_southern_state = yes
                NOR = {
                    state_region = s:STATE_MARYLAND
                    state_region = s:STATE_DELAWARE
                    state_region = s:STATE_DISTRICT_OF_COLUMBIA
                    state_region = s:STATE_WEST_VIRGINIA
                }
            }
            add_radicals_in_state = { value = 0.30 strata = upper }
            add_radicals_in_state = { value = 0.20 pop_type = farmers }
            add_radicals_in_state = { value = 0.30 pop_type = officers }
        }
    }
}

# Province scope. Preserve the outgoing holder for the handover tick:
# the native transfer and the war goal's scripted effect can run in either order.
nesnd_hus_west_observe_owner = {
    if = {
        limit = { owner ?= { nesnd_hus_west_american_side = yes } }
        set_variable = { name = nesnd_hus_west_giver value = owner }
    }
    else_if = {
        limit = { exists = var:nesnd_hus_west_giver }
        set_variable = { name = nesnd_hus_west_recent_giver value = var:nesnd_hus_west_giver days = 1 }
        remove_variable = nesnd_hus_west_giver
    }
    if = {
        limit = { NOT = { owner ?= c:HUS } }
        remove_variable = nesnd_hus_west_awarded
        remove_variable = nesnd_hus_west_pending_play
    }
}

# Called only by the native return_state and conquer_state on_enforced effects.
nesnd_hus_west_territorial_award = {
    if = {
        limit = {
            c:HUS ?= this
            has_journal_entry = nesnd_hus_west_lever
            scope:target_country ?= { nesnd_hus_west_american_side = yes }
            exists = scope:target_state.state_region
            hidden_trigger = { is_target_in_variable_list = { name = nesnd_hus_west_claims target = scope:target_state.state_region } }
        }
        scope:diplomatic_play ?= { set_variable = nesnd_hus_west_has_award }
        scope:target_state.state_region = {
            every_in_list = {
                variable = nesnd_hus_west_land
                limit = {
                    OR = { owner ?= scope:target_country owner ?= c:HUS }
                    OR = {
                        owner ?= scope:target_country
                        hidden_trigger = { var:nesnd_hus_west_giver ?= scope:target_country }
                        hidden_trigger = { var:nesnd_hus_west_recent_giver ?= scope:target_country }
                    }
                }
                set_variable = nesnd_hus_west_awarded
                if = {
                    limit = { exists = scope:diplomatic_play }
                    set_variable = { name = nesnd_hus_west_pending_play value = scope:diplomatic_play }
                }
            }
        }
    }
}

nesnd_hus_west_make_claim = {
    ordered_state = {
        limit = { nesnd_hus_west_claim_candidate = yes }
        order_by = state_population
        max = 1
        owner = { save_scope_as = nesnd_hus_west_respondent }
        state_region = {
            save_scope_as = nesnd_hus_west_region
            add_claim = c:HUS
            c:HUS = { add_to_variable_list = { name = nesnd_hus_west_claims target = prev } }
        }
        # Province IDs survive split-state mergers. Only take this snapshot when a claim is made.
        every_province_in_americas = {
            limit = {
                state.state_region ?= scope:nesnd_hus_west_region
                owner ?= { nesnd_hus_west_american_side = yes }
            }
            set_variable = { name = nesnd_hus_west_giver value = owner }
            scope:nesnd_hus_west_region = { add_to_variable_list = { name = nesnd_hus_west_land target = prev } }
        }
        c:HUS = {
            je:nesnd_hus_west_lever = { add_progress = { name = nesnd_hus_west_claim_bar value = -50 } }
            set_variable = { name = nesnd_hus_west_claim_cooldown months = 6 }
            change_relations = { country = scope:nesnd_hus_west_respondent value = -10 }
        }
    }
}

nesnd_hus_west_push_east = {
    ordered_state = {
        limit = { nesnd_hus_west_press_candidate = yes }
        order_by = state_population
        max = 1
        save_scope_as = nesnd_hus_west_target
        if = { limit = { NOT = { has_claim_by = c:HUS } } state_region = { add_claim = c:HUS } }
        owner = { save_scope_as = nesnd_hus_west_respondent }
        c:HUS = {
            create_diplomatic_play = {
                name = nesnd_hus_west_confrontation
                type = dp_return_state
                target_country = scope:nesnd_hus_west_respondent
                target_state = scope:nesnd_hus_west_target
                add_war_goal = { holder = scope:nesnd_hus_west_respondent target_country = c:HUS type = humiliation }
            }
            je:nesnd_hus_west_lever = { set_bar_progress = { name = nesnd_hus_west_claim_bar value = 0 } }
            set_variable = { name = nesnd_hus_west_border_cooldown months = 24 }
        }
    }
}
