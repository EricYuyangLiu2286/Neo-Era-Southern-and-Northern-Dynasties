# COUNTRY scope (USA), during the secession crisis. Gives the seceding movement's
# supporters radicals so the engine starts its civil war; var:nesnd_acw_crisis_side
# 1 = southern secession (CSA), 2 = northern secession (FSA).
nesnd_acw_push_secession = {
    set_variable = { name = nesnd_acw_push_wait months = 6 }
    if = {
        limit = { has_variable = nesnd_acw_crisis_side var:nesnd_acw_crisis_side = 2 }
        if = {
            limit = { NOT = { any_political_movement = { is_political_movement_type = movement_anti_slavery } } }
            create_political_movement = { type = movement_anti_slavery }
        }
        every_scope_state = {
            limit = { is_slave_state = no }
            add_radicals_in_state = { value = 0.25 strata = middle }
            add_radicals_in_state = { value = 0.25 strata = upper }
        }
    }
    else = {
        if = {
            limit = { NOT = { any_political_movement = { is_political_movement_type = movement_pro_slavery } } }
            create_political_movement = { type = movement_pro_slavery }
        }
        # Same border-state exceptions as the vanilla secession weights.
        every_scope_state = {
            limit = {
                nesnd_acw_southern_state = yes
                NOR = {
                    state_region = s:STATE_MARYLAND
                    state_region = s:STATE_DELAWARE
                    state_region = s:STATE_DISTRICT_OF_COLUMBIA
                    state_region = s:STATE_WEST_VIRGINIA
                }
            }
            add_radicals_in_state = { value = 0.30 strata = upper }
            add_radicals_in_state = { value = 0.20 pop_type = farmers }
            add_radicals_in_state = { value = 0.30 pop_type = officers }
        }
    }
}
