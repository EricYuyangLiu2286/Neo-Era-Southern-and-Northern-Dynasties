law_nesnd_keju_voting = {
	group = lawgroup_distribution_of_power
	parent = law_wealth_voting

	icon = "gfx/interface/icons/law_icons/nesnd_keju_voting.dds"

	progressiveness = 0

	unlocking_technologies = {
		democracy
	}

	disallowing_laws = {
		law_chiefdom
		law_command_economy
	}

	is_visible = {
		OR = {
			has_law = law_type:law_nesnd_keju_voting
			AND = {
				any_primary_culture = { has_discrimination_trait = tradition_sinosphere }
				NOT = { ig:ig_industrialists ?= { is_powerful = yes } }
			}
		}
	}

	can_enact = {
		custom_tooltip = {
			text = nesnd_keju_can_enact_tt
			any_primary_culture = { has_discrimination_trait = tradition_sinosphere }
			NOT = { ig:ig_industrialists ?= { is_powerful = yes } }
		}
	}

	modifier = {
		country_legitimacy_govt_total_votes_add = 65
		country_legitimacy_govt_total_clout_add = 75
		country_legitimacy_govt_size_add = 1
		country_authority_add = 100
		country_electoral_confidence_impact_mult = -0.2
		country_voting_power_from_literacy_add = 50
		country_voting_power_wealth_threshold_add = 15
		country_academics_voting_power_add = 50
		country_bureaucrats_voting_power_add = -15
		interest_group_in_government_attraction_mult = 0.10
		country_legitimacy_ideological_incoherence_mult = -0.1
		political_movement_radicalism_add = -0.05
		country_institution_cost_institution_schools_mult = 0.10
		state_education_access_wealth_add = 0.01
		state_peasants_education_access_add = -0.02
		building_academics_standard_of_living_add = 1
	}

	on_activate = {
		if = {
			limit = {
				game_date > 1836.1.1
				country_has_voting_franchise = yes
			}
			initialize_electoral_confidence = yes
		}
		random_law = {
			limit = {
				type = law_type:law_nesnd_keju_voting
			}
			save_scope_as = relevant_voting_law
		}
		handle_possible_absent_caciquismo_variables = yes
		scripted_effect_parties_emergence = yes
		inherit_caciquismo_effect = yes
		inherit_free_elections_effect = yes
		if = {
			limit = {
				has_law = law_type:law_monarchy
				has_variable = king_in_parliament_possible
				NOT = {
					has_journal_entry = je_king_in_parliament
				}
			}
			custom_tooltip = {
				text = king_in_parliament_tt
				add_journal_entry = { type = je_king_in_parliament }
			}
		}
	}

	can_impose = { always = no }

	ai_will_do = {
		OR = {
			has_law = law_type:law_nesnd_keju_voting
			AND = {
				any_primary_culture = { has_discrimination_trait = tradition_sinosphere }
				NOT = { ig:ig_industrialists ?= { is_powerful = yes } }
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		add = 100

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				owner ?= {
					has_journal_entry = je_zaibatsu
					has_variable = ai_zaibatsu_cement
				}
			}
			add = 500
		}
	}
}
