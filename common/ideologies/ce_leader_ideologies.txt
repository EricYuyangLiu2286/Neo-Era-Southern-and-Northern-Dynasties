ideology_ce_qin_royalist_leader = {
	icon = "gfx/interface/icons/ideology_icons/ideology_leader/ce_qin_ideology.dds"
	
	character_ideology = yes

	lawgroup_governance_principles = {
		law_monarchy = strongly_approve
		law_presidential_republic = neutral
		law_parliamentary_republic = disapprove
		law_theocracy = neutral
		law_council_republic = strongly_disapprove
	}
	
	lawgroup_distribution_of_power = {
		law_autocracy = strongly_approve
		law_oligarchy = approve
		law_single_party_state = approve
		law_technocracy = neutral
		law_landed_voting = neutral
		law_wealth_voting = disapprove
		law_census_voting = disapprove
		law_universal_suffrage = strongly_disapprove
		law_anarchy = strongly_disapprove
	}
	
	lawgroup_economic_system = {
		law_command_economy = neutral
		law_cooperative_ownership = neutral
		law_traditionalism = strongly_approve
		law_agrarianism = approve
		law_interventionism = disapprove
		law_laissez_faire = strongly_disapprove
	}
	
	lawgroup_land_reform = {
		law_serfdom = approve
		law_tenant_farmers = strongly_disapprove
		law_peasant_proprietorship = disapprove
		law_collectivized_agriculture = strongly_approve
		law_commercialized_agriculture = neutral # Commercilized land as an aid to land monopoly by large corps or state-owned/public-sector companies
	}

	lawgroup_welfare = {
		law_old_age_pension = approve
		law_wage_subsidies = strongly_approve
		law_poor_laws = disapprove
		law_no_social_security = strongly_disapprove
	}
	
	lawgroup_health_system = {
		law_private_health_insurance = neutral
		law_no_health_system = strongly_disapprove
		law_charitable_health_system = strongly_approve
		law_public_health_insurance = approve
	}	
	
	lawgroup_education_system = {
		law_public_schools = strongly_approve
		law_private_schools = disapprove
		law_religious_schools = neutral
		law_no_schools = strongly_disapprove 	
	}

	country_trigger = {
		# Agitators should have an ideology at odds with current laws
		exists = c:QE1
		OR = {
			c:QE1 = This
			AND = {
				c:PER = This
				exists = ruler
				ruler.culture = { has_discrimination_trait = heritage_han }
			}
			civil_war_origin_country ?= c:QE1
		}

		any_interest_group = {
			has_ideology = ideology:ideology_ce_qin_royalist
		}

		NOT = {	
			has_variable = monarchy_stamped_out_var
		}
	}
	
	interest_group_leader_trigger = {
		ideology_bonapartist_valid_trigger = yes
	}

	non_interest_group_leader_trigger = {
		# Agitator ideologies should disapprove of a current law
		NOT = { has_role_of_type = agitator }
		ideology_bonapartist_valid_trigger = yes
	}

	interest_group_leader_weight = {
		value = 100
		# less likely in a theocracy
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_theocracy
				}
			}
			add = -75
		}

		# less likely if issues are not relevant
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_tenant_farmers
					has_law_or_variant = law_type:law_mass_conscription
				}
			}
			add = -75
		}

		# more likely for PB and AF
		if = {
			limit = {
				OR = {
					is_interest_group_type = ig_petty_bourgeoisie
					is_interest_group_type = ig_armed_forces
				}
			}
			add = 50
		}

		# more likely if there's something to colonize nearby
		if = {
			limit = {
				any_country = {
					is_country_type = decentralized
					has_overlapping_interests = ROOT.owner
				}
			}
			add = 75
		}

		# more likely if the IG is happy with the monarchy, and that monarchy is not Bonapartist
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_monarchy
					#NOT = { has_variable = bonapartist_restoration_var }
				}
				ig_approval >= 5
			}
			add = 75
		}

		# reduced happy modifier if it's not a monarchy
		if = {
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_monarchy }
					#NOT = { has_variable = bonapartist_restoration_var }
				}
				ig_approval >= 5
			}
			add = 25
		}

		# more likely if the IG is unhappy with the monarchy, and that monarchy is not Bonapartist
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_monarchy
					#NOT = { has_variable = bonapartist_restoration_var }
				}
				ig_approval < 0
			}
			add = -75
		}

		# reduced unhappy modifier if it's not a monarchy
		if = {
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_monarchy }
					#NOT = { has_variable = bonapartist_restoration_var }
				}
				ig_approval < 0
			}
			add = -25
		}

		# more likely if you border a rival
		if = {
			limit = {
				owner = {
					any_rival_country = {
						any_neighbouring_state = {
							owner = root.owner
						}
					}
				}
			}
			add = 55
		}

		# more likely if you're the defender in a war
		if = {
			limit = {
				owner = {
					is_at_war = yes
					any_scope_war = {
						defender_warleader = { this = ROOT.owner }
					}
				}
			}
			add = 55
		}
		
		# less likely for lower profile igs and armed forces in CNE with land draft JE
		if = {
			limit = {
				owner = {
					exists = c:CNE
					c:CNE ?= This 
					has_journal_entry = ce_cne_land_reform
				}
				OR = {
						is_interest_group_type = ig_rural_folk
						is_interest_group_type = ig_petty_bourgeoisie
						is_interest_group_type = ig_devout
						is_interest_group_type = ig_armed_forces
					}
			}
			add = -100
		}

		# conservative and/or reactionary leaders have a higher spawn chance if the Anti-Dreyfusards won.
		if = {
			limit = {
				has_variable = anti_dreyfusard_victory
				owner = {
					any_primary_culture = {
						has_discrimination_trait = heritage_han
					}
				}
			}
			add = 55
		}
		# More likely under monarchist qin
		if = {
			limit = {
				exists = c:QE1
				owner = {
					c:QE1 ?= This
					has_law_or_variant = law_type:law_monarchy
				}
			}
			add = 75
		}
	}

	non_interest_group_leader_weight = {
		value = 100
		# less likely in a theocracy
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_theocracy
				}
			}
			add = -75
		}
		# less likely if issues are not relevant
		
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_tenant_farmers
					has_law_or_variant = law_type:law_mass_conscription
				}
			}
			add = -75
		}

		if = {
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_monarchy }
				}
				interest_group.ig_approval > 0
			}
			add = -15
		}

		# more likely if there's something to colonize nearby
		if = {
			limit = {
				any_country = {
					is_country_type = decentralized
					has_overlapping_interests = ROOT.owner
				}
			}
			add = 75
		}

		# reduced unhappy modifier if it's not a monarchy
		if = {
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_monarchy }
					
				}
				interest_group.ig_approval < 0
			}
			add = 25
		}

		# more likely if you border a rival
		if = {
			limit = {
				owner = {
					any_rival_country = {
						any_neighbouring_state = {
							owner = root.owner
						}
					}
				}
			}
			add = 55
		}

		# more likely if you're the defender in a war
		if = {
			limit = {
				owner = {
					is_at_war = yes
					any_scope_war = {
						defender_warleader = { this = ROOT.owner }
					}
				}
			}
			add = 55
		}

		# less likely for lower profile igs and armed forces in CNE with land draft JE
		if = {
			limit = {
				owner = {
					exists = c:CNE
					c:CNE ?= This
					has_journal_entry = ce_cne_land_reform
				}
				interest_group ?= {
					OR = {
						is_interest_group_type = ig_rural_folk
						is_interest_group_type = ig_petty_bourgeoisie
						is_interest_group_type = ig_devout
						is_interest_group_type = ig_armed_forces
					}
				}
			}
			add = -100
		}

		# conservative and/or reactionary leaders have a higher spawn chance if the Anti-Dreyfusards won.
		if = {
			limit = {
				interest_group ?= {
					has_variable = anti_dreyfusard_victory
				}
				owner = {
					any_primary_culture = {
						has_discrimination_trait = heritage_han
					}
				}
			}
			add = 55
		}
		# More likely under monarchist qin
		if = {
			limit = {
				exists = c:QE1
				owner = {
					c:QE1 ?= This
					has_law_or_variant = law_type:law_monarchy
				}
			}
			add = 75
		}
	}
}
