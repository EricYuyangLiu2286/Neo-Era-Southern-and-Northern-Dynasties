# Called from JAP in the existing finalization branch. Every current USM/JAP
# intersection intended for another member must be assigned before completion.
# Third-country owners are not selected; a foreign controller leaves it pending.

 c:USM ?= {
  NOT = { any_scope_state = {
   NOR = { state_region = s:STATE_KYUSHU state_region = s:STATE_SHIKOKU }
   OR = {
    has_variable = nesnd_japan_before_annex
    state_region = s:STATE_KANTO state_region = s:STATE_KYOTO state_region = s:STATE_KANSAI
    state_region = s:STATE_CHUGOKU state_region = s:STATE_TOHOKU state_region = s:STATE_TOKAI
    state_region = s:STATE_HOKUSHINETSU state_region = s:STATE_HOKKAIDO
   }

 }

   state_region = s:STATE_KANTO state_region = s:STATE_KYOTO state_region = s:STATE_TOHOKU

   state_region = s:STATE_KYUSHU state_region = s:STATE_SHIKOKU

 }
