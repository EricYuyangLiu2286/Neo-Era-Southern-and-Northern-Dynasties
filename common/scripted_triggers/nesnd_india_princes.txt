        OR = {
            $SENDER$ = { is_direct_subject_of = prev }
            AND = {
                exists = c:USM
                $SENDER$ = { is_direct_subject_of = c:USM }

}

# A country's reported stake includes private investors. Even if every non-government
# level belongs to its investors, the remaining levels must belong to its government.
nesnd_india_workshop_stake_met = {
    # Ownership can be read only on a building that stands (level > 0, as in vanilla
    # 06_portugal_politics.txt:55); a workshop still in the construction queue is skipped.
    any_scope_building = {
        is_building_type = building_tooling_workshop
        level > 0
        levels_owned_by_country = {
            target = scope:india_partner
            value >= {
                value = level
                subtract = nesnd_india_government_levels
                add = scope:india_sender.nesnd_india_workshop_levels
            }
        }
    }
}
nesnd_india_workshop_site = {
    OR = {
        nesnd_india_workshop_stake_met = yes
        AND = {
            can_construct_building = building_tooling_workshop
            # Writing the promised stake is safe if either total caps the donor's
            # existing government stake at the promised number of levels.
            NOT = {
                any_scope_building = {
                    is_building_type = building_tooling_workshop
                    level > 0
                    nesnd_india_government_levels > scope:india_sender.nesnd_india_workshop_levels
                    levels_owned_by_country = {
                        target = scope:india_partner
                        value > scope:india_sender.nesnd_india_workshop_levels
                    }
                }
            }
        }
    }
}

# Princes joining Mughal ("Mughal" here is any country for which nesnd_india_empire holds).
# Two ways in, both through the request channel (nesnd_india_princes.txt):
#   request kind 6   Mughal invites a prince (button nesnd_india_request_6, choice event nesnd_india_diplomacy.5)
#   request kind 8   a prince asks to join (button nesnd_india_union_ask_button, choice event nesnd_india_union.4)
# When the prince is another country's subject, that overlord's consent comes first, as matter kind 6
# of Indian affairs (nesnd_india_matters.txt); once it is given the invitation or the request goes into
# the request channel. Accepting runs the one settlement, nesnd_india_settle_union = { MODE = merge|dominion }.

# Country scope. A government that can take in a prince: independent, reform complete, community 60.
nesnd_india_union_host = {
    hidden_trigger = { nesnd_india_empire = yes }
    custom_tooltip = { text = nesnd_india_union_host_free_tt is_subject = no }
    custom_tooltip = { text = nesnd_ui_india_reform_done_tt has_variable = nesnd_india_reform_done }
    custom_tooltip = { text = nesnd_india_union_community_tt nesnd_india_community >= 60 }
}

# Prince scope. The prince can ask the government in nesnd_india_union_host (the richest Mughal,
# set monthly by nesnd_india_union_pick_host) to take it in. Mughal's conditions are those of an
# invitation, less the one that some prince is eligible.
nesnd_india_can_ask_union = {
    nesnd_india_can_send = yes
    hidden_trigger = { nesnd_india_prince = yes }
    custom_tooltip = { text = nesnd_india_union_host_exists_tt exists = var:nesnd_india_union_host }
    trigger_if = { limit = { exists = var:nesnd_india_union_host }
        var:nesnd_india_union_host = { nesnd_india_union_host = yes }
        NOT = { has_war_with = var:nesnd_india_union_host }
        trigger_if = { limit = { nesnd_india_consent_needed = { OTHER = var:nesnd_india_union_host GRANTS = yes } }
            custom_tooltip = { text = nesnd_india_union_no_project_tt NOR = { has_variable = nesnd_india_matter_kind has_variable = nesnd_india_project_type } }
        }
    }
}
# Prince scope. The prince is already a subject of the government it would join: it can only be merged.
nesnd_india_union_own_subject = {
    trigger_if = { limit = { exists = var:nesnd_india_union_host } is_subject = yes overlord ?= var:nesnd_india_union_host }
    trigger_else = { always = no }
}

# Holder scope, matter kind 6. Both governments are still what they were when the matter was sent.
nesnd_india_union_matter_valid = {
    trigger_if = { limit = { exists = var:nesnd_india_matter_to }
        trigger_if = { limit = { nesnd_india_empire = yes }
            nesnd_india_union_host = yes
            var:nesnd_india_matter_to = { nesnd_india_prince = yes is_at_war = no }
        }
        trigger_else = {
            nesnd_india_prince = yes is_at_war = no
            var:nesnd_india_matter_to = { nesnd_india_union_host = yes }
        }
    }
    trigger_else = { always = no }
}

# ---------------------------------------------------------------------------------------------
# In the reply event (answering scope, with scope:india_sender and scope:india_partner),
# request kinds 6 and 8. In kind 6 the sender is Mughal and the answer is the prince's; in kind 8
# the sender is the prince and the answer is Mughal's.
# ---------------------------------------------------------------------------------------------
nesnd_india_union_reply = { OR = { nesnd_india_incoming_kind = 6 nesnd_india_incoming_kind = 8 } }
# The prince is another country's subject: that overlord is paid compensation when it joins.
nesnd_india_union_compensation_due = {
    trigger_if = { limit = { nesnd_india_incoming_kind = 6 exists = scope:india_partner exists = scope:india_sender }
        scope:india_partner = { nesnd_india_consent_needed = { OTHER = scope:india_sender GRANTS = yes } }
    }
    trigger_else_if = { limit = { nesnd_india_incoming_kind = 8 exists = scope:india_partner exists = scope:india_sender }
        scope:india_sender = { nesnd_india_consent_needed = { OTHER = scope:india_partner GRANTS = yes } }
    }
    trigger_else = { always = no }
}
# The prince is not already Mughal's subject, so it may choose to become Mughal's dominion.
nesnd_india_union_dominion_open = {
    trigger_if = { limit = { nesnd_india_incoming_kind = 6 exists = scope:india_partner exists = scope:india_sender }
        custom_tooltip = { text = nesnd_india_union_not_own_subject_tt scope:india_partner = { NAND = { is_subject = yes overlord ?= scope:india_sender } } }
        nesnd_india_union_dominion_possible = { EMPIRE = scope:india_sender PRINCE = scope:india_partner }
    }
    trigger_else_if = { limit = { nesnd_india_incoming_kind = 8 exists = scope:india_partner exists = scope:india_sender }
        custom_tooltip = { text = nesnd_india_union_sender_not_subject_tt scope:india_sender = { NAND = { is_subject = yes overlord ?= scope:india_partner } } }
        nesnd_india_union_dominion_possible = { EMPIRE = scope:india_partner PRINCE = scope:india_sender }
    }
    trigger_else = { always = no }
}
# The dominion subject type (common/subject_types/ce_subject_types.txt) needs a recognised or colonial
# overlord of minor power rank or above, ranked higher than the subject.
nesnd_india_union_dominion_possible = {
    custom_tooltip = {
        text = nesnd_india_union_dominion_rank_tt
        $EMPIRE$ = { OR = { is_country_type = recognized is_country_type = colonial } country_rank >= rank_value:minor_power country_rank > $PRINCE$.country_rank }
        # valid_subject_ranks of the dominion: minor power and below (ce_subject_types.txt:109-114).
        $PRINCE$ = { country_rank <= rank_value:minor_power }
    }
}
# Empire scope. The princely pensions already count nesnd_india_pension_cap shares: one more prince
# only restarts the ten years (nesnd_india_add_pension).
nesnd_india_pension_full = {

    nesnd_india_pension_count_value >= nesnd_india_pension_cap
}
# Mughal can pay the quote, and the compensation as well when it is due.
nesnd_india_union_can_pay = {
    custom_tooltip = {
        text = nesnd_india_union_can_pay_tt
        trigger_if = { limit = { nesnd_india_incoming_kind = 6 exists = scope:india_sender } scope:india_sender = { gold_reserves >= nesnd_india_union_due } }
        trigger_else_if = { limit = { exists = scope:india_partner } scope:india_partner = { gold_reserves >= nesnd_india_union_due } }
        trigger_else = { always = no }
    }
}

        is_subject_type = subject_type_protectorate
        is_subject_type = subject_type_tributary
        is_subject_type = subject_type_dominion
        is_subject_type = subject_type_chartered_company
        is_subject_type = subject_type_nesnd_egypt_protectorate
    }

    hidden_trigger = { has_journal_entry = nesnd_india_indep is_subject = yes exists = overlord }
    is_at_war = no

    can_increase_autonomy = yes

    trigger_else = {
        nesnd_india_channel_free = yes
        overlord = { is_country_alive = yes }
        NOT = { has_war_with = overlord }
    }
}
