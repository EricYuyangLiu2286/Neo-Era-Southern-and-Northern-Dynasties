# 「伊犁的朝议」朝争条与朝争事件的判定。Root = country。
# 朝争就是朝争条 nesnd_qe1_intrigue_bar 的读数，由条的 monthly_progress 逐月累加，不另存变量。日志不在时为否。
nesnd_qe1_intrigue_at_least = { je:nesnd_qe1_court_journal ?= { scripted_bar_progress = { name = nesnd_qe1_intrigue_bar value >= $V$ } } }
nesnd_qe1_court_half = {
	OR = {
		nesnd_qe1_power_at_least = { WHO = crown V = 50 }
		nesnd_qe1_power_at_least = { WHO = literati V = 50 }
		nesnd_qe1_power_at_least = { WHO = army V = 50 }
	}
}
nesnd_qe1_court_dominant = {
	OR = {
		nesnd_qe1_power_at_least = { WHO = crown V = 60 }
		nesnd_qe1_power_at_least = { WHO = literati V = 60 }
		nesnd_qe1_power_at_least = { WHO = army V = 60 }
	}
}
