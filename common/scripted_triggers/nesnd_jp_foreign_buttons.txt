nesnd_jp_seek_capital_cne_available = {
    trigger_if = { limit = { exists = c:CNE }
        nesnd_jp_can_send_request = yes is_subject = no
        c:CNE = { is_country_alive = yes }
        NOT = { has_war_with = c:CNE }
        NOT = { has_variable = nesnd_jp_capital_aid_lock }
        NOT = { nesnd_jp_industry_finished = yes }
    } trigger_else = { always = no }
}

nesnd_jp_union_jap_available = {
    trigger_if = { limit = { exists = c:JAP }
        nesnd_jp_can_send_request = yes
        nesnd_jp_diplomatic_standing = yes
        NOT = { c:JAP = this }
        is_at_war = no is_active_in_diplomatic_play = no
        c:JAP = { nesnd_jp_diplomatic_standing = yes is_country_alive = yes is_at_war = no is_active_in_diplomatic_play = no }
        trigger_if = { limit = { is_subject = yes exists = overlord }
            nesnd_jp_authorized_union = yes
            c:JAP = { trigger_if = { limit = { is_subject = yes exists = overlord } overlord = root.overlord } trigger_else = { always = no } }
        } trigger_else = { is_subject = no c:JAP = { is_subject = no } }
    } trigger_else = { always = no }
}
