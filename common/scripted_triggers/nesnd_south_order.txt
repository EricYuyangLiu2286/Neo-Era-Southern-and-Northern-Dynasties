# The orders of the southern realm after the succession: Wu's hierarchy by culture, Ming's by profession.
# Rules learned from the Eastern Continent and Persian hierarchies:
#  - no cu: links inside pop_criteria (they do not resolve there); test the pop's culture key with has_pop_culture
#  - every culture test sits inside a custom_tooltip, because the class panel builds its description without a pop
#    and a bare culture = { } switch then prints broken text and fills the error log
#  - the classes of one hierarchy are mutually exclusive and cover every pop; the catch-all comes last
# Pop scope.
nesnd_order_zhuzu_pop = {
    custom_tooltip = {
        text = nesnd_order_zhuzu_pop_tt
        OR = {
            has_pop_culture = manchu
            has_pop_culture = japanese
            has_pop_culture = nesnd_jp_togoku
            has_pop_culture = nesnd_jp_ou
            has_pop_culture = nesnd_jp_saigoku
        }
    }
}
nesnd_order_ryukyu_pop = { custom_tooltip = { text = nesnd_order_ryukyu_pop_tt has_pop_culture = ryukyuan } }
nesnd_order_wu_pop = { custom_tooltip = { text = nesnd_order_wu_pop_tt OR = { has_pop_culture = ce_wuu has_pop_culture = han } } }
nesnd_order_nan_pop = { custom_tooltip = { text = nesnd_order_nan_pop_tt OR = { has_pop_culture = ce_sichuan has_pop_culture = ce_chu has_pop_culture = ce_xinan } } }
nesnd_order_hua_pop = {
    custom_tooltip = {
        text = nesnd_order_hua_pop_tt
        culture = { has_discrimination_trait = heritage_han }
        NOR = { nesnd_order_zhuzu_pop = yes nesnd_order_ryukyu_pop = yes nesnd_order_wu_pop = yes nesnd_order_nan_pop = yes }
    }
}
nesnd_order_other_pop = {
    custom_tooltip = {
        text = nesnd_order_other_pop_tt
        NOR = { nesnd_order_zhuzu_pop = yes nesnd_order_ryukyu_pop = yes nesnd_order_wu_pop = yes nesnd_order_nan_pop = yes nesnd_order_hua_pop = yes }
    }
}
