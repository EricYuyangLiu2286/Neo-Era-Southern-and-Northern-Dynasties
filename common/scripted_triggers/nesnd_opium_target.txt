# country root = producer; this = a foreign importing market.
# Keep JE completion, target selection and click-time validation identical.
nesnd_opium_target_market = {
    NOT = { owner = root }
    mg:opium ?= { market_goods_imports >= 200 }
    owner = {
        is_country_alive = yes
        NOT = { has_variable = ce_war_to_opium_experienced }
        NOT = { has_variable = can_have_je_ce_war_to_opium_variable }
        NOT = { has_journal_entry = je_ce_war_to_opium }
        any_primary_culture = { NOT = { has_cultural_obsession = opium } }
    }
}
# country scope, used only inside the event that saves these named scopes.
nesnd_opium_selected_target_valid = {
    exists = scope:opium_addict_country
    exists = scope:nesnd_opium_target_market
    scope:nesnd_opium_target_market = {
        nesnd_opium_target_market = yes
        owner = scope:opium_addict_country
    }
}

# The holder of the demand must be the country that received the crisis.
nesnd_can_target_opium_instigator = {
    has_variable = ce_war_to_opium_experienced
    exists = var:ce_opium_instigator
    var:ce_opium_instigator ?= scope:target_country
    scope:target_country = {
        is_country_alive = yes
        any_scope_state = {
            any_scope_building = { is_building_type = building_opium_plantation }
        }
    }
}
