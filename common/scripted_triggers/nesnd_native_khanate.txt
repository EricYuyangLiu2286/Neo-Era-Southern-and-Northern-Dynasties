nesnd_khanate_has_play = {
 trigger_if = { limit = { nesnd_khanate_exists = yes }
  trigger_if = { limit = { has_variable = nesnd_khanate_war_dispatched }
   trigger_if = { limit = { exists = var:nesnd_khanate_play }
    any_diplomatic_play = { this = prev.var:nesnd_khanate_play is_diplomatic_play_type = dp_nesnd_khanate_suppression initiator = prev target = c:NKH }
   }
   trigger_else = { always = no }
  }
  trigger_else = { any_diplomatic_play = { initiator = prev target = c:NKH is_diplomatic_play_type = dp_nesnd_khanate_suppression } }
 }
 trigger_else = { always = no }
}

nesnd_khanate_goal_bound_pair = {
 trigger_if = { limit = { exists = scope:target_country exists = scope:diplomatic_play }
  scope:diplomatic_play = { is_diplomatic_play_type = dp_nesnd_khanate_suppression }
  OR = {
   AND = {
    scope:target_country = { c:NKH ?= this has_variable = nesnd_native_khanate }
    scope:diplomatic_play = { initiator = root target = scope:target_country }
    root = {
     nesnd_khanate_original_opponent = yes
     NOT = { has_variable = nesnd_khanate_war_outcome }
     trigger_if = { limit = { has_variable = nesnd_khanate_war_dispatched has_variable = nesnd_khanate_war_epoch exists = var:nesnd_khanate_case exists = var:nesnd_khanate_play }
      var:nesnd_khanate_case = scope:target_country var:nesnd_khanate_play = scope:diplomatic_play
     }
     trigger_else = { always = no }
    }
    OR = {
     AND = {
      scope:target_country = {
       var:nesnd_khanate_original_usa ?= root
       trigger_if = { limit = { has_variable = nesnd_khanate_usa_epoch exists = var:nesnd_khanate_usa_play }
        trigger_if = { limit = { root = { has_variable = nesnd_khanate_war_epoch } }
         var:nesnd_khanate_usa_epoch = root.var:nesnd_khanate_war_epoch var:nesnd_khanate_usa_play = scope:diplomatic_play
        }
        trigger_else = { always = no }
       }
       trigger_else = { always = no }
      }
     }
     AND = {
      scope:target_country = {
       var:nesnd_khanate_original_hus ?= root
       trigger_if = { limit = { has_variable = nesnd_khanate_hus_epoch exists = var:nesnd_khanate_hus_play }
        trigger_if = { limit = { root = { has_variable = nesnd_khanate_war_epoch } }
         var:nesnd_khanate_hus_epoch = root.var:nesnd_khanate_war_epoch var:nesnd_khanate_hus_play = scope:diplomatic_play
        }
        trigger_else = { always = no }
       }
       trigger_else = { always = no }
      }
     }
    }
   }
   AND = {
    root = { c:NKH ?= this has_variable = nesnd_native_khanate }
    scope:diplomatic_play = { initiator = scope:target_country target = root }
    scope:target_country = {
     nesnd_khanate_original_opponent = yes
     NOT = { has_variable = nesnd_khanate_war_outcome }
     trigger_if = { limit = { has_variable = nesnd_khanate_war_dispatched has_variable = nesnd_khanate_war_epoch exists = var:nesnd_khanate_case exists = var:nesnd_khanate_play }
      var:nesnd_khanate_case = root var:nesnd_khanate_play = scope:diplomatic_play
     }
     trigger_else = { always = no }
    }
    OR = {
     AND = {
      root = {
       var:nesnd_khanate_original_usa ?= scope:target_country
       trigger_if = { limit = { has_variable = nesnd_khanate_usa_epoch exists = var:nesnd_khanate_usa_play }
        trigger_if = { limit = { scope:target_country = { has_variable = nesnd_khanate_war_epoch } }
         var:nesnd_khanate_usa_epoch = scope:target_country.var:nesnd_khanate_war_epoch var:nesnd_khanate_usa_play = scope:diplomatic_play
        }
        trigger_else = { always = no }
       }
       trigger_else = { always = no }
      }
     }
     AND = {
      root = {
       var:nesnd_khanate_original_hus ?= scope:target_country
       trigger_if = { limit = { has_variable = nesnd_khanate_hus_epoch exists = var:nesnd_khanate_hus_play }
        trigger_if = { limit = { scope:target_country = { has_variable = nesnd_khanate_war_epoch } }
         var:nesnd_khanate_hus_epoch = scope:target_country.var:nesnd_khanate_war_epoch var:nesnd_khanate_hus_play = scope:diplomatic_play
        }
        trigger_else = { always = no }
       }
       trigger_else = { always = no }
      }
     }
    }
   }
  }
 }
 trigger_else = { always = no }
}
nesnd_khanate_goal_pair = {
 trigger_if = { limit = { exists = scope:target_country exists = scope:diplomatic_play }
 scope:diplomatic_play = { is_diplomatic_play_type = dp_nesnd_khanate_suppression }
 OR = {
  AND = {
   OR = { c:USA ?= this nesnd_hus_original_country = yes } scope:target_country = { c:NKH ?= this has_variable = nesnd_native_khanate }
   scope:diplomatic_play = { initiator = root target = scope:target_country }
  }
  AND = {
   c:NKH ?= this has_variable = nesnd_native_khanate
   scope:target_country = { OR = { c:USA ?= this nesnd_hus_original_country = yes } }
   scope:diplomatic_play = { target = root initiator = scope:target_country }
  }
 }
 OR = {
  nesnd_khanate_goal_bound_pair = yes
  nesnd_khanate_creation_receipt = yes
  AND = { c:NKH ?= this scope:target_country = { nesnd_khanate_creation_receipt = yes } }
 }
 }
 trigger_else = { always = no }
}
nesnd_khanate_creation_receipt = {
 trigger_if = { limit = { nesnd_khanate_original_opponent = yes }
  trigger_if = { limit = { has_variable = nesnd_khanate_creation_open has_variable = nesnd_khanate_creation_epoch exists = var:nesnd_khanate_creation_case }
   var:nesnd_khanate_creation_case = c:NKH
   OR = {
    AND = { c:USA ?= this c:NKH = { trigger_if = { limit = { has_variable = nesnd_khanate_usa_epoch } var:nesnd_khanate_usa_epoch = prev.var:nesnd_khanate_creation_epoch } trigger_else = { always = no } } }
    AND = { nesnd_hus_original_country = yes c:NKH = { trigger_if = { limit = { has_variable = nesnd_khanate_hus_epoch } var:nesnd_khanate_hus_epoch = prev.var:nesnd_khanate_creation_epoch } trigger_else = { always = no } } }
   }
  }
  trigger_else = { always = no }
 }
 trigger_else = { always = no }
}
