    has_law_or_variant = law_type:law_monarchy
    has_variable = bonapartist_restoration_var
    OR = {
        has_gov_regency = yes
        ruler_is_bonapartist = yes
        ruler ?= { OR = { has_ideology = ideology:ideology_bonapartist has_variable = house_bonaparte } }
    }
}
