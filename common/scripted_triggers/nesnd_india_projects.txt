# Local projects of the Indian princes: a university, a tooling workshop or a railway in one state,
# paid for by the prince, by a neighbouring Indian government or by a foreign government.
# Variables are listed in common/script_values/nesnd_india_projects.txt.

# ---------------------------------------------------------------------------------------------
# The prince who plans the project (requester scope)
# ---------------------------------------------------------------------------------------------
nesnd_india_project_can_start = {
    hidden_trigger = { nesnd_india_prince = yes }
    custom_tooltip = { text = nesnd_india_project_idle_tt NOT = { has_variable = nesnd_india_project_type } }
    custom_tooltip = { text = nesnd_small_no_emergency_tt is_at_war = no is_active_in_diplomatic_play = no }
    in_default = no
}

# State scope.
nesnd_india_project_site_1 = { can_construct_building = building_university }
nesnd_india_project_site_2 = { can_construct_building = building_tooling_workshop }
nesnd_india_project_site_3 = { can_construct_building = building_railway }

nesnd_india_project_has_site_1 = { custom_tooltip = { text = nesnd_india_project_site_exists_tt any_scope_state = { nesnd_india_project_site_1 = yes } } }
nesnd_india_project_has_site_2 = { custom_tooltip = { text = nesnd_india_project_site_exists_tt any_scope_state = { nesnd_india_project_site_2 = yes } } }
nesnd_india_project_has_site_3 = { custom_tooltip = { text = nesnd_india_project_site_exists_tt any_scope_state = { nesnd_india_project_site_3 = yes } } }

# The chosen state is still ours and the building can still be raised there.
nesnd_india_project_site_valid = {
    trigger_if = { limit = { nesnd_india_project_type_value = 4 } always = yes }
    trigger_else_if = { limit = { nesnd_india_project_type_value = 1 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_site_1 = yes } }
    trigger_else_if = { limit = { nesnd_india_project_type_value = 2 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_site_2 = yes } }
    trigger_else_if = { limit = { nesnd_india_project_type_value = 3 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_site_3 = yes } }
    trigger_else = { always = no }
}
nesnd_india_project_site_owned = {
    trigger_if = { limit = { exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev } }
    trigger_else = { always = no }
}
# One more level stands at the site than when the work was ordered.
nesnd_india_project_built = {
    trigger_if = { limit = { nesnd_india_project_type_value = 1 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_level_1 > prev.nesnd_india_project_base_value } }
    trigger_else_if = { limit = { nesnd_india_project_type_value = 2 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_level_2 > prev.nesnd_india_project_base_value } }
    trigger_else_if = { limit = { nesnd_india_project_type_value = 3 exists = var:nesnd_india_project_site } var:nesnd_india_project_site = { owner = prev nesnd_india_project_level_3 > prev.nesnd_india_project_base_value } }
    trigger_else = { always = no }
}

# State scope. A funding agreement made by the present owner is still within its five years.
nesnd_india_deal_active = {
    save_temporary_scope_as = india_deal_state
    state_region = {
        has_variable = nesnd_india_deal_term
        trigger_if = { limit = { exists = var:nesnd_india_deal_host } var:nesnd_india_deal_host = scope:india_deal_state.owner }
        trigger_else = { always = no }
    }
}
# State scope. The five years of an agreement have run out and its record is still there.
nesnd_india_deal_lapsed = {
    state_region = { has_variable = nesnd_india_deal_host NOT = { has_variable = nesnd_india_deal_term } }
}

# Before asking another government: no agreement is running at the site, and a government that
# takes a stake gets a new building rather than a share of an old one.
nesnd_india_project_site_free = {
    trigger_if = { limit = { exists = var:nesnd_india_project_site }
        custom_tooltip = { text = nesnd_india_project_no_deal_tt var:nesnd_india_project_site = { NOT = { nesnd_india_deal_active = yes } } }
        trigger_if = { limit = { nesnd_india_project_type_value = 2 } custom_tooltip = { text = nesnd_india_project_fresh_tt var:nesnd_india_project_site = { NOT = { has_building = building_tooling_workshop } } } }
        trigger_if = { limit = { nesnd_india_project_type_value = 3 } custom_tooltip = { text = nesnd_india_project_fresh_tt var:nesnd_india_project_site = { NOT = { has_building = building_railway } } } }
    }
    trigger_else = { always = no }
}

# FUNDER is the government asked for money or for its engineers.
nesnd_india_project_can_ask = {
    trigger_if = { limit = { exists = $FUNDER$ }
        nesnd_india_channel_free = yes
        trigger_if = { limit = { nesnd_india_project_type_value = 4 } gold_reserves >= nesnd_india_survey_fee }
        trigger_else = {
            $FUNDER$ = { nesnd_india_partner_can_fund = { SENDER = prev } }
            nesnd_india_project_site_free = yes
        }
    }
    trigger_else = { always = no }
}

# The overlord must agree when a puppet deals with any other government, and when any other
# subject lets another government take a stake. Nothing is asked when the overlord itself pays.
nesnd_india_project_consent_needed = {
    trigger_if = { limit = { is_subject = yes exists = overlord exists = $FUNDER$ }
        NOT = { overlord = $FUNDER$ }
        OR = {
            is_subject_type = subject_type_puppet
            nesnd_india_project_type_value = 2
            nesnd_india_project_type_value = 3
        }
    }
    trigger_else = { always = no }
}

# The request still stands: both governments exist and are at peace, the overlord whose consent
# is awaited is still the overlord, and the site is still fit.
nesnd_india_project_request_valid = {
    is_at_war = no
    trigger_if = { limit = { exists = var:nesnd_india_project_funder }
        var:nesnd_india_project_funder = { is_country_alive = yes }
        NOT = { has_war_with = var:nesnd_india_project_funder }
    }
    trigger_else = { always = no }
    trigger_if = { limit = { nesnd_india_project_stage_value = 2 }
        trigger_if = { limit = { is_subject = yes exists = overlord exists = var:nesnd_india_project_permit }
            var:nesnd_india_project_permit = { is_country_alive = yes }
            overlord = var:nesnd_india_project_permit
        }
        trigger_else = { always = no }
    }
    nesnd_india_project_site_valid = yes
}

# ---------------------------------------------------------------------------------------------
# Governments that may be asked (candidate scope, OWNER is the prince)
# ---------------------------------------------------------------------------------------------
# A puppet cannot sign for itself. Money is only asked of a government that has it.
nesnd_india_project_funder_ok = {
    is_country_alive = yes
    NOT = { this = $OWNER$ }
    is_at_war = no
    NOT = { has_war_with = $OWNER$ }
    trigger_if = { limit = { is_subject = yes } NOT = { is_subject_type = subject_type_puppet } }
    trigger_if = { limit = { $OWNER$ = { nesnd_india_project_type_value <= 3 } } nesnd_india_partner_can_fund = { SENDER = $OWNER$ } }
}
nesnd_india_project_neighbour = {
    nesnd_india_diplomacy_live = yes
    nesnd_india_project_funder_ok = { OWNER = $OWNER$ }
}
nesnd_india_project_foreigner = {
    NOT = { nesnd_india_diplomacy_live = yes }
    nesnd_india_mission_partner = { SENDER = $OWNER$ }
    nesnd_india_project_funder_ok = { OWNER = $OWNER$ }
}

# ---------------------------------------------------------------------------------------------
# The government that must answer (answering scope). It holds nesnd_india_matter_from.
# ---------------------------------------------------------------------------------------------
# Its consent is asked as overlord.
nesnd_india_matter_consent_role = {
    hidden_trigger = {
        save_temporary_scope_as = india_matter_me
        trigger_if = { limit = { exists = var:nesnd_india_matter_from }
            var:nesnd_india_matter_from = {
                is_country_alive = yes
                nesnd_india_project_stage_value = 2
                trigger_if = { limit = { exists = var:nesnd_india_project_permit } var:nesnd_india_project_permit = scope:india_matter_me }
                trigger_else = { always = no }
            }
        }
        trigger_else = { always = no }
    }
}
# Its money, or its engineers, are asked.
nesnd_india_matter_fund_role = {
    hidden_trigger = {
        save_temporary_scope_as = india_matter_me
        trigger_if = { limit = { exists = var:nesnd_india_matter_from }
            var:nesnd_india_matter_from = {
                is_country_alive = yes
                nesnd_india_project_stage_value = 3
                trigger_if = { limit = { exists = var:nesnd_india_project_funder } var:nesnd_india_project_funder = scope:india_matter_me }
                trigger_else = { always = no }
            }
        }
        trigger_else = { always = no }
    }
}
nesnd_india_matter_live = { OR = { nesnd_india_matter_consent_role = yes nesnd_india_matter_fund_role = yes } }

nesnd_india_matter_is_survey = {
    trigger_if = { limit = { exists = var:nesnd_india_matter_from } var:nesnd_india_matter_from = { nesnd_india_project_type_value = 4 } }
    trigger_else = { always = no }
}
nesnd_india_matter_is_grant = {
    trigger_if = { limit = { exists = var:nesnd_india_matter_from } var:nesnd_india_matter_from = { nesnd_india_project_type_value = 1 } }
    trigger_else = { always = no }
}
# The answering government can meet the terms today.
nesnd_india_matter_can_fund = {
    trigger_if = { limit = { exists = var:nesnd_india_matter_from }
        NOT = { has_war_with = var:nesnd_india_matter_from }
        trigger_if = { limit = { nesnd_india_matter_is_survey = no }
            custom_tooltip = {
                text = nesnd_india_matter_funds_tt
                gold_reserves >= nesnd_india_matter_quote
                in_default = no
                NOT = { has_modifier = declared_bankruptcy }
            }
        }
        custom_tooltip = {
            text = nesnd_india_matter_site_tt
            var:nesnd_india_matter_from = { nesnd_india_project_site_valid = yes }
        }
    }
    trigger_else = { always = no }
}
