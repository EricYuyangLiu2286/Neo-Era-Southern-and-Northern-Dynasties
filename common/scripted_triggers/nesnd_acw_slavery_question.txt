# Slavery question (je_acw_countdown replacement) and the 东洲 lever. Task 58.

# COUNTRY scope (USA). Slavery still legal and no secession crisis under way.
nesnd_acw_pre_abolition = {
    NOT = { has_law_or_variant = law_type:law_slavery_banned }
    custom_tooltip = { text = nesnd_ui_not_acw_crisis_tt NOT = { has_variable = nesnd_acw_crisis } }
}

# STATE scope. Before abolition: slave states. After: the old South (Dixie homeland).
nesnd_acw_southern_state = {
    trigger_if = {
        limit = { owner = { has_law_or_variant = law_type:law_slavery_banned } }
        state_region = { is_homeland = cu:dixie }
    }
    trigger_else = { is_slave_state = yes }
}

# INTEREST GROUP scope.
nesnd_acw_proslavery_ig = {
    ig_counts_as_marginal = no
    OR = {
        has_ideology = ideology:ideology_pro_slavery
        has_ideology = ideology:ideology_pro_slavery_colonial
    }
}
nesnd_acw_abolitionist_ig = {
    ig_counts_as_marginal = no
    has_ideology = ideology:ideology_anti_slavery
}

# COUNTRY scope (HUS). The lever journal exists while the USA question or war is open.
nesnd_hus_west_eligible = {
    c:HUS ?= this
    custom_tooltip = { text = nesnd_ui_not_hus_native_successor_tt NOT = { has_variable = nesnd_hus_native_successor } }
    custom_tooltip = { text = nesnd_ui_not_hus_west_closed_tt NOT = { has_variable = nesnd_hus_west_closed } }
    exists = c:USA
    c:USA = { OR = { has_journal_entry = je_acw_countdown has_journal_entry = je_acw_war } }
    any_scope_state = { any_neighbouring_state = { owner = c:USA } }
}

# POP scope. People who can flee west: slaves, and Native American pops.
nesnd_hus_west_escapee = {
    total_size >= 100
    OR = {
        is_pop_type = slaves
        culture = { nesnd_hus_native_culture = yes }
    }
}

# STATE scope (USA-owned). A USA state on the 东洲 border holding escapees.
nesnd_hus_west_departure = {
    owner = c:USA
    any_neighbouring_state = { owner = c:HUS is_split_state = no }
    any_scope_pop = { nesnd_hus_west_escapee = yes }
}

# Both secession tags continue to hold their own states after a divided peace.
nesnd_hus_west_american_side = {
    OR = { c:USA ?= this c:CSA ?= this c:FSA ?= this }
}

nesnd_hus_west_question_over = {
    NOT = { c:USA ?= { has_journal_entry = je_acw_countdown } }
    NOT = { c:USA ?= { has_journal_entry = je_acw_war } }
}

nesnd_hus_west_claim_candidate = {
    is_in_geographic_region = geographic_region_americas
    owner = { nesnd_hus_west_american_side = yes }
    state_population > 0
    any_neighbouring_state = { owner = c:HUS }
    NOT = { has_claim_by = c:HUS }
    c:HUS = { NOT = { is_target_in_variable_list = { name = nesnd_hus_west_claims target = prev.state_region } } }
}

nesnd_hus_west_press_candidate = {
    owner = {
        nesnd_hus_west_american_side = yes
        NOT = { has_truce_with = c:HUS }
        "relations(c:HUS)" < relations_threshold:cordial
        trigger_if = {
            limit = { is_subject = yes }
            overlord = { "relations(c:HUS)" < relations_threshold:cordial }
            top_overlord = { "relations(c:HUS)" < relations_threshold:cordial }
        }
        NOT = { c:HUS = { is_subject_of = prev } }
    }
    num_provinces > 0
    is_treaty_port = no
    any_neighbouring_state = { owner = c:HUS }
    c:HUS = { is_target_in_variable_list = { name = nesnd_hus_west_claims target = prev.state_region } }
}

# Apply the same checks to manual orders and to the monthly orders.
nesnd_hus_west_can_claim = {
    custom_tooltip = {
        text = nesnd_hus_west_claim_bar_tt
        "je:nesnd_hus_west_lever.scripted_bar_progress(nesnd_hus_west_claim_bar)" >= 50
    }
    custom_tooltip = {
        text = nesnd_hus_west_quota_tt
        NOT = { any_in_list = { variable = nesnd_hus_west_claims count >= 10 always = yes } }
    }
    custom_tooltip = { text = nesnd_hus_west_claim_cooldown_tt NOT = { has_variable = nesnd_hus_west_claim_cooldown } }
    custom_tooltip = {
        text = nesnd_hus_west_claim_candidate_tt
        OR = {
            c:USA ?= { any_scope_state = { nesnd_hus_west_claim_candidate = yes } }
            c:CSA ?= { any_scope_state = { nesnd_hus_west_claim_candidate = yes } }
            c:FSA ?= { any_scope_state = { nesnd_hus_west_claim_candidate = yes } }
        }
    }
}

# STATE_REGION scope. A region's original American provinces must all be held under settled
# territorial awards. Provinces cannot hold variables: awards are kept in region lists.
nesnd_hus_west_region_won = {
    any_in_list = { variable = nesnd_hus_west_land always = yes }
    NOT = {
        any_in_list = {
            variable = nesnd_hus_west_land
            OR = {
                NOT = { owner ?= c:HUS }
                prev = { NOT = { has_variable_list = nesnd_hus_west_awarded_land is_target_in_variable_list = { name = nesnd_hus_west_awarded_land target = prev } } }
            }
        }
    }
    trigger_if = {
        limit = { has_variable_list = nesnd_hus_west_award_plays }
        NOT = {
            any_in_list = {
                variable = nesnd_hus_west_award_plays
                is_war = yes
                NOT = { has_variable = nesnd_hus_west_peace_signed }
            }
        }
    }
    NOT = { any_scope_state = { owner = { nesnd_hus_west_american_side = yes } } }
}
