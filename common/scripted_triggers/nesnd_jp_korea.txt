nesnd_jp_korea_available = {
    nesnd_jp_country = yes
    OR = { nesnd_jp_owned_korea = yes AND = { nesnd_jp_core_united = yes is_subject = no NOT = { has_variable = nesnd_jp_korea_settled } c:KOR ?= { is_country_alive = yes NOT = { is_direct_subject_of = prev } } } }
}
nesnd_jp_can_change_korea = {
    nesnd_jp_owned_korea = yes
    NOT = { has_variable = nesnd_jp_korea_policy_cooldown }
    gold_reserves >= nesnd_jp_negotiation_cost
}
nesnd_jp_korean_envoy_available = {
    trigger_if = { limit = { exists = c:KOR }
        nesnd_jp_can_send_request = yes is_subject = no is_at_war = no is_active_in_diplomatic_play = no
        NOT = { any_scope_state = { OR = { state_region = s:STATE_BUSAN state_region = s:STATE_YANGHO } NOT = { controller = owner } } }
        NOT = { has_variable = nesnd_jp_korea_settled }
        c:KOR = { is_country_alive = yes NOT = { is_direct_subject_of = prev } }
        NOT = { has_war_with = c:KOR }
        any_scope_state = { NOR = { state_region = s:STATE_BUSAN state_region = s:STATE_YANGHO } }
        trigger_if = { limit = { c:KOR = { is_subject = yes exists = overlord } }
            NOT = { has_war_with = c:KOR.overlord }
        }
    } trigger_else = { always = no }
}
