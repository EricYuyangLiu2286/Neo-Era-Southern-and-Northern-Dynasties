# Compatibility name used by existing JE status declarations. The ongoing southern war also includes its actual overseas opposition.
nesnd_south_mainland_holdout = { nesnd_south_campaign_holdout = yes }
nesnd_south_campaign_holdout = {
    is_country_alive = yes num_states > 0 NOT = { c:USM ?= this }
    custom_tooltip = { text = nesnd_ui_not_postwar_allegiance_answered_tt NOT = { has_variable = nesnd_postwar_allegiance_answered } }
    OR = {
        AND = {
            OR = { c:HNA ?= this c:SIH ?= this c:GNG ?= this }
            OR = { custom_tooltip = { text = nesnd_ui_restoration_side_tt has_variable = nesnd_restoration_side } custom_tooltip = { text = nesnd_ui_ming_court_tt has_variable = nesnd_ming_court } nesnd_board_stance_value = 1 }
            OR = { custom_tooltip = { text = nesnd_ui_usm_opposition_country_tt has_variable = usm_opposition_country } custom_tooltip = { text = nesnd_ui_ming_court_tt has_variable = nesnd_ming_court } }
        }
        AND = {
            custom_tooltip = { text = nesnd_ui_usm_opposition_country_tt has_variable = usm_opposition_country }
            OR = { c:HNA ?= this c:SIH ?= this c:GNG ?= this c:PHI ?= this c:DEI ?= this c:YN1 ?= this }
            OR = { custom_tooltip = { text = nesnd_ui_restoration_side_tt has_variable = nesnd_restoration_side } custom_tooltip = { text = nesnd_ui_ce_usm_republican_revolution_tag_tt has_variable = ce_usm_republican_revolution_tag }
                has_journal_entry = ce_usm_restoration_of_ming custom_tooltip = { text = nesnd_ui_usm_opposition_country_leader_var_tt has_variable = usm_opposition_country_leader_var }
                custom_tooltip = { text = nesnd_ui_south_answered_tt has_variable = nesnd_south_answered } }
        }
    }
    # Japan is not a holdout: its war with Wu is the Eastern Intervention (nesnd_open_japan_jingnan_war) and is settled apart (author, 2026-11-17).
}

# Diplomatic-play scope. Only the southern campaign can issue a military defeat receipt.
# A concluded play stays in the save for a while with no war and full escalation (observer game 2026-11-17: the civil war
# ended 1841.10.12 and its play was still there three days later). Only a live play counts.
# Japan's war in the Jingnan campaign is the Eastern Intervention marked by nesnd_open_japan_jingnan_war (2026-11-17).
nesnd_south_named_play = {
    OR = { is_diplomatic_play_type = dp_usm_civil_war AND = { is_diplomatic_play_type = dp_nesnd_eastern_intervention custom_tooltip = { text = nesnd_ui_jingnan_japan_play_tt has_variable = nesnd_jingnan_japan_play } } }
    initiator_is = c:USM OR = { is_war = yes escalation < 100 }
}

# An anonymous capitulation notice does not identify the war. Accept only a typed supplied play/war.
nesnd_south_callback_in_campaign = {
    OR = {
        AND = { exists = scope:diplomatic_play scope:diplomatic_play ?= { nesnd_south_named_play = yes } }
        AND = { exists = scope:war scope:war ?= { exists = diplomatic_play diplomatic_play ?= { nesnd_south_named_play = yes } } }
    }
}

# A consumed answer with no live destination, war or outstanding follow-up in the same round is an abandoned old transaction.
nesnd_holdout_abandoned_answer = {
    nesnd_holdout_peace_eligible = yes has_variable = nesnd_holdout_answered_round
    nesnd_holdout_answered_value >= nesnd_holdout_round_value
    NOR = { has_variable = nesnd_holdout_choice_pending has_variable = nesnd_holdout_north_awaited
        has_variable = nesnd_holdout_resistance_pending }
    NOT = { any_diplomatic_play = { nesnd_south_named_play = yes } }
    trigger_if = { limit = { has_variable = nesnd_holdout_recovered_answer_round }
        NOT = { var:nesnd_holdout_recovered_answer_round = nesnd_holdout_round_value }
    }
}

# The play it answered is still a war, and it is at war with Wu. Comparing the war's play with prev.var: failed in game
# ("Invalid right side during comparison 'var'", 943 times on 2026-11-17), so the play is not compared directly.
nesnd_jingnan_may_receive_rebel_journal = {
    custom_tooltip = { text = nesnd_ui_exists_eq_var_nesnd_jingnan_locked_play_tt exists = var:nesnd_jingnan_locked_play }
    # The 2026-11-17 form (var:nesnd_jingnan_locked_play = { is_war = yes } + has_war_with) let no mainland rebel

        exists = diplomatic_play
        diplomatic_play ?= { nesnd_south_named_play = yes }
    }
}

# Country scope. A Wu opponent the war has not reached: not fighting Wu (a war with anyone else does not count),
# not in a play against Wu, not settled or waiting on a holdout answer. A truce does not protect it:
# nesnd_continue_war_open ends the truce before the war.
nesnd_continue_war_candidate = {
    nesnd_south_campaign_holdout = yes
    is_subject = no
    NOT = { has_variable = nesnd_holdout_resolved }
    NOT = { has_variable = nesnd_capital_surrender_pending }
    NOT = { has_variable = nesnd_holdout_choice_pending }
    NOT = { has_variable = nesnd_holdout_north_awaited }
    trigger_if = { limit = { exists = c:USM }
        NOT = { has_war_with = c:USM }
        NOT = { is_diplomatic_play_enemy_of = c:USM }
    }
    trigger_else = { always = no }
}
