nesnd_cne_republic_west_region = { OR = { state_region = s:STATE_SHANXI state_region = s:STATE_JINBEI state_region = s:STATE_SHAANBEI state_region = s:STATE_HINGGAN } }
nesnd_cne_republic_northeast_region = { OR = { state_region = s:STATE_SHENGJING state_region = s:STATE_SOUTHERN_MANCHURIA } }
nesnd_cne_republic_north_region = { OR = { state_region = s:STATE_SHANDONG state_region = s:STATE_ZHILI state_region = s:STATE_BEIJING } }
# NRP country scope; count only states controlled by this claimant, not third-party enemies.
nesnd_cne_republic_controls_empire = {
 trigger_if = { limit = { exists = c:CNE exists = c:NRP }
  has_war_with = c:CNE
  c:CNE = { any_scope_state = { percent >= 0.8 controller = c:NRP } }
 }
 trigger_else = { always = no }
}
nesnd_cne_republic_has_play = {
 trigger_if = { limit = { exists = c:CNE exists = c:NRP }
  any_diplomatic_play = { is_diplomatic_play_type = dp_nesnd_cne_republic initiator = c:CNE target = c:NRP }
 }
 trigger_else = { always = no }
}
nesnd_cne_republic_at_war = {
 trigger_if = { limit = { exists = c:CNE } has_war_with = c:CNE }
 trigger_else = { always = no }
}
nesnd_cne_republic_target_available = {
 trigger_if = { limit = { exists = c:NRP }
  c:NRP = { has_variable = nesnd_cne_republican_challenger }
  NOT = { has_war_with = c:NRP }
 }
 trigger_else = { always = no }
}

nesnd_rep_district_available = {
    any_scope_state = { $REGION_TEST$ = yes }
    any_scope_state = { $REGION_TEST$ = no }
}
nesnd_rep_northeast_unclaimed = {
    NOT = { any_scope_state = { nesnd_cne_republic_northeast_region = yes nesnd_yan_contact_state = yes } }
}
nesnd_rep_reserved_state = {
    exists = c:CNE
    c:CNE ?= { is_target_in_variable_list = { name = nesnd_rep_states target = prev } }
}
nesnd_rep_may_prepare = {
    exists = c:CNE
    nesnd_cne_programs_live = yes
    has_journal_entry = nesnd_cne_confidence_journal
    NOR = {
        exists = c:NRP
        has_variable = nesnd_cne_republican_rose
        any_diplomatic_play = { is_diplomatic_play_type = nesnd_sha_yan_suppression is_war = yes initiator = c:CNE }
    }
}
nesnd_rep_warning_valid = {
    nesnd_rep_may_prepare = yes
    NOT = { has_variable = nesnd_rep_interrupted }
    nesnd_rep_confidence < 575
    exists = var:nesnd_rep_seed
    var:nesnd_rep_seed ?= { owner = c:CNE }
    has_variable = nesnd_rep_count
    nesnd_rep_state_count = var:nesnd_rep_count
    num_states > var:nesnd_rep_count
}
