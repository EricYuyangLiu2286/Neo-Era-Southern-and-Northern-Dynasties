# Mu Muzi and the book chain (flavorCNE.12-.16).

# The history character, or the stand-in made when she is gone by publication time.
nesnd_is_mu_muzi = {
    OR = {
        has_template = CNE_mu_muzi
        has_variable = nesnd_mu_muzi_stand_in
    }
}

# Country scope. Where she belongs when the book is due: CNE if she never went east,
# a non-CNE holder of California if she did; each side falls back to the other.
nesnd_mu_book_home = {
    OR = {
        AND = {
            c:CNE ?= this
            OR = {
                NOT = { has_global_variable = nesnd_mu_went_east }
                NOT = {
                    any_country = {
                        NOT = { c:CNE ?= this }
                        any_scope_state = { state_region = s:STATE_CALIFORNIA }
                    }
                }
            }
        }
        AND = {
            NOT = { c:CNE ?= this }
            any_scope_state = { state_region = s:STATE_CALIFORNIA }
            OR = {
                has_global_variable = nesnd_mu_went_east
                NOT = { c:CNE ?= { is_country_alive = yes } }
            }
        }
    }
}
