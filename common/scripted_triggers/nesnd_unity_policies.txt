nesnd_unity_policy_live = {
    is_country_alive = yes
    OR = {
        nesnd_qe1_unity_live = yes
        nesnd_jp_cohesion_live = yes
        AND = { nesnd_india_society_live = yes has_journal_entry = nesnd_india_society }
        nesnd_ottoman_unity_live = yes
    }
}
nesnd_unity_material_active = {
    OR = {
        has_variable = nesnd_unity_material_term
        has_modifier = nesnd_unity_material_expense
        has_modifier = nesnd_unity_material_assistance
    }
}
nesnd_unity_political_active = {
    OR = { has_variable = nesnd_unity_political_term has_modifier = nesnd_unity_political_assistance }
}
nesnd_unity_material_available = {
    nesnd_unity_policy_live = yes
    NOT = { nesnd_unity_material_active = yes }
}
nesnd_unity_political_available = {
    nesnd_unity_policy_live = yes
    NOT = { nesnd_unity_political_active = yes }
}
