nesnd_sha_russian_borderland = {
    OR = { state_region = s:STATE_IRKUTSK state_region = s:STATE_UPPER_YENISEYSK }
}
# Country scope: an uprising country that rose in the borderland, whatever its people.
# The Liaodong check keeps the original case when the new country holds no borderland state yet.
nesnd_sha_borderland_uprising = {
    OR = {
        any_scope_state = { nesnd_sha_russian_borderland = yes }
        country_has_primary_culture = cu:ce_liaodong
    }
}
# Political movement scope: a Russian minority movement of a people living in the borderland.
nesnd_sha_borderland_movement = {
    is_political_movement_type = movement_cultural_minority
    owner ?= c:RUS
    culture ?= {
        NOT = { this = cu:russian }
        save_temporary_scope_as = nesnd_sha_movement_culture
    }
    c:RUS ?= { any_scope_state = { nesnd_sha_russian_borderland = yes any_scope_pop = { culture = scope:nesnd_sha_movement_culture } } }
}

nesnd_sha_has_eastern_partner = {
    trigger_if = {
        limit = { exists = c:HUS }
        OR = {
            market = c:HUS.market
            any_scope_treaty = { binds = c:HUS any_scope_article = { has_type = foreign_investment_rights } }
        }
    }
    trigger_else = { always = no }
}
nesnd_sha_lake_republic = {
    NOT = { exists = s:STATE_IRKUTSK.region_state:RUS }
    NOT = { exists = s:STATE_IRKUTSK.region_state:SHA }
    exists = var:nesnd_sha_seceder
    var:nesnd_sha_seceder ?= {
        is_country_alive = yes
        is_subject = no
        is_secessionist = no
        any_scope_state = { state_region = s:STATE_IRKUTSK }
    }
}
nesnd_sha_can_take_lake = {
    nesnd_sha_lake_republic = yes
    has_variable = nesnd_sha_lake_independent
    trigger_if = {
        limit = { exists = var:nesnd_sha_seceder }
        NOT = { has_war_with = var:nesnd_sha_seceder }
        NOT = { has_truce_with = var:nesnd_sha_seceder }
        NOT = { is_diplomatic_play_enemy_of = var:nesnd_sha_seceder }
        trigger_if = {
            limit = { is_ai = yes }
            is_at_war = no
            army_size > 0
            army_size >= var:nesnd_sha_seceder.army_size
        }
    }
    trigger_else = { always = no }
}
