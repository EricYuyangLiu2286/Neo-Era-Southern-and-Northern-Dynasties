# Foreign intervention in the Second Jingnan War: who intervened, and the Wu morale bar.

# Country scope. Sent troops against Wu: the Chinese Empire by nesnd_crisis.20 (option c),
# the Western Qin and Huaxia Dongzhou by nesnd_external.1.
nesnd_jingnan_is_intervener = {
    OR = {
        custom_tooltip = { text = nesnd_ui_north_intervention_tt has_variable = nesnd_north_intervention }
        custom_tooltip = { text = nesnd_ui_jingnan_intervener_tt has_variable = nesnd_jingnan_intervener }
    }
}

# Country scope. An intervener that actually entered a Jingnan war on the anti-Wu side
# (nesnd_fulfil_jingnan_call sets nesnd_joined_south_play only for interveners).
nesnd_jingnan_aid_eligible = {
    is_country_alive = yes
    is_subject = no
    custom_tooltip = { text = nesnd_ui_joined_south_play_tt has_variable = nesnd_joined_south_play }
}

# USM scope. Wu is fighting a Second Jingnan war.
nesnd_wu_jingnan_at_war = {
    any_scope_war = { diplomatic_play ?= { nesnd_south_named_play = yes } }
}

# Country scope. A foreign enemy of Wu in a Jingnan war. Rebel signatories (Japan included) do not count.
nesnd_wu_resolve_foreign_enemy = {
    is_country_alive = yes
    NOT = { has_variable = usm_opposition_country }
    exists = c:USM
    NOT = { c:USM ?= this }
    has_war_with = c:USM
    any_scope_war = {
        is_war_participant = c:USM
        diplomatic_play ?= { nesnd_south_named_play = yes }
    }
}
