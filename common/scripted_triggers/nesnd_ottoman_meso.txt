# Triggers for Ottoman B9 "Mesopotamia" journal (nesnd_ottoman_meso).
# Country scope unless noted.

# State region scope. Target regions of the Mesopotamia contest (Rule 1).
nesnd_ottoman_meso_is_target_region = {
    OR = {
        this = s:STATE_BAGHDAD
        this = s:STATE_MOSUL
        this = s:STATE_BASRA
    }
}

# Country scope (evaluated country). The country or any of its subjects at any depth holds a state in the target regions (Rule 2).
# Uses any_subject_or_below (GAME common/achievements/ip4_iberian_twilight_achievements.txt:105), equivalent to checking top_overlord for an independent country.
nesnd_ottoman_meso_holds_target_region = {
    OR = {
        any_scope_state = {
            state_region = {
                nesnd_ottoman_meso_is_target_region = yes
            }
        }
        any_subject_or_below = {
            any_scope_state = {
                state_region = {
                    nesnd_ottoman_meso_is_target_region = yes
                }
            }
        }
    }
}

# Country scope (evaluated country). Eligible to be selected as the Mesopotamia foe (Rule 2).
nesnd_ottoman_meso_foe_candidate = {
    is_country_alive = yes
    exists = c:TUR
    NOT = { c:TUR ?= this }
    is_subject = no
    NOT = { is_country_type = decentralized }
    NOT = {
        has_treaty_alliance_with = {
            TARGET = c:TUR
        }
    }
    NOT = {
        has_treaty_defensive_pact_with = {
            TARGET = c:TUR
        }
    }
    nesnd_ottoman_meso_holds_target_region = yes
}

# Any scope. The stored global_var:nesnd_meso_foe is still alive, independent, not allied with TUR, and still holds target land (Rule 4).
nesnd_ottoman_meso_foe_valid = {
    exists = c:TUR
    has_global_variable = nesnd_meso_foe
    global_var:nesnd_meso_foe ?= {
        is_country_alive = yes
        is_subject = no
        NOT = { c:TUR ?= this }
        NOT = {
            has_treaty_alliance_with = {
                TARGET = c:TUR
            }
        }
        NOT = {
            has_treaty_defensive_pact_with = {
                TARGET = c:TUR
            }
        }
        nesnd_ottoman_meso_holds_target_region = yes
    }
}

# Any scope. At least one state in the three target regions lies outside Ottoman / Ottoman-subject hands (Rule 6).
nesnd_ottoman_meso_any_outside_hands = {
    OR = {
        s:STATE_BAGHDAD = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
        s:STATE_MOSUL = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
        s:STATE_BASRA = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
    }
}

# Any scope. All states in the three target regions are held by TUR or its subjects (Rule 9.3).
nesnd_ottoman_meso_all_held = {
    s:STATE_BAGHDAD = {
        any_scope_state = {
            always = yes
        }
        NOT = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
    }
    s:STATE_MOSUL = {
        any_scope_state = {
            always = yes
        }
        NOT = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
    }
    s:STATE_BASRA = {
        any_scope_state = {
            always = yes
        }
        NOT = {
            any_scope_state = {
                nesnd_ottoman_state_outside_ottoman_hands = yes
            }
        }
    }
}

# Country scope (TUR). Entry / monthly fallback trigger for nesnd_ottoman_meso (Rule 6).
nesnd_ottoman_meso_fallback_due = {
    c:TUR ?= this
    nesnd_ottoman_recovery_open = yes
    has_variable = nesnd_ottoman_route
    nesnd_ottoman_can_act_abroad = yes
    nesnd_ottoman_meso_any_outside_hands = yes
    any_country = {
        nesnd_ottoman_meso_foe_candidate = yes
    }
    NOT = { has_journal_entry = nesnd_ottoman_meso }
    NOT = { has_variable = nesnd_ottoman_meso_closed }
    NOT = { has_variable = nesnd_ottoman_meso_reopen_cd }
}

# Country scope (TUR). Any snapshotted home region (Diyarbakir / Deir ez-Zor) now has a state outside Ottoman hands (Rule 9.1).
nesnd_ottoman_meso_home_lost = {
    has_variable_list = nesnd_ottoman_meso_home_regions
    any_in_list = {
        variable = nesnd_ottoman_meso_home_regions
        any_scope_state = {
            nesnd_ottoman_state_outside_ottoman_hands = yes
        }
    }
}

# Country scope (TUR). Currently at war with global_var:nesnd_meso_foe (Rules 9, 10, 13).
nesnd_ottoman_meso_at_war_with_foe = {
    has_global_variable = nesnd_meso_foe
    global_var:nesnd_meso_foe ?= {
        has_war_with = prev
    }
}

# Any scope. Active diplomatic play opposing TUR and global_var:nesnd_meso_foe (Rules 9, 10, 13).
nesnd_ottoman_meso_play_in_progress = {
    exists = c:TUR
    has_global_variable = nesnd_meso_foe
    global_var:nesnd_meso_foe ?= {
        OR = {
            is_diplomatic_play_enemy_of = c:TUR
            any_diplomatic_play = {
                OR = {
                    AND = {
                        initiator_is = c:TUR
                        target_is = prev
                    }
                    AND = {
                        initiator_is = prev
                        target_is = c:TUR
                    }
                }
            }
        }
    }
}

# Country scope (TUR). Foe became invalid and TUR is not currently at war with it (R12, Rule 9.5).
nesnd_ottoman_meso_situation_changed = {
    nesnd_ottoman_meso_foe_valid = no
    nesnd_ottoman_meso_at_war_with_foe = no
}

# Country scope (TUR). Holds a claim on at least one of the three Mesopotamia regions (Rules 11, 12).
nesnd_ottoman_meso_has_target_claim = {
    OR = {
        has_claim = s:STATE_BAGHDAD
        has_claim = s:STATE_MOSUL
        has_claim = s:STATE_BASRA
    }
}

# Country scope (evaluated country). Direct holder of at least one target state outside Ottoman hands (Rule 11).
nesnd_ottoman_meso_state_holder = {
    is_country_alive = yes
    NOT = { is_country_type = decentralized }
    NOT = { c:TUR ?= this }
    NOT = { top_overlord ?= c:TUR }
    any_scope_state = {
        state_region = {
            nesnd_ottoman_meso_is_target_region = yes
        }
    }
}

# Country scope (recipient of event nesnd_ottoman_recovery.24). The demarcation offer is still valid (Rule 13).
nesnd_ottoman_meso_offer_open = {
    exists = c:TUR
    c:TUR = {
        is_country_alive = yes
        has_journal_entry = nesnd_ottoman_meso
    }
    global_var:nesnd_meso_foe ?= this
    nesnd_ottoman_meso_foe_valid = yes
    NOT = { has_war_with = c:TUR }
}
