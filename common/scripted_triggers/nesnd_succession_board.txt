# Country scope. Previous participants remain visible after leaving a market/overlord.
nesnd_board_country_eligible = {
    is_country_alive = yes
    OR = {
        has_variable = nesnd_board_registered
        c:USM ?= this c:HNA ?= this c:SIH ?= this c:GNG ?= this
        c:JAP ?= this c:JTK ?= this c:JIM ?= this
        c:CNE ?= this c:QE1 ?= this c:HUS ?= this c:FRA ?= this c:MUG ?= this
        c:SHA ?= this c:GBR ?= this c:RUS ?= this
        AND = { exists = c:USM is_subject_of = c:USM }
    }
}
nesnd_board_can_choose = {
	OR = {
		c:USM ?= this
		nesnd_south_political_member = yes
	}
	is_involved_in_journal_entry = nesnd_south_succession_board
	NOT = {
		has_global_variable = nesnd_south_settled
	}
	NOT = {
		has_variable = nesnd_board_stance_cooldown
	}
	NOT = {
		has_global_variable = usm_civilwar_occured
	}
	NOT = {
		nesnd_board_fixed_ming_allegiance = yes
	}
}
