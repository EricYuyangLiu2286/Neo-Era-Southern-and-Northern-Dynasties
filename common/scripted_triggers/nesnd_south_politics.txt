nesnd_south_member_can_integrate = {
	nesnd_south_integration_member = yes
	is_at_war = no
	OR = {
		is_subject = no
		is_subject_of = c:USM
	}
	nesnd_south_accord_valid = yes
	nesnd_south_accord_side_value = 2
	has_variable = nesnd_south_accepts_integration
	trigger_if = {
		limit = {
			nesnd_south_protected_overseas = yes
		}
		any_scope_state = {
			NOT = {
				nesnd_south_chinese_state = yes
			}
			controller = owner
		}
	}
	trigger_if = {
		limit = {
			any_scope_state = {
				NOT = {
					nesnd_south_chinese_state = yes
				}
			}
		}
		any_scope_state = {
			NOT = {
				nesnd_south_chinese_state = yes
			}
			controller = owner
		}
	}
	NOT = {
		any_scope_state = {
			nesnd_south_chinese_state = yes
			NOT = {
				controller = owner
			}
		}
	}
	trigger_if = {
		limit = {
			is_player = yes
		}
		c:USM = {
			is_player = no
		}
	}
}

nesnd_south_ming_peace_ready = {
 custom_tooltip = { text = nesnd_south_peace_stability_tt
    nesnd_south_pressure_value < 60
    NOT = { has_modifier = declared_bankruptcy }
 }
 custom_tooltip = { text = nesnd_south_ming_supply_tt nesnd_south_ming_supply_ready = yes }

	c:USM ?= this
	nesnd_board_stance_value = 1
	NOT = {
		has_variable = tag_WUU_var
	}
	NOT = {
		has_variable = nesnd_south_peace_committed
	}
	NOT = {
		has_global_variable = nesnd_south_settled
	}
	NOT = {
		has_global_variable = usm_civilwar_occured
	}
	is_at_war = no
	in_default = no
	bureaucracy >= 0
	has_law_or_variant = law_type:law_monarchy
	exists = ruler
	ruler = {
		is_character_alive = yes
		nesnd_ming_authorized_dynast = yes
		has_variable = nesnd_ming_claimant
	}
	nesnd_south_ming_strength >= 85
	nesnd_south_ming_lead >= 20
	nesnd_south_ming_accord >= 75
	any_country = {
		count >= 2
		nesnd_south_mainland_member = yes
		nesnd_south_accord_valid = yes
		nesnd_south_accord_side_value = 1
	}
}
