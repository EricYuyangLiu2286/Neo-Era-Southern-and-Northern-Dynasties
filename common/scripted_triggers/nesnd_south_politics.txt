nesnd_south_accord_valid = {
	has_variable = nesnd_south_accord_active
	exists = c:USM
	NOT = {
		has_war_with = c:USM
	}
	NOT = {
		has_variable = nesnd_south_accord_broken
	}
	nesnd_south_political_member = yes
	OR = {
		is_subject = no
		is_subject_of = c:USM
	}
	OR = {
		AND = {
			nesnd_south_accord_side_value = 1
			nesnd_board_stance_value = 1
		}
		AND = {
			nesnd_south_accord_side_value = 2
			nesnd_board_stance_value = 2
		}
	}
}

nesnd_south_member_can_integrate = {
	nesnd_south_integration_member = yes
	is_at_war = no
	OR = {
		is_subject = no
		is_subject_of = c:USM
	}
	nesnd_south_accord_valid = yes
	nesnd_south_accord_side_value = 2
	has_variable = nesnd_south_accepts_integration
	trigger_if = {
		limit = {
			nesnd_south_protected_overseas = yes
		}
		any_scope_state = {
			NOT = {
				nesnd_south_chinese_state = yes
			}
			controller = owner
		}
	}
	trigger_if = {
		limit = {
			any_scope_state = {
				NOT = {
					nesnd_south_chinese_state = yes
				}
			}
		}
		any_scope_state = {
			NOT = {
				nesnd_south_chinese_state = yes
			}
			controller = owner
		}
	}
	NOT = {
		any_scope_state = {
			nesnd_south_chinese_state = yes
			NOT = {
				controller = owner
			}
		}
	}
	trigger_if = {
		limit = {
			is_player = yes
		}
		c:USM = {
			is_player = no
		}
	}
}
