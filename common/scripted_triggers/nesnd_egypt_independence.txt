# Country-only patron contract; valid both before and after Egypt becomes a subject.
nesnd_egypt_patron_country = {
    hidden_trigger = { nesnd_egypt_signatory = yes } is_country_alive = yes is_subject = no
    NOT = { is_country_type = decentralized } can_have_subjects = yes
}
# Creation-only contract; never use for an already established relationship.
nesnd_egypt_patron_can_take_egypt = {
    nesnd_egypt_patron_country = yes
    trigger_if = { limit = { exists = c:EGY }
        c:EGY = { is_country_alive = yes is_subject = no }
        NOT = { has_war_with = c:EGY }
        OR = {
            can_have_as_subject = { who = c:EGY type = subject_type_protectorate }
            can_have_as_subject = { who = c:EGY type = subject_type_nesnd_egypt_protectorate }
        }
    }
    trigger_else = { always = no }
}
