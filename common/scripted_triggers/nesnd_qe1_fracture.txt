	nesnd_qe1_agenda_live = yes num_states >= 5
	NOT = { has_variable = nesnd_qe1_civil_raised }
	NOT = { exists = c:QRC }
	trigger_if = { limit = { has_variable = nesnd_qe1_balance }
		OR = {
			AND = { var:nesnd_qe1_balance <= -35 ig:ig_armed_forces = { ig_clout >= 0.25 ig_approval <= -5 } }
			AND = { var:nesnd_qe1_balance >= 35 ig:ig_intelligentsia = { ig_clout >= 0.2 ig_approval <= -5 } }
		}
	}
	trigger_else = { always = no }
}

	trigger_else = { always = no }
