nesnd_ottoman_recovery_open = {
    c:TUR ?= this is_country_alive = yes
}
nesnd_ottoman_tanzimat_success = {
    custom_tooltip = { text = nesnd_ui_not_failed_sick_man_main_tt NOT = { has_variable = failed_sick_man_main } }
    # sick_man_complete is set by both ending events (sick_man.2, sick_man.3) and kept; sick_man_var is removed by
    # je_sick_man_main_cleanup after sick_man.2 (GAME 00_victoria_cleanup_scripted_effects.txt:1694-1697), so it cannot be read here.
    hidden_trigger = { has_variable = sick_man_complete }
}
nesnd_ottoman_tanzimat_failed = {
    OR = { custom_tooltip = { text = nesnd_ui_failed_sick_man_main_tt has_variable = failed_sick_man_main } AND = { hidden_trigger = { has_variable = sick_man_complete } NOT = { nesnd_ottoman_tanzimat_success = yes } } }
}
nesnd_ottoman_reform_active = {
    OR = { has_modifier = nesnd_ottoman_qin_reform has_modifier = nesnd_ottoman_qin_reform_success has_modifier = nesnd_ottoman_qin_reform_failed }
}
nesnd_ottoman_claim_target = {
    trigger_if = {
        limit = { exists = c:TUR }
        c:TUR = {
            is_country_alive = yes
            NOT = { has_claim = prev }
            nesnd_ottoman_enthusiasm_value >= nesnd_ottoman_claim_cost
        }
        any_scope_state = {
            owner = {
                NOT = { c:TUR ?= this }
                # Egypt's land stays on offer while Egypt is independent or under a third country; only an Ottoman Egypt hides it.
                trigger_if = { limit = { c:EGY ?= this } NOT = { is_subject_of = c:TUR } }
            }
            any_neighbouring_state = { owner = c:TUR }
        }
    }
    trigger_else = { always = no }
}
nesnd_ottoman_military_building = {
    OR = { is_building_type = building_barrack is_building_type = building_conscription_center }
}

nesnd_ottoman_qinist_candidate = {
    # Interest group scope: one of the three groups that can turn Qinist during the army reform (event nesnd_ottoman_recovery.2).
    OR = { is_interest_group_type = ig_petty_bourgeoisie is_interest_group_type = ig_landowners is_interest_group_type = ig_armed_forces }
    leader ?= {
        NOT = { has_ideology = ideology:ideology_ce_qin_royalist_leader }
        is_ruler_of_own_country = no
        is_heir_of_own_country = no
    }
}
nesnd_qinist_origin_country = {
    # Country scope: the countries the Qinist leader ideology was written for (same identity conditions as its country_trigger in
    # common/ideologies/ce_leader_ideologies.txt). Elsewhere only leaders marked nesnd_qinist_allowed may hold the ideology.
    OR = {
        c:QE1 ?= this
        AND = {
            c:PER ?= this
            exists = ruler
            ruler.culture = { has_discrimination_trait = heritage_han }
        }
        AND = { exists = c:QE1 civil_war_origin_country ?= c:QE1 }
    }
}
nesnd_ottoman_eastern_frontier_held = {
    # Country scope: part of the frontier states that "Press East" claims (event nesnd_ottoman_recovery.3) lies outside Ottoman hands,
    # whoever holds it.
    OR = {
        s:STATE_HERAT = { any_scope_state = { nesnd_ottoman_state_outside_ottoman_hands = yes } }
        s:STATE_MERZ = { any_scope_state = { nesnd_ottoman_state_outside_ottoman_hands = yes } }
        s:STATE_TURKMENIA = { any_scope_state = { nesnd_ottoman_state_outside_ottoman_hands = yes } }
        s:STATE_KHIVA = { any_scope_state = { nesnd_ottoman_state_outside_ottoman_hands = yes } }
        s:STATE_BALKH = { any_scope_state = { nesnd_ottoman_state_outside_ottoman_hands = yes } }
    }
}
nesnd_ottoman_tanzimat_settled = {
    # Country scope: the vanilla closing event (sick_man.2 or sick_man.3) has been answered. sick_man.1 sets sick_man_var at the start
    # (GAME events/sick_man_events.txt:25-28); both closing events set sick_man_complete in immediate (:93, :200) and remove
    # sick_man_var in their after block via je_sick_man_main_cleanup (:174-175, :252-254; 00_victoria_cleanup_scripted_effects.txt:1694-1697).
    has_variable = sick_man_complete
    NOT = { has_variable = sick_man_var }
}
nesnd_ottoman_qin_reform_known = {
    # Country scope: the Qin-style army reform has been carried out at least once. The last three branches cover saves from before
    # nesnd_ottoman_qin_reform_begun existed (reform running now, or a Qinist leader event from B1 has fired).
    OR = {
        has_variable = nesnd_ottoman_qin_reform_begun
        nesnd_ottoman_reform_active = yes
        any_interest_group = { has_variable = nesnd_qinist_allowed }
        any_interest_group = { leader ?= { has_variable = nesnd_qinist_allowed } }
    }
}

    culture_percent_state = { target = cu:turkish value < 0.5 }
