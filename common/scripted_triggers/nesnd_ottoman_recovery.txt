nesnd_ottoman_recovery_open = {
    c:TUR ?= this is_country_alive = yes
}
nesnd_ottoman_tanzimat_success = {
    custom_tooltip = { text = nesnd_ui_not_failed_sick_man_main_tt NOT = { has_variable = failed_sick_man_main } }
    trigger_if = {
        limit = { hidden_trigger = { has_variable = sick_man_complete } hidden_trigger = { has_variable = sick_man_var } }
        hidden_trigger = { var:sick_man_var >= 4 }
    }
    trigger_else = { always = no }
}
nesnd_ottoman_tanzimat_failed = {
    OR = { custom_tooltip = { text = nesnd_ui_failed_sick_man_main_tt has_variable = failed_sick_man_main } AND = { hidden_trigger = { has_variable = sick_man_complete } NOT = { nesnd_ottoman_tanzimat_success = yes } } }
}
nesnd_ottoman_reform_active = {
    OR = { has_modifier = nesnd_ottoman_qin_reform has_modifier = nesnd_ottoman_qin_reform_success has_modifier = nesnd_ottoman_qin_reform_failed }
}
nesnd_ottoman_claim_target = {
    trigger_if = {
        limit = { exists = c:TUR }
        c:TUR = {
            is_country_alive = yes
            NOT = { has_claim = prev }
            nesnd_ottoman_enthusiasm_value >= nesnd_ottoman_claim_cost
        }
        any_scope_state = {
            owner = {
                NOT = { c:TUR ?= this }
                trigger_if = { limit = { c:EGY ?= this } is_subject = no }
            }
            any_neighbouring_state = { owner = c:TUR }
        }
    }
    trigger_else = { always = no }
}
nesnd_ottoman_military_building = {
    OR = { is_building_type = building_barrack is_building_type = building_conscription_center }
}

nesnd_ottoman_qinist_candidate = {
    # Interest group scope: one of the three groups that can turn Qinist during the army reform (event nesnd_ottoman_recovery.2).
    OR = { is_interest_group_type = ig_petty_bourgeoisie is_interest_group_type = ig_landowners is_interest_group_type = ig_armed_forces }
    leader ?= { NOT = { has_ideology = ideology:ideology_ce_qin_royalist_leader } }
}
nesnd_qinist_origin_country = {
    # Country scope: the countries the Qinist leader ideology was written for (same identity conditions as its country_trigger in
    # common/ideologies/ce_leader_ideologies.txt). Elsewhere only leaders marked nesnd_qinist_allowed may hold the ideology.
    exists = c:QE1
    OR = {
        c:QE1 = This
        AND = {
            c:PER = This
            exists = ruler
            ruler.culture = { has_discrimination_trait = heritage_han }
        }
        civil_war_origin_country ?= c:QE1
    }
}
