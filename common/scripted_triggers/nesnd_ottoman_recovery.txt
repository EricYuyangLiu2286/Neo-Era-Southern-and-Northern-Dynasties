nesnd_ottoman_tanzimat_success = {
    NOT = { has_variable = failed_sick_man_main }
    trigger_if = {
        limit = { has_variable = sick_man_complete has_variable = sick_man_var }
        var:sick_man_var >= 4
    }
    trigger_else = { always = no }
}
nesnd_ottoman_tanzimat_failed = {
    OR = { has_variable = failed_sick_man_main AND = { has_variable = sick_man_complete NOT = { nesnd_ottoman_tanzimat_success = yes } } }
}
nesnd_ottoman_reform_active = {
    OR = { has_modifier = nesnd_ottoman_qin_reform has_modifier = nesnd_ottoman_qin_reform_success has_modifier = nesnd_ottoman_qin_reform_failed }
}
nesnd_ottoman_claim_target = {
    trigger_if = {
        limit = { exists = c:TUR }
        c:TUR = {
            is_country_alive = yes
            NOT = { has_claim = prev }
            NOT = { owns_entire_state_region = prev }
        }
        any_scope_state = {
            owner = {
                NOT = { c:TUR ?= this }
                trigger_if = { limit = { c:EGY ?= this } is_subject = no }
            }
        }
    }
    trigger_else = { always = no }
}
