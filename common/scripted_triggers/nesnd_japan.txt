# Country-scope predicates. All named-country accesses have alive guards.
nesnd_japan_partition_possible = {
    trigger_if = {
        limit = { exists = c:JAP exists = c:USM }
        OR = {
            AND = {
                NOT = {
                    has_global_variable = nesnd_japan_partitioned
                }
                NOT = {
                    exists = c:JTK
                }
                NOT = {
                    exists = c:JIM
                }
                c:USM = {
                    is_country_alive = yes
                    has_variable = tag_WUU_var
                }
                c:JAP = {
                    is_country_alive = yes
                    OR = {
                        AND = { has_law = law_type:law_monarchy nesnd_japan_three_houses = yes }
                        AND = { is_subject_of = c:USM c:USM = { has_variable = nesnd_eastern_submission_accepted } }
                    }
                    OR = {
                        is_subject = no
                        is_subject_of = c:USM
                    }
                    owns_entire_state_region = STATE_KANTO
                    owns_entire_state_region = STATE_KYOTO
                    owns_entire_state_region = STATE_KANSAI
                    owns_entire_state_region = STATE_CHUGOKU
                    NOT = { any_scope_state = {
                        nesnd_eastern_archipelago_state = yes
                        NOR = { controller = owner controller = c:USM }
                    } }
                }
            }
            AND = {
                NOT = {
                    has_global_variable = nesnd_japan_partitioned
                }
                nesnd_japan_recovery_partition_possible = yes
            }
        }
    }
    trigger_else = { always = no }
}
nesnd_japan_fiscal_ready = { bureaucracy >= 0 net_total_income >= 0 }
nesnd_japan_army_ready = {
    government_legitimacy >= 50
    army_power_projection > 0
    c:JAP ?= { army_power_projection <= prev.army_power_projection }
}
nesnd_japan_navy_ready = {
    naval_power_projection > 0
    military_ship_maintenance_fulfillment >= 0.9
    c:JAP ?= { naval_power_projection <= prev.naval_power_projection }
}
nesnd_japan_can_send = {
    nesnd_japan_diplomatic_window = yes
    NOT = { has_global_variable = nesnd_japan_offer_sent }
}
nesnd_japan_is_united = { nesnd_jp_core_united = yes is_subject = no }

nesnd_japan_three_houses = {
    trigger_if = {
        limit = { exists = ruler }
        ruler = { OR = { nesnd_character_of_dynasty = { HOUSE = house_toyotomi } nesnd_character_of_dynasty = { HOUSE = house_tokugawa } nesnd_character_of_dynasty = { HOUSE = house_yamato } } }
    }
    trigger_else = { always = no }
}

        has_global_variable = nesnd_south_settled
        NOT = { has_variable = nesnd_japan_recovery_pending }
        NOT = { has_global_variable = nesnd_japan_dossier_closed }
        nesnd_japan_partition_possible = yes

        c:JAP = {
            is_country_alive = yes

}

# AI country scope with a target_country supplied by the native strategy database.
nesnd_japan_rival_claim_available = {
    trigger_if = {
        limit = { has_variable = nesnd_japan_partition_state exists = scope:target_country }
        has_global_variable = nesnd_japan_partitioned
        nesnd_jp_country = yes
        is_subject = no
        is_at_war = no
        is_active_in_diplomatic_play = no
        aggressive_diplomatic_plays_permitted = yes
        NOT = { has_variable = nesnd_jp_request_clock }
        NOT = { has_truce_with = scope:target_country }
        NOT = { is_in_same_power_bloc = scope:target_country }
        NOT = { has_friendly_diplomatic_pact = { TARGET = scope:target_country } }
        scope:target_country = {
            NOT = { this = root }
            nesnd_jp_country = yes
            has_variable = nesnd_japan_partition_state
            is_subject = no
            is_at_war = no
            is_active_in_diplomatic_play = no
            NOT = { has_variable = nesnd_jp_request_clock }
            relations:root < relations_threshold:cordial
            any_scope_state = { nesnd_japan_claimed_rival_seat = yes }
        }
    }
    trigger_else = { always = no }
}
nesnd_japan_claimed_rival_seat = {
    has_claim_by = root
    OR = {
        AND = { state_region = s:STATE_KANTO owner = { c:JTK ?= this } }
        AND = { state_region = s:STATE_KYOTO owner = { c:JIM ?= this } }
        AND = { state_region = s:STATE_CHUGOKU owner = { c:JAP ?= this } }
    }
}

nesnd_japan_legacy_claims_available = {
    trigger_if = {
        limit = { exists = c:JAP exists = c:JTK exists = c:JIM }
        c:JAP = {
            nesnd_japan_legacy_seat_live = yes
            has_variable = nesnd_toyotomi_remnant
            trigger_if = { limit = { exists = ruler } ruler = { nesnd_character_of_dynasty = { HOUSE = house_toyotomi } } }
            trigger_else = { always = no }
            NOT = { any_scope_state = { state_region = s:STATE_CHUGOKU OR = { has_claim_by = c:JTK has_claim_by = c:JIM } } }
        }
        c:JTK = {
            nesnd_japan_legacy_seat_live = yes
            trigger_if = { limit = { exists = ruler } ruler = { nesnd_character_of_dynasty = { HOUSE = house_tokugawa } } }
            trigger_else = { always = no }
            NOT = { any_scope_state = { state_region = s:STATE_KANTO OR = { has_claim_by = c:JAP has_claim_by = c:JIM } } }
        }
        c:JIM = {
            nesnd_japan_legacy_seat_live = yes
            OR = {
                AND = {
                    trigger_if = { limit = { exists = ruler } ruler = { nesnd_character_of_dynasty = { HOUSE = house_yamato } } }
                    trigger_else = { always = no }
                }
                AND = { has_variable = nesnd_japan_lin_regency_appointed nesnd_jp_emperor_valid = yes }
            }
            NOT = { any_scope_state = { state_region = s:STATE_KYOTO OR = { has_claim_by = c:JAP has_claim_by = c:JTK } } }
        }
    }
    trigger_else = { always = no }
}

# The tributary relationship at the opening is historical. A new submission ends the dispute.
nesnd_eastern_target_valid = {
    exists = c:JAP
    c:JAP ?= { is_country_alive = yes }
    NOR = {
        has_variable = nesnd_eastern_submission_accepted
        AND = { has_global_variable = nesnd_japan_war_started c:JAP ?= { is_subject_of = c:USM } }
    }
}

# Native war goal callbacks supply the play as a named scope. A late peace cannot reopen a closed expedition.
nesnd_eastern_reorganization_valid = {
    c:USM ?= this
    nesnd_japan_partition_possible = yes
    trigger_if = {
        limit = { exists = scope:diplomatic_play scope:diplomatic_play ?= { is_diplomatic_play_type = dp_nesnd_eastern_intervention } }
        NOT = { has_global_variable = nesnd_japan_dossier_closed }
        nesnd_eastern_target_valid = yes
    }
}

# Honshu, Hokkaido, Kyushu and Shikoku use the same boundary for the expedition and the exodus.
nesnd_eastern_archipelago_state = {
    OR = {
        state_region = s:STATE_KANTO state_region = s:STATE_KYOTO state_region = s:STATE_KANSAI
        state_region = s:STATE_CHUGOKU state_region = s:STATE_TOHOKU state_region = s:STATE_TOKAI
        state_region = s:STATE_HOKUSHINETSU state_region = s:STATE_HOKKAIDO
        state_region = s:STATE_KYUSHU state_region = s:STATE_SHIKOKU
    }
}
