# ==== The centre and the states (formerly nesnd_india_order.txt) ====
# The journal nesnd_india_order: Mughal ("Mughal" here is any country for which nesnd_india_empire holds)
# chooses, for its Indian subjects, a central or a local arrangement in three matters:
#   tax       nesnd_india_order_tax      1 raise_payments with each Indian subject, tax capacity at home
#                                        2 decrease_payments with each Indian subject
#   army      nesnd_india_order_army     1 authority per subject at home, liberty desire on each subject
#                                        2 exempt_from_service with each Indian subject
#   school    nesnd_india_order_school   1 da_knowledge_sharing with each Indian subject, qualifications at home
#                                        2 qualifications and less liberty desire on each subject
# Variables are listed in common/script_values/nesnd_india_empire.txt.

# Mughal scope. The journal is held.
nesnd_india_order_live = {
    save_temporary_scope_as = india_order_self
    nesnd_india_empire = yes
    custom_tooltip = { text = nesnd_ui_india_reform_done_tt has_variable = nesnd_india_reform_done }
    custom_tooltip = {
        text = nesnd_india_order_has_subject_tt
        any_subject_or_below = { nesnd_india_country = yes is_direct_subject_of = scope:india_order_self }
    }
}

# Country scope. One of Mughal's Indian subjects (any subject type). LORD is Mughal.
nesnd_india_order_member = { nesnd_india_country = yes is_direct_subject_of = $LORD$ }

# Mughal scope. ISSUE is tax, army or school; END 1 central, 2 local. True when that matter is settled that way.
nesnd_india_order_is = { nesnd_india_order_$ISSUE$_value = $END$ }
# Mughal scope. The five years since the last choice in that matter have run out.
nesnd_india_order_free = {
    trigger_if = { limit = { has_variable = nesnd_india_order_$ISSUE$_wait }
        custom_tooltip = { text = nesnd_india_order_$ISSUE$_wait_tt NOT = { has_variable = nesnd_india_order_$ISSUE$_wait } }
    }
}
# Mughal scope. A button for ISSUE and END can be pressed.
nesnd_india_order_can_choose = {
    nesnd_india_order_live = yes
    custom_tooltip = { text = nesnd_india_order_already_tt NOT = { nesnd_india_order_is = { ISSUE = $ISSUE$ END = $END$ } } }
    nesnd_india_order_free = { ISSUE = $ISSUE$ }
}

# Subject scope, with scope:india_order_lord (Mughal) and scope:india_order_member (this subject) saved.
# Whether Mughal could sign each treaty with this subject today (the treaty's own conditions decide).
nesnd_india_order_can_raise = { scope:india_order_lord = { can_create_diplomatic_pact = { target = scope:india_order_member type = raise_payments } } }
nesnd_india_order_can_decrease = { scope:india_order_lord = { can_create_diplomatic_pact = { target = scope:india_order_member type = decrease_payments } } }
nesnd_india_order_can_exempt = { scope:india_order_lord = { can_create_diplomatic_pact = { target = scope:india_order_member type = exempt_from_service } } }
nesnd_india_order_can_share = { scope:india_order_lord = { can_create_diplomatic_pact = { target = scope:india_order_member type = da_knowledge_sharing } } }

# Subject scope. LORD is its Mughal. In tax, the treaty of the end LORD has chosen now stands between
# LORD (as initiator) and us, whoever made it.
nesnd_india_order_tax_stands = {
    save_temporary_scope_as = india_order_counted
    OR = {
        $LORD$ = { nesnd_india_order_is = { ISSUE = tax END = 1 } has_diplomatic_pact = { who = scope:india_order_counted type = raise_payments is_initiator = yes } }
        $LORD$ = { nesnd_india_order_is = { ISSUE = tax END = 2 } has_diplomatic_pact = { who = scope:india_order_counted type = decrease_payments is_initiator = yes } }
    }
}
# Subject scope. Our tax marker is LORD's and is for the end LORD has chosen now. A marker for the other
# end is one whose treaty outlived its removal when the matter was changed; a marker of another Mughal
# (the subject changed hands while that Mughal's treaty was in its forced duration) is not LORD's.
nesnd_india_order_tax_marked = {
    var:nesnd_india_order_lord ?= $LORD$
    OR = {
        AND = { nesnd_india_order_tax_pact_value = 1 $LORD$ = { nesnd_india_order_is = { ISSUE = tax END = 1 } } }
        AND = { nesnd_india_order_tax_pact_value = 2 $LORD$ = { nesnd_india_order_is = { ISSUE = tax END = 2 } } }
    }
}

# Subject scope, from its own monthly refresh. It was given something by this journal and is no longer
# an Indian subject of a Mughal that holds the journal.
nesnd_india_order_left = {
    has_variable = nesnd_india_order_lord
    trigger_if = { limit = { exists = var:nesnd_india_order_lord }
        NAND = {
            var:nesnd_india_order_lord = { is_country_alive = yes nesnd_india_order_live = yes }
            nesnd_india_order_member = { LORD = var:nesnd_india_order_lord }
        }
    }
}

# Mughal scope. AI: more than half of its Indian subjects have liberty desire of 50 or more.
nesnd_india_order_ai_restless = { nesnd_india_order_restless_count > nesnd_india_order_half_members }

# ==== Mughal's old agreement, the council of princes, famine relief (formerly nesnd_india_mughal.txt) ====
# Mughal's own affairs. "Mughal" is only the country holding nesnd_india_mughal_court. Variables and
# sums are listed in common/script_values/nesnd_india_empire.txt.

# ---------------------------------------------------------------------------------------------
# Who holds the two journals
# ---------------------------------------------------------------------------------------------
# Country scope. We are a subject, and the capital of our overlord is not in an Indian state. France
# holds Indian states, but its capital is outside India, so it is such an overlord. Used by the journal
# nesnd_mug_france, its loan and redemption, and the matters of kinds 7 and 8.
nesnd_mug_foreign_overlord = {
    trigger_if = { limit = { is_subject = yes exists = overlord }
        overlord = {
            trigger_if = { limit = { exists = capital } capital = { NOT = { nesnd_india_state = yes } } }
            trigger_else = { always = no }
        }
    }
    trigger_else = { always = no }
}
# The old agreement with the overlord: Mughal is a subject of a foreign overlord.
nesnd_mug_france_live = {
    hidden_trigger = {
        is_country_alive = yes
        has_variable = nesnd_india_mughal_court
        nesnd_mug_foreign_overlord = yes
    }
}
# The council of princes: Mughal has finished its reform.
nesnd_mug_council_live = {
    hidden_trigger = {
        is_country_alive = yes
        has_variable = nesnd_india_mughal_court
        has_variable = nesnd_india_reform_done
    }
}

# The loan
# ---------------------------------------------------------------------------------------------
# Article scope, with scope:mug_loan_debtor and scope:mug_loan_creditor saved. The repayment article.
nesnd_mug_loan_article = {
    trigger_if = { limit = { exists = source_country exists = target_country }
        has_type = money_transfer
        source_country = scope:mug_loan_debtor
        target_country = scope:mug_loan_creditor
        input_quantity = nesnd_mug_loan_weekly
    }
    trigger_else = { always = no }
}
# Mughal scope. Our repayment treaty with PARTNER is in force.
nesnd_mug_loan_treaty = {
    save_temporary_scope_as = mug_loan_debtor
    $PARTNER$ = { save_temporary_scope_as = mug_loan_creditor }
    any_scope_treaty = {
        binds = scope:mug_loan_creditor
        any_scope_article = { nesnd_mug_loan_article = yes }
    }
}
# Any scope. The repayment treaty and the trade privilege owed on default can be created.
nesnd_mug_loan_possible = {
    can_create_treaty = {
        first_country = $DEBTOR$
        second_country = $CREDITOR$
        is_draft = no
        binding_period = { years = 5 }
        articles_to_create = {
            { article = money_transfer source_country = $DEBTOR$ target_country = $CREDITOR$ inputs = { { quantity = 460 } } }
        }
    }
}
nesnd_mug_trade_possible = {
    can_create_treaty = {
        first_country = $DEBTOR$
        second_country = $CREDITOR$
        is_draft = no
        binding_period = { years = 10 }
        articles_to_create = {
            { article = trade_privilege source_country = $DEBTOR$ target_country = $CREDITOR$ }
        }
    }
}

# State region scope. A harvest condition that cuts the throughput of farms.
nesnd_mug_food_harvest = {
    OR = {
        has_harvest_condition = drought
        has_harvest_condition = flood
        has_harvest_condition = frost
        has_harvest_condition = wildfire
        has_harvest_condition = hailstorm
        has_harvest_condition = locust_swarm
        has_harvest_condition = heatwave
        has_harvest_condition = disease_outbreak
        has_harvest_condition = extreme_winds
        has_harvest_condition = torrential_rains
    }
}
# State scope. A populated Indian state in such a region.
nesnd_mug_famine_struck = {
    nesnd_india_state = yes
    state_population > 0
    state_region = { nesnd_mug_food_harvest = yes }
}
# Country scope. An Indian government that asks for help as a prince: a prince, or an Indian empire
# (BHT or DRA formed by a prince) that is not Mughal.
nesnd_mug_famine_asker = {
    NOT = { has_variable = nesnd_india_mughal_court }
    OR = { nesnd_india_prince = yes nesnd_india_empire = yes }
}
nesnd_mug_famine_due = {
    is_country_alive = yes
    OR = {
        AND = { has_variable = nesnd_india_mughal_court nesnd_india_country = yes }
        nesnd_mug_famine_asker = yes
    }
    NOT = { has_variable = nesnd_mug_famine_wait }
    # A plea for famine aid of ours is still on its way (matter kind 9): no new famine events until it is answered.
    NOT = { nesnd_india_matter_kind_value = 9 }
    any_scope_state = { nesnd_mug_famine_struck = yes }
}
# Country scope. We can pay the relief of our struck states ourselves.
nesnd_mug_can_relieve = {
    custom_tooltip = {
        text = nesnd_mug_can_relieve_tt
        nesnd_mug_famine_states > 0
        gold_reserves >= nesnd_mug_famine_cost
        in_default = no
    }
}

# Matters of kinds 7 to 9 (Indian affairs, common/scripted_triggers/nesnd_india_matters.txt)
# ---------------------------------------------------------------------------------------------
# Holder scope: the matter still stands.
nesnd_mug_matter_valid = {
    trigger_if = { limit = { exists = var:nesnd_india_matter_to }
        trigger_if = { limit = { nesnd_india_matter_kind_value = 9 }
            nesnd_mug_famine_asker = yes
            any_scope_state = { has_variable = nesnd_mug_famine_state }
            OR = {
                var:nesnd_india_matter_to = { has_variable = nesnd_india_mughal_court }
                overlord ?= var:nesnd_india_matter_to
            }
        }
        trigger_else = {
            nesnd_mug_france_live = yes
            overlord ?= var:nesnd_india_matter_to
            trigger_if = { limit = { nesnd_india_matter_kind_value = 7 } NOT = { has_variable = nesnd_mug_loan_from } }
        }
    }
    trigger_else = { always = no }
}
# Answering scope: we can agree today.

    trigger_else_if = { limit = { nesnd_india_matter_is = { KIND = 8 } }

    }
    trigger_else = {
        custom_tooltip = { text = nesnd_mug_we_can_aid_tt gold_reserves >= nesnd_india_matter_quote in_default = no }
    }
}
