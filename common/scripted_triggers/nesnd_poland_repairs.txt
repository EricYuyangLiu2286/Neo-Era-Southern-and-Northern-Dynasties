# Country scope; no reference to an absent private journal is evaluated.
nesnd_poland_can_issue_ultimatum = {
	c:KRA ?= { is_subject = no is_at_war = no is_diplomatic_play_committed_participant = no }
	nesnd_poland_active = yes
	is_at_war = no
	custom_tooltip = { text = nesnd_poland_no_pending_demand_tt NOT = { has_variable = nesnd_polish_pending_state } }
	trigger_if = { limit = { has_journal_entry = je_ce_polish_war_preparation }
		je:je_ce_polish_war_preparation = { "scripted_bar_progress(ce_polish_reconquest_war_prep_bar)" >= 100 }
	}
	trigger_else_if = { limit = { has_journal_entry = je_ce_polish_territory_claim }
		je:je_ce_polish_territory_claim = { "scripted_bar_progress(ce_polish_territory_claim_bar)" >= 100 }
	}
	trigger_else = { always = no }
}
# State scope. Shared by availability, selection and reply-time validation.
nesnd_poland_ultimatum_state = {
	trigger_if = { limit = { exists = c:KRA exists = owner }
		NOT = { owner = c:KRA }
		has_claim_by = c:KRA
		owner = {
			OR = { is_at_war = no AND = { custom_tooltip = { text = nesnd_ui_poland_civil_ultimatum_target_tt has_variable = nesnd_poland_civil_ultimatum_target } any_civil_war = { always = yes } } }
			NOR = {
				has_subject_relation_with = c:KRA
				has_truce_with = c:KRA

				custom_tooltip = { text = nesnd_ui_poland_request_cooldown_tt has_variable = nesnd_poland_request_cooldown }
				any_scope_treaty = { binds = c:KRA any_scope_article = { has_type = alliance } }
				any_scope_treaty = { binds = c:KRA any_scope_article = { has_type = defensive_pact } }
				any_scope_treaty = { binds = c:KRA any_scope_article = { has_type = guarantee_independence } }
			}
		}
	}
	trigger_else = { always = no }
}
nesnd_poland_ultimatum_valid = {
	trigger_if = { limit = { exists = c:KRA exists = scope:target_diplo_play_state exists = scope:ce_from_country }
		nesnd_poland_active = yes
		OR = { is_at_war = no AND = { custom_tooltip = { text = nesnd_ui_poland_civil_ultimatum_target_tt has_variable = nesnd_poland_civil_ultimatum_target } any_civil_war = { always = yes } } }
		c:KRA = { is_subject = no is_at_war = no is_diplomatic_play_committed_participant = no }
		scope:ce_from_country = { is_country_alive = yes nesnd_poland_matching_pending = yes }
		scope:target_diplo_play_state = { owner = root nesnd_poland_ultimatum_state = yes }
	}
	trigger_else = { always = no }
}

# Country scope. Restoration = the 1809 Duchy of Warsaw: capital at Warsaw and the five
# Polish-majority states held whole (WORK reports/2026-11-17-poland-redesign/map_poland.md).
# Readers: nesnd_poland_can_declare_victory (victory button).
nesnd_poland_restoration_met = {
	capital ?= { state_region = s:STATE_GREATER_POLAND }
	owns_entire_state_region = STATE_GREATER_POLAND
	owns_entire_state_region = STATE_MAZOVIA
	owns_entire_state_region = STATE_LESSER_POLAND
	owns_entire_state_region = STATE_WEST_GALICIA
	owns_entire_state_region = STATE_POSEN
}

# No scope needed. Partition: Krakow gone, a subject of one of the three partition powers,
# or holding no part of Greater Poland, Mazovia or Lesser Poland.
# Readers: partition outcome of je_ce_the_polish_problem.
nesnd_poland_partitioned = {
	OR = {
		custom_tooltip = {
			text = nesnd_poland_partition_gone_tt
			NOT = { exists = c:KRA }
		}
		custom_tooltip = {
			text = nesnd_poland_partition_subject_tt
			c:KRA ?= {
				OR = {
					trigger_if = { limit = { exists = c:RUS } is_subject_of = c:RUS } trigger_else = { always = no }
					trigger_if = { limit = { exists = c:PRU } is_subject_of = c:PRU } trigger_else = { always = no }
					trigger_if = { limit = { exists = c:AUS } is_subject_of = c:AUS } trigger_else = { always = no }
				}
			}
		}
		custom_tooltip = {
			text = nesnd_poland_partition_core_lost_tt
			c:KRA ?= {
				NOT = {
					any_scope_state = {
						OR = {
							state_region = s:STATE_GREATER_POLAND
							state_region = s:STATE_MAZOVIA
							state_region = s:STATE_LESSER_POLAND
						}
					}
				}
			}
		}
	}
}
