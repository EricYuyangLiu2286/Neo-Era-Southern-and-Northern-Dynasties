# Country scope: the barracks agenda meets both winning conditions (same counts as the journal outcome, without tooltips).
nesnd_ottoman_autocracy_barracks_agenda_met = {
    nesnd_ottoman_autocracy_barracks_score >= 5
    calc_true_if = {
        amount >= 4
        nesnd_ottoman_autocracy_m1 = yes
        nesnd_ottoman_autocracy_m2 = yes
        nesnd_ottoman_autocracy_m3 = yes
        nesnd_ottoman_autocracy_m4 = yes
        nesnd_ottoman_autocracy_m5 = yes
        nesnd_ottoman_autocracy_m6 = yes
    }
}

# Country scope (R10): when both agendas are met in the same month the higher score wins; a tie goes to the side
# chosen in event nesnd_ottoman_recovery.6. Read by the Sultan outcome, which is written first.
nesnd_ottoman_autocracy_sultan_wins_tie = {
    OR = {
        NOT = { nesnd_ottoman_autocracy_barracks_agenda_met = yes }
        nesnd_ottoman_autocracy_lead > 0
        AND = {
            nesnd_ottoman_autocracy_lead = 0
            NOT = { AND = { has_variable = nesnd_ottoman_route_lean var:nesnd_ottoman_route_lean = flag:barracks } }
        }
    }
}
