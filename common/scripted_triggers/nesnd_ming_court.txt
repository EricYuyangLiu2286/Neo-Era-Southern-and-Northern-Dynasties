    is_country_alive = yes is_player = no num_states > 0 exists = capital
    capital = { controller = prev }
    OR = { c:HNA ?= this c:SIH ?= this }
    OR = { nesnd_board_stance_value = 0 nesnd_board_stance_value = 1 }
    custom_tooltip = { text = nesnd_ui_not_ce_usm_republican_revolution_tag_tt NOT = { has_variable = ce_usm_republican_revolution_tag } }
    custom_tooltip = { text = nesnd_ui_not_tag_wuu_var_tt NOT = { has_variable = tag_WUU_var } }
}

}
