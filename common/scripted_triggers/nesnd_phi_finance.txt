nesnd_phi_relation_valid = {
    c:PHI ?= this is_country_alive = yes
    c:USM ?= { is_country_alive = yes }
    is_subject_of = c:USM
    overlord = c:USM
    market = c:USM.market
    NOT = { has_war_with = c:USM }
}
# Read B's accepted political accord; never create a political accord here.
nesnd_phi_political_accord_valid = {
    has_variable = nesnd_south_accord_active
    NOT = { has_variable = nesnd_south_accord_broken }
    has_variable = nesnd_south_accord_side
    has_variable = nesnd_south_stance
    c:USM ?= { has_variable = nesnd_south_stance }
    OR = {
        AND = { var:nesnd_south_accord_side = 1 var:nesnd_south_stance = 1 c:USM = { var:nesnd_south_stance = 1 } }
        AND = { var:nesnd_south_accord_side = 2 var:nesnd_south_stance = 2 c:USM = { var:nesnd_south_stance = 2 } }
    }
}

# Exact stored treaty identity; an unrelated USM-PHI treaty never satisfies this.
nesnd_phi_owned_treaty_live = {
    exists = var:nesnd_phi_treaty
    any_scope_treaty = { this = c:PHI.var:nesnd_phi_treaty binds = c:USM binds = c:PHI }
}
nesnd_phi_owned_treaty_unamended = {
    nesnd_phi_owned_treaty_live = yes
    var:nesnd_phi_treaty = {
        binding_period = c:PHI.var:nesnd_phi_original_binding
        NOT = { any_scope_article = { count > 1 always = yes } }
        any_scope_article = { has_type = money_transfer source_country = c:USM target_country = c:PHI input_quantity = c:PHI.var:nesnd_phi_quote }
    }
}
