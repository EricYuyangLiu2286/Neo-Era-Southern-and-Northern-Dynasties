nesnd_external_country = { OR = { c:HUS ?= this c:FRA ?= this c:QE1 ?= this c:SHA ?= this c:BUR ?= this c:MUG ?= this c:GBR ?= this c:RUS ?= this } }
nesnd_external_crisis_settled = {
    OR = {
        has_global_variable = nesnd_south_settled
        AND = {
            exists = c:USM
            c:USM ?= {
                OR = {
                    has_variable = nesnd_south_peace_committed
                    has_variable = nesnd_south_inconclusive_settlement
                    has_variable = usm_rebel_winner_of_the_civil_war
                    AND = {
                        has_variable = usm_government_winner_civilwar
                        NOT = { nesnd_south_unresolved_holdouts = yes }
                    }
                    AND = {
                        has_global_variable = usm_civilwar_status_quo
                        nesnd_south_campaign_resolved = yes
                    }
                }
            }
        }
    }
}

nesnd_external_partner_survives = {
    trigger_if = { limit = { c:MUG ?= this }
        exists = c:FRA
        c:FRA ?= { is_country_alive = yes }
    }
    trigger_else_if = { limit = { c:SHA ?= this }
        exists = c:HUS
        c:HUS ?= { is_country_alive = yes }
    }
    trigger_else = {
        OR = {
            AND = { exists = c:USM c:USM ?= { is_country_alive = yes } }
            AND = { c:QE1 ?= this exists = c:SIH c:SIH ?= { is_country_alive = yes } }
            any_country = {
                is_country_alive = yes
                has_variable = usm_opposition_country_leader_var
                NOT = { this = prev }
            }
        }
    }
}

nesnd_external_agenda_over = {
    OR = {
        nesnd_external_crisis_settled = yes
        NOT = { nesnd_external_partner_survives = yes }
        AND = {
            NOT = { AND = { exists = c:USM c:USM ?= { is_country_alive = yes } } }
            NOT = { any_country = { is_country_alive = yes has_variable = usm_opposition_country_leader_var } }
        }
    }
}

nesnd_external_open_contact = {
    custom_tooltip = { text = nesnd_external_agenda_live_tt nesnd_external_agenda_over = no NOT = { has_variable = nesnd_external_dismissed } }
    custom_tooltip = { text = nesnd_external_no_target_tt NOT = { has_variable = nesnd_external_target } }
    custom_tooltip = { text = nesnd_external_no_pending_tt NOT = { has_variable = nesnd_external_pending } }
    custom_tooltip = { text = nesnd_external_no_commitment_tt NOT = { has_variable = nesnd_external_aid_committed } }
}

nesnd_external_can_contact = {
    nesnd_external_open_contact = yes
    custom_tooltip = {
        text = $PARTNER_TT$
        trigger_if = { limit = { exists = $PARTNER$ }
            $PARTNER$ = { is_country_alive = yes }
            NOT = { has_war_with = $PARTNER$ }
        }
        trigger_else = { always = no }
    }
}

nesnd_external_can_contact_opposition = {
    nesnd_external_open_contact = yes
    is_subject = no
    custom_tooltip = { text = nesnd_external_opposition_available_tt any_country = { is_country_alive = yes has_variable = usm_opposition_country_leader_var NOT = { this = root } NOT = { has_war_with = root } } }
}

nesnd_external_target_valid = {
    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
        }
        exists = var:nesnd_external_target
        var:nesnd_external_target = {
            is_country_alive = yes
        }
        NOT = {
            this = var:nesnd_external_target
        }
        NOT = {
            has_war_with = var:nesnd_external_target
        }
        OR = {
            AND = { NOR = { c:SHA ?= this c:MUG ?= this } c:USM ?= var:nesnd_external_target }
            AND = {
                NOR = { c:SHA ?= this c:MUG ?= this }
                var:nesnd_external_target = {
                    has_variable = usm_opposition_country_leader_var
                }
                is_subject = no
            }
            AND = {
                c:QE1 ?= this
                c:SIH ?= var:nesnd_external_target
            }
            AND = {
                c:SHA ?= this
                c:HUS ?= var:nesnd_external_target
            }
            AND = {
                c:MUG ?= this
                c:FRA ?= var:nesnd_external_target
            }
        }
    }
    trigger_else = {
        always = no
    }
}
nesnd_external_request_valid = {
    trigger_if = {
        limit = {
            exists = scope:nesnd_external_requester
        }
        is_country_alive = yes
        scope:nesnd_external_requester = {
            trigger_if = {
                limit = {
                    exists = var:nesnd_external_pending
                    exists = var:nesnd_external_target
                }
                is_country_alive = yes
                nesnd_external_agenda_over = no
                trigger_if = { limit = { OR = { c:MUG ?= this c:SHA ?= this } }
                    NOT = { has_variable = nesnd_india_request_clock }
                    NOT = { has_variable = nesnd_india_aid_lock }
                    root = { is_at_war = no }
                }
                trigger_else = { is_at_war = no }
                has_journal_entry = nesnd_external_policy
                NOT = {
                    has_variable = nesnd_external_dismissed
                }
                NOT = {
                    has_variable = nesnd_external_aid_committed
                }
                var:nesnd_external_pending = root
                var:nesnd_external_target = root
                nesnd_external_target_valid = yes
                nesnd_external_pending_politics_valid = yes
            }
            trigger_else = {
                always = no
            }
        }
    }
    trigger_else = {
        always = no
    }
}

nesnd_external_money_deliverable = {
    $DONOR$ = {
        custom_tooltip = { text = nesnd_external_donor_at_peace_tt is_at_war = no }
        in_default = no
        NOT = { has_modifier = declared_bankruptcy }
        net_fixed_income >= 5000
        net_total_income >= 1000
        gold_reserves >= 26000
    }
    custom_tooltip = {
        text = nesnd_external_money_terms_tt
        can_create_treaty = {
            first_country = $DONOR$
            second_country = $RECEIVER$
            is_draft = no
            binding_period = { years = 1 }
            articles_to_create = {
                {
                    article = money_transfer
                    source_country = $DONOR$
                    target_country = $RECEIVER$
                    inputs = { { quantity = 500 } }
                }
            }
        }
    }
}

nesnd_external_arms_deliverable = {
    $DONOR$ = {
        custom_tooltip = { text = nesnd_external_donor_at_peace_tt is_at_war = no }
        in_default = no
        NOT = { has_modifier = declared_bankruptcy }
        market = {
            mg:small_arms = {
                market_goods_pricier <= 0.25
                NOT = { country_has_local_shortage = $DONOR$ }
                custom_tooltip = { text = nesnd_external_export_room_tt
                    market_export_cap >= { value = treaty_exports add = 10 }
                }
            }
        }
    }
    $RECEIVER$ = {
        market = {
            mg:small_arms = {
                custom_tooltip = { text = nesnd_external_import_room_tt
                    market_import_cap >= { value = treaty_imports add = 10 }
                }
            }
        }
    }
    custom_tooltip = {
        text = nesnd_external_arms_terms_tt
        can_create_treaty = {
            first_country = $DONOR$
            second_country = $RECEIVER$
            is_draft = no
            binding_period = { years = 1 }
            articles_to_create = {
                {
                    article = goods_transfer
                    source_country = $DONOR$
                    target_country = $RECEIVER$
                    inputs = { { quantity = 10 } { good = g:small_arms } }
                }
            }
        }
    }
}

                    c:SHA ?= this

                    nesnd_external_can_request_arms = yes

        always = no
    }
}

nesnd_external_pending_politics_valid = {
    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
            exists = var:nesnd_external_requester_sponsor
            exists = var:nesnd_external_receiver_sponsor
        }
        trigger_if = {
            limit = {
                is_subject = yes
            }
            overlord = var:nesnd_external_requester_sponsor
        }
        trigger_else = {
            this = var:nesnd_external_requester_sponsor
        }
        var:nesnd_external_target = {
            trigger_if = {
                limit = {
                    is_subject = yes
                }
                overlord = prev.var:nesnd_external_receiver_sponsor
            }
            trigger_else = {
                this = prev.var:nesnd_external_receiver_sponsor
            }
        }
    }
    trigger_else = {
        always = no
    }
}
