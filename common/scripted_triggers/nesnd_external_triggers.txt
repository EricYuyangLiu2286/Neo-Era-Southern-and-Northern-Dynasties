nesnd_external_country = { OR = { c:HUS ?= this c:FRA ?= this c:QE1 ?= this c:SHA ?= this c:BUR ?= this c:MUG ?= this c:GBR ?= this c:RUS ?= this } }
nesnd_external_target_valid = {
    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
        }
        exists = var:nesnd_external_target
        var:nesnd_external_target = {
            is_country_alive = yes
        }
        NOT = {
            this = var:nesnd_external_target
        }
        NOT = {
            has_war_with = var:nesnd_external_target
        }
        OR = {
            c:USM ?= var:nesnd_external_target
            AND = {
                var:nesnd_external_target = {
                    has_variable = usm_opposition_country_leader_var
                }
                is_subject = no
            }
            AND = {
                c:QE1 ?= this
                c:SIH ?= var:nesnd_external_target
            }
            AND = {
                c:SHA ?= this
                c:HUS ?= var:nesnd_external_target
            }
            AND = {
                c:MUG ?= this
                c:FRA ?= var:nesnd_external_target
            }
        }
    }
    trigger_else = {
        always = no
    }
}
nesnd_external_request_valid = {
    trigger_if = {
        limit = {
            exists = scope:nesnd_external_requester
        }
        is_country_alive = yes
        NOT = {
            has_global_variable = nesnd_south_settled
        }
        scope:nesnd_external_requester = {
            trigger_if = {
                limit = {
                    exists = var:nesnd_external_pending
                    exists = var:nesnd_external_target
                }
                is_country_alive = yes
                is_at_war = no
                has_journal_entry = nesnd_external_policy
                NOT = {
                    has_variable = nesnd_external_dismissed
                }
                NOT = {
                    has_variable = nesnd_external_aid_committed
                }
                var:nesnd_external_pending = root
                var:nesnd_external_target = root
                nesnd_external_target_valid = yes
                nesnd_external_pending_politics_valid = yes
            }
            trigger_else = {
                always = no
            }
        }
    }
    trigger_else = {
        always = no
    }
}

    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
        }
        nesnd_external_target_valid = yes
        is_at_war = no
        NOT = {
            has_variable = nesnd_external_cooldown
        }
        NOT = {
            has_variable = nesnd_external_pending
        }
        NOT = {
            has_variable = nesnd_external_aid_committed
        }
        NOT = {
            has_global_variable = nesnd_south_settled
        }
        is_at_war = no
        in_default = no
        NOT = {
            has_modifier = declared_bankruptcy
        }
        net_fixed_income >= 5000
        net_total_income >= 1000
        gold_reserves >= 26000
        can_create_treaty = {
            first_country = root

            binding_period = {
                years = 1
            }
            articles_to_create = {

                    article = money_transfer
                    source_country = root
                    target_country = var:nesnd_external_target
                    inputs = {
                        {
                            quantity = 500
                        }
                    }
                }
            }
        }
    }
    trigger_else = {
        always = no
    }
}

nesnd_external_any_legal_aid = {
    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
        }
        exists = var:nesnd_external_target
        OR = {
            AND = {
                NOR = {
                    c:MUG ?= this
                    c:SHA ?= this
                }
                OR = {
                    nesnd_external_can_offer_money = yes
                    nesnd_external_can_offer_arms = yes
                }
            }
            AND = {
                OR = {
                    c:MUG ?= this
                    c:SHA ?= this
                }
                OR = {
                    nesnd_external_can_request_money = yes
                    nesnd_external_can_request_arms = yes
                }
            }
        }
    }
    trigger_else = {
        always = no
    }
}
nesnd_external_any_partner_available = {
    trigger_if = {
        limit = {
            c:MUG ?= this
        }
        trigger_if = {
            limit = {
                exists = c:FRA
            }
            c:FRA = {
                is_country_alive = yes
            }
            NOT = {
                has_war_with = c:FRA
            }
        }
        trigger_else = {
            always = no
        }
    }
    trigger_else_if = {
        limit = {
            c:SHA ?= this
        }
        trigger_if = {
            limit = {
                exists = c:HUS
            }
            c:HUS = {
                is_country_alive = yes
            }
            NOT = {
                has_war_with = c:HUS
            }
        }
        trigger_else = {
            always = no
        }
    }
    trigger_else = {
        OR = {
            AND = {
                trigger_if = {
                    limit = {
                        exists = c:USM
                    }
                    c:USM = {
                        is_country_alive = yes
                    }
                    NOT = {
                        has_war_with = c:USM
                    }
                }
                trigger_else = {
                    always = no
                }
            }
            AND = {
                c:QE1 ?= this
                trigger_if = {
                    limit = {
                        exists = c:SIH
                    }
                    c:SIH = {
                        is_country_alive = yes
                    }
                    NOT = {
                        has_war_with = c:SIH
                    }
                }
                trigger_else = {
                    always = no
                }
            }
            AND = {
                is_subject = no
                any_country = {
                    has_variable = usm_opposition_country_leader_var
                    is_country_alive = yes
                    NOT = {
                        has_war_with = root
                    }
                }
            }
        }
    }
}
nesnd_external_pending_politics_valid = {
    trigger_if = {
        limit = {
            exists = var:nesnd_external_target
            exists = var:nesnd_external_requester_sponsor
            exists = var:nesnd_external_receiver_sponsor
        }
        trigger_if = {
            limit = {
                is_subject = yes
            }
            overlord = var:nesnd_external_requester_sponsor
        }
        trigger_else = {
            this = var:nesnd_external_requester_sponsor
        }
        var:nesnd_external_target = {
            trigger_if = {
                limit = {
                    is_subject = yes
                }
                overlord = prev.var:nesnd_external_receiver_sponsor
            }
            trigger_else = {
                this = prev.var:nesnd_external_receiver_sponsor
            }
        }
    }
    trigger_else = {
        always = no
    }
}
