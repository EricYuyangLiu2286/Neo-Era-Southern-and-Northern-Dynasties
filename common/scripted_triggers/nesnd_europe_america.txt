# Austrian country actor, named current DP side/enemy/region. Mirrors the native scripted-rule body,
# with separate relation, truce and already-committed safeguards for the scripted action.
nesnd_canon_can_back_target = {
    trigger_if = { limit = { exists = scope:canon_side exists = scope:canon_enemy exists = scope:canon_play_region }
        is_country_alive = yes is_subject = no is_at_war = no is_diplomatic_play_committed_participant = no
        NOT = { this = scope:canon_side } NOT = { this = scope:canon_enemy }
        NOT = { has_war_with = scope:canon_side } NOT = { has_truce_with = scope:canon_enemy }
        NOT = { has_treaty_alliance_with = { TARGET = scope:canon_enemy } }
        NOT = { has_treaty_defensive_pact_with = { TARGET = scope:canon_enemy } }
        scope:canon_side = { is_country_alive = yes }
        scope:canon_enemy = { is_country_alive = yes NOT = { is_indirect_subject_of = prev } }
        NOT = { AND = {
            is_in_same_power_bloc = scope:canon_side is_in_same_power_bloc = scope:canon_enemy
            power_bloc ?= { modifier:power_bloc_disallow_war_bool = yes }
            scope:canon_side = { is_direct_subject_of = scope:canon_enemy }
        } }
        OR = {
            has_strategic_region_interest_tier = { strategic_region = scope:canon_play_region value >= interest_tier_effect_can_join_diplomatic_plays_min_tier }
            has_treaty_alliance_with = { TARGET = scope:canon_side }
            has_treaty_defensive_pact_with = { TARGET = scope:canon_side }
            scope:canon_side = { has_treaty_guarantee_independence_with = { TARGET = c:AUS } }
            scope:canon_side = { has_treaty_support_independence_with = { TARGET = c:AUS } }
            is_direct_subject_of = scope:canon_enemy
            scope:canon_enemy = { is_direct_subject_of = prev }
            is_direct_subject_of = scope:canon_side
            scope:canon_side = { is_direct_subject_of = prev }
        }
    }
    trigger_else = { always = no }
}

    trigger_else = { always = no }
}
