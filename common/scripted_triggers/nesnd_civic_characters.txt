# State scope
nesnd_c2_academy_state = { has_building = building_university }
nesnd_c2_sichuan_academy = {
    state_region = s:STATE_SICHUAN
    nesnd_c2_academy_state = yes
}

# Character scope
nesnd_c2_duan_politician = {
    has_template = nesnd_usm_duan
    is_character_alive = yes
    has_role_of_type = politician
}

# Country scope
nesnd_c2_duan_country = {
    OR = {
        nesnd_usm_civic_country = yes
        AND = { c:SIH ?= this is_country_alive = yes is_subject = no }
    }
    any_scope_state = { nesnd_c2_sichuan_academy = yes }
}
nesnd_c2_duan_introduction = {
    nesnd_c2_duan_country = yes
    game_date >= 1840.1.1
    game_date < 1863.1.1
    NOT = { has_global_variable = nesnd_introduced_usm_duan }
}
nesnd_c2_duan_commission = {
    nesnd_usm_civic_country = yes
    any_scope_state = { nesnd_c2_sichuan_academy = yes }
    has_global_variable = nesnd_introduced_usm_duan
    has_variable = nesnd_c2_usm_constitution_done
    OR = {
        var:nesnd_c2_usm_constitution_choice ?= 2
        var:nesnd_c2_usm_constitution_choice ?= 3
    }
    any_scope_character = {
        nesnd_c2_duan_politician = yes
        NOR = { has_variable = nesnd_c2_delegate_qualified has_variable = nesnd_c2_duan_commission_decided }
    }
}
nesnd_c2_duan_present = {
    exists = scope:c2_duan
    exists = scope:c2_duan_school
    scope:c2_duan ?= { owner = root nesnd_c2_duan_politician = yes }
    scope:c2_duan_school ?= { owner = root nesnd_c2_sichuan_academy = yes }
}

nesnd_c2_dayuan_school_available = {
    exists = scope:c2_dayuan
    scope:c2_dayuan ?= { owner = root state_population > 0 }
}
