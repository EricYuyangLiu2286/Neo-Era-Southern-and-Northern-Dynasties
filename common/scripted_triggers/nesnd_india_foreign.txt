# ==== Shared regional identity ====
# Bhutan and Nepal. Used by the independence journal (nesnd_india_indep_holder) and the monthly refresh.
nesnd_india_himalaya_country = { OR = { c:BHU ?= this c:NEP ?= this } }

# ==== Overseas dealings (India batch 17) ====
# Held by an independent, recognised Indian imperial government (nesnd_india_empire).
# Tightened from NOT puppet to is_subject = no so becoming any subject triggers the failure outcome
# ("Overseas Lost") once rather than reopening the journal every month.
nesnd_india_overseas_holder = {
	nesnd_india_empire = yes
	NOT = { is_country_type = unrecognized }
	is_subject = no
}

# Target country scope. An existing non-decentralised country whose capital lies in strategic region REGION,
# other than WHO and WHO's subjects.
nesnd_india_overseas_region_country = {
	is_country_alive = yes
	NOT = { is_country_type = decentralized }
	NOT = { this = $WHO$ }
	NOT = { is_subject_of = $WHO$ }
	exists = capital
	capital.region = sr:$REGION$
}

# Indian country scope. Another major or great power (not us or our subjects) has interest tier >= 3
# in REGION while we are below tier 3.
nesnd_india_overseas_rival_pressure = {
	save_temporary_scope_as = india_overseas_self
	NOT = {
		has_strategic_region_interest_tier = {
			strategic_region = sr:$REGION$
			value >= 3
		}
	}
	any_country = {
		is_country_alive = yes
		NOT = { this = scope:india_overseas_self }
		NOT = { is_subject_of = scope:india_overseas_self }
		country_rank >= rank_value:major_power
		has_strategic_region_interest_tier = {
			strategic_region = sr:$REGION$
			value >= 3
		}
	}
}

# Indian country scope. We are at war with at least one regional country of REGION.
nesnd_india_overseas_at_war_in_region = {
	save_temporary_scope_as = india_overseas_self
	any_country = {
		nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
		has_war_with = scope:india_overseas_self
	}
}

# Indian country scope. Raw check (no UI tooltips) that we can sign 10-year foreign investment rights
# where TARGET is source_country (host) and this is target_country (investor).
nesnd_india_overseas_can_sign_fir_raw = {
	$TARGET$ = { is_country_alive = yes exists = capital }
	NOT = { has_war_with = $TARGET$ }
	NOT = { has_treaty_foreign_investment_rights_with = { TARGET = $TARGET$ } }
	has_strategic_region_interest_tier = {
		strategic_region = $TARGET$.capital.region
		value >= 2
	}
	aggressive_diplomatic_plays_permitted = yes
	$TARGET$ = { aggressive_diplomatic_plays_permitted = yes }
	can_create_treaty = {
		first_country = this
		second_country = $TARGET$
		is_draft = no
		binding_period = { years = 10 }
		articles_to_create = {
			{
				article = foreign_investment_rights
				source_country = $TARGET$
				target_country = this
			}
		}
	}
}

# Indian country scope. Player-facing check on event nesnd_india_overseas.1 options, naming each missing item.
nesnd_india_overseas_can_sign_fir = {
	trigger_if = {
		limit = { exists = $TARGET$ }
		$TARGET$ = { is_country_alive = yes }
		NOT = { has_war_with = $TARGET$ }
		custom_tooltip = {
			text = nesnd_india_overseas_no_existing_fir_tt
			NOT = { has_treaty_foreign_investment_rights_with = { TARGET = $TARGET$ } }
		}
		custom_tooltip = {
			text = nesnd_india_overseas_interest_2_tt
			$TARGET$ = { exists = capital }
			has_strategic_region_interest_tier = {
				strategic_region = $TARGET$.capital.region
				value >= 2
			}
		}
		custom_tooltip = {
			text = nesnd_india_overseas_autonomy_tt
			aggressive_diplomatic_plays_permitted = yes
			$TARGET$ = { aggressive_diplomatic_plays_permitted = yes }
		}
		custom_tooltip = {
			text = nesnd_india_overseas_treaty_allowed_tt
			can_create_treaty = {
				first_country = this
				second_country = $TARGET$
				is_draft = no
				binding_period = { years = 10 }
				articles_to_create = {
					{
						article = foreign_investment_rights
						source_country = $TARGET$
						target_country = this
					}
				}
			}
		}
	}
	trigger_else = { always = no }
}

# Indian country scope. At least one regional country in REGION can sign foreign investment rights with us.
nesnd_india_overseas_has_fir_target = {
	custom_tooltip = {
		text = nesnd_india_overseas_has_fir_target_tt
		save_temporary_scope_as = india_overseas_self
		any_country = {
			nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
			save_temporary_scope_as = india_overseas_cand
			scope:india_overseas_self = { nesnd_india_overseas_can_sign_fir_raw = { TARGET = scope:india_overseas_cand } }
		}
	}
}

# Indian country scope. Possible trigger for the five Sign Investment Rights buttons.

}

# Indian country scope. The direction selected in scope:india_overseas_dir (1-5) has bar >= VAL.
nesnd_india_overseas_dir_bar_at_least = {
	trigger_if = {
		limit = { exists = scope:india_overseas_dir }
		trigger_if = { limit = { scope:india_overseas_dir = 1 } nesnd_india_overseas_bar_himalayas >= $VAL$ }
		trigger_else_if = { limit = { scope:india_overseas_dir = 2 } nesnd_india_overseas_bar_indonesia >= $VAL$ }
		trigger_else_if = { limit = { scope:india_overseas_dir = 3 } nesnd_india_overseas_bar_east_africa >= $VAL$ }
		trigger_else_if = { limit = { scope:india_overseas_dir = 4 } nesnd_india_overseas_bar_arabia >= $VAL$ }
		trigger_else = { nesnd_india_overseas_bar_persia >= $VAL$ }
	}
	trigger_else = { always = no }
}

# Indian country scope. Option trigger for event nesnd_india_overseas.1.
nesnd_india_overseas_fir_option_ok = {
	custom_tooltip = {
		text = nesnd_india_overseas_bar_50_tt

	}

	}

}

# Indian country scope. Option trigger for event nesnd_india_overseas.3.
nesnd_india_overseas_claim_option_ok = {
	custom_tooltip = {
		text = nesnd_india_overseas_bar_100_tt
		nesnd_india_overseas_dir_bar_at_least = { VAL = 100 }
	}
	trigger_if = {
		limit = { exists = $PICK$ }
		$PICK$ = { is_country_alive = yes }
		custom_tooltip = {
			text = nesnd_india_overseas_has_unclaimed_state_tt
			exists = $STATE$
			$STATE$ = {
				owner = $PICK$
				NOT = { has_claim_by = root }
			}
		}
	}
	trigger_else = { always = no }
}

# Indian country scope. We hold active foreign investment rights with at least one regional country
# of REGION and our interest tier in REGION is at least 2.
nesnd_india_overseas_dir_established = {
	save_temporary_scope_as = india_overseas_self
	has_strategic_region_interest_tier = {
		strategic_region = sr:$REGION$
		value >= 2
	}
	any_country = {
		nesnd_india_overseas_region_country = { REGION = $REGION$ WHO = scope:india_overseas_self }
		save_temporary_scope_as = india_overseas_partner
		scope:india_overseas_self = { has_treaty_foreign_investment_rights_with = { TARGET = scope:india_overseas_partner } }
	}
}

# Indian country scope. Milestone "Indian Ocean Trade": three distinct directions established, once only.
nesnd_india_overseas_ocean_trade_ready = {
	NOT = { has_variable = nesnd_india_ocean_trade }
	calc_true_if = {
		amount >= 3
		nesnd_india_overseas_dir_established = { REGION = region_himalayas }
		nesnd_india_overseas_dir_established = { REGION = region_indonesia }
		nesnd_india_overseas_dir_established = { REGION = region_east_africa }
		nesnd_india_overseas_dir_established = { REGION = region_arabia }
		nesnd_india_overseas_dir_established = { REGION = region_greater_persia }
	}
}
