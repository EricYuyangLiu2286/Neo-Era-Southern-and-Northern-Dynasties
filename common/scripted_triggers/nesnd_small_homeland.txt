# Fixed homeland groups are independent of the current choice of generic targets.
nesnd_small_native_one = {
 trigger_if = { limit = { owner = { nesnd_small_far_south = yes } } OR = { state_region = s:STATE_NEW_SOUTH_WALES state_region = s:STATE_VICTORIA state_region = s:STATE_QUEENSLAND state_region = s:STATE_SOUTH_AUSTRALIA state_region = s:STATE_WESTERN_AUSTRALIA state_region = s:STATE_NORTHERN_TERRITORY } }
 trigger_else_if = { limit = { owner = { nesnd_small_burma = yes } } OR = { state_region = s:STATE_MANDALAY state_region = s:STATE_KACHIN state_region = s:STATE_SHAN_STATES } }
 trigger_else_if = { limit = { owner = { nesnd_small_siam = yes } } state_region = s:STATE_BANGKOK }
 trigger_else = { always = no }
}
nesnd_small_native_two = {
 trigger_if = { limit = { owner = { nesnd_small_far_south = yes } } OR = { state_region = s:STATE_NORTH_ISLAND state_region = s:STATE_SOUTH_ISLAND } }
 trigger_else_if = { limit = { owner = { nesnd_small_burma = yes } } state_region = s:STATE_PEGU }
 trigger_else_if = { limit = { owner = { nesnd_small_siam = yes } }
 trigger_if = { limit = { owner = { nesnd_small_sia_interior_pop >= nesnd_small_sia_outer_pop } } OR = { state_region = s:STATE_NAKHON_RATCHASIMA state_region = s:STATE_LAOS } }
 trigger_else = { OR = { state_region = s:STATE_CAMBODIA state_region = s:STATE_TENASSERIM state_region = s:STATE_MALAYA } }
 }
 trigger_else = { always = no }
}
nesnd_small_restore_valid = {
 nesnd_small_population > 0 nesnd_small_populated_states >= 2
 nesnd_small_region_numbers_valid = { ONE_POP = nesnd_small_native_one_pop TWO_POP = nesnd_small_native_two_pop ONE_SHARE = nesnd_small_native_one_share TWO_SHARE = nesnd_small_native_two_share }
 any_scope_state = { owner = prev nesnd_small_native_one = yes state_population >= owner.nesnd_small_native_anchor_min }
 any_scope_state = { owner = prev nesnd_small_native_two = yes state_population >= owner.nesnd_small_native_anchor_min }
}
