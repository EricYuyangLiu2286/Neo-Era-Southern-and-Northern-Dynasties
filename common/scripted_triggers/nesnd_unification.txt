# Country scope: the capital lies in the 95-region unification scope.
nesnd_uni_capital_in_scope = {
	capital ?= {
		state_region = {
			nesnd_uni_scope_region = yes
		}
	}
}

# Country scope: takes part in the unification trend (capital in scope, or joined by owning 3 regions). Subjects count.
nesnd_uni_qualified = {
	NOT = { is_country_type = decentralized }
	OR = {
		nesnd_uni_capital_in_scope = yes
		has_variable = nesnd_uni_joined
	}
}

# Country scope: a power bloc member whose bloc leader is a qualified, independent country follows that leader and does not contend.
nesnd_uni_bloc_follower = {
	is_power_bloc_leader = no
	power_bloc ?= {
		power_bloc_leader ?= {
			nesnd_uni_qualified = yes
			is_subject = no
		}
	}
}

# Country scope: independent participants that follow no bloc leader are the contenders; subjects and bloc followers take part but do not contend.
nesnd_uni_contender = {
	nesnd_uni_qualified = yes
	is_subject = no
	nesnd_uni_bloc_follower = no
}

# Country scope: a Chinese power has its capital in the scope and is not wholly Japanese, Vietnamese, Ryukyuan or Ainu.
nesnd_uni_chinese_power = {
	nesnd_uni_capital_in_scope = yes
	NOT = {
		any_primary_culture = {
			count = all
			OR = {
				nesnd_jp_heritage_culture = yes
				this = cu:vietnamese
				this = cu:ryukyuan
				this = cu:ainu
			}
		}
	}
}

# State region scope, root = the claiming country: a claimed region of the scope that is not yet held whole (QE1 keeps Gansu and Hexi).
nesnd_uni_fading_claim_region = {
	nesnd_uni_scope_region = yes
	$C$ = { has_claim = prev }
	any_scope_state = {
		NOT = { owner = $C$ }
		NOT = {
			owner = { is_subject_of = $C$ }
		}
	}
	trigger_if = {
		limit = {
			$C$ = { c:QE1 ?= this }
		}
		NOR = {
			this = s:STATE_GANSU
			this = s:STATE_HEXI
		}
	}
}
