# Country scope: the capital lies in the 98-region unification scope.
nesnd_uni_capital_in_scope = {
	capital ?= {
		state_region = {
			nesnd_uni_scope_region = yes
		}
	}
}

# Country scope: the capital lies in modern China (the 58 regions).
nesnd_uni_capital_in_china = {
	capital ?= {
		state_region = {
			nesnd_uni_china_region = yes
		}
	}
}

# Country scope: takes part in the unification trend (capital in scope, or joined by owning 3 regions). Subjects count.
nesnd_uni_qualified = {
	NOT = { is_country_type = decentralized }
	OR = {
		nesnd_uni_capital_in_scope = yes
		has_variable = nesnd_uni_joined
	}
}

# Country scope: primary culture belongs to the Chinese world (Han and peoples of modern China).
nesnd_uni_chinese_culture = {
	any_primary_culture = {
		OR = {
			has_discrimination_trait = heritage_han
			has_discrimination_trait = heritage_tibetan
			has_discrimination_trait = heritage_mongolian
			has_discrimination_trait = heritage_uighur
			has_discrimination_trait = heritage_manchu
			has_discrimination_trait = heritage_miao
			has_discrimination_trait = heritage_yi
			has_discrimination_trait = heritage_formosan
			this = cu:zhuang
		}
	}
}

# State region scope, root = the claiming country: a claimed region of the scope that is not yet held whole (QE1 keeps Gansu and Hexi).
nesnd_uni_fading_claim_region = {
	nesnd_uni_scope_region = yes
	$C$ = { has_claim = prev }
	any_scope_state = {
		NOT = { owner = $C$ }
		NOT = {
			owner = { is_subject_of = $C$ }
		}
	}
	trigger_if = {
		limit = {
			$C$ = { c:QE1 ?= this }
		}
		NOR = {
			this = s:STATE_GANSU
			this = s:STATE_HEXI
		}
	}
}
