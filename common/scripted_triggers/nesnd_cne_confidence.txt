nesnd_cne_native_civil_war = {
    trigger_if = { limit = { exists = c:CNE }
        OR = {
            custom_tooltip = { text = nesnd_ui_cne_native_revolution_began_tt has_variable = nesnd_cne_native_revolution_began }
            any_diplomatic_play = {
                is_war = yes
                OR = { initiator = c:CNE target = c:CNE }
                OR = { is_diplomatic_play_type = dp_revolution is_diplomatic_play_type = dp_secession }
            }
        }
    }
    trigger_else = { always = no }
}
nesnd_cne_reform_failed = {
    OR = {
        nesnd_reform_extreme_crisis = yes
        nesnd_cne_coup_has_succeeded = yes
        NOT = { country_has_monarchy_law = yes }
        nesnd_cne_native_civil_war = yes
        nesnd_cne_confidence_exhausted = yes
    }
}
nesnd_cne_confidence_exhausted = {
    trigger_if = { limit = { hidden_trigger = { has_variable = nesnd_cne_confidence } } custom_tooltip = { text = nesnd_ui_var_nesnd_cne_confidence_le_0_tt var:nesnd_cne_confidence <= 0 } }
    trigger_else = { always = no }
}

nesnd_cne_reforms_in_progress = {
    OR = { has_variable = nesnd_cne_land_program has_variable = nesnd_cne_urban_program nesnd_cne_active_camp_count > 0 }
}
nesnd_cne_respite_confidence_ready = {
    trigger_if = { limit = { has_variable = nesnd_cne_confidence } var:nesnd_cne_confidence <= 900 }
    trigger_else = { always = no }
}
nesnd_cne_respite_confidence_urgent = {
    trigger_if = { limit = { has_variable = nesnd_cne_confidence } var:nesnd_cne_confidence <= 600 }
    trigger_else = { always = no }
}
nesnd_cne_respite_ready = {
    nesnd_cne_programs_live = yes
    nesnd_cne_reforms_in_progress = yes
    nesnd_cne_respite_confidence_ready = yes
    NOT = { has_variable = nesnd_cne_respite_used }
    NOT = { has_variable = nesnd_cne_respite_active }
}

# 身上有任一“在局部改革上让步”修正（改革完成时据此替换奖励）。
nesnd_cne_has_concession = {
    OR = {
        has_modifier = nesnd_cne_concession_landowners
        has_modifier = nesnd_cne_concession_armed_forces
        has_modifier = nesnd_cne_concession_industrialists
    }
}
# 「做出让步」依附的当权者。「宫廷力量」「执政者与派系」「统治者与内阁」是同一套日志的三种情形：
# 前两者共用 nesnd_power_holder / nesnd_power_holder_ig / nesnd_power_balance，后者用 nesnd_premier_pm / nesnd_premier_pm_ig / nesnd_premier_balance。
nesnd_cne_concession_court = {
    OR = { has_journal_entry = nesnd_palace_journal has_journal_entry = nesnd_cabinet_journal has_journal_entry = nesnd_premier_journal }
}
nesnd_cne_concession_has_holder = {
    trigger_if = { limit = { has_journal_entry = nesnd_premier_journal } nesnd_premier_has_pm = yes }
    trigger_else = { nesnd_power_has_holder = yes }
}
# $IG$ = ig_landowners | ig_armed_forces | ig_industrialists
nesnd_cne_concession_holder_ig = {
    trigger_if = { limit = { has_journal_entry = nesnd_premier_journal } hidden_trigger = { var:nesnd_premier_pm_ig ?= ig:$IG$ } }
    trigger_else = { hidden_trigger = { var:nesnd_power_holder_ig ?= ig:$IG$ } }
}
