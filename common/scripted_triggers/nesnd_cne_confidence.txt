nesnd_cne_native_civil_war = {
 trigger_if = { limit = { exists = c:CNE }
  OR = {
   has_variable = nesnd_cne_native_revolution_began
   any_diplomatic_play = {
    is_war = yes
    OR = { initiator = c:CNE target = c:CNE }
    OR = { is_diplomatic_play_type = dp_revolution is_diplomatic_play_type = dp_secession }
   }
  }
 }
 trigger_else = { always = no }
}
nesnd_cne_reform_failed = {
 OR = {
  nesnd_reform_extreme_crisis = yes
  nesnd_cne_coup_has_succeeded = yes
  NOT = { country_has_monarchy_law = yes }
  nesnd_cne_native_civil_war = yes
  nesnd_cne_confidence_exhausted = yes
 }
}
nesnd_cne_confidence_exhausted = {
 trigger_if = { limit = { has_variable = nesnd_cne_confidence } var:nesnd_cne_confidence <= 0 }
 trigger_else = { always = no }
}
