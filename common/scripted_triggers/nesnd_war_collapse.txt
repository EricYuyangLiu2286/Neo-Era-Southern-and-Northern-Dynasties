# Country scope. Restore the original capital OR occupation rule; Japan has its own two-centre rule.
nesnd_south_capital_defeat = {
    trigger_if = {
        limit = { exists = c:USM }
        is_country_alive = yes
        has_global_variable = usm_civilwar_occured
        nesnd_south_wu_survives = yes
        NOT = { has_variable = nesnd_holdout_resolved }
        nesnd_south_campaign_holdout = yes
        NOT = { c:JAP ?= this }
        NOT = { c:USM ?= this }
        is_subject = no
        OR = {
            nesnd_south_verified_wu_defeat = yes
            AND = {
                nesnd_south_defender_in_campaign_war = yes
                OR = {
                    AND = {
                        enemy_occupation > 0.8
                        NOT = { any_scope_war = { NOT = { exists = diplomatic_play diplomatic_play ?= { nesnd_south_named_play = yes } } } }
                    }
                    AND = {
                        trigger_if = { limit = { exists = capital }
                            capital = { OR = { controller = c:USM controller = { is_subject_of = c:USM } } }
                        } trigger_else = { always = no }
                    }
                }
            }
        }
    }
    trigger_else = { always = no }
}
# Called only after a living JAP and USM are established. A friendly controller is not occupation.
nesnd_japan_centre_occupied_by_wu_side = {
    owner = c:JAP
    exists = controller
    c:USM.var:nesnd_japan_jingnan_play.war ?= {
        is_war_participant = prev.controller
        NOT = { is_on_defending_side = prev.controller }
    }
}
nesnd_japan_two_centres_lost = {
    s:STATE_KYOTO = { any_scope_state = { nesnd_japan_centre_occupied_by_wu_side = yes } }
    s:STATE_KANTO = { any_scope_state = { nesnd_japan_centre_occupied_by_wu_side = yes } }
}
# USM scope. An enforced demand / restored losing-side settlement is an independent defeat fact.
nesnd_japan_collapse_ready = {
    trigger_if = {
        limit = { exists = c:JAP }
        c:USM ?= this
        is_country_alive = yes
        has_variable = tag_WUU_var
        c:JAP = { is_country_alive = yes has_variable = nesnd_japan_fought_against_wu NOT = { has_variable = nesnd_japan_recovery_created } }
        NOT = { has_global_variable = nesnd_japan_partitioned }
        NOT = { has_global_variable = nesnd_japan_collapse_committed }
        OR = {
            AND = { nesnd_japan_fighting_jingnan = yes nesnd_japan_two_centres_lost = yes }
            has_variable = nesnd_japan_defeat_confirmed
        }
    }
    trigger_else = { always = no }
}
nesnd_japan_collapse_viewer = {
    nesnd_japan_fighting_jingnan = yes
    OR = {
        c:JAP ?= this
        any_scope_war = {
            is_war_participant = prev
            NOT = { is_on_defending_side = prev }
            diplomatic_play ?= c:USM.var:nesnd_japan_jingnan_play
        }
    }
}

# Contextless: a real defeat must finish its country / recovery transaction before any status-quo verdict.
nesnd_south_defeat_resolution_pending = {
    custom_tooltip = { text = nesnd_sweep_defeat_pending
    trigger_if = {
        limit = { exists = c:USM }
        OR = {
            any_country = {
                is_country_alive = yes
                has_variable = usm_opposition_country
                NOT = { c:USM ?= this }
                NOT = { is_subject_of = c:USM }
                nesnd_south_capital_defeat = yes
            }
            AND = {
                NOT = { has_global_variable = nesnd_japan_partitioned }
                c:USM = {
                    OR = {
                        has_variable = nesnd_japan_recovery_pending
                        AND = {
                            has_variable = nesnd_japan_defeat_confirmed
                            exists = c:JAP
                            c:JAP ?= { is_country_alive = yes NOT = { has_variable = nesnd_japan_recovery_created } }
                        }
                    }
                }
            }
        }
    }
    trigger_else = { always = no }
    }
}
