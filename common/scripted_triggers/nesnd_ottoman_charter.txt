# Country scope: the Qin agenda meets both winning conditions (same counts as the journal outcome, without tooltips).
nesnd_ottoman_qin_agenda_met = {
    nesnd_ottoman_qin_score >= 5
    calc_true_if = {
        amount >= 4
        nesnd_ottoman_charter_q1 = yes
        nesnd_ottoman_charter_q2 = yes
        nesnd_ottoman_charter_q3 = yes
        nesnd_ottoman_charter_q4 = yes
        nesnd_ottoman_charter_q5 = yes
        nesnd_ottoman_charter_q6 = yes
    }
}
# Country scope (R10, L-14): when both agendas are met in the same month the higher score wins; a tie goes to the side
# chosen in event nesnd_ottoman_recovery.5. Read by the constitutional outcome, which is written first.
nesnd_ottoman_charter_wins_tie = {
    OR = {
        NOT = { nesnd_ottoman_qin_agenda_met = yes }
        nesnd_ottoman_charter_lead > 0
        AND = {
            nesnd_ottoman_charter_lead = 0
            NOT = { AND = { has_variable = nesnd_ottoman_route_lean var:nesnd_ottoman_route_lean = flag:barracks } }
        }
    }
}

    any_interest_group = { leader ?= { has_ideology = ideology:ideology_ce_qin_royalist_leader } }
}
