nesnd_sha_yan_open = {
    nesnd_sha_native = yes
    has_variable = nesnd_sha_northern_stage_ended
    NOT = { has_variable = nesnd_sha_yan_ended }
    OR = { var:nesnd_sha_route ?= flag:nesnd_sha_war var:nesnd_sha_route ?= flag:nesnd_sha_peace }
}
nesnd_sha_yan_preparing = {
    nesnd_sha_yan_open = yes
    OR = { has_variable = nesnd_sha_yan_preparation_window has_variable = nesnd_yan_notice }
    NOT = { has_variable = nesnd_sha_yan_rose }
}
# The northeastern states (Outer Manchuria is split into Haishenwai and Boli). Liaodong and Jilin join the rising
# only while CNE still holds them; the rest are the minimum (nesnd_sha_yan_four_states).
nesnd_sha_yan_region = {
    OR = {
        state_region = s:STATE_SHENGJING
        state_region = s:STATE_SOUTHERN_MANCHURIA
        state_region = s:STATE_SONGJIANG
        state_region = s:STATE_NORTHERN_MANCHURIA
        state_region = s:STATE_AMUR
        state_region = s:STATE_OUTER_MANCHURIA
        state_region = s:STATE_BOLI
    }
}
nesnd_sha_yan_four_states = {
    exists = s:STATE_SONGJIANG.region_state:CNE
    exists = s:STATE_NORTHERN_MANCHURIA.region_state:CNE
    exists = s:STATE_AMUR.region_state:CNE
    exists = s:STATE_OUTER_MANCHURIA.region_state:CNE
    exists = s:STATE_BOLI.region_state:CNE
}
# SHA scope. All five core garrisons are needed for both settlements and the Yan army.
nesnd_yan_conditions = {
    nesnd_sha_yan_preparing = yes
    has_variable = nesnd_sha_yan_contacts
    nesnd_sha_yan_four_states = yes
    exists = c:CNE
    has_journal_entry = nesnd_sha_restore_yan
    NOT = { exists = c:YAN }
    s:STATE_SONGJIANG.region_state:CNE ?= { nesnd_yan_contact_state = yes }
    s:STATE_OUTER_MANCHURIA.region_state:CNE ?= { nesnd_yan_contact_state = yes }
    s:STATE_NORTHERN_MANCHURIA.region_state:CNE ?= { nesnd_yan_contact_state = yes }
    s:STATE_AMUR.region_state:CNE ?= { nesnd_yan_contact_state = yes }
    s:STATE_BOLI.region_state:CNE ?= { nesnd_yan_contact_state = yes }
    c:CNE ?= {
        OR = { is_at_war = yes has_revolution = yes has_variable = nesnd_land_failed }
        any_scope_state = { nesnd_sha_yan_region = no }
    }
    trigger_if = {
        limit = { var:nesnd_sha_route ?= flag:nesnd_sha_peace }
        NOT = { has_modifier = declared_bankruptcy }
        c:CNE ?= { NOT = { has_war_with = c:SHA } }
    }
}
nesnd_yan_can_agree = {
    nesnd_yan_conditions = yes
    has_variable = nesnd_sha_yan_preparation_window
    NOR = { has_variable = nesnd_yan_cooldown has_variable = nesnd_yan_notice }
    c:CNE ?= { NOT = { has_journal_entry = nesnd_yan_warning } }
    je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 95 }
    OR = {
        NOT = { has_variable = nesnd_yan_wait_for_full }
        je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 100 }
    }
}
# State scope. Entries are individual state pieces, never a claim on a whole split region.
nesnd_yan_contact_state = {
    exists = c:SHA
    c:SHA ?= { is_target_in_variable_list = { name = nesnd_yan_contacts_by_state target = prev } }
}
nesnd_yan_announced_state = {
    exists = c:CNE
    c:CNE ?= { is_target_in_variable_list = { name = nesnd_yan_states target = prev } }
}
# CNE scope.
nesnd_yan_warning_valid = {
    c:CNE ?= this
    exists = c:SHA
    NOT = { has_variable = nesnd_yan_interrupted }
    c:SHA ?= {
        has_variable = nesnd_yan_notice
        nesnd_yan_conditions = yes
        je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 85 }
    }
    has_variable = nesnd_yan_count
    nesnd_yan_valid_state_count = var:nesnd_yan_count
    num_states > var:nesnd_yan_count
}
nesnd_sha_yan_can_rise = {
    exists = c:CNE
    has_journal_entry = nesnd_sha_restore_yan
    c:CNE ?= { nesnd_yan_warning_valid = yes }
    je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 100 }
}
nesnd_yan_can_contact = {
    nesnd_sha_yan_preparing = yes
    has_variable = nesnd_sha_yan_contacts
    has_variable = nesnd_yan_network_started
    NOR = { has_variable = nesnd_yan_notice has_variable = nesnd_yan_contact_cooldown }
    has_journal_entry = nesnd_sha_restore_yan
    je:nesnd_sha_restore_yan = { "scripted_bar_progress(nesnd_sha_yan_banner)" >= 50 }
    exists = s:$REGION$.region_state:CNE
    exists = s:STATE_SONGJIANG.region_state:CNE
    s:$REGION$.region_state:CNE ?= { nesnd_yan_contact_state = no nesnd_rep_reserved_state = no }
    s:STATE_SONGJIANG.region_state:CNE ?= { nesnd_yan_contact_state = yes }
}

nesnd_sha_yan_holds_terms = {
    exists = s:STATE_AMUR.region_state:SHA
    NOT = { exists = s:STATE_AMUR.region_state:YAN }
    # Haishenwai (Vladivostok) is Denglai's under both settlements; the first also cedes Boli.
    exists = s:STATE_OUTER_MANCHURIA.region_state:SHA
    NOT = { exists = s:STATE_OUTER_MANCHURIA.region_state:YAN }
    trigger_if = {
        limit = { NOT = { var:nesnd_sha_yan_partition ?= flag:nesnd_sha_yan_port } }
        exists = s:STATE_BOLI.region_state:SHA
        NOT = { exists = s:STATE_BOLI.region_state:YAN }
    }
}
