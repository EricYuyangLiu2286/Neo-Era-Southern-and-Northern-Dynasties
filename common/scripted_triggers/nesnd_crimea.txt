# Triggers for "Leader of the Islamic World" (nesnd_ottoman_islam_leader) and the Crimean / Northern War journals.
# Country scope unless noted.

nesnd_ottoman_islam_leader_fallback_due = {
    nesnd_ottoman_recovery_open = yes
    has_variable = nesnd_ottoman_route
    var:nesnd_ottoman_route = flag:constitutional
    nesnd_ottoman_tanzimat_success = yes
    NOT = { has_journal_entry = nesnd_ottoman_islam_leader }
    NOT = { has_variable = nesnd_ottoman_islam_leader_closed }
}

# Country scope (TUR). Any great power is allied with or in a defensive pact with the Ottoman state (Rule 7).
nesnd_ottoman_has_major_ally = {
    any_country = {
        is_country_alive = yes
        NOT = { this = root }
        country_rank >= rank_value:great_power
        OR = {
            has_treaty_alliance_with = { TARGET = root }
            has_treaty_defensive_pact_with = { TARGET = root }
        }
    }
}

# Any scope. The Crimean / Northern War can still be opened (either undecided, or reopened after 24m following flag:changed).
nesnd_crimea_can_open = {
    NOT = { has_global_variable = nesnd_crimea_active }
    OR = {
        NOT = { has_global_variable = nesnd_crimea_result }
        AND = {
            global_var:nesnd_crimea_result ?= flag:changed
            c:TUR ?= { NOT = { has_variable = nesnd_north_reopen_cd } }
        }
    }
}

# Any scope. The current northern foe holds at least one northern target state (Rule 7).
nesnd_crimea_rus_holds_target = {
    nesnd_north_foe_valid = yes
    global_var:nesnd_north_foe ?= {
        any_scope_state = {
            state_region = { nesnd_north_is_north_target_region = yes }
        }
    }
}

# Any scope. The Ottoman state holds at least one southern target state (Rule 7).
nesnd_crimea_tur_holds_balkan_or_blacksea = {
    exists = c:TUR
    c:TUR = {
        any_scope_state = {
            state_region = { nesnd_north_is_south_target_region = yes }
        }
    }
}

nesnd_crimea_can_sign_defensive_pact_with = {
    exists = $PARTNER$
    NOT = { $PARTNER$ = root }
    is_subject = no
    $PARTNER$ = { is_subject = no }
    NOT = { has_war_with = $PARTNER$ }
    NOT = { is_diplomatic_play_enemy_of = $PARTNER$ }
    NOT = { has_treaty_defensive_pact_with = { TARGET = $PARTNER$ } }
    NOT = { has_treaty_alliance_with = { TARGET = $PARTNER$ } }
    can_create_treaty = {
        first_country = root
        second_country = $PARTNER$
        is_draft = no
        binding_period = { years = 5 }
        articles_to_create = {
            { article = defensive_pact }
        }
    }
}

# Country scope (northern foe). Eligibility for the yearly pre-emptive check against TUR (Rules 7, 20.5).
nesnd_crimea_rus_yearly_eligible = {
    global_var:nesnd_north_foe ?= this
    nesnd_north_foe_valid = yes
    exists = c:TUR
    c:TUR = {
        has_journal_entry = nesnd_ottoman_islam_leader
        nesnd_ottoman_can_act_abroad = yes
        NOT = { has_variable = nesnd_crimea_tur_closed }
    }
    nesnd_crimea_can_open = yes
    NOT = { has_global_variable = nesnd_rtw_active }
    NOT = { has_war_with = c:TUR }
    NOT = { is_diplomatic_play_enemy_of = c:TUR }
    NOT = { is_diplomatic_play_ally_of = c:TUR }
    NOT = { has_truce_with = c:TUR }
    NOT = { has_treaty_alliance_with = { TARGET = c:TUR } }
    NOT = { has_treaty_defensive_pact_with = { TARGET = c:TUR } }
    nesnd_crimea_tur_holds_balkan_or_blacksea = yes
}
