# Triggers for "Leader of the Islamic World" (nesnd_ottoman_islam_leader) and the Crimean War journals. Country scope unless noted.

nesnd_ottoman_islam_leader_fallback_due = {
    nesnd_ottoman_recovery_open = yes
    has_variable = nesnd_ottoman_route
    var:nesnd_ottoman_route = flag:constitutional
    nesnd_ottoman_tanzimat_success = yes
    NOT = { has_journal_entry = nesnd_ottoman_islam_leader }
    NOT = { has_variable = nesnd_ottoman_islam_leader_closed }
}

nesnd_ottoman_has_major_ally = {
    OR = {
        AND = {
            exists = c:GBR
            OR = {
                has_treaty_alliance_with = { TARGET = c:GBR }
                has_treaty_defensive_pact_with = { TARGET = c:GBR }
            }
        }
        AND = {
            exists = c:FRA
            OR = {
                has_treaty_alliance_with = { TARGET = c:FRA }
                has_treaty_defensive_pact_with = { TARGET = c:FRA }
            }
        }
        AND = {
            exists = c:AUS
            OR = {
                has_treaty_alliance_with = { TARGET = c:AUS }
                has_treaty_defensive_pact_with = { TARGET = c:AUS }
            }
        }
        AND = {
            exists = c:PRU
            OR = {
                has_treaty_alliance_with = { TARGET = c:PRU }
                has_treaty_defensive_pact_with = { TARGET = c:PRU }
            }
        }
    }
}

nesnd_crimea_rus_holds_target = {
    exists = c:RUS
    c:RUS = {
        any_scope_state = {
            OR = {
                state_region = s:STATE_KARS
                state_region = s:STATE_CRIMEA
                state_region = s:STATE_GREATER_CAUCASUS
                state_region = s:STATE_ARMENIA
            }
        }
    }
}

nesnd_crimea_tur_holds_balkan_or_blacksea = {
    exists = c:TUR
    c:TUR = {
        any_scope_state = {
            OR = {
                state_region = s:STATE_DOBRUDJA
                state_region = s:STATE_KARS
                state_region = s:STATE_BULGARIA
                state_region = s:STATE_NORTHERN_THRACE
                state_region = s:STATE_TRABZON
            }
        }
    }
}

nesnd_crimea_play_in_progress = {
    exists = c:TUR
    exists = c:RUS
    any_diplomatic_play = {
        OR = {
            AND = { initiator = c:TUR target = c:RUS }
            AND = { initiator = c:RUS target = c:TUR }
        }
    }
}

nesnd_crimea_play_before_war = {
    exists = c:TUR
    exists = c:RUS
    any_diplomatic_play = {
        is_war = no
        OR = {
            AND = { initiator = c:TUR target = c:RUS }
            AND = { initiator = c:RUS target = c:TUR }
        }
    }
}

nesnd_crimea_tur_won = {
    OR = {
        NOT = { exists = c:RUS }
        AND = {
            NOT = { has_war_with = c:RUS }
            NOT = { is_diplomatic_play_enemy_of = c:RUS }
            OR = {
                global_var:nesnd_crimea_war_winner ?= flag:ottoman
                AND = { global_var:nesnd_crimea_target ?= flag:crimea owns_entire_state_region = STATE_CRIMEA }
                AND = { global_var:nesnd_crimea_target ?= flag:kars global_var:nesnd_crimea_initiator ?= flag:ottoman owns_entire_state_region = STATE_KARS }
                AND = { global_var:nesnd_crimea_target ?= flag:greater_caucasus owns_entire_state_region = STATE_GREATER_CAUCASUS }
                AND = { global_var:nesnd_crimea_target ?= flag:armenia owns_entire_state_region = STATE_ARMENIA }
            }
        }
    }
}

nesnd_crimea_rus_won = {
    exists = c:RUS
    NOT = { has_war_with = c:RUS }
    NOT = { is_diplomatic_play_enemy_of = c:RUS }
    global_var:nesnd_crimea_war_winner ?= flag:russia
}

nesnd_crimea_can_sign_defensive_pact_with = {
    exists = $PARTNER$
    NOT = { $PARTNER$ = root }
    is_subject = no
    $PARTNER$ = { is_subject = no }
    NOT = { has_war_with = $PARTNER$ }
    NOT = { is_diplomatic_play_enemy_of = $PARTNER$ }
    NOT = { has_treaty_defensive_pact_with = { TARGET = $PARTNER$ } }
    NOT = { has_treaty_alliance_with = { TARGET = $PARTNER$ } }
    can_create_treaty = {
        first_country = root
        second_country = $PARTNER$
        is_draft = no
        binding_period = { years = 5 }
        articles_to_create = {
            { article = defensive_pact }
        }
    }
}

nesnd_crimea_rus_yearly_eligible = {
    c:RUS ?= this
    is_subject = no
    exists = c:TUR
    c:TUR = {
        has_journal_entry = nesnd_ottoman_islam_leader
        is_subject = no
    }
    NOT = { has_global_variable = nesnd_crimea_active }
    NOT = { has_global_variable = nesnd_crimea_result }
    NOT = { has_war_with = c:TUR }
    NOT = { is_diplomatic_play_enemy_of = c:TUR }
    NOT = { is_diplomatic_play_ally_of = c:TUR }
    NOT = { has_truce_with = c:TUR }
    NOT = { has_treaty_alliance_with = { TARGET = c:TUR } }
    NOT = { has_treaty_defensive_pact_with = { TARGET = c:TUR } }
    nesnd_crimea_tur_holds_balkan_or_blacksea = yes
}
