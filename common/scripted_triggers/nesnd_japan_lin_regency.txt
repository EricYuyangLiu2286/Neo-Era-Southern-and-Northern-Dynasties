nesnd_japan_lin_candidate_eligible = {
    has_template = USM_lin_hanhan is_character_alive = yes owner = c:USM
    is_ruler_of_own_country = no is_heir_of_own_country = no is_in_exile_pool = no
}
nesnd_japan_lin_realm_ready = {
    c:USM ?= this is_country_alive = yes has_variable = tag_WUU_var
    has_global_variable = nesnd_japan_partitioned
    c:JAP ?= { is_country_alive = yes is_subject_of = c:USM }
    c:JTK ?= { is_country_alive = yes is_subject_of = c:USM }
    c:JIM ?= {
        is_country_alive = yes has_variable = nesnd_japan_partition_state
        is_direct_subject_of = c:USM country_has_monarchy_law = yes
        NOT = { has_war_with = c:USM }
        OR = { ruler ?= { has_variable = house_yamato } ruler ?= { has_template = USM_lin_hanhan } }
    }
}
nesnd_japan_lin_pending_identity_valid = {
    trigger_if = { limit = { exists = var:nesnd_japan_lin_pending_country exists = var:nesnd_japan_lin_pending_officer }
        var:nesnd_japan_lin_pending_country = c:JIM
        var:nesnd_japan_lin_pending_officer = {
            has_template = USM_lin_hanhan is_character_alive = yes is_in_exile_pool = no is_heir_of_own_country = no
            OR = { AND = { owner = c:USM is_ruler_of_own_country = no } owner = c:JIM }
        }
    } trigger_else = { always = no }
}
nesnd_japan_lin_can_be_appointed = {
    nesnd_japan_lin_realm_ready = yes
    NOT = { has_variable = nesnd_japan_lin_commission_completed }
    NOT = { has_variable = nesnd_jp_legacy_lin }
    c:JIM ?= { NOT = { has_variable = nesnd_japan_lin_regency_appointed } }
    OR = { nesnd_japan_lin_pending_identity_valid = yes any_scope_character = { nesnd_japan_lin_candidate_eligible = yes } }
}
nesnd_japan_lin_regency_is_active = {
    trigger_if = { limit = { exists = var:nesnd_japan_lin_regent }
        c:JIM ?= this is_country_alive = yes country_has_monarchy_law = yes
        has_variable = nesnd_japan_lin_regency_active
        var:nesnd_japan_lin_regent = {
            has_template = USM_lin_hanhan is_character_alive = yes owner = c:JIM
            is_ruler_of_own_country = yes has_variable = nesnd_japan_lin_commissioned
        }
    } trigger_else = { always = no }
}
nesnd_jp_emperor_valid = {
    trigger_if = { limit = { exists = var:nesnd_jp_emperor }
        var:nesnd_jp_emperor = { is_character_alive = yes owner = c:JIM is_in_exile_pool = no has_variable = house_yamato }
    } trigger_else = { always = no }
}

nesnd_jp_story_succession_valid = {
    trigger_if = { limit = { exists = scope:japanese_heir exists = scope:japanese_shogun }
        c:JAP ?= this country_has_monarchy_law = yes
        scope:japanese_shogun = { is_character_alive = yes owner = c:JAP is_ruler_of_own_country = yes }
        scope:japanese_heir = { is_character_alive = yes owner = c:JAP is_in_exile_pool = no is_ruler_of_own_country = no }
    } trigger_else = { always = no }
}
