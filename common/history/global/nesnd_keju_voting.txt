GLOBAL = {
	# 00_global.txt grants republican paternalism to non-monarchies at game start.
	c:YN1 ?= {
		ig:ig_landowners ?= {
			remove_ideology = ideology_republican_paternalistic
		}
	}
	c:HNA ?= {
		ig:ig_landowners ?= {
			remove_ideology = ideology_republican_paternalistic
		}
	}

	every_country = {
		limit = { any_primary_culture = { has_discrimination_trait = tradition_sinosphere } }
		ig:ig_intelligentsia ?= {
			if = {
				limit = { has_ideology = ideology:ideology_constitutionalist }
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_nesnd_confucian_constitutionalist
			}
			else_if = {
				limit = { has_ideology = ideology:ideology_republican }
				remove_ideology = ideology_republican
				add_ideology = ideology_nesnd_confucian_republican
			}
		}
	}
}
