# NESND compatibility override for EACR 1.0.4 (3704408897).
# EACR must load first. Its selection logic below remains unchanged.
# Our EAC_hairstyles slot is special/nonpersistent: add the display accessory;
# keep replace only for the persistent vanilla hairstyle that is suppressed.
# No accessory definition, model, texture, animation or asset is bundled here.
@EACR_PORTRAIT_MODIFIER_WEIGHT_LOW  = 10000000
@EACR_PORTRAIT_MODIFIER_WEIGHT_HIGH = 20000000

INJECT:hairstyles = {

    EACR_topknot_hairstyles = {
        dna_modifiers = {
            accessory = {
                mode = add
                gene = EAC_hairstyles
                template = topknot_hairstyles
                value = 0
            }

            # Override vanilla
            accessory = {
                mode = replace
                gene = hairstyles
                template = no_hairstyles
                value = 0
            }
        }
        weight = {
            base = 0
            modifier = {
                add = @EACR_PORTRAIT_MODIFIER_WEIGHT_LOW

                # Unwesternized Sinosphere & Confucian people
                # Filial piety and all that
                EACR_st_is_unwesternized = yes
                OR = {
                    scope:pop ?= {
                        save_temporary_scope_as = person
                    }
                    scope:character ?= {
                        save_temporary_scope_as = person
                    }
                }
                scope:person ?= {
                    OR = {
                        culture = {
                            has_discrimination_trait = tradition_sinosphere
                        }
                        religion = rel:confucian
                    }
                }

                # Do not override Manchu queue
                chinese_manchu_queue_hairstyle_character_trigger = no
                chinese_manchu_queue_hairstyle_pops_trigger = no

                # Buddhist monks are bald
                NOT = {
                    scope:person ?= {
                        religion = {
                            has_discrimination_trait = heritage_dharmic
                        }
                        OR = {
                            clergy_clothes_pop_trigger = yes
                            AND = {
                                politician_clothes_trigger = yes
                                devout_clothes_trigger = yes
                            }
                        }
                    }
                }
            }
        }
    }

    EACR_topknot_hairstyles_chinese = {
        dna_modifiers = {
            accessory = {
                mode = add
                gene = EAC_hairstyles
                template = topknot_hairstyles
                value = 1
            }

            # Override vanilla
            accessory = {
                mode = replace
                gene = hairstyles
                template = no_hairstyles
                value = 0
            }
        }
        weight = {
            base = 0
            modifier = {
                add = @EACR_PORTRAIT_MODIFIER_WEIGHT_HIGH

                # Unwesternized Chinese people
                EACR_st_is_unwesternized = yes
                OR = {
                    scope:pop ?= {
                        save_temporary_scope_as = person
                    }
                    scope:character ?= {
                        save_temporary_scope_as = person
                    }
                }
                scope:person ?= {
                    culture = {
                        EACR_st_culture_is_chinese = yes
                    }
                }

                # Do not override Manchu queue
                chinese_manchu_queue_hairstyle_character_trigger = no
                chinese_manchu_queue_hairstyle_pops_trigger = no

                # Buddhist monks are bald
                NOT = {
                    scope:person ?= {
                        religion = {
                            has_discrimination_trait = heritage_dharmic
                        }
                        OR = {
                            clergy_clothes_pop_trigger = yes
                            AND = {
                                politician_clothes_trigger = yes
                                devout_clothes_trigger = yes
                            }
                        }
                    }
                }
            }
        }
    }

}
